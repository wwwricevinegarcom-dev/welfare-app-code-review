'use client';

import { useState } from 'react';
import { Card, CardContent, CardHeader, CardTitle, CardDescription } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Textarea } from "@/components/ui/textarea";
import { 
  AlertOctagon, 
  AlertCircle, 
  MessageSquareWarning, 
  Plus, 
  Printer, 
  Trash2,
  CheckCircle2,
  Edit
} from "lucide-react";
import { saveIncidentReport, deleteIncidentReport } from "../actions";
import { toast } from "sonner";
import { toLocalISODateTime } from "@/lib/utils";

interface IncidentsClientProps {
  reports: any[];
  clients: any[];
}

const TYPE_CONFIG = {
  accident: { label: '事故・怪我', color: 'bg-red-100 text-red-800 border-red-200', icon: AlertOctagon },
  near_miss: { label: 'ヒヤリハット', color: 'bg-amber-100 text-amber-800 border-amber-200', icon: AlertCircle },
  complaint: { label: '苦情・要望', color: 'bg-blue-100 text-blue-800 border-blue-200', icon: MessageSquareWarning },
};

export function IncidentsClient({ reports, clients }: IncidentsClientProps) {
  const [selectedType, setSelectedType] = useState<string>('all');
  const [isModalOpen, setIsModalOpen] = useState(false);
  const [editingReport, setEditingReport] = useState<any>(null);
  const [loading, setLoading] = useState(false);

  const filteredReports = selectedType === 'all'
    ? reports
    : reports.filter(r => r.report_type === selectedType);

  const handleOpenCreate = () => {
    setEditingReport(null);
    setIsModalOpen(true);
  };

  const handleOpenEdit = (r: any) => {
    setEditingReport(r);
    setIsModalOpen(true);
  };

  const handleSave = async (e: React.FormEvent<HTMLFormElement>) => {
    e.preventDefault();
    setLoading(true);
    const formData = new FormData(e.currentTarget);
    if (editingReport?.id) {
      formData.append('id', editingReport.id);
    }

    try {
      const res = await saveIncidentReport(formData);
      if (res?.success === false) {
        toast.error(res.error || '保存に失敗しました');
        return;
      }
      toast.success(editingReport?.id ? '報告書を更新しました' : '報告書を保存しました');
      setIsModalOpen(false);
      setEditingReport(null);
    } catch (err: any) {
      toast.error(err.message || '保存に失敗しました');
    } finally {
      setLoading(false);
    }
  };

  const handleDelete = async (id: string) => {
    if (!confirm('この報告書を削除しますか？')) return;
    try {
      const res = await deleteIncidentReport(id);
      if (res?.success === false) {
        toast.error(res.error || '削除に失敗しました');
        return;
      }
      toast.success('削除しました');
    } catch (err: any) {
      toast.error(err.message || '削除に失敗しました');
    }
  };

  return (
    <div className="space-y-6">
      {/* フィルタ & アクション */}
      <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center gap-4 print:hidden">
        <div className="flex flex-wrap gap-2">
          <Button 
            size="sm" 
            variant={selectedType === 'all' ? 'default' : 'outline'}
            onClick={() => setSelectedType('all')}
          >
            すべて ({reports.length})
          </Button>
          <Button 
            size="sm" 
            variant={selectedType === 'accident' ? 'default' : 'outline'}
            onClick={() => setSelectedType('accident')}
          >
            事故 ({reports.filter(r => r.report_type === 'accident').length})
          </Button>
          <Button 
            size="sm" 
            variant={selectedType === 'near_miss' ? 'default' : 'outline'}
            onClick={() => setSelectedType('near_miss')}
          >
            ヒヤリハット ({reports.filter(r => r.report_type === 'near_miss').length})
          </Button>
          <Button 
            size="sm" 
            variant={selectedType === 'complaint' ? 'default' : 'outline'}
            onClick={() => setSelectedType('complaint')}
          >
            苦情・相談 ({reports.filter(r => r.report_type === 'complaint').length})
          </Button>
        </div>

        <div className="flex gap-2 w-full sm:w-auto">
          <Button variant="outline" size="sm" onClick={() => window.print()}>
            <Printer className="h-4 w-4 mr-1" /> 印刷
          </Button>
          <Button size="sm" onClick={handleOpenCreate}>
            <Plus className="h-4 w-4 mr-1" /> 報告書を作成
          </Button>
        </div>
      </div>

      {/* レポート一覧 */}
      <div className="space-y-4">
        {filteredReports.length === 0 ? (
          <Card className="p-8 text-center text-muted-foreground">
            記録がありません。「報告書を作成」ボタンから事故やヒヤリハット、苦情対応を記録してください。
          </Card>
        ) : (
          filteredReports.map((r) => {
            const config = TYPE_CONFIG[r.report_type as keyof typeof TYPE_CONFIG] || TYPE_CONFIG.accident;
            const Icon = config.icon;

            return (
              <Card key={r.id} className="shadow-sm">
                <CardHeader className="bg-muted/20 pb-3">
                  <div className="flex justify-between items-start">
                    <div>
                      <div className="flex items-center gap-2 mb-1">
                        <span className={`text-xs px-2.5 py-0.5 rounded-full font-bold border flex items-center gap-1 ${config.color}`}>
                          <Icon className="h-3.5 w-3.5" />
                          {config.label}
                        </span>
                        {r.is_reported_to_gov === 1 && (
                          <span className="text-xs px-2 py-0.5 rounded bg-red-600 text-white font-bold">
                            自治体報告済
                          </span>
                        )}
                        <span className="text-sm font-semibold text-muted-foreground">
                          発生日時: {r.occurred_at}
                        </span>
                      </div>
                      <CardTitle className="text-lg">
                        {r.client_name ? `対象利用者: ${r.client_name} 様` : '対象者: なし / 施設全体'}
                      </CardTitle>
                    </div>
                    <div className="flex items-center gap-1 print:hidden">
                      <Button 
                        variant="ghost" 
                        size="sm" 
                        className="h-8 px-2 text-muted-foreground hover:text-blue-600 gap-1 text-xs"
                        onClick={() => handleOpenEdit(r)}
                        title="編集"
                      >
                        <Edit className="h-3.5 w-3.5" />
                        <span>編集</span>
                      </Button>
                      <Button 
                        variant="ghost" 
                        size="icon" 
                        className="h-8 w-8 text-muted-foreground hover:text-destructive"
                        onClick={() => handleDelete(r.id)}
                        title="削除"
                      >
                        <Trash2 className="h-4 w-4" />
                      </Button>
                    </div>
                  </div>
                </CardHeader>
                <CardContent className="pt-4 space-y-4 text-sm">
                  <div className="grid grid-cols-1 md:grid-cols-2 gap-2 text-xs text-muted-foreground">
                    <p><span className="font-semibold text-foreground">発生場所:</span> {r.location || '事業所内'}</p>
                    <p><span className="font-semibold text-foreground">報告者:</span> {r.reporter_name}</p>
                  </div>

                  <div>
                    <p className="text-xs font-bold text-muted-foreground">【発生状況・詳細】</p>
                    <p className="mt-0.5 whitespace-pre-wrap">{r.situation_details}</p>
                  </div>

                  <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                    {r.cause_analysis && (
                      <div className="p-3 bg-muted/20 rounded border">
                        <p className="text-xs font-bold text-muted-foreground">【原因分析・背景】</p>
                        <p className="mt-1 text-xs whitespace-pre-wrap">{r.cause_analysis}</p>
                      </div>
                    )}
                    {r.immediate_response && (
                      <div className="p-3 bg-muted/20 rounded border">
                        <p className="text-xs font-bold text-muted-foreground">【直後の応急対応・処置】</p>
                        <p className="mt-1 text-xs whitespace-pre-wrap">{r.immediate_response}</p>
                      </div>
                    )}
                  </div>

                  {r.preventive_measures && (
                    <div className="p-3 bg-blue-50/50 rounded border border-blue-200">
                      <p className="text-xs font-bold text-blue-900">【再発防止策・改善措置】</p>
                      <p className="mt-1 text-xs text-blue-950 whitespace-pre-wrap">{r.preventive_measures}</p>
                    </div>
                  )}

                  {r.family_notified_at && (
                    <div className="p-3 bg-amber-50/60 rounded border border-amber-200 text-xs">
                      <p className="font-bold text-amber-900 mb-1">【ご家族・緊急連絡先への通知エビデンス】</p>
                      <div className="grid grid-cols-1 sm:grid-cols-3 gap-2 text-amber-950">
                        <p><span className="font-semibold">通知日時:</span> {r.family_notified_at}</p>
                        <p><span className="font-semibold">通知担当者:</span> {r.family_notified_by || '未入力'}</p>
                        <p><span className="font-semibold">連絡手段:</span> {r.family_contact_method || '電話等'}</p>
                      </div>
                    </div>
                  )}

                  {r.medical_institution_visited === 1 && (
                    <div className="p-3 bg-blue-50/60 rounded border border-blue-200 text-xs">
                      <p className="font-bold text-blue-900 mb-1">【医療機関受診・医師指示エビデンス】</p>
                      <div className="grid grid-cols-1 sm:grid-cols-2 gap-2 text-blue-950">
                        <p><span className="font-semibold">受診先医療機関:</span> {r.medical_institution_name || '受診済'}</p>
                        <p><span className="font-semibold">医師指示事項:</span> {r.doctor_instructions || '指示記録あり'}</p>
                      </div>
                    </div>
                  )}

                  {r.is_reported_to_gov === 1 && (
                    <div className="p-3 bg-red-50/60 rounded border border-red-200 text-xs">
                      <p className="font-bold text-red-900 mb-1">【指定権者（自治体）への事故報告エビデンス】</p>
                      <div className="grid grid-cols-1 sm:grid-cols-3 gap-2 text-red-950">
                        <p><span className="font-semibold">報告日時:</span> {r.gov_reported_at || '記録あり'}</p>
                        <p><span className="font-semibold">報告先自治体・窓口:</span> {r.gov_reported_to || '未入力'}</p>
                        <p><span className="font-semibold">報告担当職員:</span> {r.gov_reported_by || '未入力'}</p>
                      </div>
                    </div>
                  )}

                  {r.manager_comment && (
                    <div className="p-3 bg-slate-50 rounded border text-xs">
                      <span className="font-semibold text-foreground">管理者コメント・承認確認:</span> {r.manager_comment}
                    </div>
                  )}
                </CardContent>
              </Card>
            );
          })
        )}
      </div>

      {/* 新規登録・編集モーダル */}
      {isModalOpen && (
        <div className="fixed inset-0 bg-black/50 z-50 flex items-center justify-center p-4">
          <Card key={editingReport?.id || 'new_incident'} className="w-full max-w-2xl max-h-[90vh] flex flex-col shadow-xl">
            <CardHeader className="border-b bg-muted/20">
              <CardTitle>{editingReport ? '事故・ヒヤリハット報告書の編集' : '事故・ヒヤリハット報告書の作成'}</CardTitle>
              <CardDescription>
                法令遵守および再発防止のため、発生状況と家族・行政への報告エビデンスを客観的かつ具体的に記録してください。
              </CardDescription>
            </CardHeader>
            <form onSubmit={handleSave} className="overflow-y-auto flex-1">
              <CardContent className="pt-4 space-y-4">
                <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
                  <div className="space-y-1">
                    <Label htmlFor="reportType">種別 *</Label>
                    <select id="reportType" name="reportType" required defaultValue={editingReport?.report_type || 'accident'} className="w-full h-10 px-3 rounded-md border bg-background text-sm">
                      <option value="accident">事故・怪我（要通院等）</option>
                      <option value="near_miss">ヒヤリハット（未然防止）</option>
                      <option value="complaint">苦情・要望対応</option>
                    </select>
                  </div>
                  <div className="space-y-1">
                    <Label htmlFor="occurredAt">発生日時 *</Label>
                    <Input id="occurredAt" name="occurredAt" type="datetime-local" required defaultValue={editingReport?.occurred_at ? editingReport.occurred_at.slice(0, 16) : toLocalISODateTime()} />
                  </div>
                  <div className="space-y-1">
                    <Label htmlFor="clientId">対象利用者</Label>
                    <select id="clientId" name="clientId" defaultValue={editingReport?.client_id || ''} className="w-full h-10 px-3 rounded-md border bg-background text-sm">
                      <option value="">（対象者なし・全般）</option>
                      {clients.map(c => (
                        <option key={c.id} value={c.id}>{c.full_name}</option>
                      ))}
                    </select>
                  </div>
                </div>

                <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                  <div className="space-y-1">
                    <Label htmlFor="location">発生場所</Label>
                    <Input id="location" name="location" defaultValue={editingReport?.location || '作業室内'} />
                  </div>
                  <div className="space-y-1">
                    <Label htmlFor="reporterName">報告者名 *</Label>
                    <Input id="reporterName" name="reporterName" required placeholder="氏名" defaultValue={editingReport?.reporter_name || ''} />
                  </div>
                </div>

                <div className="space-y-1">
                  <Label htmlFor="situationDetails">発生状況・詳細 *</Label>
                  <Textarea id="situationDetails" name="situationDetails" rows={3} required placeholder="いつ、どこで、誰が、何をして、どうなったかを客観的に記載" defaultValue={editingReport?.situation_details || ''} />
                </div>

                <div className="space-y-1">
                  <Label htmlFor="immediateResponse">直後の応急処置・対応</Label>
                  <Textarea id="immediateResponse" name="immediateResponse" rows={2} placeholder="応急手当、病院搬送、家族・関係機関への連絡など" defaultValue={editingReport?.immediate_response || ''} />
                </div>

                {/* 家族連絡エビデンス（実地指導・安全配慮義務） */}
                <div className="p-3 border rounded-md bg-amber-50/40 space-y-2">
                  <div className="flex items-center justify-between">
                    <Label className="font-bold text-amber-950 text-xs">ご家族・緊急連絡先への連絡記録（安全配慮義務・指導監査対応）</Label>

                      <Label htmlFor="familyNotifiedAt" className="text-xs">連絡日時</Label>
                      <Input id="familyNotifiedAt" name="familyNotifiedAt" type="datetime-local" className="h-8 text-xs bg-white" defaultValue={editingReport?.family_notified_at ? editingReport.family_notified_at.slice(0, 16) : ''} />

                    <Label htmlFor="isReportedToGov" className="text-sm font-semibold text-red-950">
                      指定権者（自治体・障害福祉窓口）へ事故報告書を提出済（実地指導・法令遵守）
                    </Label>
                  </div>
                  <div className="grid grid-cols-1 md:grid-cols-3 gap-3 pt-1">
                    <div className="space-y-1">
                      <Label htmlFor="govReportedAt" className="text-xs">報告日時</Label>
                      <Input id="govReportedAt" name="govReportedAt" type="datetime-local" className="h-8 text-xs bg-white" defaultValue={editingReport?.gov_reported_at ? editingReport.gov_reported_at.slice(0, 16) : ''} />
                    </div>
                    <div className="space-y-1">
                      <Label htmlFor="govReportedTo" className="text-xs">報告先自治体・担当窓口</Label>
                      <Input id="govReportedTo" name="govReportedTo" placeholder="例: 市障害福祉課 ○○様" className="h-8 text-xs bg-white" defaultValue={editingReport?.gov_reported_to || ''} />
                    </div>
                    <div className="space-y-1">
                      <Label htmlFor="govReportedBy" className="text-xs">報告担当職員</Label>
                      <Input id="govReportedBy" name="govReportedBy" placeholder="例: 管理者 ○○" className="h-8 text-xs bg-white" defaultValue={editingReport?.gov_reported_by || ''} />
                    </div>
                  </div>
                </div>
              </CardContent>

              <div className="p-4 border-t flex justify-end gap-2 bg-muted/20">
                <Button type="button" variant="outline" onClick={() => { setIsModalOpen(false); setEditingReport(null); }}>キャンセル</Button>
                <Button type="submit" disabled={loading}>{loading ? '保存中...' : (editingReport ? '変更を保存する' : '保存する')}</Button>
              </div>
            </form>
          </Card>
        </div>
      )}
    </div>
  );
}

import { Header } from "@/components/Header";
import { SupportSubNav } from "@/components/NavigationTabs";
import { getRole } from "@/lib/auth";
import { redirect } from "next/navigation";
import { getIncidentReports } from "../actions";
import { IncidentsClient } from "./IncidentsClient";
import { db } from "@/lib/db";

export default async function IncidentsPage() {
  const role = await getRole();
  if (!role) {
    redirect("/login");
  }

  const reports = await getIncidentReports();
  const clients = db.prepare('SELECT id, full_name FROM clients WHERE deleted_at IS NULL ORDER BY full_name ASC').all() as any[];

  return (
    <div className="flex min-h-screen flex-col bg-background/50">
      <div className="print:hidden">
        <Header activePage="records/incidents" />
        <SupportSubNav />
      </div>

      <main className="flex-1 p-3.5 sm:p-6 md:p-12">
        <div className="mx-auto max-w-5xl space-y-6">
          <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
            <div>
              <h1 className="text-2xl sm:text-3xl font-bold tracking-tight text-foreground">
                事故・ヒヤリハット・苦情対応記録
              </h1>
              <p className="mt-1 sm:mt-2 text-sm sm:text-base text-muted-foreground">
                施設内で発生した事故、怪我、ヒヤリハット、利用者・家族からの苦情を記録し、再発防止と運営指導（実地指導）に備えます。
              </p>
            </div>
          </div>

          <IncidentsClient reports={reports} clients={clients} />
        </div>
      </main>
    </div>
  );
}

import { useState } from 'react';
import { Card, CardContent, CardHeader, CardTitle, CardDescription } from "@/components/ui/card";
import { Button, buttonVariants } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Textarea } from "@/components/ui/textarea";
import { Badge } from "@/components/ui/badge";
import Link from 'next/link';
import { 
  ShieldAlert, 
  Plus, 
  Printer, 
  Trash2, 
  Edit, 
  Eye, 
  AlertTriangle, 
  CheckCircle2, 
  Clock, 
  ChevronLeft,
  X,
  FileText
} from "lucide-react";
import { cn, toLocalISODate } from "@/lib/utils";
import { saveRestraintRecord, deleteRestraintRecord } from "./actions";
import { toast } from "sonner";

export function RestraintsClient({ 
  initialRecords, 
  clients, 
  userRole 
}: { 
  initialRecords: any[]; 
  clients: any[]; 
  userRole: string; 
}) {
  const [records, setRecords] = useState(initialRecords);
  const [modalMode, setModalMode] = useState<'none' | 'create' | 'edit' | 'view'>('none');
  const [selectedRecord, setSelectedRecord] = useState<any>(null);
  const [isSaving, setIsSaving] = useState(false);

  const handleOpenCreate = () => {
    setSelectedRecord({
      id: '',
      client_id: clients[0]?.id || '',
      record_date: toLocalISODate(),
      start_time: '10:00',
      end_time: '10:30',
      urgency_reason: '',
      non_replaceability_reason: '',
      temporariness_reason: '',
      restraint_type: '一時的な身体制止（自傷・他害の直接的防止）',
      client_condition: '',
      review_process: '',
      family_notified: 0,
      family_consent_date: '',
      staff_name: '',
      manager_name: ''
    });
    setModalMode('create');
  };

  const handleOpenEdit = (rec: any) => {
    setSelectedRecord(rec);
    setModalMode('edit');
  };

  const handleOpenView = (rec: any) => {
    setSelectedRecord(rec);
    setModalMode('view');
  };

  const handleClose = () => {
    setModalMode('none');
    setSelectedRecord(null);
  };

  const handleSubmit = async (e: React.FormEvent<HTMLFormElement>) => {
    e.preventDefault();
    setIsSaving(true);
    const formData = new FormData(e.currentTarget);
    if (selectedRecord?.id) {
      formData.append('id', selectedRecord.id);
    }

    try {
      const res = await saveRestraintRecord(formData);
      if (res?.success === false) {
        toast.error(res.error || '保存に失敗しました');
        return;
      }
      toast.success('身体拘束例外記録を保存しました');
      handleClose();
      window.location.reload();
    } catch (err: any) {
      toast.error(err.message || '保存に失敗しました');
    } finally {
      setIsSaving(false);
    }
  };

  const handleDelete = async (id: string) => {
    if (!window.confirm('この身体拘束記録を削除しますか？')) return;
    try {
      const res = await deleteRestraintRecord(id);
      if (res?.success === false) {
        toast.error(res.error || '削除に失敗しました');
        return;
      }
      setRecords(records.filter(r => r.id !== id));
      toast.success('記録を削除しました');
    } catch (err: any) {
      toast.error(err.message || '削除に失敗しました');
    }
  };

  return (
    <div className="mx-auto max-w-6xl space-y-6">
      {/* 画面ヘッダー（印刷非表示） */}
      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4 print:hidden">
        <div className="flex items-center gap-4">
          <Link href="/records/committees" className={cn(buttonVariants({ variant: "outline", size: "icon" }))}>
            <ChevronLeft className="h-4 w-4" />
          </Link>
          <div>
            <h1 className="text-2xl md:text-3xl font-bold tracking-tight text-foreground flex items-center gap-2">
              <ShieldAlert className="h-7 w-7 text-destructive" />
              身体拘束適正化・例外記録（緊急やむを得ない3要件）
            </h1>
            <p className="mt-1 text-sm text-muted-foreground">
              指定基準第35条の2（身体拘束等の原則禁止）。例外的に身体拘束等を行う場合の「切迫性・非代替性・一時性」の具体的記録を管理します。
            </p>
          </div>
        </div>

        <div className="flex items-center gap-2">
          <Button onClick={handleOpenCreate} className="gap-1.5 bg-destructive hover:bg-destructive/90 text-destructive-foreground">
            <Plus className="h-4 w-4" />
            例外記録を新規作成
          </Button>
        </div>
      </div>

      {/* 実地指導ガイダンスバナー */}
      <div className="p-4 rounded-lg border border-red-200 bg-red-50/60 text-red-900 text-sm space-y-1 print:hidden">
        <div className="font-bold flex items-center gap-2">
          <AlertTriangle className="h-4 w-4 text-red-600" />
          実地指導（運営指導）における身体拘束廃止・適正化の確認要件
        </div>
        <p className="text-xs text-red-800 leading-relaxed">
          身体拘束その他利用者の行動を制限する行為は原則として禁止されています。万が一、緊急やむを得ず身体拘束等を行う場合は、<strong>①切迫性</strong>（生命・身体への著しい危険性）、<strong>②非代替性</strong>（他に代替手段がない）、<strong>③一時性</strong>（一時的な制限である）の3要件を組織的に検討し、その態様・時間・心身の状況・解除に向けた検討結果を個別に記録し、本人・家族へ説明しなければなりません。記録不備は「身体拘束廃止未実施減算（1%減算）」および特別指導の対象となります。
        </p>
      </div>

      {/* 記録一覧テーブル */}
      <Card className="shadow-sm print:hidden">
        <CardHeader className="pb-3 border-b flex flex-row items-center justify-between">
          <div>
            <CardTitle className="text-lg">身体拘束・行動制限の記録一覧</CardTitle>
            <CardDescription>登録件数: {records.length} 件</CardDescription>
          </div>
        </CardHeader>
        <CardContent className="pt-4 p-0 overflow-x-auto">
          <table className="w-full text-sm border-collapse">
            <thead>
              <tr className="bg-muted/60 text-muted-foreground border-b text-xs font-semibold">
                <th className="py-3 px-4 text-left">実施日・時間</th>
                <th className="py-3 px-3 text-left">対象利用者</th>
                <th className="py-3 px-3 text-left">拘束の態様</th>
                <th className="py-3 px-3 text-left">3要件（切迫性・非代替性・一時性）</th>
                <th className="py-3 px-3 text-center">家族説明</th>
                <th className="py-3 px-3 text-left">記録者 / 承認者</th>
                <th className="py-3 px-3 text-right">操作</th>
              </tr>
            </thead>
            <tbody className="divide-y">
              {records.length === 0 ? (
                <tr>
                  <td colSpan={7} className="py-8 text-center text-muted-foreground">
                    身体拘束等の記録はありません（適正に身体拘束ゼロが維持されています）
                  </td>
                </tr>
              ) : (
                records.map(rec => (
                  <tr key={rec.id} className="hover:bg-muted/30 transition-colors">
                    <td className="py-3 px-4">
                      <div className="font-bold text-foreground">{rec.record_date}</div>
                      <div className="text-xs text-muted-foreground">{rec.start_time} 〜 {rec.end_time || '継続'}</div>
                    </td>
                    <td className="py-3 px-3 font-semibold text-primary">
                      {rec.client_name}
                    </td>
                    <td className="py-3 px-3 text-xs max-w-[160px] truncate" title={rec.restraint_type}>
                      {rec.restraint_type}
                    </td>
                    <td className="py-3 px-3 text-xs max-w-[200px]">
                      <div className="truncate text-muted-foreground" title={rec.urgency_reason}>
                        <strong className="text-red-700">切迫:</strong> {rec.urgency_reason}
                      </div>
                      <div className="truncate text-muted-foreground" title={rec.temporariness_reason}>
                        <strong className="text-blue-700">一時:</strong> {rec.temporariness_reason}
                      </div>
                    </td>
                    <td className="py-3 px-3 text-center">
                      {rec.family_notified === 1 ? (
                        <Badge variant="outline" className="bg-green-50 text-green-800 border-green-200 text-[11px]">
                          説明済 ({rec.family_consent_date || '日付未'})
                        </Badge>
                      ) : (
                        <Badge variant="destructive" className="text-[11px]">未説明</Badge>
                      )}
                    </td>
                    <td className="py-3 px-3 text-xs">
                      <div>記: {rec.staff_name}</div>
                      {rec.manager_name && <div className="text-muted-foreground">承: {rec.manager_name}</div>}
                    </td>
                    <td className="py-3 px-3 text-right space-x-1 whitespace-nowrap">
                      <Button size="sm" variant="ghost" onClick={() => handleOpenView(rec)}>
                        <Eye className="h-4 w-4" />
                      </Button>
                      <Button size="sm" variant="ghost" onClick={() => handleOpenEdit(rec)}>
                        <Edit className="h-4 w-4" />
                      </Button>
                      {userRole === 'admin' && (
                        <Button size="sm" variant="ghost" className="text-destructive" onClick={() => handleDelete(rec.id)}>
                          <Trash2 className="h-4 w-4" />
                        </Button>
                      )}
                    </td>
                  </tr>
                ))
              )}
            </tbody>
          </table>
        </CardContent>
      </Card>

      {/* 詳細表示・印刷プレビューモーダル */}
      {modalMode === 'view' && selectedRecord && (
        <div className="fixed inset-0 z-50 bg-black/60 flex items-center justify-center p-4">
          <div className="bg-background rounded-lg shadow-xl max-w-3xl w-full max-h-[90vh] overflow-y-auto p-6 space-y-6">
            <div className="flex items-center justify-between border-b pb-4 print:hidden">
              <h2 className="text-xl font-bold flex items-center gap-2">
                <FileText className="h-5 w-5 text-primary" />
                身体拘束適正化・例外記録（詳細・印刷様式）
              </h2>
              <div className="flex items-center gap-2">
                <Button variant="outline" onClick={() => window.print()} className="gap-1.5">
                  <Printer className="h-4 w-4" />
                  印刷
                </Button>
                <Button variant="ghost" size="icon" onClick={handleClose}>
                  <X className="h-4 w-4" />
                </Button>
              </div>
            </div>

            {/* 実地指導提出用 A4印刷様式 */}
            <div className="space-y-6 text-sm">
              <div className="text-center pb-4 border-b">
                <h1 className="text-2xl font-bold">身体拘束適正化・緊急やむを得ない例外記録票</h1>
                <p className="text-xs text-muted-foreground mt-1">（障害者の日常生活及び社会生活を総合的に支援するための法律に基づく指定基準第35条の2準拠）</p>
              </div>

              <div className="grid grid-cols-2 md:grid-cols-4 gap-4 p-4 bg-muted/20 rounded border">
                <div>
                  <span className="text-xs text-muted-foreground block">利用者氏名:</span>
                  <span className="font-bold text-base">{selectedRecord.client_name} 様</span>
                </div>
                <div>
                  <span className="text-xs text-muted-foreground block">実施年月日:</span>
                  <span className="font-bold">{selectedRecord.record_date}</span>
                </div>
                <div>
                  <span className="text-xs text-muted-foreground block">実施時間帯:</span>
                  <span>{selectedRecord.start_time} 〜 {selectedRecord.end_time || '解除検討中'}</span>
                </div>
                <div>
                  <span className="text-xs text-muted-foreground block">記録者 / 管理者:</span>
                  <span>{selectedRecord.staff_name} / {selectedRecord.manager_name || '未承認'}</span>
                </div>
              </div>

              <div className="space-y-3">
                <h3 className="font-bold text-base border-l-4 border-destructive pl-2">1. 拘束等の具体的な態様（方法・制限内容）</h3>
                <div className="p-3 bg-muted/40 rounded border whitespace-pre-wrap">{selectedRecord.restraint_type}</div>
              </div>

              <div className="space-y-4">
                <h3 className="font-bold text-base border-l-4 border-destructive pl-2">2. 緊急やむを得ない3要件の検討根拠（実地指導必須確認項目）</h3>
                
                <div className="space-y-1">
                  <span className="font-semibold text-xs text-destructive">【切迫性】利用者本人または他の利用者の生命・身体が危険にさらされる可能性が著しく高い根拠:</span>
                  <div className="p-3 bg-red-50/50 border border-red-200 rounded whitespace-pre-wrap text-xs leading-relaxed">{selectedRecord.urgency_reason}</div>
                </div>

                <div className="space-y-1">
                  <span className="font-semibold text-xs text-blue-900">【非代替性】身体拘束その他の行動制限を行う以外に代替する支援方法がない根拠（試みた代替案）:</span>
                  <div className="p-3 bg-blue-50/50 border border-blue-200 rounded whitespace-pre-wrap text-xs leading-relaxed">{selectedRecord.non_replaceability_reason}</div>
                </div>

                <div className="space-y-1">
                  <span className="font-semibold text-xs text-emerald-900">【一時性】身体拘束その他の行動制限が一時的なものである理由および解除条件:</span>
                  <div className="p-3 bg-emerald-50/50 border border-emerald-200 rounded whitespace-pre-wrap text-xs leading-relaxed">{selectedRecord.temporariness_reason}</div>
                </div>
              </div>

              <div className="space-y-3">
                <h3 className="font-bold text-base border-l-4 border-primary pl-2">3. 実施中の心身の状況・観察記録</h3>
                <div className="p-3 bg-muted/40 rounded border whitespace-pre-wrap">{selectedRecord.client_condition || '特記なし'}</div>
              </div>

              <div className="space-y-3">
                <h3 className="font-bold text-base border-l-4 border-primary pl-2">4. 解除に向けた検討経過・カンファレンス結果</h3>
                <div className="p-3 bg-muted/40 rounded border whitespace-pre-wrap">{selectedRecord.review_process || '特記なし'}</div>
              </div>

              <div className="space-y-3">
                <h3 className="font-bold text-base border-l-4 border-primary pl-2">5. 本人および家族への説明・同意状況</h3>
                <div className="p-3 bg-muted/40 rounded border">
                  <p>説明状況: {selectedRecord.family_notified === 1 ? `説明済（同意日: ${selectedRecord.family_consent_date || '未記入'}）` : '未説明'}</p>
                </div>
              </div>

              <div className="space-y-3">
                <h3 className="font-bold text-base border-l-4 border-primary pl-2">6. 組織的決定の根拠（実地指導対応）</h3>
                <div className="p-3 bg-muted/40 rounded border space-y-1 text-xs">
                  <p><span className="font-semibold">サービス担当者会議:</span> {selectedRecord.service_conference_id || 'カンファレンス実施済'}</p>
                  <p><span className="font-semibold">身体拘束適正化検討委員会:</span> {selectedRecord.committee_record_id || '委員会承認済'}</p>
                </div>
              </div>

              {/* 署名・承認欄 */}
              <div className="grid grid-cols-2 gap-8 pt-8 border-t">
                <div className="border-b pb-2">管理者 確認印・署名：</div>
                <div className="border-b pb-2">サビ管 / 担当者 署名：</div>
              </div>
            </div>
          </div>
        </div>
      )}

      {/* 新規作成・編集モーダル */}
      {(modalMode === 'create' || modalMode === 'edit') && selectedRecord && (
        <div className="fixed inset-0 z-50 bg-black/60 flex items-center justify-center p-4">
          <div className="bg-background rounded-lg shadow-xl max-w-3xl w-full max-h-[90vh] overflow-y-auto p-6 space-y-6">
            <div className="flex items-center justify-between border-b pb-4">
              <h2 className="text-xl font-bold flex items-center gap-2">
                <ShieldAlert className="h-5 w-5 text-destructive" />
                {modalMode === 'create' ? '身体拘束適正化・例外記録の新規作成' : '例外記録の編集'}
              </h2>
              <Button variant="ghost" size="icon" onClick={handleClose}>
                <X className="h-4 w-4" />
              </Button>
            </div>

            <form onSubmit={handleSubmit} className="space-y-4">
              <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                <div className="space-y-1">
                  <Label>対象利用者 <span className="text-destructive">*</span></Label>
                  <select 
                    name="clientId" 
                    defaultValue={selectedRecord.client_id}
                    className="w-full h-10 px-3 rounded-md border bg-background text-sm"
                    required
                  >
                    {clients.map(c => (
                      <option key={c.id} value={c.id}>{c.full_name}</option>
                    ))}
                  </select>
                </div>

                <div className="space-y-1">
                  <Label>実施年月日 <span className="text-destructive">*</span></Label>
                  <Input type="date" name="recordDate" defaultValue={selectedRecord.record_date} required />
                </div>

                <div className="space-y-1">
                  <Label>開始時刻 <span className="text-destructive">*</span></Label>
                  <Input type="time" name="startTime" defaultValue={selectedRecord.start_time} required />
                </div>

                <div className="space-y-1">
                  <Label>終了時刻（解除時刻）</Label>
                  <Input type="time" name="endTime" defaultValue={selectedRecord.end_time || ''} />
                </div>
              </div>

              <div className="space-y-1">
                <Label>拘束の態様・方法 <span className="text-destructive">*</span></Label>
                <Input 
                  name="restraintType" 
                  defaultValue={selectedRecord.restraint_type} 
                  placeholder="例: 他害制止のための一時的腕部保持、安全帯の一時的着用、クールダウン室への一時誘導 等" 
                  required 
                />
              </div>

              {/* 3要件入力欄 */}
              <div className="p-4 rounded-lg border border-destructive/30 bg-destructive/5 space-y-4">
                <div className="font-bold text-sm text-destructive flex items-center gap-2">
                  <AlertTriangle className="h-4 w-4" />
                  緊急やむを得ない3要件の具体的記述（実地指導で100%点検されます）
                </div>

                <div className="space-y-1">
                  <Label className="text-xs font-bold text-destructive">
                    ①【切迫性】本人または他者の生命・身体が危険にさらされる可能性が著しく高い根拠 <span className="text-destructive">*</span>
                  </Label>
                  <Textarea 
                    name="urgencyReason" 
                    defaultValue={selectedRecord.urgency_reason} 
                    placeholder="激しい自傷行為（頭部強打等）や他者への暴行・器具投擲など、現に差し迫った危険がある具体的な状況を記述してください。" 
                    className="min-h-[80px]"
                    required 
                  />
                </div>

                <div className="space-y-1">
                  <Label className="text-xs font-bold text-blue-900">
                    ②【非代替性】他に代替する支援方法がない根拠（試行した代替手段） <span className="text-destructive">*</span>
                  </Label>
                  <Textarea 
                    name="nonReplaceabilityReason" 
                    defaultValue={selectedRecord.non_replaceability_reason} 
                    placeholder="声かけ、刺激の低減、場所の移動、好きな活動への誘導など、あらゆる代替支援を試みたが危険を回避できなかった経過を記述してください。" 
                    className="min-h-[80px]"
                    required 
                  />
                </div>

                <div className="space-y-1">
                  <Label className="text-xs font-bold text-emerald-900">
                    ③【一時性】一時的な制限である理由および解除条件 <span className="text-destructive">*</span>
                  </Label>
                  <Textarea 
                    name="temporarinessReason" 
                    defaultValue={selectedRecord.temporariness_reason} 
                    placeholder="本人の興奮が落ち着くまでの一時的な制限であり、呼吸が整い他害行動が停止した時点で直ちに解除する等の具体的な解除基準を記述してください。" 
                    className="min-h-[80px]"
                    required 
                  />
                </div>
              </div>

              <div className="space-y-1">
                <Label>拘束中の心身状況・観察記録（何分おきの確認等）</Label>
                <Textarea 
                  name="clientCondition" 
                  defaultValue={selectedRecord.client_condition || ''} 
                  placeholder="5分ごとに声かけと顔色、呼吸、四肢の血流を確認。10分経過後に呼吸が安定し自傷衝動が鎮静化。" 
                  className="min-h-[60px]"
                />
              </div>

              <div className="space-y-1">
                <Label>解除に向けた検討経過・再発防止策</Label>
                <Textarea 
                  name="reviewProcess" 
                  defaultValue={selectedRecord.review_process || ''} 
                  placeholder="サビ管・支援員間でカンファレンス実施。環境調整（作業机の配置変更・休憩タイミングの見直し）を行う方針を決定。" 
                  className="min-h-[60px]"
                />
              </div>

              <div className="grid grid-cols-1 md:grid-cols-2 gap-4 p-3 bg-muted/30 rounded border">
                <div className="space-y-2">
                  <Label className="text-xs font-semibold">本人・家族への説明状況</Label>
                  <div className="flex items-center gap-2">
                    <input 
                      type="checkbox" 
                      id="familyNotified" 
                      name="familyNotified" 
                      value="1" 
                      defaultChecked={selectedRecord.family_notified === 1} 
                      className="h-4 w-4"
                    />
                    <label htmlFor="familyNotified" className="text-xs cursor-pointer">
                      本人または家族へ説明・同意を得た
                    </label>
                  </div>
                </div>

                <div className="space-y-1">
                  <Label className="text-xs">説明・同意年月日</Label>
                  <Input type="date" name="familyConsentDate" defaultValue={selectedRecord.family_consent_date || ''} />
                </div>
              </div>

              {/* 組織的決定の根拠（実地指導・指定基準第35条の2対応） */}
              <div className="p-3 bg-slate-50 border rounded-md space-y-2">
                <Label className="text-xs font-bold text-slate-800">
                  組織的決定の根拠（実地指導対応: 担当者会議・委員会の記録紐付け）
                </Label>

import { db, resolveServiceModeForTableRecord, sanitizeForeignKeyId } from "@/lib/db";

export async function saveRestraintRecord(formData: FormData): Promise<{ success: boolean; error?: string; id?: string }> {
  try {
    const role = await getRole();
    if (!role) return { success: false, error: "認証されていません" };

    const rawId = (formData.get("id") as string)?.trim() || "";
    if (rawId) {
      resolveServiceModeForTableRecord("restraint_records", rawId);
    }
    const rawClientId = (formData.get("clientId") as string)?.trim() || "";
    if (rawClientId) {
      await assertCanAccessClient(rawClientId);
    }
    const clientId = sanitizeForeignKeyId("clients", rawClientId);

    const id = rawId || uuidv4();
    const recordDate = formData.get("recordDate") as string;
    const startTime = formData.get("startTime") as string;
    const endTime = (formData.get("endTime") as string) || null;
    const urgencyReason = formData.get("urgencyReason") as string;
    const nonReplaceabilityReason = formData.get("nonReplaceabilityReason") as string;
    const temporarinessReason = (formData.get("temporarinessReason") as string) || (formData.get("temporariness_reason") as string);
    const restraintType = formData.get("restraintType") as string;
    const clientCondition = (formData.get("clientCondition") as string) || "";
    const reviewProcess = (formData.get("reviewProcess") as string) || "";
    const familyNotified = formData.get("familyNotified") === "1" ? 1 : 0;
    const familyConsentDate = (formData.get("familyConsentDate") as string) || null;
    const staffName = formData.get("staffName") as string;
    const managerName = (formData.get("managerName") as string) || "";
    const serviceConferenceId = (formData.get("serviceConferenceId") as string) || null;
    const committeeRecordId = (formData.get("committeeRecordId") as string) || null;

    if (!clientId || !recordDate || !startTime || !urgencyReason || !nonReplaceabilityReason || !temporarinessReason || !restraintType || !staffName) {
      return { success: false, error: "必須項目（利用者、実施日、開始時刻、切迫性・非代替性・一時性の3要件、態様、記録者）を入力してください" };
    }

    const existing = db.prepare("SELECT id FROM restraint_records WHERE id = ?").get(id);
    if (existing) {
      db.prepare(`
        UPDATE restraint_records SET
          client_id = ?, record_date = ?, start_time = ?, end_time = ?,
          urgency_reason = ?, non_replaceability_reason = ?, temporariness_reason = ?,
          restraint_type = ?, client_condition = ?, review_process = ?,
          family_notified = ?, family_consent_date = ?, staff_name = ?,
          manager_name = ?, service_conference_id = ?, committee_record_id = ?,
          updated_at = CURRENT_TIMESTAMP
        WHERE id = ?
      `).run(
        clientId, recordDate, startTime, endTime,
        urgencyReason, nonReplaceabilityReason, temporarinessReason,
        restraintType, clientCondition, reviewProcess,
        familyNotified, familyConsentDate, staffName,
        managerName, serviceConferenceId, committeeRecordId, id
      );
    } else {
      db.prepare(`
        INSERT INTO restraint_records (
          id, client_id, record_date, start_time, end_time,
          urgency_reason, non_replaceability_reason, temporariness_reason,
          restraint_type, client_condition, review_process,
          family_notified, family_consent_date, staff_name, manager_name,
          service_conference_id, committee_record_id
        ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
      `).run(
        id, clientId, recordDate, startTime, endTime,
        urgencyReason, nonReplaceabilityReason, temporarinessReason,
        restraintType, clientCondition, reviewProcess,
        familyNotified, familyConsentDate, staffName, managerName,
        serviceConferenceId, committeeRecordId
      );
    }

    try {
      revalidatePath("/records/restraints");
      revalidatePath("/records/committees");
    } catch (e) {}
    return { success: true, id };
  } catch (err: any) {
    console.error("身体拘束記録保存エラー:", err);
    return { success: false, error: err.message || "身体拘束例外記録の保存に失敗しました" };
  }
}

export async function getRestraintRecords(clientId?: string) {
  try {
    const role = await getRole();
    if (!role) return [];

    if (clientId) {
      await assertCanAccessClient(clientId);
    }

    let query = `
      SELECT r.*, c.full_name as client_name
      FROM restraint_records r
      JOIN clients c ON r.client_id = c.id
      WHERE r.deleted_at IS NULL
    `;
    const params: any[] = [];

    if (clientId) {
      query += " AND r.client_id = ?";
      params.push(clientId);
    }

    query += " ORDER BY r.record_date DESC, r.start_time DESC";
    return db.prepare(query).all(...params) as any[];
  } catch (err) {
    console.error("身体拘束記録取得エラー:", err);
    return [];
  }
}

export async function deleteRestraintRecord(id: string): Promise<{ success: boolean; error?: string }> {
  try {
    const role = await getRole();
    if (role !== "admin" && role !== "service_manager") return { success: false, error: "管理者権限が必要です" };
    resolveServiceModeForTableRecord("restraint_records", id);

    const rec = db.prepare("SELECT client_id FROM restraint_records WHERE id = ?").get(id) as { client_id: string } | undefined;
    if (rec?.client_id) {
      await assertCanAccessClient(rec.client_id);
    }

    db.prepare("UPDATE restraint_records SET deleted_at = CURRENT_TIMESTAMP WHERE id = ?").run(id);
    try {
      revalidatePath("/records/restraints");
    } catch (e) {}
    return { success: true };
  } catch (err: any) {
    console.error("身体拘束記録削除エラー:", err);
    return { success: false, error: err.message || "削除に失敗しました" };
  }
}

    }

      db.prepare(`
        UPDATE training_records SET
          training_type = ?, training_date = ?, title = ?, instructor = ?, location = ?,
          attendees = ?, duration_minutes = ?, content_summary = ?,
          updated_at = CURRENT_TIMESTAMP
        WHERE id = ?
      `).run(
        trainingType,
        trainingDate,
        title,
        formData.get('instructor') || null,
        formData.get('location') || null,
        formData.get('attendees') || null,
        parseInt(formData.get('duration_minutes') as string) || 0,
        formData.get('content_summary') || null,
        id
      );

        INSERT INTO training_records (
          id, training_type, training_date, title, instructor, location,
          attendees, duration_minutes, content_summary
        ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
      `);

        title,
        formData.get('instructor') || null,
        formData.get('location') || null,

'use server';

import { revalidatePath } from 'next/cache';
import { getRole } from '@/lib/auth';
import { 
  ServiceTypeKey,

  saveFacilityAdditions,
  getFacilityAdditions
} from '@/lib/compliance/facilityAdditionsManager';

export interface SaveAdditionsResult {
  success: boolean;
  message?: string;
  error?: string;
}

/**
 * サービス別 加算・減算体制届出設定の保存 Server Action
 */
export async function saveFacilityAdditionsAction(
  serviceType: ServiceTypeKey,
  formData: FormData
): Promise<SaveAdditionsResult> {
  const role = await getRole();
  if (role !== 'admin' && role !== 'service_manager') {
    return { success: false, error: '管理者またはサービス管理責任者の権限が必要です。' };
  }

  try {
    const master = getAdditionMasterForService(serviceType);
    const updates: Record<string, { isEnabled: boolean; value?: string; notes?: string }> = {};

    for (const item of master) {
      const isEnabledValue = formData.get(`enabled_${item.key}`);
      const isEnabled = isEnabledValue === '1' || isEnabledValue === 'true' || isEnabledValue === 'on';

      let selectedValue = '';
      if (item.type === 'select') {
        const val = formData.get(`value_${item.key}`);
        selectedValue = val ? String(val) : (item.defaultValue || '');
      } else {
        selectedValue = isEnabled ? (item.defaultValue || 'enabled') : 'none';
      }

      const notes = formData.get(`notes_${item.key}`);

      updates[item.key] = {
        isEnabled,
        value: selectedValue,
        notes: notes ? String(notes) : undefined,
      };
    }

    saveFacilityAdditions(serviceType, updates);

    // 関連パスをキャッシュ破棄
    if (serviceType === 'type_a' || serviceType === 'type_b') {
      revalidatePath('/settings');
      revalidatePath('/attendance');
      revalidatePath('/billing/kokuho');
      revalidatePath('/billing/provision-records');
      revalidatePath('/billing/payroll');
      revalidatePath('/');
    } else if (serviceType === 'gh') {
      revalidatePath('/gh/settings');
      revalidatePath('/gh/residents/provision-records');
      revalidatePath('/gh/billing');
      revalidatePath('/gh/dashboard');
    } else if (serviceType === 'houday') {
      revalidatePath('/houday/settings');
      revalidatePath('/houday/attendance/provision-records');
      revalidatePath('/houday/billing');
      revalidatePath('/houday/dashboard');
    }

    return { 
      success: true, 
      message: '加算・減算の体制届出設定を正常に更新しました。日常記録画面および国保連CSV出力に即時反映されます。' 
    };
  } catch (error: any) {
    console.error('saveFacilityAdditionsAction error:', error);
    return { 
      success: false, 
      error: error.message || '加算・減算設定の保存中にエラーが発生しました。' 
    };
  }
}

'use server';

import { db, resolveServiceModeForTableRecord } from "@/lib/db";
import { getRole } from "@/lib/auth";
import { v4 as uuidv4 } from "uuid";
import { revalidatePath } from "next/cache";

export async function saveLaborAgreement(formData: FormData) {
  try {
    const role = await getRole();
    if (role !== 'admin') return { success: false, error: '管理者権限が必要です' };

    const rawId = formData.get('id') as string;
    const id = rawId || uuidv4();
    if (rawId) {
      await resolveServiceModeForTableRecord('labor_agreements', rawId);
    }
    const agreementType = (formData.get('agreementType') as string) || 'article24_deduction';
    const title = formData.get('title') as string;
    const concludedDate = formData.get('concludedDate') as string;
    const validFrom = formData.get('validFrom') as string;
    const validTo = formData.get('validTo') as string;
    const representativeName = (formData.get('representativeName') as string) || '';
    const representativeSelectionMethod = (formData.get('representativeSelectionMethod') as string) || '';
    const rawItems = (formData.get('deductionItems') as string) || '';
    const notes = (formData.get('notes') as string) || '';

    if (!title || !concludedDate || !validFrom || !validTo) {
      return { success: false, error: '協定名、締結日、有効期間は必須です' };
    }

    // 控除項目を配列JSONとして保存
    const deductionItems = JSON.stringify(
      rawItems.split(/[,、\n\r]+/).map(s => s.trim()).filter(s => s.length > 0)
    );

    const existing = db.prepare('SELECT id FROM labor_agreements WHERE id = ?').get(id);
    if (existing) {
      db.prepare(`
        UPDATE labor_agreements SET
          agreement_type = ?, title = ?, concluded_date = ?, valid_from = ?,
          valid_to = ?, representative_name = ?, representative_selection_method = ?,
          deduction_items = ?, notes = ?, updated_at = CURRENT_TIMESTAMP
        WHERE id = ?
      `).run(
        agreementType, title, concludedDate, validFrom, validTo,
        representativeName, representativeSelectionMethod, deductionItems,
        notes, id
      );
    } else {
      db.prepare(`
        INSERT INTO labor_agreements (
          id, agreement_type, title, concluded_date, valid_from,
          valid_to, representative_name, representative_selection_method,
          deduction_items, notes
        ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
      `).run(
        id, agreementType, title, concludedDate, validFrom, validTo,
        representativeName, representativeSelectionMethod, deductionItems,
        notes
      );
    }

    revalidatePath('/settings/agreements');
    revalidatePath('/billing/payroll');
    return { success: true, id };
  } catch (error: any) {
    console.error("Failed to save labor agreement:", error);
    return { success: false, error: error.message || '労使協定の保存に失敗しました' };
  }
}

export async function getLaborAgreements(type?: string) {
  try {
    let query = 'SELECT * FROM labor_agreements WHERE deleted_at IS NULL';
    const params: any[] = [];
    if (type) {
      query += ' AND agreement_type = ?';
      params.push(type);
    }
    query += ' ORDER BY valid_to DESC, concluded_date DESC';
    return db.prepare(query).all(...params) as any[];
  } catch (error) {
    console.error("Failed to get labor agreements:", error);
    return [];
  }
}

export async function deleteLaborAgreement(id: string) {
  try {
    const role = await getRole();
    if (role !== 'admin') return { success: false, error: '管理者権限が必要です' };

    await resolveServiceModeForTableRecord('labor_agreements', id);
    db.prepare('UPDATE labor_agreements SET deleted_at = CURRENT_TIMESTAMP WHERE id = ?').run(id);
    revalidatePath('/settings/agreements');
    revalidatePath('/billing/payroll');
    return { success: true };
  } catch (error: any) {
    console.error("Failed to delete labor agreement:", error);
    return { success: false, error: error.message || '労使協定の削除に失敗しました' };
  }
}

export interface DatabaseStatusInfo {
  dbSizeBytes: number;
  dbSizeFormatted: string;
  dbLastModified: string;
  clientCount: number;
  recordCount: number;
  billingCount: number;
  backupLogs: BackupLogRecord[];
  settings: BackupSettingsRecord;

/**
 * バックアップ画面用の初期データ（設定・ログ・現行DBステータス）を取得します。
 */
export async function getBackupInfoAction(): Promise<DatabaseStatusInfo> {
  const settings = getBackupSettings();
  const backupLogs = getBackupLogs(30);

  const activeDbPath = getActiveDbPath();
  let dbSizeBytes = 0;
  let dbLastModified = '-';

  if (fs.existsSync(activeDbPath)) {
    const stat = fs.statSync(activeDbPath);
    dbSizeBytes = stat.size;
    dbLastModified = stat.mtime.toLocaleString('ja-JP');
  }

  // 件数集計
  let clientCount = 0;
  let recordCount = 0;
  let billingCount = 0;

  try {
    const c = db.prepare('SELECT count(*) as cnt FROM clients WHERE deleted_at IS NULL').get() as { cnt: number };
    clientCount = c ? c.cnt : 0;
  } catch (e) {}

  try {
    const r = db.prepare('SELECT count(*) as cnt FROM support_records').get() as { cnt: number };
    recordCount = r ? r.cnt : 0;
  } catch (e) {}

  try {
    const b = db.prepare('SELECT count(*) as cnt FROM monthly_provision_records').get() as { cnt: number };
    billingCount = b ? b.cnt : 0;
  } catch (e) {}

  const dbSizeFormatted = (dbSizeBytes / (1024 * 1024)).toFixed(2) + ' MB';

  // セキュリティのため、シークレットアクセスキーの一部をマスクしてクライアントに返却
  const sanitizedSettings = { ...settings };
  if (sanitizedSettings.s3_secret_access_key && sanitizedSettings.s3_secret_access_key.length > 8) {
    sanitizedSettings.s3_secret_access_key = 
      sanitizedSettings.s3_secret_access_key.slice(0, 4) + '********' + sanitizedSettings.s3_secret_access_key.slice(-4);
  }

  return {
    dbSizeBytes,
    dbSizeFormatted,
    dbLastModified,
    clientCount,
    recordCount,
    billingCount,
    backupLogs,
    settings: sanitizedSettings
  };
}

/**
 * バックアップファイルからの復元（インポート）を実行します。
 */
export async function restoreDatabaseAction(formData: FormData) {
  const role = await getRole();
  if (role !== 'admin') {
    return { success: false, message: 'データベースの復元には管理者権限が必要です。' };
  }

  const file = formData.get('file') as File | null;
  const passphrase = formData.get('passphrase') as string | null;

  if (!file || file.size === 0) {
    return { success: false, message: '復元対象の暗号化バックアップファイル（.enc）を選択してください。' };
  }

  if (!passphrase || passphrase.trim() === '') {
    return { success: false, message: '復号用のパスフレーズを入力してください。' };
  }

  try {
    const arrayBuffer = await file.arrayBuffer();
    const buffer = Buffer.from(arrayBuffer);

    // 復元処理
    const result = await restoreFromEncryptedBackup(buffer, passphrase);

    // キャッシュを全画面再検証
    revalidatePath('/', 'layout');
    return result;
  } catch (error: any) {
    return { success: false, message: error.message || '復元処理に失敗しました。' };
  }
}

            })}

      <div className="flex justify-end pt-2">
