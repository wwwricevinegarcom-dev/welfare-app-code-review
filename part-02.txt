import React from 'react';
import Link from 'next/link';
import { getHoudayDb } from '../../lib/db';
import { PrintButton } from '@/components/ui/PrintButton';
import { Button } from '@/components/ui/button';
import { Card, CardContent } from '@/components/ui/card';
import { ArrowLeft, Printer, Users } from 'lucide-react';

export const dynamic = 'force-dynamic';

export default async function HoudayProvisionRecordsPage(props: {
  searchParams?: Promise<{ month?: string; childId?: string }>;
}) {
  const searchParams = await props.searchParams;
  const currentMonthStr = new Date().toISOString().slice(0, 7);
  const targetMonth = searchParams?.month || currentMonthStr;
  const db = getHoudayDb();

  const settings = db.prepare('SELECT * FROM houday_settings LIMIT 1').get() as any || {
    facility_name: '放課後等デイサービス事業所',
    capacity: 10
  };

  const children = db.prepare(`
    SELECT * FROM houday_children
    WHERE status = 'active'
    ORDER BY name ASC
  `).all() as any[];

  const selectedChildId = searchParams?.childId || (children[0]?.id || '');
  const selectedChild = children.find(c => c.id === selectedChildId) || children[0];

  // 日付の配列生成
  const [yearStr, monthStr] = targetMonth.split('-');
  const year = parseInt(yearStr, 10);
  const month = parseInt(monthStr, 10);
  const daysInMonth = new Date(year, month, 0).getDate();

  // 通所記録の取得
  const attendances = selectedChild ? db.prepare(`
    SELECT * FROM houday_attendance
    WHERE child_id = ? AND target_date LIKE ?
    ORDER BY target_date ASC
  `).all(selectedChild.id, `${targetMonth}-%`) as any[] : [];

  // カレンダーの取得
  const calendars = db.prepare(`
    SELECT * FROM houday_calendar
    WHERE target_date LIKE ?
  `).all(`${targetMonth}-%`) as any[];

  let totalDays = 0;
  let schoolDaysCount = 0;
  let vacationDaysCount = 0;
  let pickupCount = 0;
  let snackCount = 0;

  const rows = [];
  for (let d = 1; d <= daysInMonth; d++) {
    const dStr = String(d).padStart(2, '0');
    const dateStr = `${targetMonth}-${dStr}`;
    const dateObj = new Date(year, month - 1, d);
    const dayOfWeek = ['日', '月', '火', '水', '木', '金', '土'][dateObj.getDay()];
    const isWeekend = dayOfWeek === '日' || dayOfWeek === '土';

    const cal = calendars.find(c => c.target_date === dateStr);
    const isSchoolDay = cal ? cal.day_type === 'school_day' : !isWeekend;

    const att = attendances.find(a => a.target_date === dateStr);

    let dayTypeLabel = isSchoolDay ? '開学' : '休業';
    let serviceStatus = '-';
    let serviceLabel = '-';
    let startTime = '';
    let endTime = '';
    let duration = '';
    let transportLabel = '-';
    let snack = '-';
    let remarks = '';

    if (att && att.status === 'present') {
      totalDays++;
      if (isSchoolDay) schoolDaysCount++;
      else vacationDaysCount++;

      serviceStatus = '1';
      serviceLabel = '通所';
      startTime = att.arrival_time || (isSchoolDay ? '14:30' : '10:00');
      endTime = att.departure_time || (isSchoolDay ? '17:30' : '16:00');
      duration = isSchoolDay ? '3.0h' : '6.0h';

      // 送迎
      if (att.pickup_school === 1 && att.dropoff_home === 1) {
        transportLabel = '往復';
        pickupCount += 2;
      } else if (att.pickup_school === 1) {
        transportLabel = '迎え';
        pickupCount += 1;
      } else if (att.dropoff_home === 1) {
        transportLabel = '送り';
        pickupCount += 1;
      }

      if (att.snack_consumed === 1) {
        snack = '〇';
        snackCount++;
      }

      remarks = att.support_note || (isSchoolDay ? '放課後活動' : '休業日創作活動');
    } else if (att && att.status === 'absent') {
      serviceStatus = '2';
      serviceLabel = '加算欠席';
      remarks = '欠席時対応加算算定';
      totalDays++;
    }

    rows.push({
      day: d,
      dayOfWeek,
      dayTypeLabel,
      serviceStatus,
      serviceLabel,
      startTime,
      endTime,
      duration,
      transportLabel,
      snack,
      remarks,
    });
  }

  return (
    <div className="space-y-6">
      {/* 画面ヘッダー（印刷時非表示） */}
      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4 print:hidden">
        <div className="flex items-center gap-3">
          <Link
            href="/houday/attendance"
            className="text-xs text-muted-foreground hover:text-foreground flex items-center gap-1 font-medium"
          >
            <ArrowLeft className="h-3.5 w-3.5" />
            通所記録へ戻る
          </Link>
          <div>
            <h1 className="text-2xl font-black text-slate-900">
              放課後等デイサービス 障害児通所給付費実績記録票（様式第７）
            </h1>
            <p className="text-xs text-slate-500">
              児童福祉法・こども家庭庁標準様式（様式第7）準拠。開学/休業日区分、送迎加算、指導実績の法定記録原本
            </p>
          </div>
        </div>

        {selectedChild && (
          <PrintButton
            label="実績記録票を印刷 / PDF"
            className="bg-sky-600 hover:bg-sky-700 text-white font-bold h-9 text-xs shadow-sm"
          />
        )}
      </div>

      {/* 絞り込みツールバー（印刷時非表示） */}
      <Card className="border-sky-200 bg-sky-50/20 shadow-xs print:hidden">
        <CardContent className="pt-4 pb-4">
          <form className="flex flex-wrap gap-4 items-end" method="GET">
            <div className="space-y-1">
              <label className="text-xs font-bold text-gray-700">対象月</label>
              <input
                type="month"
                name="month"
                defaultValue={targetMonth}
                className="flex h-9 rounded-md border border-input bg-background px-3 py-1 text-sm shadow-xs"
              />
            </div>

            <div className="space-y-1 min-w-[200px]">
              <label className="text-xs font-bold text-gray-700">児童氏名</label>
              <select
                name="childId"
                defaultValue={selectedChildId}
                className="flex h-9 w-full rounded-md border border-input bg-background px-3 py-1 text-sm shadow-xs"
              >
                {children.map(c => (
                  <option key={c.id} value={c.id}>
                    {c.name}（{c.grade} / {c.school_name}）
                  </option>
                ))}
              </select>
            </div>

            <Button type="submit" className="h-9 px-4 bg-sky-600 hover:bg-sky-700 font-bold text-white">
              表示更新
            </Button>
          </form>
        </CardContent>
      </Card>

      {/* A4印刷用紙コンテナ */}
      {selectedChild && (
        <div className="print-a4-sheet bg-white w-[210mm] min-h-[297mm] h-[297mm] p-[9mm_10mm_9mm_10mm] box-border text-black mx-auto shadow-2xl border border-gray-300 print:shadow-none print:border-none print:m-0 flex flex-col justify-between overflow-hidden">
          {/* ヘッダー */}
          <div>
            <div className="text-center mb-3 print:mb-1.5">
              <h2 className="text-xl md:text-2xl font-bold tracking-tight print:text-[18px] print:mb-0.5">
                放課後等デイサービス 障害児通所給付費実績記録票（様式第７）
              </h2>
              <p className="text-sm font-semibold print:text-[12px]">
                【 令和{year - 2018}年（{year}年） {month} 月分 】
              </p>
            </div>

            <div className="flex justify-between items-end mb-3 print:mb-1.5 text-xs print:text-[10px] leading-tight">
              <div className="space-y-0.5">
                <p>受給者証番号： <span className="font-mono font-semibold text-sm print:text-[11px]">{selectedChild.recipient_number || '2810099999'}</span></p>
                <p className="text-sm print:text-[12px] border-b border-black inline-block min-w-[220px] pb-0.5 font-bold">
                  支給決定児童氏名： {selectedChild.name}　様（保護者: {selectedChild.guardian_name} 様）
                </p>
                <p>学校・学年： <span className="font-bold">{selectedChild.school_name}（{selectedChild.grade}）</span></p>
              </div>
              <div className="text-right space-y-0.5">
                <p>支給決定量： <span className="font-bold text-sm print:text-[11px]">{selectedChild.grant_days || 23}</span> 日/月</p>
                <p>事業所番号： <span className="font-mono font-semibold">2810099999</span></p>
                <p>事業者及びその事業所名： <span className="font-semibold">{settings.facility_name}</span></p>
              </div>
            </div>

            {/* テーブル */}
            <table className="w-full border-collapse border border-black text-xs print:text-[10px] text-center table-fixed">
              <thead>
                <tr className="bg-gray-100 print:bg-gray-200 font-bold h-7 print:h-[6.5mm]">

                  <th className="border border-black p-0.5 w-[6%]">曜日</th>
                  <th className="border border-black p-0.5 w-[9%] text-[10px] print:text-[9px]">開学区分</th>
                  <th className="border border-black p-0.5 w-[11%] text-[10px] print:text-[9px]">提供状況</th>
                  <th className="border border-black p-0.5 w-[9%] text-[10px] print:text-[9px]">開始時間</th>

              <tbody>

'use server';

import { getHoudayDb } from '../lib/db';
import { revalidatePath } from 'next/cache';

export async function recalculateHoudayMonthlyBillings(formData: FormData) {
  try {
    const db = getHoudayDb();
    const targetMonth = (formData.get('target_month') as string)?.trim() || new Date().toISOString().slice(0, 7);

    // 設定取得
    const settings = db.prepare('SELECT * FROM houday_settings WHERE id = ?').get('houday_default') as any || {
      snack_price: 100,
      activity_material_cost: 1000,
    };

    const children = db.prepare("SELECT * FROM houday_children WHERE status = 'active'").all() as any[];

    db.transaction(() => {
      for (const c of children) {
        // 通所実績集計
        const atts = db.prepare(`
          SELECT * FROM houday_attendance
          WHERE child_id = ? AND target_date LIKE ? AND status = 'present'
        `).all(c.id, `${targetMonth}%`) as any[];

        const attendanceDays = atts.length;
        const schoolDayCount = atts.filter(a => a.service_type === 'school_day').length;
        const vacationDayCount = atts.filter(a => a.service_type === 'vacation').length;
        const snackCount = atts.filter(a => a.snack_consumed === 1).length;
        const snackTotalCost = snackCount * settings.snack_price;
        const materialCost = attendanceDays > 0 ? settings.activity_material_cost : 0;
        const actualCost = snackTotalCost + materialCost;

        // 送迎回数
        let transportCount = 0;
        for (const a of atts) {
          if (a.pickup_school) transportCount += 1;
          if (a.dropoff_home) transportCount += 1;
        }

        // 国保連単位数
        // 平日放課後: 610単位/日, 休業日: 714単位/日, 送迎: 54単位/回
        const baseUnits = (schoolDayCount * 610) + (vacationDayCount * 714);
        const transportUnits = transportCount * 54;
        const kokuhoUnits = baseUnits + transportUnits;

        const grandTotal = actualCost; // 自己負担無償化（0円）想定

        const id = `hbill_${c.id}_${targetMonth}`;
        db.prepare(`
          INSERT INTO houday_monthly_billings (
            id, child_id, target_month, attendance_days,
            school_day_count, vacation_day_count,
            snack_count, snack_total_cost, material_cost,
            transport_count, kokuho_units,
            guardian_copay_amount, guardian_actual_cost, guardian_grand_total
          ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, 0, ?, ?)
          ON CONFLICT(child_id, target_month) DO UPDATE SET
            attendance_days = excluded.attendance_days,
            school_day_count = excluded.school_day_count,
            vacation_day_count = excluded.vacation_day_count,
            snack_count = excluded.snack_count,
            snack_total_cost = excluded.snack_total_cost,
            material_cost = excluded.material_cost,
            transport_count = excluded.transport_count,
            kokuho_units = excluded.kokuho_units,
            guardian_actual_cost = excluded.guardian_actual_cost,
            guardian_grand_total = excluded.guardian_grand_total
        `).run(
          id, c.id, targetMonth, attendanceDays,
          schoolDayCount, vacationDayCount,
          snackCount, snackTotalCost, materialCost,
          transportCount, kokuhoUnits,
          actualCost, grandTotal
        );
      }
    })();

    revalidatePath('/houday/billing');
    revalidatePath('/houday/dashboard');
  } catch (error: any) {
    console.error('recalculateHoudayMonthlyBillings error:', error);
  }
}

export async function createHoudayChild(formData: FormData) {
  try {
    const db = getHoudayDb();
    const name = (formData.get('name') as string)?.trim();
    const furigana = (formData.get('furigana') as string)?.trim() || '';
    const dateOfBirth = (formData.get('date_of_birth') as string)?.trim() || '';
    const grade = (formData.get('grade') as string)?.trim() || '小1';
    const schoolId = (formData.get('school_id') as string)?.trim() || null;
    const recipientNumber = (formData.get('recipient_number') as string)?.trim() || '';
    const grantDays = parseInt(formData.get('grant_days') as string, 10) || 23;
    const hasMedicalCare = formData.get('has_medical_care') === '1' ? 1 : 0;
    const guardianName = (formData.get('guardian_name') as string)?.trim();
    const guardianContact = (formData.get('guardian_contact') as string)?.trim();
    const pickupNeeded = formData.get('pickup_needed') === '1' ? 1 : 0;
    const allergies = (formData.get('allergies') as string)?.trim() || 'なし';
    const notes = (formData.get('notes') as string)?.trim() || '';

    if (!name || !guardianName || !guardianContact) {
      return;
    }

    // 学校名取得
    let schoolName = '';
    if (schoolId) {
      const sch = db.prepare('SELECT school_name FROM houday_schools WHERE id = ?').get(schoolId) as any;
      if (sch) schoolName = sch.school_name;
    }

    const id = `ch_${Date.now()}`;
    db.prepare(`
      INSERT INTO houday_children (
        id, name, furigana, date_of_birth, grade, school_id, school_name,
        recipient_number, grant_days, has_medical_care, guardian_name,
        guardian_contact, pickup_needed, allergies, notes, status
      ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, 'active')
    `).run(
      id, name, furigana, dateOfBirth, grade, schoolId, schoolName,
      recipientNumber, grantDays, hasMedicalCare, guardianName,
      guardianContact, pickupNeeded, allergies, notes
    );

    revalidatePath('/houday/children');
    revalidatePath('/houday/dashboard');
  } catch (error: any) {
    console.error('createHoudayChild error:', error);
  }
}

export async function deleteHoudayChild(childId: string) {
  try {
    if (!childId) {
      return { success: false, error: '児童IDが指定されていません' };
    }
    const db = getHoudayDb();

    db.transaction(() => {
      // 関連データのクリーンアップ（子テーブル全件）
      db.prepare('DELETE FROM houday_attendance WHERE child_id = ?').run(childId);
      db.prepare('DELETE FROM houday_plans WHERE child_id = ?').run(childId);
      db.prepare('DELETE FROM houday_monthly_billings WHERE child_id = ?').run(childId);
      try {
        db.prepare('DELETE FROM houday_coordination_logs WHERE child_id = ?').run(childId);
      } catch (e) {
        // テーブルがない場合は無視
      }
      // 児童本体の削除
      db.prepare('DELETE FROM houday_children WHERE id = ?').run(childId);
    })();

    try {
      revalidatePath('/houday/children');
      revalidatePath('/houday/dashboard');
      revalidatePath('/houday/attendance');
      revalidatePath('/houday/plans');
    } catch (e) {}
    return { success: true };
  } catch (error: any) {
    console.error('deleteHoudayChild error:', error);
    return { success: false, error: error?.message || '児童の削除に失敗しました' };
  }
}

let isHoudayDbInitialized = false;

export function getHoudayDb(): Database.Database {
  const db = getDbConnection('houday');
  if (!isHoudayDbInitialized) {
    initHoudayDatabase(db);
    isHoudayDbInitialized = true;
  }
  return db;
}

function initHoudayDatabase(db: Database.Database) {
  db.exec(`
    -- 1. 放デイ事業所設定
    CREATE TABLE IF NOT EXISTS houday_settings (
      id TEXT PRIMARY KEY,
      facility_name TEXT NOT NULL,
      capacity INTEGER DEFAULT 10,                 -- 定員10名

export async function createHoudayPlan(formData: FormData) {
  try {
    const db = getHoudayDb();
    const childId = formData.get('child_id') as string;
    const planStartDate = (formData.get('plan_start_date') as string)?.trim();
    const planEndDate = (formData.get('plan_end_date') as string)?.trim();
    const creatorName = (formData.get('creator_name') as string)?.trim() || '児発管 井上';
    const overallGoal = (formData.get('overall_goal') as string)?.trim();
    const shortTermGoal = (formData.get('short_term_goal') as string)?.trim();
    const domainHealthLife = (formData.get('domain_health_life') as string)?.trim() || '';
    const domainMotorSensory = (formData.get('domain_motor_sensory') as string)?.trim() || '';
    const domainCognitionBehavior = (formData.get('domain_cognition_behavior') as string)?.trim() || '';
    const domainLanguageCommunication = (formData.get('domain_language_communication') as string)?.trim() || '';
    const domainSocialRelationship = (formData.get('domain_social_relationship') as string)?.trim() || '';
    const schoolCooperation = (formData.get('school_cooperation') as string)?.trim() || '';
    const familySupport = (formData.get('family_support') as string)?.trim() || '';
    const guardianAgreedDate = (formData.get('guardian_agreed_date') as string)?.trim() || null;
    const monitoringDate = (formData.get('monitoring_date') as string)?.trim() || null;

    if (!childId || !planStartDate || !planEndDate || !overallGoal || !shortTermGoal) {
      return;
    }

    const id = `plan_${childId}_${Date.now()}`;
    db.prepare(`
      INSERT INTO houday_plans (
        id, child_id, status, plan_start_date, plan_end_date, creator_name,
        overall_goal, short_term_goal,
        domain_health_life, domain_motor_sensory, domain_cognition_behavior,
        domain_language_communication, domain_social_relationship,
        school_cooperation, family_support, guardian_agreed_date, monitoring_date
      ) VALUES (?, ?, 'active', ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
    `).run(
      id, childId, planStartDate, planEndDate, creatorName,
      overallGoal, shortTermGoal,
      domainHealthLife, domainMotorSensory, domainCognitionBehavior,
      domainLanguageCommunication, domainSocialRelationship,
      schoolCooperation, familySupport, guardianAgreedDate, monitoringDate
    );

    revalidatePath('/houday/plans');
    revalidatePath('/houday/dashboard');
  } catch (error: any) {
    console.error('createHoudayPlan error:', error);
  }
}

export async function updateHoudaySettings(formData: FormData) {
  try {
    const db = getHoudayDb();

    const capacity = parseInt(formData.get('capacity') as string, 10) || 10;
    const snackPrice = parseInt(formData.get('snack_price') as string, 10) || 100;
    const activityMaterialCost = parseInt(formData.get('activity_material_cost') as string, 10) || 1000;

    db.prepare(`
      UPDATE houday_settings
      SET facility_name = ?, capacity = ?, snack_price = ?,
          activity_material_cost = ?, updated_at = CURRENT_TIMESTAMP
      WHERE id = 'houday_default'
    `).run(facilityName, capacity, snackPrice, activityMaterialCost);

    revalidatePath('/houday/settings');
    revalidatePath('/houday/dashboard');
    revalidatePath('/houday/billing');
  } catch (error: any) {
    console.error('updateHoudaySettings error:', error);
  }
}

export async function addHoudayVehicle(formData: FormData) {
  try {
    const db = getHoudayDb();
    const vehicleName = (formData.get('vehicle_name') as string)?.trim();
    const plateNumber = (formData.get('plate_number') as string)?.trim() || '';
    const capacity = parseInt(formData.get('capacity') as string, 10) || 6;
    const hasSafetyAlarm = formData.get('has_safety_alarm') === '1' ? 1 : 0;
    const notes = (formData.get('notes') as string)?.trim() || '';

    if (!vehicleName) {
      return;
    }

    const id = `veh_${Date.now()}`;
    db.prepare(`
      INSERT INTO houday_vehicles (id, vehicle_name, plate_number, capacity, has_safety_alarm, notes)
      VALUES (?, ?, ?, ?, ?, ?)
    `).run(id, vehicleName, plateNumber, capacity, hasSafetyAlarm, notes);

    revalidatePath('/houday/settings');
    revalidatePath('/settings');
    revalidatePath('/houday/transport');
  } catch (error: any) {
    console.error('addHoudayVehicle error:', error);
  }
}

export async function deleteHoudayVehicle(vehicleId: string) {
  try {
    if (!vehicleId) {
      return { success: false, error: '車両IDが指定されていません' };
    }
    const db = getHoudayDb();

    db.transaction(() => {
      // 該当車両に紐づく送迎運行記録の安全な整理
      db.prepare('DELETE FROM houday_transport_logs WHERE vehicle_id = ?').run(vehicleId);
      // 車両本体の削除
      db.prepare('DELETE FROM houday_vehicles WHERE id = ?').run(vehicleId);
    })();

    try {
      revalidatePath('/houday/settings');
      revalidatePath('/settings');
      revalidatePath('/houday/transport');
    } catch (e) {}
    return { success: true };
  } catch (error: any) {
    console.error('deleteHoudayVehicle error:', error);
    return { success: false, error: error?.message || '車両の削除に失敗しました' };
  }
}

/**
 * AIが提案した法改正ルール差分をワンクリックでDBに一括適用
 */
export async function applyBillingRuleUpdates(updates: LegalAdditionItem[]) {
  const role = await getRole();
  if (role !== 'admin' && role !== 'service_manager') {
    return { success: false, error: '権限がありません' };
  }

  if (!updates || updates.length === 0) {
    return { success: false, error: '適用対象のルールがありません' };
  }

  try {
    const updateStmt = db.prepare(`
      INSERT INTO billing_rules (rule_key, category, rule_name, rule_value, unit_label, description, effective_from, updated_at)
      VALUES (?, ?, ?, ?, ?, ?, ?, CURRENT_TIMESTAMP)
      ON CONFLICT(rule_key) DO UPDATE SET
        rule_value = excluded.rule_value,
        description = excluded.description,
        effective_from = excluded.effective_from,
        updated_at = CURRENT_TIMESTAMP
    `);

    const logStmt = db.prepare(`
      INSERT INTO audit_logs (id, user_id, user_name, action, target_table, target_id, details)
      VALUES (?, ?, ?, ?, ?, ?, ?)
    `);

    db.transaction(() => {
      for (const item of updates) {
        // category推定（既存があれば流用、無ければキーから推測）
        const existing = db.prepare(`SELECT category, unit_label FROM billing_rules WHERE rule_key = ?`).get(item.rule_key) as any;
        const category = existing?.category || (item.rule_key.startsWith('deduction_') ? 'deduction' : item.rule_key.startsWith('treatment_') ? 'treatment_addition' : 'base_addition');

        // 率（%）項目の安全正規化: 9.4 等のパーセント入力値が渡された場合は 0.094 の小数に変換
        const isRate = category === 'treatment_addition' || category === 'deduction' || 
                       item.unit_label === '率' || item.unit_label === '%' || 
                       existing?.unit_label === '率' || existing?.unit_label === '%';
        let safeValue = item.proposed_value;
        if (isRate && safeValue > 1) {
          safeValue = parseFloat((safeValue / 100).toFixed(4));
        }

        updateStmt.run(
          item.rule_key,
          category,
          item.rule_name,
          safeValue,
          existing?.unit_label || (isRate ? '率' : item.unit_label),
          item.description,
          item.effective_from
        );

        logStmt.run(
          `audit_${Date.now()}_${Math.random().toString(36).substring(2, 7)}`,
          role,
          role === 'admin' ? '管理者' : 'サービス管理責任者',
          'APPLY_LEGAL_RULE_UPDATE',
          'billing_rules',
          item.rule_key,
          JSON.stringify({
            old_value: item.current_value,
            new_value: safeValue,
            description: item.description,
            effective_from: item.effective_from
          })
        );
      }
    })();

    revalidatePath('/');
    revalidatePath('/settings');
    revalidatePath('/billing');
    revalidatePath('/billing/simulation');
    
    return { success: true, count: updates.length };
  } catch (error: any) {
    console.error('Failed to apply billing rule updates:', error);
    return { success: false, error: '加算ルールの更新に失敗しました: ' + (error.message || String(error)) };
  }
}

  const role = await getRole();

    let safeValue = rule_value;
    if (isRate && safeValue > 1) {

    }

    }

    // File extension check
    const originalExt = path.extname(file.name).toLowerCase();
    const allowedExtensions = ['.pdf', '.jpg', '.jpeg', '.png', '.doc', '.docx', '.xlsx'];
    if (!allowedExtensions.includes(originalExt)) {
      return NextResponse.json({ error: "Invalid file extension. Only .pdf, .jpg, .jpeg, .png, .doc, .docx, .xlsx are allowed." }, { status: 400 });
    }

    // Sanitize filename
    const sanitizedFileName = file.name.replace(/\.\./g, '');

    await ensureStorageDir();

    const fileBuffer = Buffer.from(await file.arrayBuffer());
    // We already checked originalExt, use it
    const fileExtension = originalExt || (file.type === 'application/pdf' ? '.pdf' : '.png');
    const documentId = uuidv4();
    const savedFileName = `${documentId}${fileExtension}`;
    const filePath = path.join(STORAGE_DIR, savedFileName);

    // Save to disk
    await fs.writeFile(filePath, fileBuffer);

    // Save to DB (relative path or just file name)
    const relativePath = `storage/documents/${savedFileName}`;
    
    db.prepare(`
      INSERT INTO client_documents (id, client_id, file_name, file_path, file_type)
      VALUES (?, ?, ?, ?, ?)
    `).run(documentId, clientId, sanitizedFileName, relativePath, file.type);

    return NextResponse.json({ 
      success: true, 
      document: {
        id: documentId,
        file_name: file.name,
        file_type: file.type,
      }
    });
  } catch (error: any) {
    console.error("Failed to upload document:", error);
    return NextResponse.json({ error: "Failed to upload document" }, { status: 500 });
  }
  });
}

import { NextRequest, NextResponse } from "next/server";
import { db, withServiceModeFromRequest } from "@/lib/db";
import { getRole, assertCanAccessClient } from "@/lib/auth";
import fs from "fs/promises";
import path from "path";

// GET: Stream the file content for viewing/downloading
export async function GET(
  request: NextRequest,
  { params }: { params: Promise<{ docId: string }> }
) {
  return withServiceModeFromRequest(request, async () => {
    try {
      const role = await getRole();
      if (!role) {
        return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
      }

      const { docId } = await params;

      const document = db.prepare(`
        SELECT client_id, file_name, file_path, file_type 
        FROM client_documents 
        WHERE id = ?
      `).get(docId) as { client_id: string; file_name: string; file_path: string; file_type: string } | undefined;

      if (!document) {
        return NextResponse.json({ error: "Document not found" }, { status: 404 });
      }

      await assertCanAccessClient(document.client_id);

      const baseDir = path.join(process.cwd(), 'storage', 'documents');
      const storageDirWithSep = baseDir.endsWith(path.sep) ? baseDir : baseDir + path.sep;
      const fullPath = path.resolve(/*turbopackIgnore: true*/ process.cwd(), document.file_path);
      
      if (!fullPath.startsWith(storageDirWithSep)) {
        return NextResponse.json({ error: "Access denied" }, { status: 403 });
      }
      
      try {
        const fileBuffer = await fs.readFile(fullPath);
        
        const MIME_MAP: Record<string, string> = {
          '.pdf': 'application/pdf',
          '.png': 'image/png',
          '.jpg': 'image/jpeg',
          '.jpeg': 'image/jpeg',
          '.gif': 'image/gif',
          '.webp': 'image/webp',
          '.doc': 'application/msword',
          '.docx': 'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
          '.xls': 'application/vnd.ms-excel',
          '.xlsx': 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
          '.txt': 'text/plain; charset=utf-8',
          '.csv': 'text/csv; charset=utf-8',
        };
        const ext = path.extname(document.file_path).toLowerCase();
        const safeContentType = MIME_MAP[ext] || 'application/octet-stream';
        const isInlineSafe = ['.pdf', '.png', '.jpg', '.jpeg', '.gif', '.webp', '.txt'].includes(ext);

        const headers = new Headers();
        headers.set("Content-Type", safeContentType);
        headers.set("Content-Disposition", `${isInlineSafe ? 'inline' : 'attachment'}; filename*=UTF-8''${encodeURIComponent(document.file_name)}`);
        headers.set("X-Content-Type-Options", "nosniff");
        headers.set("Content-Security-Policy", "default-src 'none'; style-src 'unsafe-inline'; sandbox");

        return new NextResponse(fileBuffer, { status: 200, headers });
      } catch (err) {
        console.error("File reading error:", err);
        return NextResponse.json({ error: "File not found on disk" }, { status: 404 });
      }

    } catch (error: any) {
      console.error("Failed to fetch document:", error);
      return NextResponse.json({ error: error.message || "Failed to fetch document" }, { status: 500 });
    }
  });
}

export async function DELETE(
  request: NextRequest,
  { params }: { params: Promise<{ docId: string }> }
) {
  return withServiceModeFromRequest(request, async () => {
    try {
      const role = await getRole();
      if (!role) {
        return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
      }

      const { docId } = await params;

      const document = db.prepare(`
        SELECT client_id, file_path FROM client_documents WHERE id = ?
      `).get(docId) as { client_id: string; file_path: string } | undefined;

      if (!document) {
        return NextResponse.json({ error: "Document not found" }, { status: 404 });
      }

      await assertCanAccessClient(document.client_id);

      const baseDir = path.join(process.cwd(), 'storage', 'documents');
      const storageDirWithSep = baseDir.endsWith(path.sep) ? baseDir : baseDir + path.sep;
      const fullPath = path.resolve(/*turbopackIgnore: true*/ process.cwd(), document.file_path);
      if (!fullPath.startsWith(storageDirWithSep)) {
        return NextResponse.json({ error: "Access denied" }, { status: 403 });
      }

      // Delete from database
      db.prepare(`DELETE FROM client_documents WHERE id = ?`).run(docId);

      // Delete from disk
      try {
        await fs.unlink(fullPath);
      } catch (err) {
        console.warn("Failed to delete file from disk (it might already be deleted):", err);
      }

      return NextResponse.json({ success: true });
    } catch (error: any) {
      console.error("Failed to delete document:", error);
      return NextResponse.json({ error: error.message || "Failed to delete document" }, { status: 500 });
    }
  });
}

import { NextRequest, NextResponse } from 'next/server';
import path from 'path';
import fs from 'fs';
import { v4 as uuidv4 } from 'uuid';
import { db, getServiceModeSync, withServiceModeFromRequest } from '@/lib/db';
import { getRole } from '@/lib/auth';
import { revalidatePath } from 'next/cache';
import { extractTextFromPdfBuffer, extractTextFromPlainBuffer } from '@/lib/compliance/pdfTextExtractor';

const MAX_FILE_SIZE_BYTES = 10 * 1024 * 1024; // 10MB
const ALLOWED_EXTENSIONS = new Set(['.pdf', '.txt', '.md']);
const ALLOWED_CATEGORIES = new Set(['mhlw_notice', 'local_guideline', 'internal_rule', 'qa']);

export async function POST(req: NextRequest) {
  return withServiceModeFromRequest(req, async () => {
    try {
      const role = await getRole();
      if (role !== 'admin' && role !== 'service_manager') {
        return NextResponse.json(
          { success: false, error: '公式資料のアップロード権限がありません（管理者またはサビ管のみ可能）' },
          { status: 403 }
        );
      }

      const formData = await req.formData();
      const file = formData.get('file') as File | null;
      const rawTitle = (formData.get('title') as string | null)?.trim() || '';
      const rawCategory = (formData.get('category') as string | null)?.trim() || 'mhlw_notice';
      const manualText = (formData.get('extractedText') as string | null)?.trim() || '';

      if (!file) {
        return NextResponse.json(
          { success: false, error: 'アップロードするファイル（.pdf, .txt, .md）を選択してください' },
          { status: 400 }
        );
      }

      if (file.size <= 0) {
        return NextResponse.json(
          { success: false, error: '空のファイルはアップロードできません' },
          { status: 400 }
        );
      }

      if (file.size > MAX_FILE_SIZE_BYTES) {
        return NextResponse.json(
          { success: false, error: 'ファイルサイズが上限（20MB）を超えています' },
          { status: 400 }
        );
      }

      const originalName = path.basename(file.name || 'document.txt');
      const ext = path.extname(originalName).toLowerCase();
      if (!ALLOWED_EXTENSIONS.has(ext)) {
        return NextResponse.json(
          { success: false, error: '対応していないファイル形式です。.pdf, .txt, .md のみアップロード可能です' },
          { status: 400 }
        );
      }

      const category = ALLOWED_CATEGORIES.has(rawCategory) ? rawCategory : 'mhlw_notice';
      const serviceMode = getServiceModeSync(); // 'type_a' or 'type_b'

      // パストラバーサル防止: モード別の専用ディレクトリを厳格に解決
      const storageRoot = path.resolve(process.cwd(), 'storage', 'knowledge', serviceMode);
      if (!fs.existsSync(storageRoot)) {
        fs.mkdirSync(storageRoot, { recursive: true });
      }

      const docId = `know_${serviceMode === 'type_b' ? 'b' : 'a'}_${uuidv4().slice(0, 8)}`;
      const safeFileName = originalName.replace(/[^a-zA-Z0-9.\-_ぁ-んァ-ヶｱ-ﾝﾞﾟ一-龠々ー（）()]/g, '_');
      const storedFileName = `${docId}_${safeFileName}`;
      const resolvedTargetPath = path.resolve(storageRoot, storedFileName);

      // 境界チェック（末尾 path.sep 付き）
      if (!resolvedTargetPath.startsWith(storageRoot + path.sep)) {
        return NextResponse.json(
          { success: false, error: '不正なファイルパスが検出されました' },
          { status: 400 }
        );
      }

      const arrayBuffer = await file.arrayBuffer();
      const buffer = Buffer.from(arrayBuffer);

      // PDFマジックナンバー検査 (.pdf の場合は先頭 %PDF- を確認)
      if (ext === '.pdf') {
        const header = buffer.subarray(0, 5).toString('ascii');
        if (!header.startsWith('%PDF-')) {
          return NextResponse.json(
            { success: false, error: '有効なPDFファイルではありません（ヘッダ不正）' },
            { status: 400 }
          );
        }
      }

      // ディスクへ安全に保存
      fs.writeFileSync(resolvedTargetPath, buffer);

      // テキスト抽出（PDFまたはプレーンテキスト）
      let extractedText = manualText;
      if (!extractedText) {
        if (ext === '.pdf') {
          extractedText = await extractTextFromPdfBuffer(buffer, originalName);
        } else {
          extractedText = extractTextFromPlainBuffer(buffer);
        }
      }

      const title = rawTitle || originalName.replace(/\.[^.]+$/, '');
      const relFilePath = path.join('storage', 'knowledge', serviceMode, storedFileName).replace(/\\/g, '/');

      db.prepare(`
        INSERT INTO knowledge_documents (
          id, title, category, file_name, file_path, extracted_text, is_active, uploaded_at, deleted_at
        ) VALUES (?, ?, ?, ?, ?, ?, 1, CURRENT_TIMESTAMP, NULL)
      `).run(docId, title, category, originalName, relFilePath, extractedText);

      // 監査ログ記録
      try {
        db.prepare(`
          INSERT INTO audit_logs (id, user_id, user_name, action, target_table, target_id, details)
          VALUES (?, ?, ?, ?, ?, ?, ?)
        `).run(
          `audit_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`,
          role,
          role === 'admin' ? '管理者' : 'サービス管理責任者',
          'UPLOAD_KNOWLEDGE_DOCUMENT',
          'knowledge_documents',
          docId,
          JSON.stringify({
            service_mode: serviceMode,
            title,
            category,
            file_name: originalName,
            size_bytes: buffer.byteLength,
            extracted_chars: extractedText.length
          })
        );
      } catch (e) {}

      revalidatePath('/compliance/notebook');
      revalidatePath('/');

      const created = db.prepare(`
        SELECT id, title, category, file_name, file_path, extracted_text, is_active, uploaded_at
        FROM knowledge_documents
        WHERE id = ?
      `).get(docId);

      return NextResponse.json({
        success: true,
        document: created
      });
    } catch (error: any) {
      console.error('Failed to upload knowledge document:', error);
      return NextResponse.json(
        { success: false, error: '資料のアップロード処理中にエラーが発生しました: ' + (error.message || String(error)) },
        { status: 500 }
      );
    }
  });
}
