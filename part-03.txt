function TimeSelect({ 
  name, 
  label, 
  defaultValue, 
  value: controlledValue, 
  onChange: onControlledChange, 
  minHour = 8, 
  maxHour = 18 
}: { 
  name: string, 
  label: string, 
  defaultValue?: string, 
  value?: string, 
  onChange?: (val: string) => void, 
  minHour?: number, 
  maxHour?: number 
}) {
  const initialTime = controlledValue ?? defaultValue ?? "";
  const [internalHour, setInternalHour] = useState(initialTime ? initialTime.split(':')[0] : "")
  const [internalMinute, setInternalMinute] = useState(initialTime ? initialTime.split(':')[1] : "")
  
  const hour = controlledValue !== undefined ? (controlledValue ? controlledValue.split(':')[0] : "") : internalHour;
  const minute = controlledValue !== undefined ? (controlledValue ? controlledValue.split(':')[1] : "") : internalMinute;
  const value = hour && minute ? `${hour}:${minute}` : ""

  const hours = Array.from({ length: maxHour - minHour + 1 }, (_, i) => (i + minHour).toString().padStart(2, '0'));

  const handleHourChange = (newH: string) => {
    if (controlledValue === undefined) setInternalHour(newH);
    const newMin = minute || "00";
    if (onControlledChange) {
      onControlledChange(newH ? `${newH}:${newMin}` : "");
    }
  };

  const handleMinuteChange = (newM: string) => {
    if (controlledValue === undefined) setInternalMinute(newM);
    const newHr = hour || "09";
    if (onControlledChange) {
      onControlledChange(newHr ? `${newHr}:${newM}` : "");
    }
  };

  return (
    <div className="space-y-1">
      <Label className="text-xs font-semibold">{label}</Label>
      <input type="hidden" name={name} value={value} />
      <div className="flex items-center gap-1">
        <select 
          className="flex h-9 w-full rounded-md border border-input bg-transparent px-2 py-1 text-sm shadow-sm transition-colors focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring"
          value={hour} 
          onChange={e => handleHourChange(e.target.value)}
          aria-label={`${label} 時`}
        >
          <option value="">--</option>
          {hours.map(h => (
            <option key={h} value={h}>{h}</option>
          ))}
        </select>
        <span className="font-bold">:</span>
        <select 
          className="flex h-9 w-full rounded-md border border-input bg-transparent px-2 py-1 text-sm shadow-sm transition-colors focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring"
          value={minute} 
          onChange={e => handleMinuteChange(e.target.value)}
          aria-label={`${label} 分`}
        >
          <option value="">--</option>
          {Array.from({ length: 60 }, (_, i) => i.toString().padStart(2, '0')).map(m => (
            <option key={m} value={m}>{m}</option>
          ))}
        </select>
      </div>
    </div>
  )
}

export function AttendanceForm({ 
  clients, 
  todayStr,
  initialData,
  records = [],
  offsiteWorkplaces = [],
  staffList = [],
  clientShifts = [],
  hasMeal = true,
  hasTransportation = true,
  hasVisitSupport = true,
  hasAbsenceSupport = true
}: { 
  clients: any[], 
  todayStr: string,
  initialData?: any,
  records?: any[],
  offsiteWorkplaces?: any[],
  staffList?: any[],
  clientShifts?: any[],
  hasMeal?: boolean,
  hasTransportation?: boolean,
  hasVisitSupport?: boolean,
  hasAbsenceSupport?: boolean
}) {

  const router = useRouter()
  const [status, setStatus] = useState(initialData?.status || "present")
  const [isAbsenceAddition, setIsAbsenceAddition] = useState(initialData?.is_absence_addition === 1)
  const [isVisitSupport, setIsVisitSupport] = useState(initialData?.visit_support_used === 1)
  const [workType, setWorkType] = useState(initialData?.work_type || "in_facility")
  const [recordDate, setRecordDate] = useState(initialData?.record_date || todayStr)
  const [clientId, setClientId] = useState(initialData?.client_id || "")

  // 時間管理
  const [clockIn, setClockIn] = useState(initialData?.clock_in || "09:00")
  const [clockOut, setClockOut] = useState(initialData?.clock_out || "16:00")
  const [breakTimeMinutes, setBreakTimeMinutes] = useState(initialData?.break_time_minutes ?? 60)

  // 施設外・内勤 分割時間管理
  const [offsiteStartTime, setOffsiteStartTime] = useState(initialData?.offsite_start_time || "10:00")
  const [offsiteEndTime, setOffsiteEndTime] = useState(initialData?.offsite_end_time || "12:00")
  const [inFacilityStartTime, setInFacilityStartTime] = useState(initialData?.in_facility_start_time || "13:00")
  const [inFacilityEndTime, setInFacilityEndTime] = useState(initialData?.in_facility_end_time || "15:00")
  const [offsiteWorkplaceId, setOffsiteWorkplaceId] = useState(initialData?.offsite_workplace_id || "")
  const [offsiteStaffId, setOffsiteStaffId] = useState(initialData?.offsite_staff_id || "")

  // シフト自動判定案内
  const [shiftNotice, setShiftNotice] = useState<string>("")

  // 初期データ読み込み
  useEffect(() => {
    if (initialData) {
      setStatus(initialData.status || "present")
      setIsAbsenceAddition(initialData.is_absence_addition === 1)
      setIsVisitSupport(initialData.visit_support_used === 1)
      setWorkType(initialData.work_type || "in_facility")
      setRecordDate(initialData.record_date || todayStr)
      setClientId(initialData.client_id || "")
      setClockIn(initialData.clock_in || "09:00")
      setClockOut(initialData.clock_out || "16:00")
      setBreakTimeMinutes(initialData.break_time_minutes ?? 60)
      setOffsiteStartTime(initialData.offsite_start_time || "10:00")
      setOffsiteEndTime(initialData.offsite_end_time || "12:00")
      setInFacilityStartTime(initialData.in_facility_start_time || "13:00")
      setInFacilityEndTime(initialData.in_facility_end_time || "15:00")
      setOffsiteWorkplaceId(initialData.offsite_workplace_id || "")
      setOffsiteStaffId(initialData.offsite_staff_id || "")
      setShiftNotice("")
    }
  }, [initialData, todayStr])

  // 利用者や日付が変更されたとき、新規作成ならシフト情報または曜日別パターンから公休判定・予定時間を自動反映
  useEffect(() => {
    if (!initialData && clientId && recordDate) {
      const shift = clientShifts?.find(s => s.client_id === clientId && s.target_date === recordDate);
      const client = clients?.find(c => c.id === clientId);

      if (shift) {
        if (shift.is_holiday === 1) {
          setStatus('holiday');
          setShiftNotice(`※ シフト表の【${shift.holiday_type || '公休'}】設定に基づき、自動判定されました`);
          return;
        } else {
          // 出勤予定
          setStatus('present');
          const wt = shift.work_type || (shift.is_offsite === 1 ? 'out_facility' : 'in_facility');
          setWorkType(wt);
          if (shift.scheduled_in) setClockIn(shift.scheduled_in);
          if (shift.scheduled_out) setClockOut(shift.scheduled_out);
          if (shift.offsite_start_time) setOffsiteStartTime(shift.offsite_start_time);
          if (shift.offsite_end_time) setOffsiteEndTime(shift.offsite_end_time);
          if (shift.in_facility_start_time) setInFacilityStartTime(shift.in_facility_start_time);
          if (shift.in_facility_end_time) setInFacilityEndTime(shift.in_facility_end_time);

          // 就労先IDの補完（シフト側になければ曜日別パターンから参照）
          if (shift.offsite_workplace_id) {
            setOffsiteWorkplaceId(shift.offsite_workplace_id);
          } else if (client?.weekly_schedule_pattern) {
            try {
              const p = typeof client.weekly_schedule_pattern === 'string'
                ? JSON.parse(client.weekly_schedule_pattern)
                : client.weekly_schedule_pattern;
              const dObj = new Date(recordDate);
              const dayP = p[dObj.getDay().toString()];
              if (dayP?.offsite_workplace_id) {
                setOffsiteWorkplaceId(dayP.offsite_workplace_id);
              }
            } catch (e) {}
          }

          setShiftNotice(`※ シフト表の出勤予定（${shift.scheduled_in || '10:00'}〜${shift.scheduled_out || '15:00'}、${wt === 'both' ? '内勤＋施設外' : wt === 'out_facility' ? '施設外' : '内勤'}）を自動反映しました`);
          return;
        }
      } else if (client) {
        // 個別シフト未登録の場合: 曜日別定例パターン（weekly_schedule_pattern）をチェック
        let matchedPattern: any = null;
        if (client.weekly_schedule_pattern) {
          try {
            const parsed = typeof client.weekly_schedule_pattern === 'string'
              ? JSON.parse(client.weekly_schedule_pattern)
              : client.weekly_schedule_pattern;
            const dObj = new Date(recordDate);
            matchedPattern = parsed[dObj.getDay().toString()];
          } catch (e) {}
        }

        if (matchedPattern) {
          if (matchedPattern.is_holiday) {
            setStatus('holiday');
            setShiftNotice(`※ 曜日別定例パターンの【${matchedPattern.holiday_type || '公休'}】設定に基づき、自動判定されました`);
            return;
          } else {
            setStatus('present');
            const wt = matchedPattern.work_type || 'in_facility';
            setWorkType(wt);
            const sTime = matchedPattern.start_time || client.work_start_time || '10:00';
            const eTime = matchedPattern.end_time || client.work_end_time || (wt === 'out_facility' ? '16:00' : '15:00');
            setClockIn(sTime);
            setClockOut(eTime);
            if (matchedPattern.offsite_workplace_id) {
              setOffsiteWorkplaceId(matchedPattern.offsite_workplace_id);
            }
            setBreakTimeMinutes(client.break_time_minutes ?? 60);
            const typeLabel = wt === 'both' ? '内勤＋施設外' : wt === 'out_facility' ? '施設外就労' : '内勤';
            setShiftNotice(`※ 利用者台帳の曜日別設定（${sTime}〜${eTime}、${typeLabel}）を初期反映しました`);
            return;
          }
        }

        // 利用者台帳（定休日・定例設定）フォールバック判定
        const holidayInfo = getClientDayHolidayInfo(recordDate, client);
        if (holidayInfo.isHoliday) {
          setStatus('holiday');
          setShiftNotice(`※ 利用者台帳の【${holidayInfo.holidayLabel || '公休'}】設定に基づき、自動で公休判定されました`);
          return;
        }
        
        // デフォルト出勤
        setStatus('present');
        setClockIn(client.work_start_time || '10:00');
        setClockOut(client.work_end_time || '15:00');
        setBreakTimeMinutes(client.break_time_minutes ?? 60);
        setShiftNotice('');
      }
    }
  }, [clientId, recordDate, initialData, clientShifts, clients]);

  const showAbsenceAdditionForm = status === "absent" && isAbsenceAddition
  const isPresent = status === "present"
  
  // 選択された利用者の今月の欠席時対応加算の取得回数を計算 (編集中のレコードは除外)
  const absenceAdditionCount = useMemo(() => {
    if (!clientId || !records) return 0;
    return records.filter(r => 
      r.client_id === clientId && 
      r.is_absence_addition === 1 &&
      r.id !== initialData?.id
    ).length;
  }, [clientId, records, initialData]);

  const isOverLimit = absenceAdditionCount >= 4;

  const applyExampleSplitTimes = () => {
    setWorkType('both');
    setOffsiteStartTime('10:00');
    setOffsiteEndTime('12:00');
    setInFacilityStartTime('13:00');
    setInFacilityEndTime('15:00');
    setClockIn('10:00');
    setClockOut('15:00');
    setBreakTimeMinutes(60);
    toast.info('「10:00〜12:00 施設外 / 13:00〜15:00 内勤」をセットしました');
  };

  const applyExampleSplitTimesReverse = () => {
    setWorkType('both');
    setInFacilityStartTime('10:00');
    setInFacilityEndTime('12:00');
    setOffsiteStartTime('13:00');
    setOffsiteEndTime('15:00');
    setClockIn('10:00');
    setClockOut('15:00');
    setBreakTimeMinutes(60);
    toast.info('「10:00〜12:00 内勤 / 13:00〜15:00 施設外」をセットしました');
  };

  const handleSubmit = async (formData: FormData) => {
    try {
      const res = await saveAttendanceRecord(formData);
      if (res && !res.success) {
        toast.error(res.error);
        return;
      }
      toast.success("勤怠データを保存しました");
      if (res?.recordDate) {
        router.push(`/attendance?date=${res.recordDate}`);
        router.refresh();
      }
    } catch (e: any) {
      toast.error(e.message || "エラーが発生しました");
    }
  };

  return (
    <form action={handleSubmit} className="space-y-6">
      <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
        <div className="space-y-2">
          <Label htmlFor="recordDate">日付</Label>
          <Input type="date" id="recordDate" name="recordDate" required value={recordDate} onChange={e => setRecordDate(e.target.value)} />
        </div>
        <div className="space-y-2">
          <Label htmlFor="clientId">利用者</Label>
          <select 
            id="clientId" 
            name="clientId" 
            required 
            className="flex h-9 w-full rounded-md border border-input bg-transparent px-3 py-1 text-sm shadow-sm transition-colors focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring disabled:cursor-not-allowed disabled:opacity-50"
            value={clientId}
            onChange={e => setClientId(e.target.value)}
          >
            <option value="">選択してください</option>
            {clients?.map(c => (
              <option key={c.id} value={c.id}>{c.full_name}</option>
            ))}
          </select>
        </div>

        <div className="space-y-2">
          <div className="flex items-center justify-between">
            <Label htmlFor="statusSelect">勤怠ステータス</Label>
            {status === 'holiday' && (
              <span className="text-[11px] font-semibold text-blue-700 bg-blue-50 px-1.5 py-0.5 rounded border border-blue-200">
                公休
              </span>
            )}
            {status === 'absent' && !isAbsenceAddition && (
              <span className="text-[11px] font-semibold text-amber-800 bg-amber-50 px-1.5 py-0.5 rounded border border-amber-300">
                欠勤（加算なし・無給）
              </span>
            )}
            {status === 'absent' && isAbsenceAddition && (
              <span className="text-[11px] font-semibold text-rose-700 bg-rose-50 px-1.5 py-0.5 rounded border border-rose-300">
                欠席時加算 算定中
              </span>
            )}
            {status === 'pto' && (
              <span className="text-[11px] font-semibold text-purple-700 bg-purple-50 px-1.5 py-0.5 rounded border border-purple-200">
                有給
              </span>
            )}
          </div>
          <select 
            id="statusSelect" 
            className="flex h-9 w-full rounded-md border border-input bg-transparent px-3 py-1 text-sm shadow-sm transition-colors focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring"
            value={
              status === 'present' ? 'present' :
              status === 'holiday' ? 'holiday' :
              status === 'pto' ? 'pto' :
              (status === 'absent' && isAbsenceAddition) ? 'absent_addition' : 'absent_normal'
            }
            onChange={(e) => {
              const val = e.target.value;
              if (val === 'present') {
                setStatus('present');
                setIsAbsenceAddition(false);
              } else if (val === 'holiday') {
                setStatus('holiday');
                setIsAbsenceAddition(false);
              } else if (val === 'pto') {
                setStatus('pto');
                setIsAbsenceAddition(false);
              } else if (val === 'absent_normal') {
                setStatus('absent');
                setIsAbsenceAddition(false);
              } else if (val === 'absent_addition') {
                setStatus('absent');
                setIsAbsenceAddition(true);
              }
            }}
          >
            <option value="present">出勤</option>
            <option value="holiday">公休（事業所の休日・個人の所定休日）</option>
            <option value="absent_normal">欠勤（有給を使わない休み・無給欠勤 / 加算なし）</option>
            {hasAbsenceSupport && (
              <option value="absent_addition">欠席（欠席時対応加算を算定 / 94単位）</option>
            )}
            <option value="pto">有給（年次有給休暇取得）</option>
          </select>

          <input type="hidden" name="status" value={status} />
          <input type="hidden" name="isAbsenceAddition" value={isAbsenceAddition ? "true" : "false"} />
        </div>
      </div>

      {/* シフト自動判定のアラート/案内 */}
      {shiftNotice && (
        <div className="p-2.5 rounded-md bg-blue-50 border border-blue-200 text-xs text-blue-900 flex items-center gap-2">
          <CheckCircle2 className="w-4 h-4 text-blue-600 shrink-0" />
          <span>{shiftNotice}</span>
        </div>
      )}

      {/* 公休・有休・欠勤（加算なし）時の案内 */}
      {(status === 'holiday' || status === 'pto' || (status === 'absent' && !isAbsenceAddition)) && (
        <div className="p-4 border rounded-md bg-amber-50/50 border-amber-200 text-sm text-amber-900 space-y-1">
          <div className="font-semibold flex items-center gap-1.5">
            <AlertCircle className="w-4 h-4 text-amber-600" />
            {status === 'holiday' 
              ? '本日は「公休」として記録されます' 
              : status === 'pto'
                ? '本日は「有給休暇」として記録されます'
                : '本日は「欠勤（有給を使わない休み・加算なし）」として記録されます'}
          </div>
          <p className="text-xs text-amber-800 leading-relaxed">
            {status === 'holiday' 
              ? '出退勤時間は不要です。シフト連動により、本体事業所の必要人員配置の分母から除外されます。'
              : status === 'pto'
                ? '有給休暇管理簿に消化実績が反映されます。保存ボタンを押して記録を確定してください。'
                : '出退勤時間は記録されず、有給休暇も消化されません。欠席時対応加算は算定（請求）されませんのでご安心ください。'}
          </p>
        </div>
      )}

      {isPresent && (
        <div className="p-4 border rounded-md bg-muted/10 space-y-6">
          <div className="grid grid-cols-1 sm:grid-cols-2 md:grid-cols-4 gap-4">
            <TimeSelect 
              name="clockIn" 
              label="出勤時間" 
              value={clockIn} 
              onChange={setClockIn} 
            />
            <TimeSelect 
              name="clockOut" 
              label="退勤時間" 
              value={clockOut} 
              onChange={setClockOut} 
            />
            <div className="space-y-1">
              <Label htmlFor="breakTimeMinutes" className="text-xs font-semibold">休憩(分)</Label>
              <Input 
                type="number" 
                id="breakTimeMinutes" 
                name="breakTimeMinutes" 
                min="0" 
                value={breakTimeMinutes} 
                onChange={e => setBreakTimeMinutes(parseInt(e.target.value) || 0)} 
                className="h-9"
                onWheel={(e) => e.currentTarget.blur()} 
              />
            </div>
            <div className="space-y-1">
              <Label htmlFor="workType" className="text-xs font-semibold">勤務形態</Label>
              <select 
                id="workType" 
                name="workType" 
                value={workType} 
                onChange={e => setWorkType(e.target.value)} 
                className="flex h-9 w-full rounded-md border border-input bg-transparent px-3 py-1 text-sm shadow-sm transition-colors focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring disabled:cursor-not-allowed disabled:opacity-50"
              >
                <option value="in_facility">施設内（内勤のみ）</option>
                <option value="out_facility">施設外（終日施設外就労）</option>
                <option value="both">施設内＋施設外（時間分割・併用）</option>
                <option value="remote">在宅</option>
              </select>
            </div>
          </div>

          {/* 施設内＋施設外 併用（時間分割）の設定カード */}
          {workType === 'both' && (
            <div className="p-4 rounded-md bg-amber-50/70 border border-amber-300 space-y-4 shadow-sm">
              <div className="flex flex-wrap items-center justify-between gap-2 border-b border-amber-200 pb-2">
                <div className="flex items-center gap-2 text-sm font-bold text-amber-950">
                  <Clock className="w-4 h-4 text-amber-700" />
                  施設外就労と内勤の時間分割設定（指定基準・常時配置連動）
                </div>
                <div className="flex flex-wrap items-center gap-1.5">
                  <Button 
                    type="button" 
                    variant="outline" 
                    size="sm" 
                    onClick={applyExampleSplitTimes}
                    className="h-7 text-xs border-amber-400 bg-white hover:bg-amber-100 text-amber-900"
                  >
                    <Sparkles className="w-3.5 h-3.5 mr-1 text-amber-600" />
                    外10~12 / 内13~15
                  </Button>
                  <Button 
                    type="button" 
                    variant="outline" 
                    size="sm" 
                    onClick={applyExampleSplitTimesReverse}
                    className="h-7 text-xs border-amber-400 bg-white hover:bg-amber-100 text-amber-900"
                  >
                    <Sparkles className="w-3.5 h-3.5 mr-1 text-amber-600" />
                    内10~12 / 外13~15
                  </Button>
                </div>
              </div>

              <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
                {/* 施設外就労ブロック */}
                <div className="p-3 bg-white rounded border border-amber-200 space-y-3">
                  <div className="text-xs font-bold text-amber-900 flex items-center gap-1.5">
                    <span className="w-2.5 h-2.5 rounded-full bg-amber-500" />
                    ① 施設外就労の時間帯
                  </div>
                  <div className="grid grid-cols-2 gap-2">
                    <TimeSelect 
                      name="offsiteStartTime" 
                      label="施設外 開始" 
                      value={offsiteStartTime} 
                      onChange={setOffsiteStartTime} 
                    />
                    <TimeSelect 
                      name="offsiteEndTime" 
                      label="施設外 終了" 
                      value={offsiteEndTime} 
                      onChange={setOffsiteEndTime} 
                    />
                  </div>

'use client';

import { useState } from 'react';
import { Button } from '@/components/ui/button';
import { Users, Loader2 } from 'lucide-react';
import { bulkRegisterPresent } from './actions';
import { toast } from 'sonner';
import { useRouter } from 'next/navigation';

export function BulkAttendanceButton({ 
  targetDate, 
  targetDepartmentId = 'all',
  activeDepartments = []
}: { 
  targetDate: string; 
  targetDepartmentId?: string; 
  activeDepartments?: any[];
}) {
  const [loading, setLoading] = useState(false);
  const router = useRouter();

  const currentDep = activeDepartments.find(d => d.id === targetDepartmentId);
  const locLabel = targetDepartmentId === 'all' ? '全' : (currentDep?.name || '指定拠点');

  const handleBulkRegister = async () => {
    if (!confirm(`【${targetDate}】の未登録利用者（${locLabel}）を通常出席（09:30〜15:30、休憩60分）として一括登録しますか？\n（既に登録済みの記録は上書きされません）`)) {
      return;
    }

    setLoading(true);
    try {
      const res = await bulkRegisterPresent(targetDate, targetDepartmentId);
      if (res.success) {
        toast.success(res.message || '一括出席登録が完了しました');
        router.refresh();
      } else {
        toast.error(res.error || '一括登録に失敗しました');
      }
    } catch (err: any) {
      toast.error(`エラーが発生しました: ${err.message}`);
    } finally {
      setLoading(false);
    }
  };

  return (
    <Button
      type="button"
      onClick={handleBulkRegister}
      disabled={loading}
      className="bg-emerald-600 hover:bg-emerald-700 text-white font-medium shadow-sm h-12 px-5 gap-2"
    >
      {loading ? (
        <>
          <Loader2 className="h-4 w-4 animate-spin" />
          <span>一括登録中...</span>
        </>
      ) : (
        <>
          <Users className="h-4 w-4" />
          <span>{targetDepartmentId !== 'all' ? `${locLabel}未登録者を一括出席` : '当日全員一括出席登録'}</span>
        </>
      )}
    </Button>
  );
}

    const d = new Date();

    d.setMonth(d.getMonth() - monthsAgo);

'use client';

import { useState } from 'react';
import { buttonVariants } from "@/components/ui/button";
import { deleteAttendanceRecord } from './actions';
import { toast } from 'sonner';

export function DeleteAttendanceButton({ recordId }: { recordId: string }) {
  const [isDeleting, setIsDeleting] = useState(false);

  const handleDelete = async () => {
    if (!window.confirm('この出退勤記録を削除してもよろしいですか？\n（月次確定済みのデータがある場合は削除できません）')) {
      return;
    }

    setIsDeleting(true);
    try {
      const res = await deleteAttendanceRecord(recordId);
      if (res.success) {
        toast.success('出退勤記録を削除しました');
      } else {
        toast.error(res.error || '削除に失敗しました');
      }
    } catch (err: any) {
      toast.error('削除処理中にエラーが発生しました');
    } finally {
      setIsDeleting(false);
    }
  };

  return (
    <button
      type="button"
      onClick={handleDelete}
      disabled={isDeleting}
      className={buttonVariants({ variant: "destructive", size: "sm" })}
    >
      {isDeleting ? '削除中...' : '削除'}
    </button>
  );
}

            const autoStaffComment = `【欠席時対応】${parsedData.contactTime ? `${parsedData.contactTime}に` : ''}${parsedData.contactPerson || '本人'}より連絡あり。状況: ${parsedData.absenceCondition || '体調不良等'}。対応内容: ${parsedData.consultationDetails || '休養を促し次回通所予定を確認'}。次回通所予定: ${parsedData.nextScheduledDate || '未定'}（対応者: ${parsedData.handledBy || '職員'}）`;

            db.prepare(`
              INSERT INTO support_records (
                id, client_id, record_date, task_content, staff_comment,
                service_location
              ) VALUES (?, ?, ?, ?, ?, ?)
            `).run(
              uuidv4(),
              parsedData.clientId,
              parsedData.recordDate,
              autoTaskContent,
              autoStaffComment,
              '電話・連絡'
            );
          }
        }
      } catch (supportErr) {
        console.warn('支援記録自動連動スキップ（計画未作成または個別制約）:', supportErr);
      }
    })() // Execute the transaction

    // 監査ログに記録
    await logAudit({
      action: 'update',
      targetTable: 'attendance_records',
      targetId: `${parsedData.clientId}:${parsedData.recordDate}`,
      details: {
        clientId: parsedData.clientId,
        date: parsedData.recordDate,
        status: parsedData.status,
        workType: parsedData.workType,
        isAbsenceAddition: parsedData.isAbsenceAddition
      }
    });

    // 有給休暇の自動消化・返還連動
    try {
      if (!prevWasPto && newIsPto) {
        await consumePtoDays(parsedData.clientId, 1);
      } else if (prevWasPto && !newIsPto) {
        await restorePtoDays(parsedData.clientId, 1);
      }
    } catch (ptoErr) {
      console.error('有給自動消化/返還エラー:', ptoErr);
    }

    try {
      revalidatePath('/attendance', 'layout');
      revalidatePath('/', 'layout');
      revalidatePath('/support-records');
      revalidatePath('/billing/payroll');
    } catch (e) {}
    return { success: true, recordDate: parsedData.recordDate }
  } catch (error) {
    if (error instanceof Error && (error.message.includes('【警告】') || error.message.startsWith('【'))) {
      return { success: false, error: error.message };
    }
    console.error('勤怠保存エラー:', error)
    return { success: false, error: '勤怠の保存に失敗しました' };
  }
}

export async function deleteAttendanceRecord(id: string) {
  try {
    const role = await getRole()
    if (!role) return { success: false, error: '認証されていません' };
    resolveServiceModeForAttendance(id);

    const record = db.prepare('SELECT client_id, record_date, status FROM attendance_records WHERE id = ?').get(id) as any;
    if (record) {
      await assertCanAccessClient(record.client_id);
      const yearMonth = record.record_date.substring(0, 7);
      if (await isMonthLocked(yearMonth)) {
        return { success: false, error: `【請求ロック済】${yearMonth}月は国保連請求データが月次確定ロックされているため、勤怠レコードを削除できません。変更が必要な場合は管理者にロック解除を依頼してください。` };
      }
      const lockedProvision = db.prepare('SELECT id FROM monthly_provision_records WHERE client_id = ? AND target_month = ? AND is_locked = 1').get(record.client_id, yearMonth);
      const lockedPayslip = db.prepare('SELECT id FROM payslips WHERE client_id = ? AND target_month = ? AND is_locked = 1').get(record.client_id, yearMonth);
      if (lockedProvision || lockedPayslip) {
        return { success: false, error: `【警告】${yearMonth}月は実績記録票または給与明細が確定ロックされているため、勤怠データを削除できません。` };
      }
    }

    db.prepare('UPDATE attendance_records SET deleted_at = CURRENT_TIMESTAMP WHERE id = ?').run(id)

    // 有給休暇レコードを削除した場合は有休残日数を自動返還
    if (record && (record.status === 'pto' || record.status === 'paid_leave')) {
      try {
        await restorePtoDays(record.client_id, 1);
      } catch (ptoErr) {
        console.error('有給自動返還エラー:', ptoErr);
      }
    }
    try {
      revalidatePath('/attendance', 'layout');
      revalidatePath('/', 'layout');
      revalidatePath('/support-records');
      revalidatePath('/billing/payroll');
    } catch (e) {}
    return { success: true }
  } catch (error) {
    console.error('勤怠削除エラー:', error)
    if (error instanceof Error && error.message.includes('【警告】')) {
      return { success: false, error: error.message };
    }
    return { success: false, error: '勤怠データの削除に失敗しました' };
  }
}

export async function bulkRegisterPresent(targetDate: string, targetDepartmentId: string = 'all') {
  try {
    const role = await getRole();
    if (!role) return { success: false, error: '認証されていません' };

    if (!targetDate || !/^\d{4}-\d{2}-\d{2}$/.test(targetDate)) {
      return { success: false, error: '日付の指定が正しくありません' };
    }

    const yearMonth = targetDate.substring(0, 7);
    if (await isMonthLocked(yearMonth)) {
      return { success: false, error: `【請求ロック済】${yearMonth}月は国保連請求がロックされているため一括登録できません。` };
    }

    let clientsQuery = `
      SELECT id, full_name, uses_meal_provision, uses_transportation, work_start_time, work_end_time, break_time_minutes, regular_holidays, weekly_schedule_pattern, department_id 
      FROM clients 
      WHERE deleted_at IS NULL AND (status IS NULL OR status = 'active')
    `;
    const queryParams: any[] = [];
    if (targetDepartmentId && targetDepartmentId !== 'all') {
      clientsQuery += ` AND department_id = ?`;
      queryParams.push(targetDepartmentId);
    }
    clientsQuery += ` ORDER BY full_name`;

    const clients = db.prepare(clientsQuery).all(...queryParams) as any[];

    let locLabel = '全体';
    if (targetDepartmentId && targetDepartmentId !== 'all') {
      const dep = db.prepare('SELECT name FROM departments WHERE id = ?').get(targetDepartmentId) as any;
      if (dep) locLabel = dep.name;
    }
    if (clients.length === 0) {
      return { success: false, error: `${locLabel}に登録されている利用者がいません` };
    }

    const existingRecords = db.prepare(`
      SELECT client_id FROM attendance_records WHERE record_date = ? AND deleted_at IS NULL
    `).all(targetDate) as { client_id: string }[];
    const existingClientIds = new Set(existingRecords.map(r => r.client_id));

    const toInsert = clients.filter(c => !existingClientIds.has(c.id));
    if (toInsert.length === 0) {
      return { success: true, message: `${locLabel}の利用者は全員勤怠が登録済みです（スキップ: 0名）`, count: 0 };
    }

    // 当日の利用者シフトを取得
    const clientShifts = db.prepare(`
      SELECT * FROM client_shifts WHERE target_date = ?
    `).all(targetDate) as any[];
    const shiftMap = new Map(clientShifts.map(s => [s.client_id, s]));

    const insertStmt = db.prepare(`
      INSERT INTO attendance_records (
        id, client_id, record_date, status,
        clock_in, clock_out, break_time_minutes, work_type,
        meal_provided, transportation_used, pickup_used, dropoff_used,
        transportation_service_used, pickup_service, dropoff_service,
        is_absence_addition, is_offsite, offsite_start_time, offsite_end_time, in_facility_start_time, in_facility_end_time, offsite_workplace_id,
        created_at, updated_at, deleted_at
      ) VALUES (
        ?, ?, ?, ?,
        ?, ?, ?, ?,
        ?, ?, ?, ?,
        ?, ?, ?,
        0, ?, ?, ?, ?, ?, ?,
        CURRENT_TIMESTAMP, CURRENT_TIMESTAMP, NULL
      )
      ON CONFLICT(client_id, record_date) DO UPDATE SET
        status = excluded.status,
        clock_in = excluded.clock_in,
        clock_out = excluded.clock_out,
        break_time_minutes = excluded.break_time_minutes,
        work_type = excluded.work_type,
        meal_provided = excluded.meal_provided,
        transportation_used = excluded.transportation_used,
        pickup_used = excluded.pickup_used,
        dropoff_used = excluded.dropoff_used,
        transportation_service_used = excluded.transportation_service_used,
        pickup_service = excluded.pickup_service,
        dropoff_service = excluded.dropoff_service,
        is_absence_addition = 0,
        is_offsite = excluded.is_offsite,
        offsite_start_time = excluded.offsite_start_time,
        offsite_end_time = excluded.offsite_end_time,
        in_facility_start_time = excluded.in_facility_start_time,
        in_facility_end_time = excluded.in_facility_end_time,
        offsite_workplace_id = excluded.offsite_workplace_id,
        updated_at = CURRENT_TIMESTAMP,
        deleted_at = NULL
    `);

    let presentCount = 0;
    let holidayCount = 0;

    db.transaction(() => {
      for (const c of toInsert) {
        const shift = shiftMap.get(c.id);
        let matchedPattern: any = null;
        if (c.weekly_schedule_pattern) {
          try {
            const p = typeof c.weekly_schedule_pattern === 'string'
              ? JSON.parse(c.weekly_schedule_pattern)
              : c.weekly_schedule_pattern;
            const dObj = new Date(targetDate);
            matchedPattern = p[dObj.getDay().toString()];
          } catch (e) {}
        }

        const holidayInfo = getClientDayHolidayInfo(targetDate, c, shift);
        const isHoliday = holidayInfo.isHoliday;

        if (isHoliday) {
          // 公休として自動判定・登録
          insertStmt.run(
            uuidv4(),
            c.id,
            targetDate,
            'holiday',
            null, null, 0, null,
            0, 0, 0, 0,
            0, 0, 0,
            0, null, null, null, null, null
          );
          holidayCount++;
        } else {
          // 出勤として登録（シフトまたは曜日別パターンの予定時間・勤務形態・就労先を自動反映）
          const usesMeal = c.uses_meal_provision ? 1 : 0;
          const usesTrans = c.uses_transportation ? 1 : 0;
          
          let clockIn = shift?.scheduled_in;
          let clockOut = shift?.scheduled_out;
          let workType = shift?.work_type || (shift?.is_offsite === 1 ? 'out_facility' : undefined);
          let offsiteWorkplaceId = shift?.offsite_workplace_id || null;

          if (!clockIn) {
            clockIn = matchedPattern?.start_time || c.work_start_time || '10:00';
          }
          if (!clockOut) {
            clockOut = matchedPattern?.end_time || c.work_end_time || (matchedPattern?.work_type === 'out_facility' ? '16:00' : '15:00');
          }
          if (!workType) {
            workType = matchedPattern?.work_type || 'in_facility';
          }
          if (!offsiteWorkplaceId && matchedPattern?.offsite_workplace_id) {
            offsiteWorkplaceId = matchedPattern.offsite_workplace_id;
          }
          const safeOffsiteWorkplaceId = sanitizeForeignKeyId('offsite_workplaces', offsiteWorkplaceId);

          const breakTime = c.break_time_minutes ?? 60;
          const isOffsite = (workType === 'out_facility' || workType === 'both' || shift?.is_offsite === 1) ? 1 : 0;
          const offsiteStart = shift?.offsite_start_time || (workType === 'out_facility' ? clockIn : null);
          const offsiteEnd = shift?.offsite_end_time || (workType === 'out_facility' ? clockOut : null);
          const inFacilityStart = shift?.in_facility_start_time || (workType === 'in_facility' ? clockIn : null);
          const inFacilityEnd = shift?.in_facility_end_time || (workType === 'in_facility' ? clockOut : null);

          insertStmt.run(
            uuidv4(),
            c.id,
            targetDate,
            'present',
            clockIn,
            clockOut,
            breakTime,
            workType,
            usesMeal,
            usesTrans,
            usesTrans,
            usesTrans,
            usesTrans,
            usesTrans,
            usesTrans,
            isOffsite,
            offsiteStart,
            offsiteEnd,
            inFacilityStart,
            inFacilityEnd,
            safeOffsiteWorkplaceId
          );
          presentCount++;
        }
      }
    })();

    try {
      revalidatePath('/attendance', 'layout');
      revalidatePath('/', 'layout');
      revalidatePath('/billing/payroll');
    } catch (e) {}
    return { 
      success: true, 
      count: toInsert.length, 
      message: `${toInsert.length}名の一括勤怠反映が完了しました（出勤: ${presentCount}名、公休自動判定: ${holidayCount}名）` 
    };
  } catch (error: any) {
    console.error('一括出席登録エラー:', error);
    return { success: false, error: `一括登録に失敗しました: ${error.message}` };
  }
}

export async function toggleQuickAttendanceField(
  clientId: string,
  targetDate: string,
  actionType: 'clock_in' | 'clock_out' | 'toggle_pickup' | 'toggle_dropoff'
) {
  try {
    const role = await getRole();
    if (!role) return { success: false, error: '認証エラー' };
    await assertCanAccessClient(clientId);

    const now = new Date();
    const timeStr = `${String(now.getHours()).padStart(2, '0')}:${String(now.getMinutes()).padStart(2, '0')}`;

    const existing = db.prepare(`
      SELECT * FROM attendance_records
      WHERE client_id = ? AND record_date = ? AND deleted_at IS NULL
    `).get(clientId, targetDate) as any;

    if (!existing) {
      // レコードがない場合は新規作成（論理削除済みの行がある場合も考慮してUPSERT）
      const client = db.prepare(`SELECT * FROM clients WHERE id = ?`).get(clientId) as any;
      const id = uuidv4();
      const clockIn = actionType === 'clock_in' ? timeStr : (client?.work_start_time || '10:00');
      const clockOut = actionType === 'clock_out' ? timeStr : null;
      const pickup = actionType === 'toggle_pickup' ? 1 : 0;
      const dropoff = actionType === 'toggle_dropoff' ? 1 : 0;
      const transUsed = pickup || dropoff ? 1 : 0;

      db.prepare(`
        INSERT INTO attendance_records (
          id, client_id, record_date, status, clock_in, clock_out,
          break_time_minutes, work_type, 
          pickup_service, dropoff_service, transportation_service_used,
          pickup_used, dropoff_used, transportation_used,
          created_at, updated_at
        ) VALUES (?, ?, ?, 'present', ?, ?, 60, 'in_facility', ?, ?, ?, ?, ?, ?, CURRENT_TIMESTAMP, CURRENT_TIMESTAMP)
        ON CONFLICT(client_id, record_date) DO UPDATE SET
          status = 'present',
          clock_in = excluded.clock_in,
          clock_out = excluded.clock_out,
          pickup_service = excluded.pickup_service,
          dropoff_service = excluded.dropoff_service,
          transportation_service_used = excluded.transportation_service_used,
          pickup_used = excluded.pickup_used,
          dropoff_used = excluded.dropoff_used,
          transportation_used = excluded.transportation_used,
          updated_at = CURRENT_TIMESTAMP,
          deleted_at = NULL
      `).run(id, clientId, targetDate, clockIn, clockOut, pickup, dropoff, transUsed, pickup, dropoff, transUsed);
    } else {
      // 既存レコードの更新
      if (actionType === 'clock_in') {
        const newClockIn = existing.clock_in ? null : timeStr;
        db.prepare(`
          UPDATE attendance_records
          SET clock_in = ?, status = 'present', updated_at = CURRENT_TIMESTAMP
          WHERE id = ?
        `).run(newClockIn, existing.id);
      } else if (actionType === 'clock_out') {
        const newClockOut = existing.clock_out ? null : timeStr;
        db.prepare(`
          UPDATE attendance_records
          SET clock_out = ?, status = 'present', updated_at = CURRENT_TIMESTAMP
          WHERE id = ?
        `).run(newClockOut, existing.id);
      } else if (actionType === 'toggle_pickup') {
        const currentPickup = (existing.pickup_service === 1 || existing.pickup_used === 1) ? 1 : 0;
        const currentDropoff = (existing.dropoff_service === 1 || existing.dropoff_used === 1) ? 1 : 0;
        const newPickup = currentPickup === 1 ? 0 : 1;
        const newUsed = newPickup === 1 || currentDropoff === 1 ? 1 : 0;
        db.prepare(`
          UPDATE attendance_records
          SET pickup_service = ?, pickup_used = ?,
              transportation_service_used = ?, transportation_used = ?,
              updated_at = CURRENT_TIMESTAMP
          WHERE id = ?
        `).run(newPickup, newPickup, newUsed, newUsed, existing.id);
      } else if (actionType === 'toggle_dropoff') {
        const currentPickup = (existing.pickup_service === 1 || existing.pickup_used === 1) ? 1 : 0;
        const currentDropoff = (existing.dropoff_service === 1 || existing.dropoff_used === 1) ? 1 : 0;
        const newDropoff = currentDropoff === 1 ? 0 : 1;
        const newUsed = currentPickup === 1 || newDropoff === 1 ? 1 : 0;
        db.prepare(`
          UPDATE attendance_records
          SET dropoff_service = ?, dropoff_used = ?,
              transportation_service_used = ?, transportation_used = ?,
              updated_at = CURRENT_TIMESTAMP
          WHERE id = ?
        `).run(newDropoff, newDropoff, newUsed, newUsed, existing.id);
      }
    }

    revalidatePath('/attendance', 'layout');
    revalidatePath('/', 'layout');
    return { success: true };
  } catch (err: any) {
    console.error('クイック勤怠更新エラー:', err);
    return { success: false, error: err.message };
  }
}

  const filteredRecords = records.filter(r => r.record_date === selectedDateStr)
  const staffingStatus = calculateDailyStaffingStatus(selectedDateStr)
  const currentMonthYm = `${currentYear}-${String(currentMonth + 1).padStart(2, '0')}`

  const quickClients = (clients || []).map(c => {

              {role !== 'staff' && (

            </div>

            {!isGeneralStaff && (
              <Link
                href={`/attendance?date=${selectedDateStr}&departmentId=all`}
                className={buttonVariants({
                  variant: activeDepartmentId === 'all' ? 'default' : 'outline',
                  size: 'sm',
                  className: `h-8 px-3.5 text-xs font-bold shrink-0 ${activeDepartmentId === 'all' ? 'shadow-xs' : ''}`

              >
                すべて表示
              </Link>
            )}
            {activeDepartments.map(dep => (
              <Link
                key={dep.id}
                href={`/attendance?date=${selectedDateStr}&departmentId=${dep.id}`}
                className={buttonVariants({
                  variant: activeDepartmentId === dep.id ? 'default' : 'outline',
                  size: 'sm',

            </div>

            {staffingStatus.unassignedOffsiteClients && staffingStatus.unassignedOffsiteClients.length > 0 && (
              <div className="mt-3 p-3 bg-amber-50/90 rounded-md border border-amber-300 shadow-sm text-xs">
                <div className="font-bold text-amber-950 flex items-center gap-1.5 mb-2">
                  <AlertCircle className="w-4 h-4 text-amber-600 shrink-0" />
                  <span>【要対応】引率職員が未配置の施設外就労利用者 ({staffingStatus.unassignedOffsiteClients.length}名)：</span>
                </div>
                <div className="flex flex-wrap gap-2">
                  {staffingStatus.unassignedOffsiteClients.map(u => (
                    <a
                      key={u.clientId}
                      href={u.attendanceId ? `/attendance?date=${selectedDateStr}&editId=${u.attendanceId}` : `/attendance?date=${selectedDateStr}`}
                      className="inline-flex items-center gap-1.5 px-2.5 py-1 rounded bg-white hover:bg-amber-100 border border-amber-300 text-amber-950 font-medium transition-colors shadow-xs group"
                    >
                      <UserX className="w-3.5 h-3.5 text-rose-500" />
                      <strong className="text-slate-900 group-hover:text-blue-700">{u.clientName}</strong>
                      <span className="text-[10px] text-amber-800 bg-amber-100 px-1 py-0.5 rounded">

            {staffingStatus.offsiteAssignments && staffingStatus.offsiteAssignments.length > 0 && (
              <div className="mt-3 pt-3 border-t border-amber-300/60 text-xs">
                <div className="font-bold text-slate-800 flex items-center gap-1.5 mb-2">
                  <Building2 className="w-4 h-4 text-slate-600" />
                  <span>【当日の施設外就労・引率体制一覧】</span>

                <div className="grid grid-cols-1 sm:grid-cols-2 md:grid-cols-3 gap-2.5">

                    <div key={idx} className={`p-2.5 rounded-md border shadow-xs ${oa.isCompliant ? 'bg-white/90 border-emerald-200' : 'bg-rose-50/90 border-rose-300'}`}>

                        <span className="text-muted-foreground">引率担当:</span>

                          <span className="font-bold text-emerald-800 flex items-center gap-1">
                            <UserCheck className="w-3.5 h-3.5 text-emerald-600" />
                            {oa.staffName}
                          </span>
                        ) : (

                            <UserX className="w-3.5 h-3.5 text-rose-500" />

                          </span>
                        )}
                      </div>
                      {oa.clients && oa.clients.length > 0 && (
                        <div className="mt-1.5 text-[11px] text-slate-600 border-t border-slate-100 pt-1">
                          <span className="text-[10px] text-muted-foreground block">参加利用者:</span>
                          <div className="flex flex-wrap gap-1 mt-0.5">
                            {oa.clients.map(c => (
                              <span key={c.clientId} className="bg-slate-100 text-slate-700 px-1 py-0.2 rounded text-[10px]">
                                {c.clientName}
                              </span>
                            ))}
                          </div>
                        </div>
                      )}
                    </div>
                  ))}
                </div>
              </div>
            )}
          </div>

          {/* 📱 現場スマホ・送迎車用 クイック出退勤＆送迎チェッカー（設定でチェックが入っている場合のみ表示） */}
          {showQuickChecker && (
            <Card className="border-2 border-primary/30 bg-gradient-to-br from-white to-blue-50/30 shadow-sm overflow-hidden">
              <CardHeader className="bg-primary/10 border-b py-3 px-4 flex flex-row items-center justify-between">
                <div>
                  <CardTitle className="text-base sm:text-lg font-bold text-primary flex items-center gap-2">
                    <span>📱 {hasTransportation ? "現場ワンタップ出退勤・送迎チェッカー" : "現場ワンタップ出退勤チェッカー"}</span>
                    <Badge variant="outline" className="bg-primary/10 text-primary border-primary/30 text-[10px] font-bold">
                      現場スマホ推奨

                  </CardTitle>
                  <p className="text-xs text-muted-foreground mt-0.5">
                    {hasTransportation 
                      ? "送迎車の中やフロア巡回中に、親指1タップで「迎」「出勤」「退勤」「送」を即時記録できます。"
                      : "フロア巡回中や現場で、親指1タップで「出勤」「退勤」を即時記録できます。"}
                  </p>
                </div>
              </CardHeader>
              <CardContent className="p-3 sm:p-4">
                <MobileQuickCheckIn 
                  clients={quickClients} 
                  todayStr={selectedDateStr} 
                  hasTransportation={hasTransportation}
                />
              </CardContent>
            </Card>
          )}

          <Card className="shadow-sm">
            <CardHeader className="bg-primary/5 border-b pb-4 flex flex-col sm:flex-row sm:items-center justify-between gap-3">
              <div>
                <CardTitle className="text-xl">実績の詳細入力・編集</CardTitle>
                <p className="text-xs text-muted-foreground mt-0.5">対象日: {selectedDateStr} の出退勤・勤務形態・加算・引率</p>
              </div>
              <div className="flex items-center gap-1.5">
                <Link href={`/attendance?date=${prevDayStr}&departmentId=${activeDepartmentId}`} className={buttonVariants({ variant: "outline", size: "sm", className: "h-8 px-2.5 text-xs" })}>
                  <ChevronLeft className="w-3.5 h-3.5 mr-1" />
                  前日 ({prevDayStr.slice(5)})
                </Link>
                {selectedDateStr !== todayStr && (
                  <Link href={`/attendance?date=${todayStr}&departmentId=${activeDepartmentId}`} className={buttonVariants({ variant: "secondary", size: "sm", className: "h-8 px-2.5 text-xs font-bold" })}>
                    今日に戻る
                  </Link>
                )}
                <Link href={`/attendance?date=${nextDayStr}&departmentId=${activeDepartmentId}`} className={buttonVariants({ variant: "outline", size: "sm", className: "h-8 px-2.5 text-xs" })}>
                  翌日 ({nextDayStr.slice(5)})
                  <ChevronRight className="w-3.5 h-3.5 ml-1" />
                </Link>
              </div>
            </CardHeader>

            <CardContent className="pt-6">
              <AttendanceForm 
                clients={clients || []} 
                todayStr={selectedDateStr} 
                initialData={initialData} 
                records={records}
                offsiteWorkplaces={offsiteWorkplaces}
                staffList={staffList}
                clientShifts={shifts}
                hasMeal={hasMeal}
                hasTransportation={hasTransportation}
                hasVisitSupport={hasVisitSupport}
                hasAbsenceSupport={hasAbsenceSupport}
              />

          </Card>

          <Card className="shadow-sm">
            <CardHeader className="bg-primary/5 border-b pb-4 space-y-3">
              <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
                <div>
                  <CardTitle className="text-xl flex items-center gap-2">
                    <CalendarIcon className="w-5 h-5 text-primary" />
                    <span>{currentYear}年{currentMonth + 1}月 の勤怠カレンダー</span>
                  </CardTitle>
                  <p className="text-xs text-muted-foreground mt-0.5">
                    日付をクリックすると当日の実績入力フォームおよび一覧へジャンプします。
                  </p>
                </div>
                <div className="flex items-center gap-3 text-xs shrink-0">
                  <span className="flex items-center gap-1 font-medium text-emerald-800">
                    <span className="w-2.5 h-2.5 rounded-full bg-emerald-500 inline-block"></span> 人員充足
                  </span>
                  <span className="flex items-center gap-1 font-medium text-rose-800">
                    <span className="w-2.5 h-2.5 rounded-full bg-rose-500 inline-block"></span> 人員不足(警告)
                  </span>
                </div>
              </div>

              {/* 年月クイックピッカー・過去月ショートカット */}
              <CalendarMonthPicker currentDateStr={selectedDateStr} todayStr={todayStr} />

              {/* 月間実績サマリーバナー */}
              <div className="grid grid-cols-2 sm:grid-cols-4 lg:grid-cols-7 gap-2 p-2.5 bg-white dark:bg-slate-900 border rounded-lg text-center text-xs shadow-2xs">
                <div className="border-r last:border-r-0 border-slate-100">
                  <span className="text-[10px] text-muted-foreground block">延べ出勤</span>
                  <span className="text-sm font-bold text-emerald-700">{monthlyStats.totalPresent} 名</span>
                </div>
                <div className="border-r last:border-r-0 border-slate-100">
                  <span className="text-[10px] text-muted-foreground block">延べ有休取得</span>
                  <span className="text-sm font-bold text-blue-700">{monthlyStats.totalPto} 日</span>

                  const isSelected = cell.dateStr === selectedDateStr
                  const isToday = cell.dateStr === todayStr
                  const dailyRecords = records.filter(r => r.record_date === cell.dateStr)
                  const presentCount = dailyRecords.filter(r => r.status === 'present').length
                  
                  // Calculate how many clients are scheduled to work today
                  let shiftCount = 0;
                  clients.forEach(c => {
                    const shift = shifts.find(s => s.client_id === c.id && s.target_date === cell.dateStr);
                    const holidayInfo = getClientDayHolidayInfo(cell.dateStr, c, shift);
                    if (!holidayInfo.isHoliday) {
                      shiftCount++;
                    }
                  });

                      key={cell.dateStr}
                      href={`/attendance?date=${cell.dateStr}`}

                        isSelected 
                          ? 'ring-2 ring-primary border-transparent bg-primary/5'

                            実績:{presentCount}名
                          </span>
                        )}
                        {hasClients && dayStaffing && (
                          dayStaffing.isCompliant ? (
                            <span className="hidden sm:inline-flex text-[8px] bg-emerald-100 text-emerald-800 border border-emerald-200 px-0.5 rounded-sm w-full justify-center">
                              ✓充足

                              ⚠️不足(-{Math.floor(dayStaffing.shortageMinutes / 60)}h)

            </CardContent>

    if (filterType === 'hasExpired') return item.totalExpired > 0;
    if (filterType === 'needsFiveDays') return item.mandatoryFiveDaysTarget?.isTarget && item.mandatoryFiveDaysTarget.remainingToFiveDays > 0;

      } finally {
        setSubmitting(false);

        setSubmitting(false);
      }

  const handleExportCsv = () => {

      const url = URL.createObjectURL(blob);

      a.href = url;
      a.download = `年次有給休暇管理簿_職員_${toLocalISODate(new Date())}.csv`;

            <span className="font-bold flex items-center gap-2 text-amber-950 text-sm">
              <AlertTriangle className="w-5 h-5 text-amber-600 shrink-0" />

          </p>
          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-2.5 pt-1">
            {mandatoryAlerts.map(item => (

                "p-2.5 rounded-lg border bg-white flex flex-col justify-between space-y-1.5 shadow-2xs",

                    残り{item.daysUntilDeadline}日

                    不足 {item.shortageDays}日

            className="font-bold gap-1.5 shadow-2xs"

            <Users className="w-4 h-4" />

            <CardDescription className="text-xs font-semibold text-blue-800">
              {isClientView ? '利用者' : '職員'} 累計付与日数

            size="sm"

          >
            <FileSpreadsheet className="w-4 h-4" />
            {isClientView ? '全社 利用者有休消化台帳（法定様式）' : '全社 職員有休消化台帳（法定様式）'}

          <Button
            variant={activeTab === 'detail' ? 'default' : 'outline'}
            size="sm"
            onClick={() => setActiveTab('detail')}
            className="gap-1.5"
          >
            {isClientView ? <Users className="w-4 h-4" /> : <Building2 className="w-4 h-4" />}
            {isClientView ? '利用者別 詳細台帳・付与履歴' : '職員別 詳細台帳・付与履歴'}

            <Printer className="w-3.5 h-3.5" />
            管理簿を印刷

      <div className="hidden print:block text-center mb-6">

          労働基準法第39条および同法施行規則第24条の7に基づく法定管理台帳 ｜ 出力日: {new Date().toLocaleDateString('ja-JP')}

/**
 * 労働基準法第115条に基づく2年時効消滅処理の自動実行
 * 有効期限（付与日＋2年）を過ぎた残日数を時効消滅（expired_days）として確定
 */
export async function processPtoExpirations(): Promise<{ expiredCount: number; totalExpiredDays: number }> {
  const todayStr = toLocalISODate(new Date());
  
  const expiredRecords = db.prepare(`
    SELECT id, client_id, remaining_days 
    FROM client_pto_records 
    WHERE expiration_date < ? AND remaining_days > 0
  `).all(todayStr) as { id: string; client_id: string; remaining_days: number }[];

  if (expiredRecords.length === 0) {
    return { expiredCount: 0, totalExpiredDays: 0 };
  }

  let totalExpiredDays = 0;
  const updateStmt = db.prepare(`
    UPDATE client_pto_records 
    SET expired_days = COALESCE(expired_days, 0) + remaining_days,
        remaining_days = 0,
        updated_at = CURRENT_TIMESTAMP
    WHERE id = ?
  `);

  const clientIdsToUpdate = new Set<string>();

  db.transaction(() => {
    for (const rec of expiredRecords) {
      updateStmt.run(rec.id);
      totalExpiredDays += rec.remaining_days;
      clientIdsToUpdate.add(rec.client_id);
    }
  })();

  // clients テーブルの pto_granted キャッシュを再同期
  for (const cId of clientIdsToUpdate) {
    db.prepare(`
      UPDATE clients 
      SET pto_granted = COALESCE((
        SELECT SUM(remaining_days) FROM client_pto_records 
        WHERE client_id = ? AND expiration_date >= date('now')
      ), 0)
      WHERE id = ?
    `).run(cId, cId);
  }

  return { expiredCount: expiredRecords.length, totalExpiredDays };
}

/**
 * 直近算定期間（直近6ヶ月または1年間）の出勤率（労基法第39条: 全労働日の8割以上出勤要件）を計算
 */
export async function calculateAttendanceRate(clientId: string, startDateStr?: string, endDateStr?: string): Promise<{ rate: number; attendedDays: number; scheduledDays: number; isEligible: boolean }> {
  const endDate = endDateStr || toLocalISODate(new Date());
  let startDate = startDateStr;
  
  if (!startDate) {
    const end = new Date(endDate);
    end.setMonth(end.getMonth() - 6);
    startDate = toLocalISODate(end);
  }

  // 期間内の実績集計
  // 労基法第39条第10項: 有給休暇取得日や業務上傷病の休業日は出勤とみなす
  const stats = db.prepare(`
    SELECT 
      SUM(CASE WHEN status IN ('present', 'out_facility', 'paid_leave', 'pto', 'special_leave') OR is_offsite = 1 THEN 1 ELSE 0 END) as attended,
      SUM(CASE WHEN status = 'absent' THEN 1 ELSE 0 END) as absent,
      COUNT(*) as total_logged
    FROM attendance_records
    WHERE client_id = ? 
      AND record_date >= ? 
      AND record_date <= ?
      AND deleted_at IS NULL
  `).get(clientId, startDate, endDate) as any;

  const attended = stats?.attended || 0;
  const absent = stats?.absent || 0;
  const scheduledDays = attended + absent;

  if (scheduledDays === 0) {
    // 期間内の勤怠記録がない場合は暫定的に100%（または新規入所判定）
    return { rate: 1.0, attendedDays: 0, scheduledDays: 0, isEligible: true };
  }

  const rate = Math.round((attended / scheduledDays) * 100) / 100;
  return {
    rate,
    attendedDays: attended,
    scheduledDays,
    isEligible: rate >= 0.8
  };
}

/**
 * 勤続年数および週所定労働時間・週所定労働日数に応じた法定年次有給休暇付与日数の計算（労基法第39条）
 */
export async function calculateStatutoryGrant(
  yearsOfServiceMonths: number,
  weeklyHours: number = 20,
  attendanceRate: number = 1.0,
  weeklyDays?: number
): Promise<number> {
  // 出勤率が8割未満の場合は付与日数0（労基法第39条）
  if (attendanceRate < 0.8) {
    return 0;
  }

  const yearsOfService = yearsOfServiceMonths / 12;
  // 週所定日数が未指定の場合は週労働時間から推定（通常就労継続A型は週20h=週5日×4h、または週4日等）
  const days = weeklyDays ?? (weeklyHours >= 30 ? 5 : Math.min(5, Math.max(1, Math.round(weeklyHours / 4))));
  return calculateStatutoryGrantDays(weeklyHours, days, yearsOfService);
}

    } catch {
      return [];
    }
  }

  // 取得前に法定付与月の自動計算付与および2年時効消滅の確定を実行
  await processAutomaticPtoGrants();

  const clientQuery = clientId 
    ? db.prepare('SELECT id, full_name, recipient_number, service_start_date, contract_date, weekly_scheduled_hours, regular_holidays FROM clients WHERE id = ? AND deleted_at IS NULL').all(clientId)
    : db.prepare('SELECT id, full_name, recipient_number, service_start_date, contract_date, weekly_scheduled_hours, regular_holidays FROM clients WHERE deleted_at IS NULL ORDER BY full_name ASC').all();

  const allClients = clientQuery as any[];
  const today = new Date();
  const oneYearAgo = new Date();
  oneYearAgo.setFullYear(today.getFullYear() - 1);
  const oneYearAgoStr = toLocalISODate(oneYearAgo);

  const threeMonthsLater = new Date();
  threeMonthsLater.setMonth(today.getMonth() + 3);
  const threeMonthsLaterStr = toLocalISODate(threeMonthsLater);
  const todayStr = toLocalISODate(today);

  const summaries: ClientPtoSummary[] = [];

  for (const c of allClients) {
    const records = db.prepare(`
      SELECT * FROM client_pto_records 
      WHERE client_id = ? 
      ORDER BY grant_date DESC
    `).all(c.id) as PtoRecord[];

    // 実際の有休取得実績日一覧（勤怠データから直接取得）
    const takenRows = db.prepare(`
      SELECT record_date FROM attendance_records
      WHERE client_id = ? AND status IN ('paid_leave', 'pto') AND deleted_at IS NULL
      ORDER BY record_date DESC
    `).all(c.id) as { record_date: string }[];
    const takenDates = takenRows.map(r => r.record_date);

    let totalGranted = 0;
    let totalUsed = 0;
    let totalRemaining = 0;
    let totalExpired = 0;
    let carriedOverDays = 0;
    let currentGrantDays = 0;
    let expiringSoonDays = 0;

    for (const r of records) {
      totalGranted += r.granted_days;
      totalUsed += r.used_days;
      totalExpired += (r.expired_days || 0);

      // 有効期限内の残日数のみ加算
      if (r.expiration_date >= todayStr) {
        totalRemaining += r.remaining_days;
        if (r.expiration_date <= threeMonthsLaterStr && r.remaining_days > 0) {
          expiringSoonDays += r.remaining_days;
        }

        // 付与から1年以上経過（前年度付与分）で有効なものは「繰越」
        if (r.grant_date < oneYearAgoStr) {
          carriedOverDays += r.remaining_days;
        } else {
          currentGrantDays += r.granted_days;
        }
      }
    }

    // 法定付与サジェストの算出
    const startDateStr = c.service_start_date || c.contract_date;
    let statutorySuggestion: ClientPtoSummary['statutorySuggestion'];

    if (startDateStr) {
      const startDate = new Date(startDateStr);
      const diffTime = today.getTime() - startDate.getTime();
      const diffMonths = Math.floor(diffTime / (1000 * 60 * 60 * 24 * 30.4375));
      const yearsOfService = (diffMonths / 12).toFixed(1);

      // 次回付与日（6か月後、1年6か月後、2年6か月後...）
      let milestoneMonths = 6;
      while (milestoneMonths <= diffMonths) {
        milestoneMonths += 12;
      }
      const nextGrantDate = new Date(startDate);
      nextGrantDate.setMonth(startDate.getMonth() + milestoneMonths);
      const nextGrantDateStr = toLocalISODate(nextGrantDate);
      let weeklyDays = 5;
      if (c.regular_holidays) {
        try {
          const holidays = JSON.parse(c.regular_holidays);
          if (Array.isArray(holidays)) {
            weeklyDays = Math.max(1, 7 - holidays.length);
          }
        } catch (e) {}
      }

      const attStats = await calculateAttendanceRate(c.id);
      const suggestedDays = await calculateStatutoryGrant(diffMonths, c.weekly_scheduled_hours || 20, attStats.rate, weeklyDays);
      const isEligibleNow = diffMonths >= 6 && records.length === 0 && attStats.isEligible;

      statutorySuggestion = {
        nextGrantDate: nextGrantDateStr,
        suggestedDays,
        yearsOfService: `${yearsOfService}年 (${diffMonths}ヶ月) [出勤率: ${(attStats.rate * 100).toFixed(1)}%]`,
        isEligibleNow
      };
    }

    summaries.push({
      clientId: c.id,
      clientName: c.full_name,
      recipientNumber: c.recipient_number,
      serviceStartDate: startDateStr,
      weeklyHours: c.weekly_scheduled_hours || 20,
      totalGranted,
      totalUsed,
      totalRemaining,
      totalExpired,
      carriedOverDays,
      currentGrantDays,
      expiringSoonDays,
      records,
      takenDates,
      statutorySuggestion
    });
  }

  return summaries;
}


/**
 * 有給休暇の付与を登録（手動または自動付与）
 */
export async function addPtoGrant(
  clientId: string,
  grantDate: string,
  grantedDays: number,
  note: string = '法定年次付与'
): Promise<{ success: boolean; message?: string; error?: string }> {
  try {
    const role = await getRole();
    if (!role) return { success: false, error: '認証されていません。ログインしてください。' };
    resolveServiceModeForClient(clientId);
    await assertCanAccessClient(clientId);

    if (grantedDays <= 0) {
      return { success: false, error: '付与日数は0日より大きい値を指定してください。' };
    }

    // 有効期限は付与日から2年間（労基法第115条）
    const gDate = new Date(grantDate);
    const expDate = new Date(gDate);
    expDate.setFullYear(gDate.getFullYear() + 2);
    const expirationDate = toLocalISODate(expDate);

    const id = randomUUID();

    db.prepare(`
      INSERT INTO client_pto_records (
        id, client_id, grant_date, granted_days, used_days, remaining_days, expiration_date, note
      ) VALUES (?, ?, ?, ?, 0, ?, ?, ?)
    `).run(id, clientId, grantDate, grantedDays, grantedDays, expirationDate, note);

    // clients テーブルの pto_granted も同期
    db.prepare(`
      UPDATE clients 
      SET pto_granted = COALESCE((
        SELECT SUM(remaining_days) FROM client_pto_records 
        WHERE client_id = ? AND expiration_date >= date('now')
      ), 0)
      WHERE id = ?
    `).run(clientId, clientId);

    safeRevalidate('/attendance/pto');
    safeRevalidate('/billing/payroll');
    return { success: true, message: `${grantedDays}日の有給休暇を付与しました（失効日: ${expirationDate}）。` };
  } catch (err: any) {
    console.error('有給付与エラー:', err);
    return { success: false, error: err.message || '有給休暇の付与に失敗しました' };
  }
}

/**
 * 有給休暇の消化を記録（出欠登録や給与確定と連携）
 * 【予防的改善】消費計算の前に必ず processPtoExpirations を実行し最新の有効残日数を確定
 */
export async function consumePtoDays(clientId: string, daysToConsume: number) {
  if (daysToConsume <= 0) return;

  // 最新の時効消滅状態を確定
  await processPtoExpirations();

  // 有効期限が近い順に消化（FIFO）
  const validRecords = db.prepare(`
    SELECT * FROM client_pto_records
    WHERE client_id = ? AND remaining_days > 0 AND expiration_date >= date('now')
    ORDER BY expiration_date ASC, grant_date ASC
  `).all(clientId) as PtoRecord[];

  let needed = daysToConsume;
  const updateStmt = db.prepare(`
    UPDATE client_pto_records 
    SET used_days = used_days + ?, remaining_days = remaining_days - ?, updated_at = CURRENT_TIMESTAMP
    WHERE id = ?
  `);

  db.transaction(() => {
    for (const r of validRecords) {
      if (needed <= 0) break;
      const consumeFromRecord = Math.min(needed, r.remaining_days);
      updateStmt.run(consumeFromRecord, consumeFromRecord, r.id);
      needed -= consumeFromRecord;
    }
    // clients テーブルの pto_granted を同一トランザクション内で更新
    db.prepare(`
      UPDATE clients 
      SET pto_granted = COALESCE((
        SELECT SUM(remaining_days) FROM client_pto_records 
        WHERE client_id = ? AND expiration_date >= date('now')
      ), 0)
      WHERE id = ?
    `).run(clientId, clientId);
  })();

  safeRevalidate('/attendance/pto');
  safeRevalidate('/billing/payroll');
}

/**
 * 有給休暇の消化返還（勤怠の有休取消や勤怠レコード削除時に残日数を復元）

export async function restorePtoDays(clientId: string, daysToRestore: number) {

  let toRestore = daysToRestore;
  const updateStmt = db.prepare(`
    UPDATE client_pto_records 
    SET used_days = MAX(0, used_days - ?), 
        remaining_days = remaining_days + ?, 
        updated_at = CURRENT_TIMESTAMP
    WHERE id = ?
  `);

  db.transaction(() => {
    for (const r of usedRecords) {
      if (toRestore <= 0) break;
      const restoreAmount = Math.min(toRestore, r.used_days);
      updateStmt.run(restoreAmount, restoreAmount, r.id);
      toRestore -= restoreAmount;
    }
    // clients テーブルの pto_granted を同一トランザクション内で更新
    db.prepare(`
      UPDATE clients 
      SET pto_granted = COALESCE((
        SELECT SUM(remaining_days) FROM client_pto_records 
        WHERE client_id = ? AND expiration_date >= date('now')
      ), 0)
      WHERE id = ?
    `).run(clientId, clientId);
  })();

  safeRevalidate('/attendance/pto');
  safeRevalidate('/billing/payroll');
}

/**
 * 有給台帳レコードの直接手動修正
 */
export async function updatePtoRecord(recordId: string, grantedDays: number, usedDays: number, note: string, expiredDays?: number): Promise<{ success: boolean; error?: string }> {
  try {
    const role = await getRole();
    if (!role) return { success: false, error: '認証されていません。ログインしてください。' };
    resolveServiceModeForTableRecord('client_pto_records', recordId);
    const existingRec = db.prepare('SELECT client_id FROM client_pto_records WHERE id = ?').get(recordId) as any;
    if (existingRec?.client_id) {
      await assertCanAccessClient(existingRec.client_id);
    }

    if (grantedDays < usedDays) {
      return { success: false, error: '消化日数は付与日数以下にしてください。' };
    }
    const expDays = expiredDays !== undefined ? expiredDays : 0;
    const remaining = Math.max(0, grantedDays - usedDays - expDays);

    db.prepare(`
      UPDATE client_pto_records
      SET granted_days = ?, used_days = ?, remaining_days = ?, expired_days = ?, note = ?, updated_at = CURRENT_TIMESTAMP
      WHERE id = ?
    `).run(grantedDays, usedDays, remaining, expDays, note, recordId);

    const rec = db.prepare('SELECT client_id FROM client_pto_records WHERE id = ?').get(recordId) as any;
    if (rec?.client_id) {
      db.prepare(`
        UPDATE clients 
        SET pto_granted = COALESCE((
          SELECT SUM(remaining_days) FROM client_pto_records 
          WHERE client_id = ? AND expiration_date >= date('now')
        ), 0)
        WHERE id = ?
      `).run(rec.client_id, rec.client_id);
    }

    safeRevalidate('/attendance/pto');
    return { success: true };
  } catch (err: any) {
    console.error('有給レコード更新エラー:', err);
    return { success: false, error: err.message || '有給レコードの更新に失敗しました' };
  }
}

/**
 * 職員および利用者の有給付与月・基準日到達判定と法定付与日数の自動計算・付与
 * 労働基準法第39条に基づき、入社/利用開始から6ヶ月後、以降1年ごとの付与月に自動付与
 */
export async function processAutomaticPtoGrants(): Promise<{
  clientGrantsCount: number;
  staffGrantsCount: number;
  details: string[];
}> {
  const today = new Date();
  const todayStr = toLocalISODate(today);
  const details: string[] = [];
  let clientGrantsCount = 0;
  let staffGrantsCount = 0;

  try {
    // 1. 失効処理を最新化
    await processPtoExpirations();
    await processStaffPtoExpirations();

    // 2. 利用者（クライアント）の自動付与判定
    const clients = db.prepare(`
      SELECT id, full_name, service_start_date, contract_date, created_at,
             weekly_scheduled_hours, regular_holidays
      FROM clients
      WHERE deleted_at IS NULL AND (status IS NULL OR status = 'active')
    `).all() as any[];

    for (const c of clients) {
      const startDateStr = c.service_start_date || c.contract_date || (c.created_at ? c.created_at.slice(0, 10) : null);
      if (!startDateStr) continue;

      const startDate = new Date(startDateStr);
      const originalDay = startDate.getDate();
      let milestoneMonths = 6;
      let prevGrantDate = new Date(startDate);
      while (true) {
        const grantTargetDate = new Date(startDate);
        grantTargetDate.setMonth(startDate.getMonth() + milestoneMonths);
        // 民法143条月末日補正（応当日がない場合は末日）
        if (grantTargetDate.getDate() !== originalDay) {
          grantTargetDate.setDate(0);
        }
        const grantTargetDateStr = toLocalISODate(grantTargetDate);

        // まだ到来していない未来の基準日はスキップして終了
        if (grantTargetDateStr > todayStr) break;

        // この基準日（grant_date）に対応する付与レコードが既に存在するか確認
        const existing = db.prepare(`
          SELECT id FROM client_pto_records
          WHERE client_id = ? AND grant_date = ?
        `).get(c.id, grantTargetDateStr);

        if (!existing) {
          // 出勤率8割要件のチェック（直前基準日〜今回基準日前日までの法定算定期間）
          const calcStartStr = toLocalISODate(prevGrantDate);
          const calcEndStr = toLocalISODate(new Date(grantTargetDate.getTime() - 86400000));
          const attStats = await calculateAttendanceRate(c.id, calcStartStr, calcEndStr);
          let weeklyDays = 5;
          if (c.regular_holidays) {
            try {
              const h = JSON.parse(c.regular_holidays);
              if (Array.isArray(h)) weeklyDays = Math.max(1, 7 - h.length);
            } catch (e) {}
          }
          const weeklyHours = c.weekly_scheduled_hours || 20;
          const grantedDays = await calculateStatutoryGrant(milestoneMonths, weeklyHours, attStats.rate, weeklyDays);

          if (grantedDays > 0) {
            const expDate = new Date(grantTargetDate);
            expDate.setFullYear(expDate.getFullYear() + 2);
            const expDateStr = toLocalISODate(expDate);

            db.prepare(`
              INSERT INTO client_pto_records (
                id, client_id, grant_date, granted_days, used_days, remaining_days, expired_days, expiration_date, note
              ) VALUES (?, ?, ?, ?, 0, ?, 0, ?, ?)
            `).run(
              randomUUID(),
              c.id,
              grantTargetDateStr,
              grantedDays,
              grantedDays,
              expDateStr,
              `自動付与: 勤続${(milestoneMonths / 12).toFixed(1)}年 (出勤率${Math.round(attStats.rate * 100)}%)`
            );

            clientGrantsCount++;
            details.push(`利用者 ${c.full_name}: ${grantTargetDateStr}付与分 ${grantedDays}日 自動付与完了`);
          }
        }

        prevGrantDate = grantTargetDate;
        milestoneMonths += 12; // 1年ごと
      }

      // clients テーブルの pto_granted を最新化
      db.prepare(`
        UPDATE clients 
        SET pto_granted = COALESCE((
          SELECT SUM(remaining_days) FROM client_pto_records 
          WHERE client_id = ? AND expiration_date >= date('now')
        ), 0)
        WHERE id = ?
      `).run(c.id, c.id);
    }

    // 3. 職員（スタッフ）の自動付与判定
    const staffList = db.prepare(`
      SELECT id, name, hire_date, weekly_hours, regular_holidays, pto_granted
      FROM staff
      WHERE deleted_at IS NULL
    `).all() as any[];

    for (const s of staffList) {
      if (!s.hire_date) {
        // hire_date未設定かつ既存のpto_grantedがある場合の初期移行
        if (s.pto_granted > 0) {
          const count = db.prepare('SELECT COUNT(*) as cnt FROM staff_pto_records WHERE staff_id = ?').get(s.id) as any;
          if (count?.cnt === 0) {
            const expDate = new Date();
            expDate.setFullYear(expDate.getFullYear() + 1);
            db.prepare(`
              INSERT INTO staff_pto_records (
                id, staff_id, grant_date, granted_days, used_days, remaining_days, expired_days, expiration_date, note
              ) VALUES (?, ?, ?, ?, 0, ?, 0, ?, ?)
            `).run(randomUUID(), s.id, todayStr, s.pto_granted, s.pto_granted, toLocalISODate(expDate), '初期登録値移行');
          }
        }
        continue;
      }

      const hireDate = new Date(s.hire_date);
      const originalStaffDay = hireDate.getDate();
      let milestoneMonths = 6;
      while (true) {
        const grantTargetDate = new Date(hireDate);
        grantTargetDate.setMonth(hireDate.getMonth() + milestoneMonths);
        if (grantTargetDate.getDate() !== originalStaffDay) {
          grantTargetDate.setDate(0); // 月末日補正
        }
        const grantTargetDateStr = toLocalISODate(grantTargetDate);

        if (grantTargetDateStr > todayStr) break;

        const existing = db.prepare(`
          SELECT id FROM staff_pto_records
          WHERE staff_id = ? AND grant_date = ?
        `).get(s.id, grantTargetDateStr);

        if (!existing) {
          let weeklyDays = 5;
          if (s.regular_holidays) {
            try {
              const h = JSON.parse(s.regular_holidays);
              if (Array.isArray(h)) weeklyDays = Math.max(1, 7 - h.length);
            } catch (e) {}
          }
          const weeklyHours = s.weekly_hours || 40;
          const grantedDays = calculateStatutoryDays(milestoneMonths / 12, weeklyDays, weeklyHours);

          if (grantedDays > 0) {
            const expDate = new Date(grantTargetDate);
            expDate.setFullYear(expDate.getFullYear() + 2);
            const expDateStr = toLocalISODate(expDate);

            db.prepare(`
              INSERT INTO staff_pto_records (
                id, staff_id, grant_date, granted_days, used_days, remaining_days, expired_days, expiration_date, note
              ) VALUES (?, ?, ?, ?, 0, ?, 0, ?, ?)
            `).run(
              randomUUID(),
              s.id,
              grantTargetDateStr,
              grantedDays,
              grantedDays,
              expDateStr,
              `自動付与: 勤続${(milestoneMonths / 12).toFixed(1)}年`
            );

            staffGrantsCount++;
            details.push(`職員 ${s.name}: ${grantTargetDateStr}付与分 ${grantedDays}日 自動付与完了`);
          }
        }

        milestoneMonths += 12;
      }

      // staff テーブルの pto_granted を最新化
      db.prepare(`
        UPDATE staff 
        SET pto_granted = COALESCE((
          SELECT SUM(remaining_days) FROM staff_pto_records 
          WHERE staff_id = ? AND expiration_date >= date('now')
        ), 0)
        WHERE id = ?
      `).run(s.id, s.id);
    }

    safeRevalidate('/attendance/pto');
    safeRevalidate('/staff');
  } catch (err) {
    console.error('自動有給付与エラー:', err);
  }
  return { clientGrantsCount, staffGrantsCount, details };
}

/**
 * 職員有給管理台帳の取得
 */
export async function getStaffPtoLedger(staffId?: string): Promise<StaffPtoSummary[]> {
  const role = await getRole();
  if (!role) return [];
  await processAutomaticPtoGrants();

  const query = staffId
    ? db.prepare('SELECT id, name, role, hire_date, weekly_hours, regular_holidays FROM staff WHERE id = ? AND deleted_at IS NULL').all(staffId)
    : db.prepare('SELECT id, name, role, hire_date, weekly_hours, regular_holidays FROM staff WHERE deleted_at IS NULL ORDER BY name ASC').all();

  const allStaff = query as any[];
  const today = new Date();
  const todayStr = toLocalISODate(today);
  const oneYearAgo = new Date();
  oneYearAgo.setFullYear(today.getFullYear() - 1);
  const oneYearAgoStr = toLocalISODate(oneYearAgo);
  const threeMonthsLater = new Date();
  threeMonthsLater.setMonth(today.getMonth() + 3);
  const threeMonthsLaterStr = toLocalISODate(threeMonthsLater);

  const summaries: StaffPtoSummary[] = [];

  for (const s of allStaff) {
    const records = db.prepare(`
      SELECT * FROM staff_pto_records
      WHERE staff_id = ?
      ORDER BY grant_date DESC
    `).all(s.id) as PtoRecord[];

    const takenRows = db.prepare(`
      SELECT target_date FROM staff_shifts
      WHERE staff_id = ? AND is_holiday = 1 AND holiday_type IN ('有給休暇', '有給', '有休')
      ORDER BY target_date DESC
    `).all(s.id) as { target_date: string }[];
    const takenDates = takenRows.map(r => r.target_date);

    let totalGranted = 0;
    let totalUsed = 0;
    let totalRemaining = 0;
    let totalExpired = 0;
    let carriedOverDays = 0;
    let currentGrantDays = 0;
    let expiringSoonDays = 0;

    for (const r of records) {
      totalGranted += r.granted_days;
      totalUsed += r.used_days;
      totalExpired += (r.expired_days || 0);

      if (r.expiration_date >= todayStr) {
        totalRemaining += r.remaining_days;
        if (r.expiration_date <= threeMonthsLaterStr && r.remaining_days > 0) {
          expiringSoonDays += r.remaining_days;
        }

        if (r.grant_date < oneYearAgoStr) {
          carriedOverDays += r.remaining_days;
        } else {
          currentGrantDays += r.granted_days;
        }
      }
    }

    // 労基法第39条第7項 年5日取得義務の進捗
    const latestRecord = records[0];
    let mandatoryFiveDaysTarget: StaffPtoSummary['mandatoryFiveDaysTarget'];

    if (latestRecord && latestRecord.granted_days >= 10) {
      const grantD = new Date(latestRecord.grant_date);
      const periodEnd = new Date(grantD);
      periodEnd.setFullYear(periodEnd.getFullYear() + 1);
      periodEnd.setDate(periodEnd.getDate() - 1);
      const periodEndStr = toLocalISODate(periodEnd);

      // 当期期間内の取得日数
      const periodTakenCount = takenDates.filter(d => d >= latestRecord.grant_date && d <= periodEndStr).length;
      const remainingToFive = Math.max(0, 5 - periodTakenCount);
      const diffDays = Math.ceil((periodEnd.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));

      mandatoryFiveDaysTarget = {
        isTarget: true,
        takenDays: periodTakenCount,
        remainingToFiveDays: remainingToFive,
        daysUntilPeriodEnd: Math.max(0, diffDays),
        periodEnd: periodEndStr,
        isWarning: diffDays <= 90 && remainingToFive > 0,
        isCritical: diffDays <= 30 && remainingToFive > 0
      };
    }

    summaries.push({
      staffId: s.id,
      staffName: s.name,
      role: s.role || 'staff',
      hireDate: s.hire_date || null,
      weeklyHours: s.weekly_hours || 40,
      totalGranted,
      totalUsed,
      totalRemaining,
      totalExpired,
      carriedOverDays,
      currentGrantDays,
      expiringSoonDays,
      records,
      takenDates,
      mandatoryFiveDaysTarget
    });
  }

  return summaries;
}

/**
 * 職員向け有給付与の手動追加
 */
export async function addStaffPtoGrant(
  staffId: string,
  grantDate: string,
  grantedDays: number,
  note: string = '手動付与'
): Promise<{ success: boolean; message?: string; error?: string }> {
  try {
    const role = await getRole();
    if (!role) return { success: false, error: '認証されていません。ログインしてください。' };
    resolveServiceModeForStaff(staffId);

    if (grantedDays <= 0) {
      return { success: false, error: '付与日数は0日より大きい値を指定してください。' };
    }

    const gDate = new Date(grantDate);
    const expDate = new Date(gDate);
    expDate.setFullYear(gDate.getFullYear() + 2);
    const expirationDate = toLocalISODate(expDate);

    const id = randomUUID();

    db.prepare(`
      INSERT INTO staff_pto_records (
        id, staff_id, grant_date, granted_days, used_days, remaining_days, expiration_date, note
      ) VALUES (?, ?, ?, ?, 0, ?, ?, ?)
    `).run(id, staffId, grantDate, grantedDays, grantedDays, expirationDate, note);

    db.prepare(`
      UPDATE staff 
      SET pto_granted = COALESCE((
        SELECT SUM(remaining_days) FROM staff_pto_records 
        WHERE staff_id = ? AND expiration_date >= date('now')
      ), 0)
      WHERE id = ?
    `).run(staffId, staffId);

    safeRevalidate('/attendance/pto');
    safeRevalidate('/staff');
    return { success: true, message: `職員に${grantedDays}日の有給休暇を付与しました（失効日: ${expirationDate}）。` };
  } catch (err: any) {
    console.error('職員有給付与エラー:', err);
    return { success: false, error: err.message || '職員有給の付与に失敗しました' };
  }
}

/**
 * 職員向け有給台帳レコードの手動修正
 */
export async function updateStaffPtoRecord(recordId: string, grantedDays: number, usedDays: number, note: string, expiredDays?: number): Promise<{ success: boolean; error?: string }> {
  try {
    const role = await getRole();
    if (!role) return { success: false, error: '認証されていません。ログインしてください。' };
    resolveServiceModeForTableRecord('staff_pto_records', recordId);

    if (grantedDays < usedDays) {
      return { success: false, error: '消化日数は付与日数以下にしてください。' };
    }
    const expDays = expiredDays !== undefined ? expiredDays : 0;
    const remaining = Math.max(0, grantedDays - usedDays - expDays);

    db.prepare(`
      UPDATE staff_pto_records
      SET granted_days = ?, used_days = ?, remaining_days = ?, expired_days = ?, note = ?, updated_at = CURRENT_TIMESTAMP
      WHERE id = ?
    `).run(grantedDays, usedDays, remaining, expDays, note, recordId);

    const rec = db.prepare('SELECT staff_id FROM staff_pto_records WHERE id = ?').get(recordId) as any;
    if (rec?.staff_id) {
      db.prepare(`
        UPDATE staff 
        SET pto_granted = COALESCE((
          SELECT SUM(remaining_days) FROM staff_pto_records 
          WHERE staff_id = ? AND expiration_date >= date('now')
        ), 0)
        WHERE id = ?
      `).run(rec.staff_id, rec.staff_id);
    }

    safeRevalidate('/attendance/pto');
    safeRevalidate('/staff');
    return { success: true };
  } catch (err: any) {
    console.error('職員有給レコード更新エラー:', err);
    return { success: false, error: err.message || '職員有給レコードの更新に失敗しました' };
  }
}
