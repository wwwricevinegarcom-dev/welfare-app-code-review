export async function savePayrollSettings(clientId: string, commute: number, health: number, pension: number, tax: number = 0) {
  try {
    const role = await getRole();
    if (role !== 'admin' && role !== 'service_manager') {
      return { success: false, error: '権限がありません' };
    }
    await resolveServiceModeForClient(clientId);
    await assertCanAccessClient(clientId);

    const existing = db.prepare('SELECT client_id FROM payroll_settings WHERE client_id = ?').get(clientId);
    
    if (existing) {
      db.prepare(`
        UPDATE payroll_settings SET commute_allowance = ?, health_insurance = ?, welfare_pension = ?, income_tax = ?
        WHERE client_id = ?
      `).run(commute, health, pension, tax, clientId);
    } else {
      db.prepare(`
        INSERT INTO payroll_settings (id, client_id, commute_allowance, health_insurance, welfare_pension, income_tax)
        VALUES (?, ?, ?, ?, ?, ?)
      `).run(randomUUID(), clientId, commute, health, pension, tax);
    }
    return { success: true };
  } catch (e: any) {
    console.error("Failed to save payroll settings:", e);
    return { success: false, error: e.message || '設定の保存に失敗しました' };
  }
}

export async function savePayslips(targetMonth: string, payslipsData: any[]) {
  try {
    const role = await getRole();
    if (role !== 'admin' && role !== 'service_manager') {
      return { success: false, error: '権限がありません' };
    }

    if (payslipsData && payslipsData.length > 0 && payslipsData[0].client_id) {
      await resolveServiceModeForClient(payslipsData[0].client_id);
    }

    for (const p of payslipsData) {
      await assertCanAccessClient(p.client_id);
    }

    const serviceMode = await getServiceMode();
    const isTypeB = serviceMode === 'type_b';

    // 0. 最低賃金および労働基準法コンプライアンス検証（A型のみ。B型は非雇用の工賃のため適用除外）
    if (!isTypeB) {
      const facility = db.prepare(`SELECT * FROM facility_settings WHERE id = 'default'`).get() as any || {};
      const minWage = getApplicableMinimumWage(facility.pref_code, facility.regional_minimum_wage);

      for (const p of payslipsData) {
        if ((p.hourly_wage || 0) < 0 || (p.total_working_hours || 0) < 0 || (p.base_salary || 0) < 0) {
          return { success: false, error: '【給与計算エラー】基本給・時給・労働時間に不正なマイナス値が指定されています。' };
        }

        if (p.hourly_wage > 0 && p.hourly_wage < minWage) {
          const client = db.prepare('SELECT full_name, min_wage_exemption_approved FROM clients WHERE id = ?').get(p.client_id) as any;
          if (!client?.min_wage_exemption_approved) {
            const clientName = client?.full_name || p.client_id;
            return {
              success: false,
              error: `【最低賃金法違反】${clientName} さんの時給（${p.hourly_wage}円）が地域別最低賃金（${minWage}円）を下回っています。最低賃金の減額特例許可（労基署長許可）を取得済みの場合は、利用者情報編集画面で「最低賃金減額特例」を有効にしてください。`
            };
          }
        }

        if ((p.overtime_allowance || 0) < 0 || (p.overtime_excess_allowance || 0) < 0 || (p.night_shift_allowance || 0) < 0 || (p.holiday_allowance || 0) < 0) {
          return { success: false, error: `【給与計算エラー】割増賃金（残業・60h超・深夜・休日手当）に不正な値（マイナス）が設定されています。` };
        }
      }
    } else {
      // B型時のバリデーション（工賃マイナス値ガード）
      for (const p of payslipsData) {
        if ((p.base_salary || 0) < 0 || (p.total_payment || 0) < 0) {
          return { success: false, error: '【工賃計算エラー】工賃額に不正なマイナス値が指定されています。' };
        }
      }
    }

    // 1. monthly_payslips に給与・工賃計算データを保存
    const insertMonthly = db.prepare(`
      INSERT INTO monthly_payslips (
        id, client_id, target_month, total_working_minutes, working_days, pto_days, pto_allowance,
        hourly_wage, base_salary, total_payment, overtime_minutes, overtime_allowance,
        night_shift_minutes, night_shift_allowance, holiday_working_minutes, holiday_allowance,
        overtime_excess_minutes, overtime_excess_allowance,
        wage_type, daily_wage_payment, piecework_payment, ability_allowance,
        updated_at
      ) VALUES (
        lower(hex(randomblob(16))), ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, CURRENT_TIMESTAMP
      )
      ON CONFLICT(client_id, target_month) DO UPDATE SET
        total_working_minutes = excluded.total_working_minutes,
        working_days = excluded.working_days,
        pto_days = excluded.pto_days,
        pto_allowance = excluded.pto_allowance,
        hourly_wage = excluded.hourly_wage,
        base_salary = excluded.base_salary,
        total_payment = excluded.total_payment,
        overtime_minutes = excluded.overtime_minutes,
        overtime_allowance = excluded.overtime_allowance,
        night_shift_minutes = excluded.night_shift_minutes,
        night_shift_allowance = excluded.night_shift_allowance,
        holiday_working_minutes = excluded.holiday_working_minutes,
        holiday_allowance = excluded.holiday_allowance,
        overtime_excess_minutes = excluded.overtime_excess_minutes,
        overtime_excess_allowance = excluded.overtime_excess_allowance,
        wage_type = excluded.wage_type,
        daily_wage_payment = excluded.daily_wage_payment,
        piecework_payment = excluded.piecework_payment,
        ability_allowance = excluded.ability_allowance,
        updated_at = CURRENT_TIMESTAMP
    `);

    // 2. payslips に確定ロックを登録
    const insertLock = db.prepare(`
      INSERT INTO payslips (id, client_id, target_month, is_locked)
      VALUES (lower(hex(randomblob(16))), ?, ?, 1)
      ON CONFLICT(client_id, target_month) DO UPDATE SET is_locked = 1
    `);

    // 3. payroll_settings の custom_deductions を UPSERT 更新
    const upsertSettings = db.prepare(`
      INSERT INTO payroll_settings (id, client_id, custom_deductions)
      VALUES (lower(hex(randomblob(16))), ?, ?)
      ON CONFLICT(client_id) DO UPDATE SET custom_deductions = excluded.custom_deductions
    `);

    let totalMonthWages = 0;

    const saveMany = db.transaction((data) => {
      for (const p of data) {
        const workingMinutes = Math.round((p.total_working_hours || 0) * 60);
        const hourlyWage = Math.max(0, p.hourly_wage || 0);
        
        // サーバーサイドでの基本給の妥当性検証・再計算（改ざん防止・基発150号端数処理準拠・浮動小数点誤差防止）
        const calculatedBaseSalary = Math.round((workingMinutes * hourlyWage) / 60);
        const baseSalary = p.base_salary !== undefined ? Math.max(0, p.base_salary) : calculatedBaseSalary;
        const ptoAllowance = Math.max(0, p.pto_allowance || 0);
        const overtimeMinutes = Math.round(Math.max(0, p.overtime_minutes || (p.overtime_normal_hours || 0) * 60));
        const overtimeAllowance = Math.round(Math.max(0, p.overtime_allowance || 0));
        const nightShiftMinutes = Math.round(Math.max(0, p.night_shift_minutes || (p.night_shift_hours || 0) * 60));
        const nightShiftAllowance = Math.round(Math.max(0, p.night_shift_allowance || 0));
        const holidayWorkingMinutes = Math.round(Math.max(0, p.holiday_working_minutes || (p.holiday_hours || 0) * 60));
        const holidayAllowance = Math.round(Math.max(0, p.holiday_allowance || 0));
        const overtimeExcessMinutes = Math.round(Math.max(0, p.overtime_excess_minutes || 0));
        const overtimeExcessAllowance = Math.round(Math.max(0, p.overtime_excess_allowance || 0));
        const commuteAllowance = Math.max(0, p.commute_allowance || 0);

        const abilityAllowance = Math.round(Math.max(0, p.ability_allowance || 0));
        const totalPayment = isTypeB 
          ? (baseSalary + abilityAllowance + commuteAllowance)
          : (baseSalary + ptoAllowance + overtimeAllowance + nightShiftAllowance + holidayAllowance + commuteAllowance);

        // 生産活動収支・工賃実績報告用（実費弁償たる通勤手当は工賃総額から除外）
        const statutoryWageAmount = isTypeB 
          ? (baseSalary + abilityAllowance)
          : (baseSalary + ptoAllowance + overtimeAllowance + nightShiftAllowance + holidayAllowance);
        totalMonthWages += statutoryWageAmount;

        insertMonthly.run(
          p.client_id,
          targetMonth,
          workingMinutes,
          p.working_days || 0,
          p.pto_days || 0,
          ptoAllowance,
          hourlyWage,
          baseSalary,
          totalPayment,
          overtimeMinutes,
          overtimeAllowance,
          nightShiftMinutes,
          nightShiftAllowance,
          holidayWorkingMinutes,
          holidayAllowance,
          overtimeExcessMinutes,
          overtimeExcessAllowance,
          p.wage_type || 'hourly',
          p.daily_wage_payment || 0,
          p.piecework_payment || 0,
          p.ability_allowance || 0
        );

        insertLock.run(p.client_id, targetMonth);

        if (p.custom_deductions) {
          upsertSettings.run(p.client_id, p.custom_deductions);
        }
      }

      // 生産活動収支（production_activities）の total_wages_paid も自動同期
      db.prepare(`
        INSERT INTO production_activities (id, target_month, total_wages_paid)
        VALUES (lower(hex(randomblob(16))), ?, ?)
        ON CONFLICT(target_month) DO UPDATE SET
          total_wages_paid = excluded.total_wages_paid,
          updated_at = CURRENT_TIMESTAMP
      `).run(targetMonth, totalMonthWages);
    });

    saveMany(payslipsData);
    try {
      revalidatePath('/billing/payroll');
      revalidatePath('/billing/payroll/ledger');
      revalidatePath('/billing/payroll/payslips');
      revalidatePath('/billing/payroll/wage-report');
      revalidatePath('/billing/production');
      revalidatePath('/');
    } catch (e) {
      // CLI実行やリクエスト外コンテキストを許容
    }
    return { success: true };
  } catch (e: any) {
    console.error("Failed to save payslips:", e);
    return { success: false, error: e.message || '給与・工賃データの保存に失敗しました' };
  }
}

export async function exportAccountingJournalCsv(targetMonth: string): Promise<{ success: boolean; csv?: string; error?: string }> {
  try {
    const role = await getRole();
    if (role !== 'admin' && role !== 'service_manager') {
      return { success: false, error: '権限がありません' };
    }

    const serviceMode = await getServiceMode();
    const isTypeB = serviceMode === 'type_b';

    const payslips = db.prepare(`
      SELECT 
        mp.*,
        c.full_name,
        ps.commute_allowance,
        ps.health_insurance,
        ps.welfare_pension,
        ps.income_tax,
        ps.custom_deductions
      FROM monthly_payslips mp
      JOIN clients c ON mp.client_id = c.id
      LEFT JOIN payroll_settings ps ON mp.client_id = ps.client_id
      WHERE mp.target_month = ?
    `).all(targetMonth) as any[];

    if (!payslips || payslips.length === 0) {
      return {
        success: false,
        error: `${targetMonth} の${isTypeB ? '工賃' : '給与'}データが存在しません。「一括計算」および「保存」を実行してください。`
      };
    }

    const [year, month] = targetMonth.split('-');
    const lastDay = new Date(Number(year), Number(month), 0).getDate();
    const txDate = `${targetMonth}-${String(lastDay).padStart(2, '0')}`;

    const rows: string[] = [
      '取引日,借方勘定科目,借方金額,貸方勘定科目,貸方金額,摘要'
    ];

    const wageAccount = isTypeB ? '工賃' : '賃金手当';
    const payableAccount = isTypeB ? '未払工賃' : '未払賃金';
    const wageLabel = isTypeB ? '工賃' : '賃金';

    for (const p of payslips) {
      const baseAndAllowances = isTypeB
        ? ((p.base_salary || 0) + (p.ability_allowance || 0))
        : ((p.base_salary || 0) + (p.overtime_allowance || 0) + (p.night_shift_allowance || 0) + (p.holiday_allowance || 0) + (p.pto_allowance || 0));
      const commute = p.commute_allowance || 0;
      
      let customDeductionTotal = 0;
      try {
        const customs = JSON.parse(p.custom_deductions || '[]');
        customDeductionTotal = customs.reduce((acc: number, cur: any) => acc + (Number(cur.amount) || 0), 0);
      } catch(e) {}

      const totalDeductions = isTypeB
        ? ((p.income_tax || 0) + customDeductionTotal)
        : ((p.health_insurance || 0) + (p.welfare_pension || 0) + (p.income_tax || 0) + customDeductionTotal);
      const netPayment = (baseAndAllowances + commute) - totalDeductions;

      // 数式インジェクション防止（先頭が =, +, -, @, タブ等の場合はプレフィックス付与）
      const escapeDescription = (desc: string) => {
        let str = desc;
        if (/^[=+\-@\t\r]/.test(str)) {
          str = "'" + str;
        }
        return `"${str.replace(/"/g, '""')}"`;
      };

      if (baseAndAllowances > 0) {
        rows.push(`"${txDate}","${wageAccount}",${baseAndAllowances},"${payableAccount}",${baseAndAllowances},${escapeDescription(`${p.full_name} ${month}月度 ${wageLabel}`)}`);
      }
      if (commute > 0) {
        rows.push(`"${txDate}","旅費交通費",${commute},"${payableAccount}",${commute},${escapeDescription(`${p.full_name} ${month}月度 通勤手当`)}`);
      }
      if (totalDeductions > 0) {
        const dedLabel = isTypeB ? '実費・税金等控除' : '社会保険料・税金等控除';
        rows.push(`"${txDate}","${payableAccount}",${totalDeductions},"預り金",${totalDeductions},${escapeDescription(`${p.full_name} ${month}月度 ${dedLabel}`)}`);
      }
      if (netPayment > 0) {
        rows.push(`"${txDate}","${payableAccount}",${netPayment},"普通預金",${netPayment},${escapeDescription(`${p.full_name} ${month}月度 差引支給額`)}`);
      }
    }

    return { success: true, csv: rows.join('\r\n') };
  } catch (e: any) {
    console.error("Failed to export accounting journal CSV:", e);
    return { success: false, error: e.message || '仕訳CSVの出力に失敗しました' };
  }
}

import { db } from "@/lib/db";

export default async function LedgerPrintPage({ searchParams }: { searchParams: Promise<{ month?: string }> }) {
  const role = await getRole();
  if (!role) redirect("/login");

  const serviceMode = await getServiceMode();
  const isTypeB = serviceMode === 'type_b';

  const params = await searchParams;
  const month = params.month;
  
  if (!month) return <div>対象月が指定されていません</div>;

  const facility = db.prepare(`SELECT * FROM facility_settings WHERE id = 'default'`).get() as any || {};
  const employmentInsuranceRate = facility.employment_insurance_rate || 0.006;

  const rawPayslips = db.prepare(`
    SELECT 
      p.id, p.client_id, p.target_month,
      c.full_name, c.gender,
      mp.total_working_minutes,
      mp.working_days,
      mp.pto_days,
      mp.pto_allowance,
      mp.hourly_wage,
      mp.base_salary,
      mp.total_payment,
      mp.overtime_minutes,
      mp.night_shift_minutes,
      mp.holiday_working_minutes,
      mp.overtime_allowance,
      mp.night_shift_allowance,
      mp.holiday_allowance,
      mp.overtime_excess_minutes,
      mp.overtime_excess_allowance,
      mp.ability_allowance,
      ps.commute_allowance,
      ps.health_insurance AS health_insurance_deduction,
      ps.welfare_pension AS welfare_pension_deduction,
      ps.income_tax,
      ps.custom_deductions
    FROM payslips p
    JOIN clients c ON p.client_id = c.id

    LEFT JOIN payroll_settings ps ON p.client_id = ps.client_id

  `).all(month) as any[];

  if (rawPayslips.length === 0) {

  }

  // 勤怠から出勤日数と有休日数を個別集計
  const attendanceStats = db.prepare(`
    SELECT 
      client_id,
      SUM(CASE WHEN status = 'present' THEN 1 ELSE 0 END) as working_days,
      SUM(CASE WHEN status IN ('paid_leave', 'pto') THEN 1 ELSE 0 END) as pto_days
    FROM attendance_records
    WHERE record_date LIKE ? AND deleted_at IS NULL
    GROUP BY client_id
  `).all(`${month}-%`) as any[];

  const payslips = rawPayslips.map(slip => {
    const att = attendanceStats.find((a: any) => a.client_id === slip.client_id);
    const working_days = (typeof slip.working_days === 'number' && slip.working_days > 0) ? slip.working_days : (att?.working_days || 0);
    const pto_days = isTypeB ? 0 : ((typeof slip.pto_days === 'number' && slip.pto_days > 0) ? slip.pto_days : (att?.pto_days || 0));
    
    const total_working_hours = (slip.total_working_minutes || 0) / 60;
    const overtime_hours = isTypeB ? 0 : ((slip.overtime_minutes || 0) / 60);
    const overtime_excess_hours = isTypeB ? 0 : ((slip.overtime_excess_minutes || 0) / 60);
    const night_hours = isTypeB ? 0 : ((slip.night_shift_minutes || 0) / 60);
    const holiday_hours = isTypeB ? 0 : ((slip.holiday_working_minutes || 0) / 60);
    
    const commute_allowance = slip.commute_allowance || 0;
    const ability_allowance = slip.ability_allowance || 0;
    const pto_allowance = isTypeB ? 0 : (slip.pto_allowance || 0);
    const overtime_allowance = isTypeB ? 0 : (slip.overtime_allowance || 0);
    const overtime_excess_allowance = isTypeB ? 0 : (slip.overtime_excess_allowance || 0);
    const night_shift_allowance = isTypeB ? 0 : (slip.night_shift_allowance || 0);
    const holiday_allowance = isTypeB ? 0 : (slip.holiday_allowance || 0);
    const base_salary = slip.base_salary || 0;
    const hourly_wage = slip.hourly_wage || 0;

    const health_insurance_deduction = isTypeB ? 0 : (slip.health_insurance_deduction || 0);
    const welfare_pension_deduction = isTypeB ? 0 : (slip.welfare_pension_deduction || 0);
    const income_tax = slip.income_tax || 0;
    
    // 総支給額（B型: 基本工賃 + 能力給 + 通勤手当 / A型: 基本給 + 有休手当 + 割増手当 + 通勤手当）
    const totalPay = isTypeB
      ? (base_salary + ability_allowance + commute_allowance)
      : (base_salary + pto_allowance + overtime_allowance + night_shift_allowance + holiday_allowance + commute_allowance);
    
    // 雇用保険（B型は雇用保険なし）
    let employment_insurance_deduction = 0;
    if (!isTypeB) {
      const rawEmpIns = totalPay * employmentInsuranceRate;
      const calculatedEmpIns = (rawEmpIns - Math.floor(rawEmpIns)) > 0.500001
        ? Math.ceil(rawEmpIns)
        : Math.floor(rawEmpIns);
      employment_insurance_deduction = typeof slip.employment_insurance_deduction === 'number'
        ? slip.employment_insurance_deduction
        : calculatedEmpIns;
    }
    
    let customDeductions = [];
    try {
      customDeductions = slip.custom_deductions ? JSON.parse(slip.custom_deductions) : [];
    } catch {
      customDeductions = [];
    }
    const customSum = customDeductions.reduce((sum: number, d: any) => sum + (d.amount || 0), 0);
    
    const total_deductions = isTypeB
      ? (income_tax + customSum)
      : (health_insurance_deduction + welfare_pension_deduction + employment_insurance_deduction + income_tax + customSum);
    const net_payment = totalPay - total_deductions;

    // 性別表示の正規化
    let genderLabel = '未設定';
    if (slip.gender === 'male' || slip.gender === '男' || slip.gender === '1') genderLabel = '男';
    else if (slip.gender === 'female' || slip.gender === '女' || slip.gender === '2') genderLabel = '女';

      pto_days,

      overtime_excess_hours,
      night_hours,
      holiday_hours,
      base_salary,
      ability_allowance,
      pto_allowance,
      overtime_allowance,
      overtime_excess_allowance,
      night_shift_allowance,
      holiday_allowance,
      commute_allowance,
      totalPay,
      health_insurance_deduction,
      welfare_pension_deduction,
      employment_insurance_deduction,
      income_tax,
      customSum,
      total_deductions,
      hourly_wage,
      net_payment,
    };
  });

  const [yearStr, monthStr] = month.split('-');

  return (

function formatHoursMinutes(decimalHours: number): string {
  const hours = Math.floor(decimalHours);
  const minutes = Math.round((decimalHours - hours) * 60);
  return `${hours}時間${minutes.toString().padStart(2, '0')}分`;
}

      } else {
        overtimeMinutes = 0;

    const setting = payrollSettings.find(s => s.client_id === client.id) || {};
    const hourlyWage = (saved && saved.hourly_wage) ? saved.hourly_wage : (contract?.hourly_wage || 0);
    
    // B型対応: 賃金・工賃体系（時間給・日給・出来高制）
    const wageType = setting.wage_type || client.wage_type || 'hourly';
    const dailyWage = (saved && typeof (saved as any).daily_wage_payment === 'number' && (saved as any).daily_wage_payment > 0)
      ? (saved as any).daily_wage_payment
      : (setting.daily_wage || contract?.daily_wage || 0);
    const pieceworkUnitPrice = setting.piecework_unit_price || contract?.piecework_unit_price || 0;
    const pieceworkQuantity = setting.piecework_quantity || 0;
    const abilityAllowance = (saved && typeof (saved as any).ability_allowance === 'number')
      ? (saved as any).ability_allowance
      : (setting.ability_allowance || 0);

    let calculatedBaseSalary = 0;
    let pieceworkPayment = 0;
    let dailyWagePayment = 0;

    if (isTypeB) {
      if (wageType === 'daily') {
        dailyWagePayment = Math.round(workingDaysCount * dailyWage);
        calculatedBaseSalary = dailyWagePayment;
      } else if (wageType === 'piecework') {
        pieceworkPayment = Math.round(pieceworkUnitPrice * pieceworkQuantity);
        calculatedBaseSalary = pieceworkPayment;
      } else {
        // 時間給制（デフォルト・実働時間 × 工賃単価）
        calculatedBaseSalary = Math.round((actualWorkingMinutes * hourlyWage) / 60);
      }
    } else {
      // 労基法第108条準拠: 基本給（実働分）と有休手当（有休分）の明確な分離計算
      calculatedBaseSalary = Math.round((actualWorkingMinutes * hourlyWage) / 60);
    }

    const calculatedPtoAllowance = isTypeB ? 0 : Math.round((ptoMinutesTotal * hourlyWage) / 60);
    const actualHours = actualWorkingMinutes / 60;
    const ptoHours = isTypeB ? 0 : (ptoMinutesTotal / 60);

    const baseSalary = (saved && typeof saved.base_salary === 'number' && saved.base_salary > 0)
      ? saved.base_salary
      : calculatedBaseSalary;

    const ptoAllowance = isTypeB ? 0 : ((saved && typeof saved.pto_allowance === 'number')
      ? saved.pto_allowance
      : calculatedPtoAllowance);
    
    // B-16: 通達（昭和63年基発150号）準拠の端数処理（30分未満切り捨て、30分以上切り上げ）
    const roundStatutoryHours = (totalMins: number) => {
      const h = Math.floor(totalMins / 60);
      const m = totalMins % 60;
      return m >= 30 ? h + 1 : h;
    };

    const roundedOvertimeHours = roundStatutoryHours(overtimeMinutes);
    const roundedNightShiftHours = roundStatutoryHours(nightShiftMinutes);
    const roundedHolidayHours = roundStatutoryHours(holidayWorkingMinutes);

    // 割増賃金（B型は非雇用のため割増賃金なし、A型は就業規則・設定動的適用）
    const normalOvertimeHours = Math.min(roundedOvertimeHours, 60);
    const excessOvertimeHours = Math.max(0, roundedOvertimeHours - 60);
    const calculatedNormalOvertime = isTypeB ? 0 : Math.round(normalOvertimeHours * hourlyWage * overtimeRate);
    const calculatedExcessOvertime = isTypeB ? 0 : Math.round(excessOvertimeHours * hourlyWage * overtimeOver60Rate);
    const calculatedOvertime = calculatedNormalOvertime + calculatedExcessOvertime;

    const calculatedNight = isTypeB ? 0 : Math.round(roundedNightShiftHours * hourlyWage * nightRate);
    const calculatedHoliday = isTypeB ? 0 : Math.round(roundedHolidayHours * hourlyWage * holidayRate);

    const overtimeAllowance = isTypeB ? 0 : ((saved && typeof saved.overtime_allowance === 'number') ? saved.overtime_allowance : calculatedOvertime);
    const overtimeExcessAllowance = isTypeB ? 0 : ((saved && typeof saved.overtime_excess_allowance === 'number') ? saved.overtime_excess_allowance : calculatedExcessOvertime);
    const overtimeNormalAllowance = isTypeB ? 0 : (overtimeAllowance - overtimeExcessAllowance);
    const nightShiftAllowance = isTypeB ? 0 : ((saved && typeof saved.night_shift_allowance === 'number') ? saved.night_shift_allowance : calculatedNight);
    const holidayAllowance = isTypeB ? 0 : ((saved && typeof saved.holiday_allowance === 'number') ? saved.holiday_allowance : calculatedHoliday);

    // Deductions
    const commute = (saved && typeof saved.commute_allowance === 'number') ? saved.commute_allowance : (setting.commute_allowance || 0);
    const totalPayment = isTypeB 
      ? (baseSalary + abilityAllowance + commute)
      : (baseSalary + ptoAllowance + overtimeAllowance + nightShiftAllowance + holidayAllowance + commute);
    
    // 社会保険・雇用保険（B型は非雇用契約のため0円）
    const standardRemuneration = contract?.standard_remuneration || 0;
    const autoHealthInsurance = standardRemuneration > 0 ? Math.round(standardRemuneration * healthInsuranceRate) : 0;
    const autoWelfarePension = standardRemuneration > 0 ? Math.round(standardRemuneration * welfarePensionRate) : 0;

    const rawEmpIns = totalPayment * employmentInsuranceRate;
    const calculatedEmpIns = (rawEmpIns - Math.floor(rawEmpIns)) > 0.500001
      ? Math.ceil(rawEmpIns)
      : Math.floor(rawEmpIns);
    const employmentInsurance = isTypeB ? 0 : ((saved && typeof saved.employment_insurance_deduction === 'number') 
      ? saved.employment_insurance_deduction 
      : calculatedEmpIns);
    const healthInsurance = isTypeB ? 0 : ((saved && typeof saved.health_insurance_deduction === 'number') ? saved.health_insurance_deduction : (setting.health_insurance || autoHealthInsurance || 0));
    const welfarePension = isTypeB ? 0 : ((saved && typeof saved.welfare_pension_deduction === 'number') ? saved.welfare_pension_deduction : (setting.welfare_pension || autoWelfarePension || 0));
    const incomeTax = (saved && typeof saved.income_tax === 'number') ? saved.income_tax : (setting.income_tax || 0);
    
    let customDeductions = [];
    try {
      customDeductions = saved && saved.custom_deductions ? JSON.parse(saved.custom_deductions) : [];
    } catch {
      customDeductions = [];
    }

    // 最低賃金減額特例許可の考慮（A型のみ）
    const isExempted = !isTypeB && hourlyWage > 0 && hourlyWage < minWage && !!client.min_wage_exemption_approved;
    const isBelowMinWage = !isTypeB && hourlyWage > 0 && hourlyWage < minWage && !client.min_wage_exemption_approved;
    const customDeductionSum = customDeductions.reduce((sum: number, d: any) => sum + (d.amount || 0), 0);
    const isExcessiveDeduction = baseSalary > 0 && customDeductionSum > (baseSalary * 0.5);

    // B-15: 36協定アラート（A型のみ）
    const monthlyOvertimeHours = overtimeMinutes / 60;
    if (!isTypeB && monthlyOvertimeHours > 45) {
      laborLawWarnings.push(`${client.full_name} さんの当月時間外労働が ${monthlyOvertimeHours.toFixed(1)} 時間となり、36協定の上限基準（月45時間）を超過しています。`);
    }

    if (!isTypeB) {
      try {
        const pastOtRow = db.prepare(`
          SELECT COALESCE(SUM(ROUND(overtime_allowance / NULLIF(hourly_wage * ?, 0))), 0) as past_ot_hours
          FROM monthly_payslips
          WHERE client_id = ? AND target_month >= ? AND target_month < ?
        `).get(overtimeRate, client.id, pastOneYearStartMonth, currentMonth) as any;
        const annualEstimatedOt = (pastOtRow?.past_ot_hours || 0) + monthlyOvertimeHours;
        if (annualEstimatedOt > 360) {
          laborLawWarnings.push(`${client.full_name} さんの過去1年間の累積時間外労働（推計${annualEstimatedOt.toFixed(1)}時間）が36協定の年間限度（年360時間）を超過しています。`);
        }
      } catch (e) {}
    }

    return {
      client_id: client.id,
      client_name: client.full_name,
      total_attendance_days: totalDays,
      working_days: workingDaysCount,
      pto_days: ptoDaysCount,
      total_working_hours: totalHours,
      formatted_working_hours: formatHoursMinutes(totalHours),
      hourly_wage: hourlyWage,
      base_salary: baseSalary,
      wage_type: wageType,
      daily_wage: dailyWage,
      piecework_unit_price: pieceworkUnitPrice,
      piecework_quantity: pieceworkQuantity,
      piecework_payment: pieceworkPayment,
      daily_wage_payment: dailyWagePayment,
      ability_allowance: abilityAllowance,
      pto_allowance: ptoAllowance,
      overtime_allowance: overtimeAllowance,
      overtime_normal_allowance: overtimeNormalAllowance,
      overtime_excess_allowance: overtimeExcessAllowance,
      overtime_normal_hours: normalOvertimeHours,
      overtime_excess_hours: excessOvertimeHours,
      overtime_excess_minutes: Math.round(excessOvertimeHours * 60),
      overtime_minutes: overtimeMinutes,
      night_shift_allowance: nightShiftAllowance,
      night_shift_hours: roundedNightShiftHours,
      night_shift_minutes: nightShiftMinutes,
      holiday_allowance: holidayAllowance,
      holiday_hours: roundedHolidayHours,
      holiday_working_minutes: holidayWorkingMinutes,
      commute_allowance: commute,
      employment_insurance_deduction: employmentInsurance,
      health_insurance_deduction: healthInsurance,
      welfare_pension_deduction: welfarePension,
      income_tax: incomeTax,
      custom_deductions: customDeductions,
      is_below_min_wage: isBelowMinWage,
      is_excessive_deduction: isExcessiveDeduction,
      saved: !!saved,
      gender: client.gender
    };
  });

  const belowMinWageClients = payrollData.filter(p => p.is_below_min_wage);
  const excessiveDeductionClients = payrollData.filter(p => p.is_excessive_deduction);

  // 年5日有給取得義務の要注意・警告者を集計
  const paidLeaveStatuses = getAllClientsPaidLeaveStatus();
  const criticalPaidLeave = paidLeaveStatuses.filter(s => s.complianceStatus === 'critical');
  const warningPaidLeave = paidLeaveStatuses.filter(s => s.complianceStatus === 'warning');

  // 労基法第24条協定（賃金控除協定）の有効性および項目照合
  const todayDateStr = new Date().toISOString().slice(0, 10);
  const activeArticle24Agreements = db.prepare(`
    SELECT * FROM labor_agreements 
    WHERE agreement_type = 'article24_deduction' 
      AND valid_from <= ? AND valid_to >= ?
  `).all(todayDateStr, todayDateStr) as any[];

  const allowedDeductionSet = new Set<string>();
  for (const agr of activeArticle24Agreements) {
    try {
      const items = JSON.parse(agr.deduction_items || '[]');
      items.forEach((it: string) => allowedDeductionSet.add(it.trim()));
    } catch(e) {}
  }

  const hasAnyCustomDeductions = payrollData.some(p => p.custom_deductions && p.custom_deductions.some((d: any) => d.amount > 0));
  const isArticle24Missing = activeArticle24Agreements.length === 0 && hasAnyCustomDeductions;

  const unagreedDeductionIssues: { clientName: string, items: string[] }[] = [];
  if (activeArticle24Agreements.length > 0) {
    for (const p of payrollData) {
      if (p.custom_deductions && p.custom_deductions.length > 0) {
        const unagreed = p.custom_deductions
          .filter((d: any) => d.amount > 0 && !allowedDeductionSet.has(d.name?.trim()))
          .map((d: any) => d.name);
        if (unagreed.length > 0) {
          unagreedDeductionIssues.push({ clientName: p.client_name, items: unagreed });
        }
      }
    }
  }

  // 社会保険・雇用保険の加入義務コンプライアンス検証（労務監査対策）
  const insuranceComplianceIssues: {
    clientId: string;
    clientName: string;
    avgWeeklyHours: number;
    issueType: 'employment_insurance' | 'social_insurance' | 'both';
    reason: string;
  }[] = [];

  for (const client of clients) {
    const pData = payrollData.find(p => p.client_id === client.id);
    if (!pData) continue;

    // 月間労働時間を4.33週（30日÷7）で割って週平均実働時間を算出（所定労働時間も加味）
    const calculatedWeeklyHours = Math.round((pData.total_working_hours / 4.33) * 10) / 10;
    const scheduledWeeklyHours = client.weekly_scheduled_hours || 0;
    const effectiveWeeklyHours = Math.max(calculatedWeeklyHours, scheduledWeeklyHours);

    const setting = payrollSettings.find(s => s.client_id === client.id) || {};
    const hasEmploymentInsurance = !!client.employment_insurance_number || pData.employment_insurance_deduction > 0;
    const hasHealthInsurance = (setting.health_insurance || 0) > 0 || pData.health_insurance_deduction > 0;
    const hasWelfarePension = (setting.welfare_pension || 0) > 0 || pData.welfare_pension_deduction > 0;

    // 週20時間以上は雇用保険の加入義務（雇用保険法第6条）
    const needsEmployment = effectiveWeeklyHours >= 20;
    // 週30時間以上は健康保険・厚生年金の当然適用（健康保険法第3条・厚生年金保険法第12条）
    const needsSocial = effectiveWeeklyHours >= 30;

    const empIssue = needsEmployment && !hasEmploymentInsurance;
    const socIssue = needsSocial && (!hasHealthInsurance || !hasWelfarePension);

    if (empIssue || socIssue) {
      insuranceComplianceIssues.push({
        clientId: client.id,
        clientName: client.full_name,
        avgWeeklyHours: effectiveWeeklyHours,
        issueType: empIssue && socIssue ? 'both' : empIssue ? 'employment_insurance' : 'social_insurance',
        reason: empIssue && socIssue
          ? `週${effectiveWeeklyHours}時間労働（雇用保険・社会保険の両方で加入要件該当、未加入疑い）`
          : empIssue
            ? `週${effectiveWeeklyHours}時間労働（週20h以上: 雇用保険加入義務該当、被保険者番号未登録または控除0円）`
            : `週${effectiveWeeklyHours}時間労働（週30h以上: 社会保険加入義務該当、健康保険・年金控除0円）`
      });
    }
  }

  // B型専用: 厚労省最新算定式による平均工賃月額・基本報酬判定
  let bTypeStats = {
    monthlyTotalWage: 0,
    monthlyTotalClients: 0,
    monthlyOpenDays: 0,
    monthlyAverageWage: 0,
    annualTotalWage: 0,
    annualTotalClients: 0,
    annualOpenDays: 0,
    officialAverageWage: 0,
    predictedServiceFeeName: '',
    predictedUnitScore: 0,
    staffingClass: facility.staffing_class || '6:1'
  };

  if (isTypeB) {
    // 厚労省工賃基準: 実費弁償（通勤手当）を除外し、純作業工賃＋能力給で算定
    const monthlyTotalWage = payrollData.reduce((sum, p) => sum + (p.base_salary || 0) + (p.ability_allowance || 0), 0);
    const monthlyTotalClients = payrollData.reduce((sum, p) => sum + (p.working_days || 0), 0);
    const monthlyOpenDaysRow = db.prepare(`
      SELECT COUNT(DISTINCT record_date) as open_days
      FROM attendance_records
      WHERE record_date LIKE ? AND status = 'present' AND deleted_at IS NULL
    `).get(`${currentMonth}-%`) as any;
    const monthlyOpenDays = Math.max(1, monthlyOpenDaysRow?.open_days || 20);
    const monthlyAverageWage = clients.length > 0 ? Math.round(monthlyTotalWage / clients.length) : 0;

    let annualTotalWage = 0;
    let annualTotalClients = 0;
    let annualOpenDays = 0;

    try {
      const pastAnnualRow = db.prepare(`
        SELECT 
          COALESCE(SUM(base_salary + COALESCE(ability_allowance, 0)), 0) as total_wages,
          COALESCE(SUM(working_days), 0) as total_clients
        FROM monthly_payslips
        WHERE target_month >= ? AND target_month <= ?
      `).get(pastOneYearStartMonth, currentMonth) as any;

      annualTotalWage = pastAnnualRow?.total_wages || (monthlyTotalWage * 12);
      annualTotalClients = pastAnnualRow?.total_clients || (monthlyTotalClients * 12);

      const pastOpenDaysRow = db.prepare(`
        SELECT COUNT(DISTINCT record_date) as annual_days
        FROM attendance_records
        WHERE record_date >= ? AND record_date <= ? AND status = 'present' AND deleted_at IS NULL
      `).get(`${pastOneYearStartMonth}-01`, `${currentMonth}-31`) as any;

      annualOpenDays = Math.max(1, pastOpenDaysRow?.annual_days || (monthlyOpenDays * 12));
    } catch(e) {
      annualTotalWage = monthlyTotalWage * 12;
      annualTotalClients = monthlyTotalClients * 12;
      annualOpenDays = monthlyOpenDays * 12;
    }

    const { averageWageMonthly } = calculateTypeBAverageWage(annualTotalWage, annualTotalClients, annualOpenDays);
    const feeInfo = determineTypeBServiceFee(facility.staffing_class || '6:1', averageWageMonthly || monthlyAverageWage);

    bTypeStats = {
      monthlyTotalWage,
      monthlyTotalClients,
      monthlyOpenDays,
      monthlyAverageWage,
      annualTotalWage,
      annualTotalClients,
      annualOpenDays,
      officialAverageWage: averageWageMonthly || monthlyAverageWage,
      predictedServiceFeeName: feeInfo.serviceName,
      predictedUnitScore: feeInfo.unitScore,
      staffingClass: facility.staffing_class || '6:1'
    };
  }

  return (
    <div className="flex min-h-screen flex-col bg-background/50">
      <Header activePage="payroll" />
      <BillingSubNav />
      <main className="flex-1 p-3.5 sm:p-6 md:p-12">
        <div className="mx-auto max-w-7xl space-y-6">
          {/* B型モード時: 厚労省最新算定式・工賃月額集計パネルおよび平均工賃月額基準管理（3,000円以上） */}
          {isTypeB && (
            <TypeBWageStatPanel 
              currentMonth={currentMonth}
              monthlyTotalWage={bTypeStats.monthlyTotalWage}
              monthlyTotalClients={bTypeStats.monthlyTotalClients}
              monthlyOpenDays={bTypeStats.monthlyOpenDays}
              monthlyAverageWage={bTypeStats.monthlyAverageWage}
              annualTotalWage={bTypeStats.annualTotalWage}
              annualTotalClients={bTypeStats.annualTotalClients}
              annualOpenDays={bTypeStats.annualOpenDays}
              officialAverageWage={bTypeStats.officialAverageWage}
              predictedServiceFeeName={bTypeStats.predictedServiceFeeName}
              predictedUnitScore={bTypeStats.predictedUnitScore}
              staffingClass={bTypeStats.staffingClass}
            />
          )}

          {/* A型モード時のみ: 労基法関連アラート（年5日有給、社保雇用保険、最低賃金、24条協定） */}
          {!isTypeB && (
            <>
              {(criticalPaidLeave.length > 0 || warningPaidLeave.length > 0) && (
                <div className="p-4 bg-amber-50 border border-amber-300 rounded-md text-amber-900 text-sm flex flex-col sm:flex-row sm:items-center justify-between gap-3">
                  <div>
                    <div className="font-bold text-base mb-1">
                      ⚠【労基法第39条】有給休暇の「年5日取得義務」期限が迫っている利用者がいます（計 {criticalPaidLeave.length + warningPaidLeave.length} 名）
                    </div>
                    <p className="text-xs text-amber-800 mb-2">
                      法定付与10日以上の労働者には、基準日から1年以内に5日以上の有給を取得させる義務があります。
                    </p>
                    <div className="flex flex-wrap gap-2 text-xs">
                      {criticalPaidLeave.map(c => (
                        <span key={c.clientId} className="px-2 py-0.5 rounded bg-red-100 text-red-800 font-bold">
                          {c.clientName} (残{c.daysUntilPeriodEnd}日 / 不足{c.remainingDaysToTarget}日)
                        </span>
                      ))}
                      {warningPaidLeave.map(c => (
                        <span key={c.clientId} className="px-2 py-0.5 rounded bg-amber-100 text-amber-800 font-medium">
                          {c.clientName} (残{c.daysUntilPeriodEnd}日 / 不足{c.remainingDaysToTarget}日)
                        </span>
                      ))}
                    </div>
                  </div>
                  <Link href="/attendance/pto" className="self-start sm:self-auto shrink-0 px-3 py-1.5 text-xs font-bold bg-amber-600 text-white rounded hover:bg-amber-700">
                    有給管理簿を確認 →
                  </Link>
                </div>
              )}

              {insuranceComplianceIssues.length > 0 && (
                <div className="p-4 bg-indigo-50 border border-indigo-200 rounded-md text-indigo-900 text-sm">
                  <div className="font-bold text-base mb-1 flex items-center gap-2">
                    <span>🛡【社労士監査対策】社会保険・雇用保険の加入要件該当アラート（計 {insuranceComplianceIssues.length} 名）</span>
                  </div>
                  <p className="text-xs text-indigo-700 mb-2">
                    労働時間基準（週20時間以上で雇用保険、週30時間以上で社会保険）を満たしているにもかかわらず、被保険者番号未登録または保険料控除が0円の利用者がいます。実地調査・年金事務所調査で遡及加入・追徴金が発生するリスクを防止するため、加入状況を確認してください。
                  </p>
                  <div className="space-y-1 text-xs">
                    {insuranceComplianceIssues.map(issue => (
                      <div key={issue.clientId} className="flex flex-col sm:flex-row sm:items-center justify-between gap-1 bg-white/80 p-2 rounded border border-indigo-100">
                        <span className="font-medium text-slate-800">{issue.clientName}</span>
                        <span className="text-indigo-900 font-semibold">{issue.reason}</span>
                        <Link href={`/clients/${issue.clientId}/edit`} className="text-blue-600 underline hover:text-blue-800 text-[11px] self-start sm:self-auto">
                          利用者・給与設定を修正 →
                        </Link>
                      </div>
                    ))}
                  </div>
                </div>
              )}

              {belowMinWageClients.length > 0 && (
                <div className="p-4 bg-red-50 border border-red-300 rounded-md text-red-800 text-sm">
                  <div className="font-bold text-base mb-1">🚨【労働基準法違反リスク】地域別最低賃金を下回っている利用者がいます</div>
                  <p className="mb-2">現在の適用最低賃金基準: <strong>{minWage.toLocaleString()}円/時</strong></p>
                  <ul className="list-disc list-inside space-y-1">
                    {belowMinWageClients.map(c => (
                      <li key={c.client_id}>
                        {c.client_name}: 現在の時給 <strong>{c.hourly_wage.toLocaleString()}円</strong> （{minWage - c.hourly_wage}円不足）
                      </li>
                    ))}
                  </ul>
                  <p className="mt-2 text-xs text-red-600">※A型事業所では最低賃金の適用除外許可を得ていない限り、最低賃金以上の支払いが義務付けられています。</p>
                </div>
              )}

              {isArticle24Missing && (
                <div className="p-4 bg-red-50 border border-red-300 rounded-md text-red-900 text-sm flex flex-col sm:flex-row sm:items-center justify-between gap-3">
                  <div>
                    <div className="font-bold text-base mb-1">🚨【労働基準法第24条違反リスク】有効な賃金控除協定（24条協定）が未登録です</div>
                    <p className="text-xs text-red-800 leading-relaxed">
                      賃金から昼食代や送迎費実費などを控除（天引き）するには、労働者の過半数代表者との「書面による労使協定（24条協定）」の締結が法令上義務付けられています。協定のない天引きは労基法第24条違反（全額払いの原則違反: 30万円以下の罰金）の対象となります。
                    </p>
                  </div>
                  <Link href="/settings/agreements" className="self-start sm:self-auto shrink-0 px-3 py-1.5 text-xs font-bold bg-red-700 text-white rounded hover:bg-red-800">
                    24条協定を登録 →
                  </Link>
                </div>
              )}

              {unagreedDeductionIssues.length > 0 && (

            </>
          )}

          <PayrollClient

export default async function WageReportPage({ searchParams }: { searchParams: Promise<{ year?: string }> }) {
  const role = await getRole();
  if (!role) redirect("/login");

  const params = await searchParams;
  const today = new Date();
  const currentMonthNum = today.getMonth() + 1;
  const currentYearNum = today.getFullYear();
  // 4月以降なら今年度、1〜3月なら前年度が直近年度
  const defaultFiscalYear = currentMonthNum >= 4 ? currentYearNum : currentYearNum - 1;
  const fiscalYear = params.year ? parseInt(params.year, 10) : defaultFiscalYear;

  const facility = db.prepare(`SELECT * FROM facility_settings WHERE id = 'default'`).get() as any || {};

  // 4月から翌年3月までの12ヶ月のリストを生成
  const months: string[] = [];
  for (let m = 4; m <= 12; m++) {
    months.push(`${fiscalYear}-${String(m).padStart(2, '0')}`);
  }
  for (let m = 1; m <= 3; m++) {
    months.push(`${fiscalYear + 1}-${String(m).padStart(2, '0')}`);
  }

  // 月別実績の集計
  let annualTotalOpenDays = 0;
  let annualTotalAttendances = 0;
  let annualTotalWage = 0;

  const monthlyData = months.map(monthStr => {
    const [y, m] = monthStr.split('-');
    const monthLabel = `${parseInt(m, 10)}月`;

    // 開所日数と延べ利用人数・実利用人数
    const attendanceStats = db.prepare(`
      SELECT 
        COUNT(DISTINCT record_date) as open_days,
        COUNT(*) as total_attendances,
        COUNT(DISTINCT client_id) as unique_users
      FROM attendance_records
      WHERE record_date LIKE ? AND status = 'present' AND deleted_at IS NULL
    `).get(`${monthStr}-%`) as any;

    const openDays = attendanceStats?.open_days || 0;
    const totalAttendances = attendanceStats?.total_attendances || 0;
    const uniqueUsers = attendanceStats?.unique_users || 0;
    const avgUsersPerDay = openDays > 0 ? (totalAttendances / openDays) : 0;

    // 当月の工賃支払総額（厚労省指定基準: 実費弁償の通勤手当等を除外した純工賃を集計）
    const wageStats = db.prepare(`
      SELECT COALESCE(SUM(mp.base_salary + COALESCE(mp.ability_allowance, 0)), 0) as total_wage
      FROM payslips p
      JOIN monthly_payslips mp ON p.client_id = mp.client_id AND p.target_month = mp.target_month
      WHERE p.target_month = ?
    `).get(monthStr) as any;

    const totalWage = wageStats?.total_wage || 0;
    const avgWagePerUser = uniqueUsers > 0 ? (totalWage / uniqueUsers) : 0;

    annualTotalOpenDays += openDays;
    annualTotalAttendances += totalAttendances;
    annualTotalWage += totalWage;

    return {
      month: monthStr,
      monthLabel,
      openDays,
      totalAttendances,
      uniqueUsers,
      avgUsersPerDay,
      totalWage,
      avgWagePerUser
    };
  });

  // 厚労省算定式による確定平均工賃月額
  const avgWageResult = calculateTypeBAverageWage({
    totalWagePaid: annualTotalWage,
    totalClientAttendances: annualTotalAttendances,
    facilityOpenDays: Math.max(1, annualTotalOpenDays),
    operatingMonths: 12
  });

  const annualSummary = {
    totalOpenDays: annualTotalOpenDays,
    totalAttendances: annualTotalAttendances,
    annualAvgUsersPerDay: avgWageResult.averageUsersPerDay,
    annualTotalWage: annualTotalWage,
    statutoryMonthlyAverageWage: avgWageResult.averageMonthlyWage,
    isBelow3000: avgWageResult.isBelowMinimum3000,
    targetWage: facility.target_wage || 20000
  };

  return (
    <WageReportClient 
      facility={facility}
      fiscalYear={fiscalYear}
      monthlyData={monthlyData}
      annualSummary={annualSummary}
    />
  );
}

import { db } from "@/lib/db";

export async function getCalculatedWagesForMonth(targetMonth: string): Promise<number | null> {
  const role = await getRole();
  if (role !== 'admin') return null;

  try {
    const row = db.prepare(`
      SELECT SUM(
        base_salary 
        + overtime_allowance 
        + night_shift_allowance 
        + holiday_allowance 
        + COALESCE(overtime_excess_allowance, 0) 
        + COALESCE(pto_allowance, 0)
        + COALESCE(ability_allowance, 0)
      ) as total_wages
      FROM monthly_payslips
      WHERE target_month = ?
    `).get(targetMonth) as any;

    return row && row.total_wages !== null ? Number(row.total_wages) : null;
  } catch (error) {
    console.error("Failed to fetch calculated wages for month:", error);
    return null;
  }
}

export async function getProductionActivities() {
  const role = await getRole();
  if (role !== 'admin') return [];

  const rows = db.prepare(`
    SELECT * FROM production_activities 
    ORDER BY target_month DESC 
    LIMIT 12
  `).all() as any[];

  // 各月の確定給与・工賃総額を取得して付与
  return rows.map(r => {
    const payslipRow = db.prepare(`
      SELECT SUM(
        base_salary 
        + overtime_allowance 
        + night_shift_allowance 
        + holiday_allowance 
        + COALESCE(overtime_excess_allowance, 0) 
        + COALESCE(pto_allowance, 0)
        + COALESCE(ability_allowance, 0)
      ) as total_wages
      FROM monthly_payslips
      WHERE target_month = ?
    `).get(r.target_month) as any;
    return {
      ...r,
      payslip_total_wages: payslipRow && payslipRow.total_wages !== null ? Number(payslipRow.total_wages) : null
    };
  });
}

export async function saveProductionActivity(formData: FormData) {
  try {
    const role = await getRole();
    if (role !== 'admin') return;

    const target_month = formData.get("targetMonth") as string;
    if (!target_month) return;
    const total_revenue = parseInt(formData.get("totalRevenue") as string, 10) || 0;
    const total_expenses = parseInt(formData.get("totalExpenses") as string, 10) || 0;
    const total_wages_paid = parseInt(formData.get("totalWagesPaid") as string, 10) || 0;

    db.prepare(`
      INSERT INTO production_activities (
        id, target_month, total_revenue, total_expenses, total_wages_paid, updated_at
      ) VALUES (
        ?, ?, ?, ?, ?, CURRENT_TIMESTAMP
      ) ON CONFLICT(target_month) DO UPDATE SET
        total_revenue = excluded.total_revenue,
        total_expenses = excluded.total_expenses,
        total_wages_paid = excluded.total_wages_paid,
        updated_at = CURRENT_TIMESTAMP
    `).run(randomUUID(), target_month, total_revenue, total_expenses, total_wages_paid);

    revalidatePath('/billing/production');
    revalidatePath('/billing/simulation');
  } catch (e: any) {
    console.error("Failed to save production activity:", e);
  }
}

export async function getProductionDeficitStatus() {
  const role = await getRole();
  if (!role) return { consecutiveDeficitMonths: 0, hasDeficitRisk: false, recentActivities: [] };

  const rows = db.prepare(`
    SELECT target_month, total_revenue, total_expenses, total_wages_paid
    FROM production_activities
    ORDER BY target_month DESC
    LIMIT 6
  `).all() as any[];

  let consecutiveDeficitMonths = 0;
  for (const r of rows) {
    const profit = r.total_revenue - r.total_expenses;
    if (profit < r.total_wages_paid) {
      consecutiveDeficitMonths++;
    } else {
      break; // 連続が途切れたら終了
    }
  }

  return {
    consecutiveDeficitMonths,
    hasDeficitRisk: consecutiveDeficitMonths >= 3,
    recentActivities: rows,
  };
}

import { getIntegrityErrors } from "@/lib/compliance/recordIntegrityChecker";

export async function toggleLockProvisionRecord(clientId: string, targetMonth: string, lockStatus: boolean) {
  const role = await getRole();
  if (!role) throw new Error("認証されていません");
  if (role !== 'admin' && role !== 'service_manager') {
    throw new Error("管理者またはサビ管のみロック操作が可能です");
  }

  // 実績確定（ロック）時のみ整合性チェックを行う
  if (lockStatus) {
    const errors = getIntegrityErrors(targetMonth, clientId);
    if (errors && errors.length > 0) {
      // 最初の1件のエラー内容を返す
      const firstError = errors[0];
      return { 
        success: false, 
        error: `勤怠と支援記録の不整合があるため確定できません。\n[${firstError.recordDate}] ${firstError.message}\nダッシュボードから詳細を確認・修正してください。` 
      };
    }
  }

  try {
    const isLocked = lockStatus ? 1 : 0;
    db.prepare(`
      UPDATE monthly_provision_records 
      SET is_locked = ?, updated_at = CURRENT_TIMESTAMP
      WHERE client_id = ? AND target_month = ?
    `).run(isLocked, clientId, targetMonth);
    
    return { success: true };
  } catch (error: any) {
    console.error("Failed to toggle lock:", error);
    return { success: false, error: error.message };
  }
}

export async function calculatePreviousYearAverageHours() {
  try {
    const role = await getRole();
    if (!role) return 0;

    const today = new Date();
    const currentMonth = today.getMonth() + 1;
    const currentYear = today.getFullYear();
    
    // 今年度の起算年は、4月以降なら今年、1～3月なら前年
    // 前年度はそのさらに1年前
    const previousFyStartYear = currentMonth >= 4 ? currentYear - 1 : currentYear - 2;
    
    const previousFyStartDate = `${previousFyStartYear}-04-01`;
    const previousFyEndDate = `${previousFyStartYear + 1}-04-01`; // exclusive

    const attendances = db.prepare(`
      SELECT a.status, a.clock_in, a.clock_out, a.break_time_minutes, c.weekly_scheduled_hours, c.regular_holidays
      FROM attendance_records a
      JOIN clients c ON a.client_id = c.id
      WHERE a.record_date >= ? AND a.record_date < ?
        AND a.deleted_at IS NULL
        AND c.deleted_at IS NULL
        AND (
          c.service_start_date IS NULL 
          OR c.service_start_date < strftime('%Y-%m-01', a.record_date) 
          OR c.service_start_date > date(strftime('%Y-%m-01', a.record_date), '+1 month', '-1 day')
        )
        AND (
          c.contract_end_date IS NULL 
          OR c.contract_end_date < strftime('%Y-%m-01', a.record_date) 
          OR c.contract_end_date > date(strftime('%Y-%m-01', a.record_date), '+1 month', '-1 day')
        )
    `).all(previousFyStartDate, previousFyEndDate) as any[];

    let totalMinutes = 0;
    let totalWorkDays = 0;

    if (attendances) {
      for (const record of attendances) {
        if (record.status === 'pto') {
          // 厚労省ルール: 年次有給休暇取得日は契約上の所定労働時間をみなし労働時間として合算
          let scheduledDays = 5;
          try {
            const holidays = record.regular_holidays ? JSON.parse(record.regular_holidays) : [];
            if (Array.isArray(holidays)) {
              scheduledDays = Math.max(1, 7 - holidays.length);
            }
          } catch {
            scheduledDays = 5;
          }
          const dailyHours = record.weekly_scheduled_hours && record.weekly_scheduled_hours > 0 
            ? (record.weekly_scheduled_hours / scheduledDays) 
            : 4.0; // デフォルト4時間
          totalMinutes += Math.round(dailyHours * 60);
          totalWorkDays++;
        } else if (record.clock_in && record.clock_out) {
          const [inH, inM] = record.clock_in.split(':').map(Number);
          const [outH, outM] = record.clock_out.split(':').map(Number);
          
          let outTotalM = outH * 60 + outM;
          const inTotalM = inH * 60 + inM;
          if (outTotalM < inTotalM) {
            outTotalM += 24 * 60;
          }
          
          const mins = outTotalM - inTotalM - (record.break_time_minutes || 0);
          if (mins > 0) {
            totalMinutes += mins;
            totalWorkDays++;
          }
        }
      }
    }

    const totalHours = totalMinutes / 60;
    // 厚労省告示基準: 1日当たり平均労働時間 = 前年度延べ労働時間 ÷ 前年度延べ就労日数
    const averageHours = totalWorkDays > 0 ? (totalHours / totalWorkDays) : 0;
    return Number(averageHours.toFixed(1));
  } catch (e) {
    console.error("Failed to calculate previous year average hours:", e);
    return 0;
  }
}

// 生産活動収支の自動判定ロジック
export async function evaluateProductionActivity(targetMonth: string) {
  const [targetY, targetM] = targetMonth.split('-').map(Number);

  // カレンダー年月ベースで直近3年度（各12ヶ月）の月キーセットを生成
  // 年1: 直近12ヶ月 (0〜11ヶ月前)
  // 年2: 12〜23ヶ月前
  // 年3: 24〜35ヶ月前
  const getMonthsForPeriod = (startOffsetMonths: number, count: number): Set<string> => {
    const set = new Set<string>();
    for (let i = startOffsetMonths; i < startOffsetMonths + count; i++) {
      const d = new Date(targetY, targetM - 1 - i, 1);
      const ym = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
      set.add(ym);
    }
    return set;
  };

  const period1Months = getMonthsForPeriod(0, 12);
  const period2Months = getMonthsForPeriod(12, 12);
  const period3Months = getMonthsForPeriod(24, 12);

  // targetMonth以前のデータを取得（最大60件）
  const activities = db.prepare(`
    SELECT * FROM production_activities
    WHERE target_month <= ?
    ORDER BY target_month DESC
    LIMIT 60
  `).all(targetMonth) as any[];

  if (activities.length === 0) {
    return "profit_exceeds_wages"; // データがない場合はデフォルト
  }

  // カレンダー基準で各年度にデータを振り分け（欠損月があっても過去年と混ざらない）
  const period1 = activities.filter(a => period1Months.has(a.target_month));
  const period2 = activities.filter(a => period2Months.has(a.target_month));
  const period3 = activities.filter(a => period3Months.has(a.target_month));

  const periods = [period1, period2, period3];

  let consecutiveDeficitYears = 0;

  for (const period of periods) {
    if (period.length === 0) break; // その年のデータが全くない場合はループ終了

    let sumProfit = 0;
    let sumWages = 0;
    for (const act of period) {
      sumProfit += (act.total_revenue - act.total_expenses);
      sumWages += act.total_wages_paid;
    }

    if (sumProfit < sumWages) {
      consecutiveDeficitYears++;
    } else {
      break; // 赤字が途切れたら連続赤字カウント終了
    }
  }

  if (consecutiveDeficitYears >= 3) return "deficit_3_years";
  if (consecutiveDeficitYears === 2) return "deficit_2_years";
  if (consecutiveDeficitYears === 1) return "deficit_1_year";
  return "profit_exceeds_wages";
}

export type SimulationInput = {
  version: string;
  averageHours: number;
  productionActivity: string;
  diversity: string;
  supportImprovement: string;
  regionalCooperation: string;
  managementImprovement: string;
  knowledgeSkill: string;
};

export async function runSimulation(input: SimulationInput) {
  try {
    const role = await getRole();
    if (!role) return { success: false, error: '認証されていません' };

    const rules = loadMasterData(input.version, 'score_rules');
    const unitVals = loadMasterData(input.version, 'unit_values');

    let totalScore = 0;
    const details = {
      working_hours: 0,
      production_activity: 0,
      diversity: 0,
      support_improvement: 0,
      regional_cooperation: 0,
      management_improvement: 0,
      knowledge_skill: 0
    };

    // 1. 労働時間のスコア判定
    const whRules = rules.categories.working_hours;
    for (const rule of whRules) {
      if ((rule.min === null || input.averageHours >= rule.min) && 
          (rule.max === null || input.averageHours < rule.max)) {
        details.working_hours = rule.score;
        break;
      }
    }

    // 2. その他の項目
    const calcCategory = (catKey: keyof typeof details, inputVal: string) => {
      const catRules = rules.categories[catKey];
      const match = catRules.find((r: any) => r.condition === inputVal);
      if (match) {
        details[catKey] = match.score;
      }
    };

    calcCategory('production_activity', input.productionActivity);
    calcCategory('diversity', input.diversity);
    calcCategory('support_improvement', input.supportImprovement);
    calcCategory('regional_cooperation', input.regionalCooperation);
    calcCategory('management_improvement', input.managementImprovement);
    calcCategory('knowledge_skill', input.knowledgeSkill);

    totalScore = Object.values(details).reduce((a, b) => a + b, 0);

    // 3. 単位数の判定
    const facility = db.prepare(`SELECT * FROM facility_settings WHERE id = 'default'`).get() as any || {};
    
    // Convert capacity_class and staffing_class to JSON keys
    let capacityKey = 'capacity_20_under';
    if (facility.capacity_class === '21人以上40人以下') capacityKey = 'capacity_21_40';
    else if (facility.capacity_class === '41人以上60人以下') capacityKey = 'capacity_41_60';
    else if (facility.capacity_class === '61人以上80人以下') capacityKey = 'capacity_61_80';
    else if (facility.capacity_class === '81人以上') capacityKey = 'capacity_81_over';

    let staffingKey = 'staff_ratio_7_5';
    if (facility.staffing_class === '10:1') staffingKey = 'staff_ratio_10';

    let baseUnit = 0;
    let unitMatrix = unitVals.unit_matrices?.[capacityKey]?.[staffingKey];
    
    // Fallback if the JSON master is incomplete
    if (!unitMatrix) {
      unitMatrix = unitVals.unit_matrices?.['capacity_20_under']?.['staff_ratio_7_5'] || [];
    }

    for (const row of unitMatrix) {
      if ((row.min_score === null || totalScore >= row.min_score) && 
          (row.max_score === null || totalScore < row.max_score)) {
        baseUnit = row.unit;
        break;
      }
    }

    return {
      success: true,
      totalScore,
      details,
      baseUnit,
      alerts: {
        deficitWarning: details.production_activity < 0,
        managementWarning: details.management_improvement === -50
      }
    };
  } catch (error: any) {
    console.error("Simulation failed:", error);
    return { success: false, error: "シミュレーションに失敗しました。マスターデータを確認してください。" };
  }
}

export async function getCurrentFiscalYearScoreProgress(): Promise<{
  success: boolean;
  data?: FiscalYearScoreProgress;
  error?: string;
}> {
  try {
    const role = await getRole();
    if (!role) return { success: false, error: '認証されていません' };
    const today = new Date();
    const currentMonth = today.getMonth() + 1; // 1-12
    const currentYear = today.getFullYear();

    // 今年度の起算年は、4月以降なら今年、1～3月なら前年
    const currentFyStartYear = currentMonth >= 4 ? currentYear : currentYear - 1;
    const startMonthStr = `${currentFyStartYear}-04`;
    const currentMonthStr = `${today.getFullYear()}-${String(currentMonth).padStart(2, '0')}`;

    // 4月から当月までの月リストを作成
    const monthsList: { year: number; month: number; ym: string; label: string }[] = [];
    const elapsedMonthsCount = currentMonth >= 4 ? (currentMonth - 4 + 1) : (currentMonth + 9);

    for (let m = 0; m < elapsedMonthsCount; m++) {
      const targetYear = currentFyStartYear + (m < 9 ? 0 : 1);
      const targetMonth = (m + 3) % 12 + 1;
      const ym = `${targetYear}-${String(targetMonth).padStart(2, '0')}`;
      monthsList.push({
        year: targetYear,
        month: targetMonth,
        ym,
        label: `${targetMonth}月`
      });
    }

    const allClients = db.prepare(`SELECT * FROM clients WHERE deleted_at IS NULL`).all() as any[];

    // 各月の労働時間と月末在籍者・延べ就労日数を計算
    const monthlyProgress: MonthlyHoursProgress[] = [];
    let cumulativeMinutes = 0;
    let cumulativeWorkDays = 0;
    let cumulativeSumEndOfMonthClients = 0;

    for (const mInfo of monthsList) {
      const startOfMonth = `${mInfo.ym}-01`;
      const endOfMonthObj = new Date(mInfo.year, mInfo.month, 0);
      const endOfMonth = `${endOfMonthObj.getFullYear()}-${String(endOfMonthObj.getMonth() + 1).padStart(2, '0')}-${String(endOfMonthObj.getDate()).padStart(2, '0')}`;
      const nextMonthStartDate = new Date(mInfo.year, mInfo.month, 1);
      const nextMonthStartStr = `${nextMonthStartDate.getFullYear()}-${String(nextMonthStartDate.getMonth() + 1).padStart(2, '0')}-01`;

      const attendances = db.prepare(`
        SELECT a.status, a.clock_in, a.clock_out, a.break_time_minutes, c.weekly_scheduled_hours, c.regular_holidays
        FROM attendance_records a
        JOIN clients c ON a.client_id = c.id
        WHERE a.record_date >= ? AND a.record_date < ?
          AND a.deleted_at IS NULL
      `).all(startOfMonth, nextMonthStartStr) as any[];

      let mMinutes = 0;
      let mWorkDays = 0;
      if (attendances) {
        for (const rec of attendances) {
          if (rec.status === 'pto') {
            let scheduledDays = 5;
            try {
              const holidays = rec.regular_holidays ? JSON.parse(rec.regular_holidays) : [];
              if (Array.isArray(holidays)) scheduledDays = Math.max(1, 7 - holidays.length);
            } catch {
              scheduledDays = 5;
            }
            const dailyHours = rec.weekly_scheduled_hours && rec.weekly_scheduled_hours > 0 
              ? (rec.weekly_scheduled_hours / scheduledDays) 
              : 4.0;
            mMinutes += Math.round(dailyHours * 60);
            mWorkDays++;
          } else if (rec.clock_in && rec.clock_out) {
            const [inH, inM] = rec.clock_in.split(':').map(Number);
            const [outH, outM] = rec.clock_out.split(':').map(Number);
            let outTotalM = outH * 60 + outM;
            const inTotalM = inH * 60 + inM;
            if (outTotalM < inTotalM) outTotalM += 24 * 60;
            const mins = outTotalM - inTotalM - (rec.break_time_minutes || 0);
            if (mins > 0) {
              mMinutes += mins;
              mWorkDays++;
            }
          }
        }
      }

      // 当月末時点での有効在籍者数（除外要件考慮）
      const activeClients = allClients.filter(c => {
        if (!c.service_start_date) return false;
        if (c.service_start_date > endOfMonth) return false;
        if (c.contract_end_date && c.contract_end_date < endOfMonth) return false;
        // 当月中開始・退所除外
        if (c.service_start_date >= startOfMonth && c.service_start_date <= endOfMonth) return false;
        if (c.contract_end_date && c.contract_end_date >= startOfMonth && c.contract_end_date <= endOfMonth) return false;
        return true;
      });

      const mActiveCount = activeClients.length;
      // 厚労省告示基準: 1日当たりの平均労働時間 = 総労働時間 ÷ 延べ就労日数
      const mAvgHours = mWorkDays > 0 ? (mMinutes / 60) / mWorkDays : 0;

      monthlyProgress.push({
        month: mInfo.ym,
        monthLabel: mInfo.label,
        totalMinutes: mMinutes,
        totalHours: Number((mMinutes / 60).toFixed(1)),
        activeClientCount: mActiveCount,
        averageHours: Number(mAvgHours.toFixed(1))
      });

      cumulativeMinutes += mMinutes;
      cumulativeWorkDays += mWorkDays;
      cumulativeSumEndOfMonthClients += mActiveCount;
    }

    const averageHoursToDate = cumulativeWorkDays > 0
      ? Number(((cumulativeMinutes / 60) / cumulativeWorkDays).toFixed(1))
      : 0;

    // ルールマスターの読み込み
    const rules = loadMasterData('2024', 'score_rules');
    const unitVals = loadMasterData('2024', 'unit_values');

    // 1. 労働時間スコア
    let workingHoursScore = 0;
    for (const rule of rules.categories.working_hours) {
      if ((rule.min === null || averageHoursToDate >= rule.min) && 
          (rule.max === null || averageHoursToDate < rule.max)) {
        workingHoursScore = rule.score;
        break;
      }
    }

    // 2. 生産活動収支
    const prodActivities = db.prepare(`
      SELECT * FROM production_activities
      WHERE target_month >= ? AND target_month <= ?
    `).all(startMonthStr, currentMonthStr) as any[];

    let totalRevenue = 0;
    let totalExpenses = 0;
    let totalWages = 0;
    for (const pa of prodActivities) {
      totalRevenue += (pa.total_revenue || 0);
      totalExpenses += (pa.total_expenses || 0);
      totalWages += (pa.total_wages_paid || 0);
    }
    const netProfit = totalRevenue - totalExpenses;
    const wageCoverageDiff = netProfit - totalWages;

    let prodStatus = "profit_exceeds_wages";
    let prodScore = 60;
    let prodLabel = "収支が賃金総額を上回る（要件達成）";

    if (prodActivities.length > 0 && wageCoverageDiff < 0) {
      // 過去3年判定
      const pastEval = await evaluateProductionActivity(currentMonthStr);
      if (pastEval === 'deficit_3_years') {
        prodStatus = 'deficit_3_years';
        prodScore = -20;
        prodLabel = '3年以上連続で下回る';
      } else if (pastEval === 'deficit_2_years') {
        prodStatus = 'deficit_2_years';
        prodScore = -10;
        prodLabel = '2年連続で下回る';
      } else {
        prodStatus = 'deficit_1_year';
        prodScore = -5;
        prodLabel = '1年連続で下回る';
      }
    }

    // 3. その他評価項目（直近確定値またはデフォルト基準値）
    const latestFacilityUnit = db.prepare(`
      SELECT * FROM monthly_facility_units
      ORDER BY target_month DESC
      LIMIT 1
    `).get() as any;

    const otherCategories = {
      diversityScore: latestFacilityUnit?.diversity_score ?? 10,
      supportImprovementScore: latestFacilityUnit?.support_improvement_score ?? 15,
      regionalCooperationScore: latestFacilityUnit?.regional_cooperation_score ?? 10,
      managementImprovementScore: latestFacilityUnit?.management_improvement_score ?? 0,
      knowledgeSkillScore: latestFacilityUnit?.knowledge_skill_score ?? 0
    };

    const currentTotalScore = workingHoursScore + prodScore +
      otherCategories.diversityScore +
      otherCategories.supportImprovementScore +
      otherCategories.regionalCooperationScore +
      otherCategories.managementImprovementScore +
      otherCategories.knowledgeSkillScore;

    // 4. 単位数の判定
    const facility = db.prepare(`SELECT * FROM facility_settings WHERE id = 'default'`).get() as any || {};
    let capacityKey = 'capacity_20_under';
    if (facility.capacity_class === '21人以上40人以下') capacityKey = 'capacity_21_40';
    else if (facility.capacity_class === '41人以上60人以下') capacityKey = 'capacity_41_60';
    else if (facility.capacity_class === '61人以上80人以下') capacityKey = 'capacity_61_80';
    else if (facility.capacity_class === '81人以上') capacityKey = 'capacity_81_over';

    let staffingKey = 'staff_ratio_7_5';
    if (facility.staffing_class === '10:1') staffingKey = 'staff_ratio_10';

    let unitMatrix = unitVals.unit_matrices?.[capacityKey]?.[staffingKey] || 
                     unitVals.unit_matrices?.['capacity_20_under']?.['staff_ratio_7_5'] || [];

    let currentBaseUnit = unitMatrix[unitMatrix.length - 1]?.unit ?? 325;
    let currentMatrixIndex = -1;

    for (let i = 0; i < unitMatrix.length; i++) {
      const row = unitMatrix[i];
      if ((row.min_score === null || currentTotalScore >= row.min_score) && 
          (row.max_score === null || currentTotalScore < row.max_score)) {
        currentBaseUnit = row.unit;
        currentMatrixIndex = i;
        break;
      }
    }

    // 5. 上位ランク情報の算出
    let nextRankInfo = null;
    if (currentMatrixIndex > 0) {
      const higherRow = unitMatrix[currentMatrixIndex - 1];
      const targetScore = higherRow.min_score || 0;
      const scoreNeeded = targetScore - currentTotalScore;

      // 労働時間であと何時間伸ばせば届くか
      let adviceMessage = `あと ${scoreNeeded} 点で上位報酬区分（${higherRow.unit}単位/日）へランクアップします。`;
      if (scoreNeeded <= 10) {
        adviceMessage += " 多様な働き方（資格取得支援やテレワーク対応）の追加認定や、平均労働時間のわずかな延長（10〜15分程度）で達成可能です。";
      } else if (scoreNeeded <= 25) {
        adviceMessage += " 平均労働時間を1段階引き上げる（例: 5時間台→6時間台）ことで一気に達成できます。";
      } else {
        adviceMessage += " 生産活動収支の黒字化（+60点）および平均労働時間の延伸を組み合わせることで大幅な増収が見込めます。";
      }

      nextRankInfo = {
        targetUnit: higherRow.unit,
        targetScore,
        scoreNeeded,
        adviceMessage
      };
    }

    // 6. 経営インパクト試算
    // 在籍利用者数（または定員20名想定）
    const activeClientCount = Math.max(1, allClients.filter(c => !c.contract_end_date || c.contract_end_date >= currentMonthStr).length);
    const monthlyOperatingDays = 22; // 標準稼働日数
    const unitPrice = getAreaUnitCost(facility?.area_cost_grade || 'other'); // 事業所の地域区分単価（動的取得）

    const currentMonthlyRevenue = Math.round(currentBaseUnit * unitPrice * activeClientCount * monthlyOperatingDays);
    const nextRankUnit = nextRankInfo?.targetUnit || currentBaseUnit;
    const nextRankMonthlyRevenue = Math.round(nextRankUnit * unitPrice * activeClientCount * monthlyOperatingDays);
    const monthlyDifference = nextRankMonthlyRevenue - currentMonthlyRevenue;
    const annualDifference = monthlyDifference * 12;

    return {
      success: true,
      data: {
        fiscalYear: currentFyStartYear,
        startMonth: startMonthStr,
        currentMonth: currentMonthStr,
        elapsedMonths: elapsedMonthsCount,
        remainingMonths: 12 - elapsedMonthsCount,
        averageHoursToDate,
        workingHoursScore,
        monthlyProgress,
        productionSummary: {
          totalRevenue,
          totalExpenses,
          totalWages,
          netProfit,
          wageCoverageDiff,
          status: prodStatus,
          score: prodScore,
          label: prodLabel
        },
        otherCategories,
        currentTotalScore,
        currentBaseUnit,
        capacityClass: facility.capacity_class || '20人以下',
        staffingClass: facility.staffing_class || '7.5:1',
        nextRankInfo,
        revenueImpact: {
          activeClientCount,
          monthlyOperatingDays,
          unitPrice,
          currentMonthlyRevenue,
          nextRankMonthlyRevenue,
          monthlyDifference,
          annualDifference
        }
      }
    };
  } catch (error: any) {
    console.error('Failed to get current fiscal year score progress:', error);
    return { success: false, error: error.message };
  }
}

export async function getTypeBWageSimulationData() {
  try {
    const role = await getRole();
    if (!role) throw new Error('認証されていません');

    const facility = db.prepare(`SELECT * FROM facility_settings WHERE id = 'default'`).get() as any || {};
    const unitPrice = getAreaUnitCost(facility.region_class || '2級地');

    // 前年度の期間算出（4月〜翌3月）
    const today = new Date();
    const curYear = today.getFullYear();
    const curMonth = today.getMonth() + 1;
    const startYear = curMonth >= 4 ? curYear - 1 : curYear - 2;
    const startMonthStr = `${startYear}-04`;
    const endMonthStr = `${startYear + 1}-03`;

    // 1. payslips / monthly_payslips から前年度の工賃支払総額を集計（厚労省指定基準: 実費弁償の通勤手当等を除外した純工賃を集計）
    let wageStats = db.prepare(`
      SELECT 
        COALESCE(SUM(mp.base_salary + COALESCE(mp.ability_allowance, 0)), 0) as total_wage,
        COUNT(DISTINCT p.client_id) as unique_clients
      FROM payslips p
      JOIN monthly_payslips mp ON p.client_id = mp.client_id AND p.target_month = mp.target_month
      WHERE p.target_month >= ? AND p.target_month <= ?
    `).get(startMonthStr, endMonthStr) as any;

    let totalWagePaid = wageStats?.total_wage || 0;
    if (totalWagePaid === 0) {
      // 直近登録された全月データからフォールバック集計
      const fallbackStats = db.prepare(`
        SELECT COALESCE(SUM(mp.base_salary + COALESCE(mp.ability_allowance, 0)), 0) as total_wage 
        FROM payslips p
        JOIN monthly_payslips mp ON p.client_id = mp.client_id AND p.target_month = mp.target_month
      `).get() as any;
      totalWagePaid = fallbackStats?.total_wage || 0;
    }

    // 2. 開所日数と延べ利用人数を集計
    let attendanceStats = db.prepare(`
      SELECT 
        COUNT(DISTINCT record_date) as facility_open_days,
        COUNT(*) as total_client_attendances
      FROM attendance_records
      WHERE record_date >= ? AND record_date < ?
        AND status = 'present' AND deleted_at IS NULL
    `).get(`${startYear}-04-01`, `${startYear + 1}-04-01`) as any;

    let facilityOpenDays = attendanceStats?.facility_open_days || 0;
    let totalClientAttendances = attendanceStats?.total_client_attendances || 0;

    if (facilityOpenDays === 0) {
      const allAttendance = db.prepare(`
        SELECT 
          COUNT(DISTINCT record_date) as open_days,
          COUNT(*) as attendances
        FROM attendance_records
        WHERE status = 'present' AND deleted_at IS NULL
      `).get() as any;
      facilityOpenDays = allAttendance?.open_days || 240;
      totalClientAttendances = allAttendance?.attendances || 3600;
    }

    // 利用者数
    const activeClientsCount = (db.prepare(`SELECT COUNT(*) as count FROM clients WHERE deleted_at IS NULL`).get() as any)?.count || 15;

    // 厚労省最新算定式による平均工賃月額算出
    const avgWageResult = calculateTypeBAverageWage({
      totalWagePaid,
      totalClientAttendances,
      facilityOpenDays: Math.max(1, facilityOpenDays),
      operatingMonths: 12
    });

    const staffRatio = (facility.staff_ratio || '6:1') as '6:1' | '7.5:1' | '10:1';
    const currentServiceFee = determineTypeBServiceFee({
      averageMonthlyWage: avgWageResult.averageMonthlyWage,
      staffRatio,
      capacity: 20
    });

    return {
      success: true,
      data: {
        fiscalYearPeriod: `${startYear}年度 (${startMonthStr}〜${endMonthStr})`,
        totalWagePaid,
        facilityOpenDays,
        totalClientAttendances,
        averageUsersPerDay: avgWageResult.averageUsersPerDay,
        averageMonthlyWage: avgWageResult.averageMonthlyWage,
        isBelowMinimum3000: avgWageResult.isBelowMinimum3000,
        staffRatio,
        currentServiceFee,
        unitPrice,
        activeClientsCount,
        brackets: TYPE_B_WAGE_BRACKETS,
        additions: TYPE_B_ADDITIONS,
        fixedFees: TYPE_B_FIXED_FEES
      }
    };
  } catch (error: any) {
    console.error('Failed to get type B wage simulation data:', error);
    return { success: false, error: error.message };
  }
}
