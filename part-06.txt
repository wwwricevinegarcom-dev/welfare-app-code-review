import { db } from "@/lib/db";
import { getRole } from "@/lib/auth";
import { redirect } from "next/navigation";
import { TransportationRecordReport, TransRecordItem } from "./TransportationRecordReport";

export const dynamic = 'force-dynamic';

export default async function TransportationRecordsPage({
  searchParams,
}: {
  searchParams: Promise<{ month?: string }>;
}) {
  const role = await getRole();
  if (!role) {
    redirect("/login");
  }

  const resolvedParams = await searchParams;
  const today = new Date();
  const defaultMonth = `${today.getFullYear()}-${String(today.getMonth() + 1).padStart(2, '0')}`;
  const targetMonth = resolvedParams.month || defaultMonth;

  const facility = db.prepare(`SELECT * FROM facility_settings WHERE id = 'default'`).get() as any || {
    facility_name: '就労継続支援A型事業所',
    manager_name: '管理者',
  };

  // 送迎利用実績を取得 (transportation_used = 1 または pickup/dropoff 使用)
  const rows = db.prepare(`
    SELECT 
      ar.id,
      ar.client_id,
      c.full_name as client_name,
      c.recipient_number,
      ar.record_date,
      ar.clock_in,
      ar.clock_out,
      ar.transportation_used,
      ar.pickup_used,
      ar.dropoff_used,
      ar.transportation_service_used,
      ar.pickup_service,
      ar.dropoff_service,
      ar.pickup_driver,
      ar.dropoff_driver
    FROM attendance_records ar
    JOIN clients c ON ar.client_id = c.id
    WHERE ar.record_date LIKE ? 
      AND ar.deleted_at IS NULL
      AND (ar.transportation_used = 1 OR ar.pickup_used = 1 OR ar.dropoff_used = 1 OR ar.pickup_service = 1 OR ar.dropoff_service = 1 OR ar.transportation_service_used = 1)
    ORDER BY ar.record_date ASC, c.full_name ASC
  `).all(`${targetMonth}-%`) as any[];

  const records: TransRecordItem[] = rows.map(r => ({
    id: r.id,
    client_id: r.client_id,
    client_name: r.client_name,
    recipient_number: r.recipient_number,
    record_date: r.record_date,
    clock_in: r.clock_in,
    clock_out: r.clock_out,
    pickup_used: (r.pickup_used === 1 || r.pickup_service === 1) ? 1 : ((r.transportation_used === 1 || r.transportation_service_used === 1) ? 1 : 0),
    dropoff_used: (r.dropoff_used === 1 || r.dropoff_service === 1) ? 1 : ((r.transportation_used === 1 || r.transportation_service_used === 1) ? 1 : 0),
    pickup_driver: r.pickup_driver,
    dropoff_driver: r.dropoff_driver,
  }));

  return (
    <div className="min-h-screen bg-slate-50 py-6 print:py-0 print:bg-white">
      <TransportationRecordReport
        facility={facility}
        targetYearMonth={targetMonth}
        records={records}
      />
    </div>
  );
}

  departments?: any[],
  todayStr: string,
  initialFilter?: 'all' | 'active' | 'terminated' | 'transitioned' | 'suspended' | 'expired',
  alertDaysCertificate?: number
}) {
  const [searchQuery, setSearchQuery] = useState('');
  const [filterType, setFilterType] = useState<'all' | 'active' | 'terminated' | 'transitioned' | 'suspended' | 'expired'>(initialFilter);

  const [sortOrder, setSortOrder] = useState<'hire_asc' | 'hire_desc' | 'name_asc' | 'created_desc'>('hire_asc');
  const todayDate = useMemo(() => new Date(todayStr), [todayStr]);

  const filteredClients = useMemo(() => {
    return initialClients.filter(client => {
      const nameMatch = client.full_name?.toLowerCase().includes(searchQuery.toLowerCase());
      const numberMatch = client.recipient_number?.toLowerCase().includes(searchQuery.toLowerCase());
      const matchesSearch = !searchQuery || nameMatch || numberMatch;

      if (!matchesSearch) return false;

      // 部署フィルター
      if (selectedDepartment !== 'all') {
        if (selectedDepartment === 'unassigned') {
          if (client.department_id) return false;
        } else {
          if (client.department_id !== selectedDepartment) return false;
        }
      }

      const validToDate = client.valid_to ? new Date(client.valid_to) : null;
      const isExpired = validToDate ? validToDate < todayDate : false;
      const daysUntilExpiry = validToDate ? Math.ceil((validToDate.getTime() - todayDate.getTime()) / (1000 * 60 * 60 * 24)) : null;
      const isExpiringSoon = !isExpired && daysUntilExpiry !== null && daysUntilExpiry <= alertDaysCertificate;
      const status = client.status || (client.transition_date ? 'transitioned' : (client.contract_end_date && new Date(client.contract_end_date) <= todayDate ? 'terminated' : 'active'));

      if (filterType === 'expired') return isExpired || isExpiringSoon;
      if (filterType === 'active') return status === 'active';
      if (filterType === 'terminated') return status === 'terminated';
      if (filterType === 'transitioned') return status === 'transitioned';
      if (filterType === 'suspended') return status === 'suspended';

      return true;

        const dateB = b.contract_date || b.service_start_date || '';
        if (!dateA && !dateB) return (a.full_name || '').localeCompare(b.full_name || '', 'ja');
        if (!dateA) return 1;
        if (!dateB) return -1;
        const cmp = dateA.localeCompare(dateB);
        return sortOrder === 'hire_asc' ? cmp : -cmp;
      }
      if (sortOrder === 'name_asc') {
        return (a.full_name || '').localeCompare(b.full_name || '', 'ja');

  const terminatedCount = initialClients.filter(c => c.status === 'terminated' || (!c.status && c.contract_end_date && new Date(c.contract_end_date) <= new Date(todayStr))).length;

    if (isInactive) return false;
    if (!c.valid_to) return false;
    const vDate = new Date(c.valid_to);
    const tDate = new Date(todayStr);

  }).length;

              const isExpired = !isInactive && validToDate ? validToDate < todayDate : false;
              const daysUntilExpiry = validToDate ? Math.ceil((validToDate.getTime() - todayDate.getTime()) / (1000 * 60 * 60 * 24)) : null;
              const isExpiringSoon = !isInactive && !isExpired && daysUntilExpiry !== null && daysUntilExpiry <= alertDaysCertificate;

                  <TableCell className="py-4 text-base">{client.allowance_days}日 / 月</TableCell>

                        href={`/clients/${client.id}/documents`} 
                        className={cn(buttonVariants({ variant: "default", size: "sm" }), "h-9 px-3 gap-1 bg-primary text-white hover:bg-primary/90")}
                      >
                        <FileText className="h-4 w-4" />
                        <span>書類カルテ</span>
                      </Link>
                      <Link 
                        href={`/clients/${client.id}/edit`} 
                        className={cn(buttonVariants({ variant: "outline", size: "sm" }), "h-9 px-3 gap-1")}
                      >
                        <Edit className="h-4 w-4 text-muted-foreground" />
                        <span>編集</span>
                      </Link>
                      <DeleteClientButton id={client.id} name={client.full_name} />

                  </TableCell>

            })}
          </TableBody>
        </Table>
        </div>
      ) : (
        <div className="p-12 text-center text-muted-foreground">
          {searchQuery || filterType !== 'all' ? '該当する利用者が条件に見つかりませんでした。' : '利用者が登録されていません。'}
        </div>
      )}
    </div>
  );
}

'use client';

import { useState } from 'react';
import { Button } from "@/components/ui/button";

  DialogFooter

import { Upload, Download, FileSpreadsheet, AlertCircle, CheckCircle2, Loader2 } from "lucide-react";
import { importClientsFromCsv, CsvClientImportItem } from "./actions";
import { toast } from "sonner";

export function CsvImportModal() {
  const [isOpen, setIsOpen] = useState(false);
  const [parsedData, setParsedData] = useState<CsvClientImportItem[]>([]);
  const [fileName, setFileName] = useState<string>('');
  const [isProcessing, setIsProcessing] = useState(false);
  const [parseErrors, setParseErrors] = useState<string[]>([]);

  // 雛形CSVのダウンロード（Excelで文字化けしないようBOM付与）
  const handleDownloadTemplate = () => {
    const headers = [
      "氏名", "フリガナ", "受給者証番号(10桁)", "負担上限月額", "有効期限開始(YYYY-MM-DD)", "有効期限終了(YYYY-MM-DD)",
      "支給量日数", "障害区分", "障害種別", "時給", "週所定労働時間",
      "所定始業(HH:MM)", "所定終業(HH:MM)", "休憩分", "生年月日(YYYY-MM-DD)", "性別",
      "電話番号", "住所", "サービス開始日(YYYY-MM-DD)"
    ];
    
    const sampleRows = [
      [
        "神戸 太郎", "コウベ タロウ", "1234567890", "0", "2026-04-01", "2027-03-31",
        "23", "区分3", "精神障害", "1052", "20",
        "09:00", "16:00", "60", "1995-05-15", "男性",
        "090-1234-5678", "兵庫県神戸市中央区...", "2026-04-01"
      ],
      [
        "三宮 花子", "サンノミヤ ハナコ", "2345678901", "9300", "2026-04-01", "2027-03-31",
        "22", "区分2", "知的障害", "1052", "20",
        "09:30", "15:30", "60", "2000-11-20", "女性",
        "080-9876-5432", "兵庫県神戸市灘区...", "2026-04-01"
      ]
    ];

    const csvContent = "\uFEFF" + [
      headers.join(","),
      ...sampleRows.map(r => r.map(c => `"${c}"`).join(","))
    ].join("\r\n");

    const blob = new Blob([csvContent], { type: "text/csv;charset=utf-8;" });
    const url = URL.createObjectURL(blob);
    const link = document.createElement("a");
    link.href = url;
    link.setAttribute("download", "利用者一括登録テンプレート.csv");
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
  };

  // CSVファイルの読み込みと簡易パース
  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;

    setFileName(file.name);
    const reader = new FileReader();

    reader.onload = (event) => {
      const text = event.target?.result as string;
      if (!text) return;

      const lines = text.split(/\r?\n/).map(l => l.trim()).filter(Boolean);
      if (lines.length < 2) {
        setParseErrors(["有効なデータ行がありません（ヘッダー行＋1件以上のデータが必要です）"]);
        setParsedData([]);
        return;
      }

      // 堅牢なCSV行パーサー（空フィールド対応・ダブルクォートエスケープ対応）
      const parseCsvLine = (line: string): string[] => {
        const result: string[] = [];
        let current = '';
        let inQuotes = false;
        for (let j = 0; j < line.length; j++) {
          const char = line[j];
          if (char === '"') {
            if (inQuotes && line[j + 1] === '"') {
              current += '"';
              j++;
            } else {
              inQuotes = !inQuotes;
            }
          } else if (char === ',' && !inQuotes) {
            result.push(current.trim());
            current = '';
          } else {
            current += char;
          }
        }
        result.push(current.trim());
        return result;
      };

      const headerRow = parseCsvLine(lines[0]);
      const hasFuriganaHeader = headerRow.some(h => h.includes('フリガナ') || h.includes('ふりがな'));

      // ヘッダー行を無視してパース
      const items: CsvClientImportItem[] = [];
      const errors: string[] = [];

      for (let i = 1; i < lines.length; i++) {
        const cleaned = parseCsvLine(lines[i]);
        const fullName = cleaned[0];
        if (!fullName) {
          errors.push(`${i + 1}行目: 氏名が未入力です`);
          continue;
        }

        if (hasFuriganaHeader) {
          items.push({
            fullName,
            furigana: cleaned[1] || undefined,
            recipientNumber: cleaned[2] || undefined,
            paymentLimit: cleaned[3] ? parseInt(cleaned[3], 10) : 0,
            validFrom: cleaned[4] || undefined,
            validTo: cleaned[5] || undefined,
            allowanceDays: cleaned[6] ? parseInt(cleaned[6], 10) : 23,
            disabilityClass: cleaned[7] || undefined,
            disabilityType: cleaned[8] || undefined,
            hourlyWage: cleaned[9] ? parseInt(cleaned[9], 10) : undefined,
            weeklyScheduledHours: cleaned[10] ? parseFloat(cleaned[10]) : undefined,
            workStartTime: cleaned[11] || undefined,
            workEndTime: cleaned[12] || undefined,
            breakTimeMinutes: cleaned[13] ? parseInt(cleaned[13], 10) : undefined,
            birthdate: cleaned[14] || undefined,
            gender: cleaned[15] || undefined,
            phoneNumber: cleaned[16] || undefined,
            address: cleaned[17] || undefined,
            serviceStartDate: cleaned[18] || undefined,
          });
        } else {
          // 旧フォーマット（フリガナなし18列）後方互換
          items.push({
            fullName,
            recipientNumber: cleaned[1] || undefined,
            paymentLimit: cleaned[2] ? parseInt(cleaned[2], 10) : 0,
            validFrom: cleaned[3] || undefined,
            validTo: cleaned[4] || undefined,
            allowanceDays: cleaned[5] ? parseInt(cleaned[5], 10) : 23,
            disabilityClass: cleaned[6] || undefined,
            disabilityType: cleaned[7] || undefined,
            hourlyWage: cleaned[8] ? parseInt(cleaned[8], 10) : undefined,
            weeklyScheduledHours: cleaned[9] ? parseFloat(cleaned[9]) : undefined,
            workStartTime: cleaned[10] || undefined,
            workEndTime: cleaned[11] || undefined,
            breakTimeMinutes: cleaned[12] ? parseInt(cleaned[12], 10) : undefined,
            birthdate: cleaned[13] || undefined,
            gender: cleaned[14] || undefined,
            phoneNumber: cleaned[15] || undefined,
            address: cleaned[16] || undefined,
            serviceStartDate: cleaned[17] || undefined,
          });
        }
      }

      setParseErrors(errors);
      setParsedData(items);
    };

    reader.readAsText(file);
  };

  // インポート実行
  const handleExecuteImport = async () => {
    if (parsedData.length === 0) return;
    setIsProcessing(true);

    try {
      const res = await importClientsFromCsv(parsedData);
      if (res.success) {
        toast.success(res.message);
        setIsOpen(false);
        setParsedData([]);
        setFileName('');
        window.location.reload();
      } else {
        toast.error(res.error || "インポートに失敗しました");
      }
    } catch (e: any) {
      toast.error(e.message || "予期せぬエラーが発生しました");
    } finally {
      setIsProcessing(false);
    }
  };

  return (
    <Dialog open={isOpen} onOpenChange={setIsOpen}>
      <DialogTrigger asChild>
        <Button variant="outline" className="h-10 px-3.5 gap-2 border-indigo-200 bg-indigo-50/50 hover:bg-indigo-100 text-indigo-900 font-semibold shadow-xs">
          <Upload className="h-4 w-4 text-indigo-600" />
          <span>CSV一括登録</span>
        </Button>
      </DialogTrigger>
      
      <DialogContent className="max-w-3xl max-h-[85vh] flex flex-col">
        <DialogHeader>
          <DialogTitle className="text-xl font-bold flex items-center gap-2 text-indigo-950">
            <FileSpreadsheet className="h-5 w-5 text-indigo-600" />
            利用者データ一括CSVインポート
          </DialogTitle>
          <DialogDescription className="text-sm">
            他社ソフト（カイポケ、ワイズマン等）やExcelで管理している利用者一覧をまとめて登録できます。
          </DialogDescription>
        </DialogHeader>

        <div className="space-y-4 my-2 overflow-y-auto pr-1 flex-1">
          {/* テンプレート案内カード */}
          <div className="p-3.5 bg-muted/40 rounded-lg border flex items-center justify-between gap-4">
            <div className="space-y-0.5">
              <div className="text-sm font-bold">1. CSVテンプレートを準備</div>
              <div className="text-xs text-muted-foreground">
                所定フォーマットの雛形CSVをダウンロードし、氏名や受給者証情報を入力してください。
              </div>
            </div>
            <Button 
              size="sm" 
              variant="outline" 
              onClick={handleDownloadTemplate}
              className="gap-1.5 shrink-0 bg-white font-medium text-xs"
            >
              <Download className="h-3.5 w-3.5" />
              テンプレートDL
            </Button>
          </div>

          {/* アップロードエリア */}
          <div className="space-y-2">
            <div className="text-sm font-bold">2. 入力済みのCSVファイルを選択</div>
            <label className="border-2 border-dashed border-input hover:border-primary/60 bg-card p-6 rounded-xl flex flex-col items-center justify-center cursor-pointer transition-colors text-center group">
              <Upload className="h-8 w-8 text-muted-foreground group-hover:text-primary mb-2 transition-colors" />
              <span className="text-sm font-semibold">
                {fileName ? fileName : "クリックしてCSVファイルを選択（またはドラッグ＆ドロップ）"}
              </span>
              <span className="text-xs text-muted-foreground mt-1">.csv ファイル対応（UTF-8またはShift-JIS）</span>
              <input 
                type="file" 
                accept=".csv" 
                className="hidden" 
                onChange={handleFileChange}
              />
            </label>
          </div>

          {/* エラー表示 */}
          {parseErrors.length > 0 && (
            <div className="p-3 bg-rose-50 border border-rose-200 rounded-lg text-rose-800 text-xs space-y-1">
              <div className="font-bold flex items-center gap-1">
                <AlertCircle className="h-3.5 w-3.5 text-rose-600" />
                以下の行で問題が検出されました（スキップされます）:
              </div>
              <ul className="list-disc list-inside space-y-0.5 max-h-24 overflow-y-auto">
                {parseErrors.map((err, i) => (
                  <li key={i}>{err}</li>
                ))}
              </ul>
            </div>
          )}

          {/* プレビューテーブル */}
          {parsedData.length > 0 && (
            <div className="space-y-2">
              <div className="flex items-center justify-between text-xs">
                <span className="font-bold text-gray-700">
                  インポート対象プレビュー（全 {parsedData.length} 件）
                </span>
                <span className="text-emerald-600 font-semibold flex items-center gap-1">
                  <CheckCircle2 className="h-3.5 w-3.5" /> 正常に解析されました
                </span>
              </div>
              <div className="border rounded-md overflow-x-auto max-h-56 text-xs">
                <table className="w-full text-left border-collapse">

import { redirect } from "next/navigation";

export async function saveOffsiteEvaluation(clientId: string, formData: FormData) {
  const role = await getRole();
  if (!role) {
    redirect("/login");
  }

  let id = '';
  try {
    await assertCanAccessClient(clientId);

    const existingId = formData.get("id") as string;
    const todayStr = new Date().toISOString().slice(0, 10);
    const evaluationDate = (formData.get("evaluationDate") as string) || todayStr;
    const periodStart = (formData.get("periodStart") as string) || todayStr;
    const periodEnd = (formData.get("periodEnd") as string) || todayStr;
    const workplaceId = sanitizeForeignKeyId('offsite_workplaces', formData.get("workplaceId") as string);
    const contractorId = sanitizeForeignKeyId('contractor_agreements', formData.get("contractorId") as string);
    const workTasks = formData.get("workTasks") as string;
    const basicHabitsScore = Number(formData.get("basicHabitsScore")) || 3;
    const workPerformanceScore = Number(formData.get("workPerformanceScore")) || 3;
    const interpersonalScore = Number(formData.get("interpersonalScore")) || 3;
    const safetyScore = Number(formData.get("safetyScore")) || 4;
    const readinessScore = Number(formData.get("readinessScore")) || 3;
    const detailedEvaluation = formData.get("detailedEvaluation") as string;
    const strengthsAndGrowth = formData.get("strengthsAndGrowth") as string;
    const challengesAndSupport = formData.get("challengesAndSupport") as string;
    const planReflection = formData.get("planReflection") as string;
    const evaluatorId = sanitizeForeignKeyId('staff', formData.get("evaluatorId") as string);
    const evaluatorName = formData.get("evaluatorName") as string;
    const supervisorId = sanitizeForeignKeyId('staff', formData.get("supervisorId") as string);
    const supervisorName = formData.get("supervisorName") as string;
    const feedbackDate = formData.get("feedbackDate") as string;
    const clientSignature = formData.get("clientSignature") as string;

    id = existingId && existingId !== 'new' ? existingId : `eval_${clientId}_${Date.now()}`;
    const existing = existingId && existingId !== 'new'
      ? db.prepare('SELECT id FROM offsite_evaluations WHERE id = ? AND client_id = ?').get(id, clientId)
      : null;

    if (existing) {
      db.prepare(`
        UPDATE offsite_evaluations SET
          workplace_id = ?, contractor_id = ?, evaluation_date = ?, period_start = ?, period_end = ?,
          work_tasks = ?, basic_habits_score = ?, work_performance_score = ?, interpersonal_score = ?, safety_score = ?, readiness_score = ?,
          detailed_evaluation = ?, strengths_and_growth = ?, challenges_and_support = ?, plan_reflection = ?,
          evaluator_id = ?, evaluator_name = ?, supervisor_id = ?, supervisor_name = ?, feedback_date = ?, client_signature = ?,
          updated_at = CURRENT_TIMESTAMP
        WHERE id = ? AND client_id = ?
      `).run(
        workplaceId, contractorId, evaluationDate, periodStart, periodEnd,
        workTasks, basicHabitsScore, workPerformanceScore, interpersonalScore, safetyScore, readinessScore,
        detailedEvaluation, strengthsAndGrowth, challengesAndSupport, planReflection,
        evaluatorId, evaluatorName, supervisorId, supervisorName, feedbackDate || null, clientSignature || '本人確認済',
        id, clientId
      );
    } else {
      db.prepare(`
        INSERT INTO offsite_evaluations (
          id, client_id, workplace_id, contractor_id, evaluation_date, period_start, period_end,
          work_tasks, basic_habits_score, work_performance_score, interpersonal_score, safety_score, readiness_score,
          detailed_evaluation, strengths_and_growth, challenges_and_support, plan_reflection,
          evaluator_id, evaluator_name, supervisor_id, supervisor_name, feedback_date, client_signature
        ) VALUES (
          ?, ?, ?, ?, ?, ?, ?,
          ?, ?, ?, ?, ?, ?,
          ?, ?, ?, ?,
          ?, ?, ?, ?, ?, ?
        )
      `).run(
        id, clientId, workplaceId, contractorId, evaluationDate, periodStart, periodEnd,
        workTasks, basicHabitsScore, workPerformanceScore, interpersonalScore, safetyScore, readinessScore,
        detailedEvaluation, strengthsAndGrowth, challengesAndSupport, planReflection,
        evaluatorId, evaluatorName, supervisorId, supervisorName, feedbackDate || null, clientSignature || '本人確認済'
      );
    }

    revalidatePath(`/clients/${clientId}/documents`);
  } catch (error: any) {
    console.error('saveOffsiteEvaluation error:', error);
    redirect(`/clients/${clientId}/documents?tab=offsite-eval`);
  }

  redirect(`/clients/${clientId}/documents/offsite-evaluations/${id}`);
}

export async function deleteOffsiteEvaluation(clientId: string, evalId: string): Promise<{ success: boolean; error?: string }> {
  try {
    await assertCanAccessClient(clientId);
    const role = await getRole();
    if (!role) return { success: false, error: '認証されていません' };

    db.prepare('UPDATE offsite_evaluations SET deleted_at = CURRENT_TIMESTAMP WHERE id = ? AND client_id = ?').run(evalId, clientId);
    revalidatePath(`/clients/${clientId}/documents`);
    return { success: true };
  } catch (error: any) {
    console.error('deleteOffsiteEvaluation error:', error);
    return { success: false, error: error?.message || '施設外評価票の削除に失敗しました' };
  }
}

'use client';

import { useState, useEffect } from "react";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Upload, File, Image as ImageIcon, Trash2 } from "lucide-react";
import { toast } from "sonner";
import { format } from "date-fns";
import { ja } from "date-fns/locale";

export function FilesTab({ clientId }: { clientId: string }) {
  const [documents, setDocuments] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);
  const [uploading, setUploading] = useState(false);

  useEffect(() => {
    fetchDocuments();
  }, [clientId]);

  const fetchDocuments = async () => {
    try {
      const res = await fetch(`/api/clients/${clientId}/documents`);
      const data = await res.json();
      if (data.success) {
        setDocuments(data.documents);
      }
    } catch (err) {
      toast.error("ファイルの取得に失敗しました");
    } finally {
      setLoading(false);
    }
  };

  const handleFileUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;

    // Check file size (e.g., limit to 10MB)
    if (file.size > 10 * 1024 * 1024) {
      toast.error("10MB以下のファイルを選択してください");
      return;
    }

    setUploading(true);
    const formData = new FormData();
    formData.append("file", file);

    try {
      const res = await fetch(`/api/clients/${clientId}/documents`, {
        method: "POST",
        body: formData,
      });

      if (res.ok) {
        toast.success("ファイルをアップロードしました");
        fetchDocuments();
      } else {
        toast.error("アップロードに失敗しました");
      }
    } catch (err) {
      toast.error("アップロード中にエラーが発生しました");
    } finally {
      setUploading(false);
      e.target.value = ''; // Reset input
    }
  };

  const handleDelete = async (docId: string) => {
    if (!confirm("本当にこのファイルを削除しますか？")) return;

    try {
      const res = await fetch(`/api/documents/${docId}`, {
        method: "DELETE",
      });

      if (res.ok) {
        toast.success("ファイルを削除しました");
        fetchDocuments();
      } else {
        toast.error("削除に失敗しました");
      }
    } catch (err) {
      toast.error("削除中にエラーが発生しました");
    }
  };

  return (
    <Card>
      <CardHeader className="flex flex-row items-center justify-between">
        <CardTitle>添付書類管理</CardTitle>
        <div className="flex items-center gap-2">
          <Input 
            type="file" 
            id="file-upload" 
            className="hidden" 
            onChange={handleFileUpload}
            accept=".pdf,.png,.jpg,.jpeg"
            disabled={uploading}
          />
            <Button variant="outline" className="cursor-pointer" disabled={uploading} type="button" onClick={() => document.getElementById('file-upload')?.click()}>
              <Upload className="w-4 h-4 mr-2" />
              {uploading ? "送信中..." : "アップロード"}
            </Button>
        </div>
      </CardHeader>
      <CardContent>
        {loading ? (
          <div className="text-center py-8 text-muted-foreground">読み込み中...</div>
        ) : documents.length === 0 ? (
          <div className="text-center py-12 text-muted-foreground border-2 border-dashed rounded-md">
            ファイルがありません。<br />
            障害福祉サービス受給者証のコピーや履歴書などをアップロードしてください。
          </div>
        ) : (
          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
            {documents.map((doc) => {
              const isImage = doc.file_type.startsWith("image/");
              return (
                <div key={doc.id} className="border rounded-md p-4 flex flex-col justify-between group hover:border-primary/50 transition-colors bg-muted/20">
                  <div className="flex items-start gap-3">
                    <div className="p-2 bg-background rounded shadow-sm">
                      {isImage ? <ImageIcon className="w-8 h-8 text-blue-500" /> : <File className="w-8 h-8 text-red-500" />}
                    </div>
                    <div className="flex-1 min-w-0">
                      <p className="text-sm font-medium truncate" title={doc.file_name}>{doc.file_name}</p>
                      <p className="text-xs text-muted-foreground mt-1">
                        {format(new Date(doc.uploaded_at), "yyyy年MM月dd日 HH:mm")}
                      </p>
                    </div>
                  </div>
                  <div className="flex gap-2 mt-4 pt-4 border-t">
                    <Button 
                      variant="default" 
                      size="sm" 
                      className="flex-1"
                      onClick={() => window.open(`/api/documents/${doc.id}`, '_blank')}
                    >
                      プレビュー
                    </Button>
                    <Button 
                      variant="destructive" 
                      size="icon" 
                      className="shrink-0"
                      onClick={() => handleDelete(doc.id)}
                    >
                      <Trash2 className="w-4 h-4" />
                    </Button>
                  </div>
                </div>
              );
            })}
          </div>
        )}
      </CardContent>
    </Card>
  );
}

    const set = new Set<string>();
    for (const r of interviewRecords) {
      if (r.interview_date) set.add(r.interview_date.slice(0, 7));

    return Array.from(set).sort().reverse();
  }, [interviewRecords]);

  // 面談種別ラベル & バッジ色

    switch (type) {

      case 'work':
        return { label: '就労・作業相談', color: 'bg-emerald-100 text-emerald-800 border-emerald-200' };
      case 'life':
        return { label: '生活・健康相談', color: 'bg-amber-100 text-amber-800 border-amber-200' };
      case 'relationship':
        return { label: '人間関係相談', color: 'bg-purple-100 text-purple-800 border-purple-200' };
      case 'family':

  const filteredInterviews = useMemo(() => {

      const matchesMonth = selectedMonth === 'all' || (item.interview_date && item.interview_date.startsWith(selectedMonth));
      const query = searchQuery.toLowerCase();
      const matchesQuery = !query ||
        (item.topic && item.topic.toLowerCase().includes(query)) ||
        (item.client_statement && item.client_statement.toLowerCase().includes(query)) ||
        (item.staff_response && item.staff_response.toLowerCase().includes(query)) ||
        (item.interviewer_name && item.interviewer_name.toLowerCase().includes(query)) ||
