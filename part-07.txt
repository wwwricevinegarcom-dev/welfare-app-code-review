'use client';

import { useState } from "react";
import { useRouter } from "next/navigation";
import { Button } from "@/components/ui/button";
import { Card, CardContent, CardHeader, CardTitle, CardDescription } from "@/components/ui/card";
import { Textarea } from "@/components/ui/textarea";
import { Label } from "@/components/ui/label";
import { Printer, Plus, Save, Sparkles, Target, MessageSquare, CheckCircle2, Trash2, BookOpen, ShieldCheck, Loader2 } from "lucide-react";
import { Input } from "@/components/ui/input";
import { saveMonitoring, deleteMonitoring } from "../actions";
import { generateGroundedMonitoringDraft } from "@/app/plans/actions";
import { Badge } from "@/components/ui/badge";
import { toast } from "sonner";

export function MonitoringTab({ 
  clientId, 
  monitorings, 
  plans, 
  clientName,
  interviewRecords = [],
  supportRecords = []
}: any) {
  const router = useRouter();
  const [activeMonitoring, setActiveMonitoring] = useState(
    monitorings?.[0] || { 
      id: "new", 
      client_id: clientId, 
      monitoring_date: new Date().toISOString().split('T')[0],
      plan_id: plans?.[0]?.id || "" 
    }
  );
  const [isSaving, setIsSaving] = useState(false);
  const [isGeneratingGrounded, setIsGeneratingGrounded] = useState(false);
  const [groundedMeta, setGroundedMeta] = useState<{
    period_start: string;
    period_end: string;
    attendance_count: number;
    support_record_count: number;
    cited_record_dates: string[];
    masked_source_preview: string;
  } | null>(null);
  const [showMaskedSource, setShowMaskedSource] = useState(false);

  const handleCreateNew = () => {
    setActiveMonitoring({
      id: "new",
      client_id: clientId,
      monitoring_date: new Date().toISOString().split('T')[0],
      plan_id: plans?.[0]?.id || "",
      evaluation: "",
      next_steps: "",
      consent_date: ""
    });
  };

  const selectedPlan = plans?.find((p: any) => p.id === activeMonitoring.plan_id) || plans?.[0];

  // 過去記録ノート（直近6ヶ月間の支援記録 support_records ＆ 勤怠実績 attendance_records）のみを根拠とした厳格グラウンディング要約
  const handleAutoGenerateSummary = async () => {
    setIsGeneratingGrounded(true);
    try {
      const res = await generateGroundedMonitoringDraft(
        clientId,
        activeMonitoring.plan_id || selectedPlan?.id,
        activeMonitoring.monitoring_date
      );
      if (!res.success || !res.data) {
        toast.error(res.error || "過去記録ノートからの要約生成に失敗しました");
        return;
      }

      const d = res.data;
      const evaluationText = `${d.goal_achievement}\n\n■ 記録された様子・本人の意向（支援記録抜粋）:\n${d.client_feedback}`;
      const nextStepsText = `【次期計画および支援方針（直近6ヶ月実績に基づく見直し事項）】\n${d.next_action_plan}`;

      setActiveMonitoring((prev: any) => ({
        ...prev,
        evaluation: evaluationText,
        next_steps: nextStepsText,
      }));

      setGroundedMeta({
        period_start: d.period_start,
        period_end: d.period_end,
        attendance_count: d.attendance_count,
        support_record_count: d.support_record_count,
        cited_record_dates: d.cited_record_dates || [],
        masked_source_preview: d.masked_source_preview || "",
      });

      toast.success(
        `直近6ヶ月間の勤怠実績（${d.attendance_count}件）・支援記録（${d.support_record_count}件）のみを根拠に要約を生成しました（PIIマスク済）`
      );
    } catch (err: any) {
      toast.error(err?.message || "要約生成中にエラーが発生しました");
    } finally {
      setIsGeneratingGrounded(false);
    }
  };

  const handleSave = async () => {
    setIsSaving(true);
    try {
      const res = await saveMonitoring(clientId, activeMonitoring);
      if (res && !res.success) {
        toast.error(res.error || "保存に失敗しました");
        setIsSaving(false);
        return;
      }
      if (res?.id) {
        setActiveMonitoring((prev: any) => ({ ...prev, id: res.id }));
      }
      toast.success("モニタリングを保存しました");
      router.refresh();
    } catch (e: any) {
      toast.error(e.message || "エラーが発生しました");
    }
    setIsSaving(false);
  };

  const handleDelete = async (monitoringId: string) => {
    if (!confirm("このモニタリング報告書を削除しますか？\n※削除すると元に戻せません。")) return;
    try {
      await deleteMonitoring(clientId, monitoringId);
      toast.success("モニタリング報告書を削除しました");
      const remaining = monitorings?.filter((m: any) => m.id !== monitoringId) || [];
      if (remaining.length > 0) {
        setActiveMonitoring(remaining[0]);
      } else {
        handleCreateNew();
      }
      router.refresh();
    } catch (e: any) {
      toast.error(e?.message || "削除に失敗しました");
    }
  };

  return (
    <div className="flex gap-6">
      {/* Sidebar for history */}
      <div className="w-64 shrink-0 print:hidden space-y-4 border-r pr-6">
        <Button className="w-full" onClick={handleCreateNew}>
          <Plus className="w-4 h-4 mr-2" />
          新規作成
        </Button>
        
        <div className="space-y-2 mt-4">
          <h3 className="font-semibold text-sm text-muted-foreground flex items-center justify-between">
            <span>過去の履歴</span>
            <span className="text-xs bg-muted px-1.5 py-0.5 rounded-full font-normal">{monitorings?.length || 0}件</span>
          </h3>
          {(!monitorings || monitorings.length === 0) && (
            <p className="text-xs text-muted-foreground py-2">履歴なし</p>
          )}
          <div className="space-y-1.5 max-h-[500px] overflow-y-auto pr-1">
            {monitorings?.map((m: any) => {
              const isSelected = activeMonitoring.id === m.id;
              const snippet = (m.evaluation || m.next_steps || '').replace(/\n/g, ' ').slice(0, 30);
              return (
                <div 
                  key={m.id}
                  onClick={() => setActiveMonitoring(m)}
                  className={`p-2.5 rounded-md cursor-pointer border text-left transition-colors group relative ${
                    isSelected 
                      ? 'bg-primary/10 border-primary/60 shadow-2xs font-semibold' 
                      : 'hover:bg-muted/70 bg-card'
                  }`}
                >
                  <div className="flex items-center justify-between text-xs">
                    <span className="font-bold text-foreground">{m.monitoring_date}</span>
                    <div className="flex items-center gap-1">
                      {isSelected && <Badge variant="outline" className="text-[9px] px-1 py-0 h-4 border-primary text-primary">選択中</Badge>}
                      <Button
                        variant="ghost"
                        size="sm"
                        className="h-5 w-5 p-0 text-muted-foreground hover:text-destructive hover:bg-destructive/10"
                        onClick={(e) => {
                          e.stopPropagation();
                          handleDelete(m.id);
                        }}
                        title="削除"
                      >
                        <Trash2 className="w-3 h-3" />
                      </Button>
                    </div>
                  </div>
                  {snippet && (
                    <p className="text-[11px] text-muted-foreground truncate mt-1">
                      {snippet}...
                    </p>
                  )}
                </div>
              );
            })}
          </div>
        </div>
      </div>

      {/* Main Form */}
      <div className="flex-1 space-y-6">
        <div className="flex justify-between items-center print:hidden">
          <div>
            <h2 className="text-2xl font-bold">モニタリング報告書</h2>
            <p className="text-xs text-muted-foreground mt-0.5">
              個別支援計画の目標に対する達成状況を評価し、次期計画へ反映します（実地指導 指定基準準拠）。
            </p>
          </div>
          <div className="flex items-center gap-2 flex-wrap">
            <Button 
              type="button" 
              variant="outline" 
              size="sm" 
              disabled={isGeneratingGrounded}
              onClick={handleAutoGenerateSummary}
              className="text-xs font-semibold text-indigo-800 border-indigo-300 bg-indigo-50/80 hover:bg-indigo-100 flex items-center gap-1.5 shadow-2xs"
            >
              {isGeneratingGrounded ? (
                <>
                  <Loader2 className="w-3.5 h-3.5 text-indigo-600 animate-spin" />
                  過去6ヶ月記録を照合・要約中...
                </>
              ) : (
                <>
                  <BookOpen className="w-3.5 h-3.5 text-indigo-600" />
                  過去6ヶ月記録ノートから事実限定要約（NotebookLM方式）
                </>
              )}
            </Button>
            <Button variant="outline" size="sm" onClick={() => window.print()}>
              <Printer className="w-4 h-4 mr-2" />
              印刷・PDF化
            </Button>
            {activeMonitoring.id !== "new" && (
              <Button 
                variant="outline" 
                size="sm" 
                onClick={() => handleDelete(activeMonitoring.id)}
                className="text-destructive border-destructive/30 hover:bg-destructive/10"
              >
                <Trash2 className="w-3.5 h-3.5 mr-1" />
                削除
              </Button>
            )}
            <Button size="sm" onClick={handleSave} disabled={isSaving} className="bg-emerald-600 hover:bg-emerald-700 text-white">
              <Save className="w-4 h-4 mr-2" />
              保存する
            </Button>
          </div>
        </div>

        {/* 過去記録ノート（Strict Grounding ＆ PIIマスキング検証パネル） */}
        {groundedMeta && (
          <div className="p-3 bg-emerald-50/80 border border-emerald-200 rounded-lg text-xs space-y-2 print:hidden">
            <div className="flex items-center justify-between flex-wrap gap-2">
              <div className="font-bold text-emerald-950 flex items-center gap-1.5">
                <ShieldCheck className="w-4 h-4 text-emerald-600" />
                過去記録ノート（Strict Grounding）参照結果 — 期間: {groundedMeta.period_start} 〜 {groundedMeta.period_end}
              </div>
              <div className="flex items-center gap-2">
                <Badge variant="outline" className="bg-white text-emerald-800 border-emerald-300 text-[10px]">
                  勤怠実績: {groundedMeta.attendance_count}件
                </Badge>
                <Badge variant="outline" className="bg-white text-emerald-800 border-emerald-300 text-[10px]">
                  日次支援記録: {groundedMeta.support_record_count}件

            {groundedMeta.cited_record_dates.length > 0 && (

        {selectedPlan && (

'use client';

import { Button, buttonVariants } from "@/components/ui/button";
import { Card, CardContent, CardDescription, CardHeader, CardTitle } from "@/components/ui/card";
import { Badge } from "@/components/ui/badge";
import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from "@/components/ui/table";
import { Award, Briefcase, Calendar, CheckCircle2, FileText, Plus, Printer, Star, Edit, Trash2 } from "lucide-react";
import Link from "next/link";
import { cn } from "@/lib/utils";
import { deleteOffsiteEvaluation } from "../offsite-evaluations/actions";
import { toast } from "sonner";

interface OffsiteEvaluation {
  id: string;
  client_id: string;
  workplace_id?: string;
  workplace_name?: string;
  company_name?: string;
  contractor_id?: string;
  evaluation_date: string;
  period_start: string;
  period_end: string;
  work_tasks?: string;
  basic_habits_score: number;
  work_performance_score: number;
  interpersonal_score: number;
  safety_score: number;
  readiness_score: number;
  detailed_evaluation?: string;
  strengths_and_growth?: string;
  challenges_and_support?: string;
  plan_reflection?: string;
  evaluator_name?: string;
  supervisor_name?: string;
  feedback_date?: string;
  client_signature?: string;
}

export function OffsiteEvaluationTab({
  clientId,
  evaluations = [],
  clientName
}: {
  clientId: string;
  evaluations: OffsiteEvaluation[];
  clientName: string;
}) {
  const renderScoreStars = (score: number) => {
    return (
      <div className="flex items-center gap-0.5 text-amber-500">
        {[1, 2, 3, 4, 5].map((star) => (
          <span key={star} className={star <= score ? "text-amber-500 font-bold" : "text-gray-300"}>
            ★
          </span>
        ))}
        <span className="ml-1 text-xs font-mono text-muted-foreground">({score}/5)</span>
      </div>
    );
  };

  const handleDelete = async (evalId: string) => {
    if (!confirm('この施設外就労評価表を削除しますか？')) return;
    try {
      await deleteOffsiteEvaluation(clientId, evalId);
      toast.success('施設外就労評価表を削除しました');
    } catch (e: any) {
      toast.error(e.message || '削除に失敗しました');
    }
  };

  return (
    <div className="space-y-6">
      <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center gap-4 bg-muted/20 p-4 rounded-lg border print:hidden">
        <div>
          <h2 className="text-2xl font-bold flex items-center gap-2">
            <Briefcase className="w-6 h-6 text-primary" />
            施設外就労評価表 台帳（厚労省指定基準第191条の2）
          </h2>
          <p className="text-muted-foreground text-sm mt-1">
            {clientName} 様の企業先施設外就労における作業遂行力、労働習慣、一般就労移行適性の定期評価記録です。実地指導（運営指導）の必須提出台帳です。
          </p>
        </div>
        <div className="flex items-center gap-2">
          <Link
            href={`/clients/${clientId}/documents/offsite-evaluations/new`}
            className={cn(buttonVariants({ variant: "default" }), "font-bold gap-1")}
          >
            <Plus className="w-4 h-4" />
            新規評価表を作成
          </Link>
        </div>
      </div>

      <Card className="shadow-sm">
        <CardHeader className="bg-primary/5 border-b pb-4">
          <div className="flex justify-between items-center">
            <div>
              <CardTitle className="text-xl">施設外就労評価シート 交付・保管台帳</CardTitle>
              <CardDescription className="mt-1">
                登録件数: {evaluations.length} 件（過去全期間の評価シートが保存されています）
              </CardDescription>
            </div>
            <div className="text-xs text-muted-foreground">
              ※ 個別支援計画（Plan）と連動し、評価結果は次期モニタリングおよび計画見直しに反映されます。
            </div>
          </div>
        </CardHeader>
        <CardContent className="p-0">
          {evaluations && evaluations.length > 0 ? (
            <Table>
              <TableHeader className="bg-muted/50">
                <TableRow>
                  <TableHead className="py-3 w-[110px]">評価作成日</TableHead>
                  <TableHead className="py-3 w-[180px]">対象期間</TableHead>
                  <TableHead className="py-3">就労先企業 / 作業内容</TableHead>
                  <TableHead className="py-3 w-[140px]">作業遂行力</TableHead>
                  <TableHead className="py-3 w-[140px]">一般就労適性</TableHead>
                  <TableHead className="py-3 w-[130px]">評価者 / サビ管</TableHead>
                  <TableHead className="py-3 w-[100px]">本人確認</TableHead>
                  <TableHead className="py-3 w-[180px] text-right">操作</TableHead>
                </TableRow>
              </TableHeader>
              <TableBody>
                {evaluations.map((ev) => (
                  <TableRow key={ev.id} className="hover:bg-muted/20">
                    <TableCell className="py-3 font-mono font-medium text-sm whitespace-nowrap">
                      {ev.evaluation_date}
                    </TableCell>
                    <TableCell className="py-3 text-xs font-mono">
                      {ev.period_start} 〜 {ev.period_end}
                    </TableCell>
                    <TableCell className="py-3">
                      <div className="text-sm font-semibold text-foreground">

import { saveSelfPlan, deleteSelfPlan } from "../actions";

export function SelfPlanTab({ clientId, selfPlans = [], clientName }: any) {
  const [activePlan, setActivePlan] = useState<any>(selfPlans[0] || null);
  const [formData, setFormData] = useState<any>(activePlan || {
    id: "new",
    creation_date: new Date().toISOString().split("T")[0],
    hopes_and_dreams: "",
    current_challenges: "",
    weekly_schedule: ""
  });
  const [isSaving, setIsSaving] = useState(false);

  const handleNew = () => {
    const newPlan = {
      id: "new",
      creation_date: new Date().toISOString().split("T")[0],
      hopes_and_dreams: "",
      current_challenges: "",
      weekly_schedule: ""
    };
    setActivePlan(newPlan);
    setFormData(newPlan);
  };

  const handleChange = (e: any) => {
    setFormData({ ...formData, [e.target.name]: e.target.value });
  };

  const handleSave = async () => {
    setIsSaving(true);
    try {
      const res = await saveSelfPlan(clientId, formData);
      if (res && !res.success) {
        toast.error(res.error || "保存に失敗しました");
        return;
      }
      const wasNew = formData.id === "new";
      if (res?.id) {
        setFormData((prev: any) => ({ ...prev, id: res.id }));
        setActivePlan((prev: any) => prev ? { ...prev, id: res.id } : { ...formData, id: res.id });
      }
      toast.success("セルフプランを保存しました");
      if (wasNew) {
        setTimeout(() => window.location.reload(), 1000);
      }
    } catch (err: any) {
      toast.error(err?.message || "保存に失敗しました");
    } finally {
      setIsSaving(false);
    }
  };

  const handleDelete = async () => {
    if (!activePlan || activePlan.id === "new") return;
    if (!confirm("このセルフプランを削除しますか？\n※削除すると元に戻せません。")) return;
    try {
      const res = await deleteSelfPlan(clientId, activePlan.id);
      if (res && !res.success) {
        toast.error(res.error || "削除に失敗しました");
        return;
      }
      toast.success("セルフプランを削除しました");
      setTimeout(() => window.location.reload(), 500);
    } catch (err: any) {
      toast.error(err?.message || "削除に失敗しました");
    }
  };

  return (
    <Card>
      <CardHeader className="flex flex-row items-center justify-between">
        <div>
          <CardTitle>セルフプラン（サービス等利用計画案）</CardTitle>

          </p>
        </div>
        <div className="flex gap-2 print:hidden">
          <Button variant="outline" onClick={handleNew}>
            <Plus className="w-4 h-4 mr-2" />
            新規作成
          </Button>
          <Button variant="outline" onClick={() => window.print()} disabled={!activePlan || activePlan.id === "new"}>
            <Printer className="w-4 h-4 mr-2" />
            印刷
          </Button>
          {activePlan && activePlan.id !== "new" && (
            <Button variant="outline" onClick={handleDelete} className="text-destructive border-destructive/30 hover:bg-destructive/10">
              <Trash2 className="w-4 h-4 mr-2" />
              削除
            </Button>
          )}
          <Button onClick={handleSave} disabled={isSaving} className="bg-emerald-600 hover:bg-emerald-700 text-white">
            <Save className="w-4 h-4 mr-2" />

'use client';

import { useState, useEffect, useRef } from "react";
import { Button } from "@/components/ui/button";
import { Card, CardContent, CardHeader, CardTitle, CardDescription } from "@/components/ui/card";
import { FileIcon, FileTextIcon, ImageIcon, Trash2Icon, DownloadIcon, Loader2Icon, UploadCloudIcon } from "lucide-react";

interface DocumentData {
  id: string;
  file_name: string;
  file_type: string;
  uploaded_at: string;
}

export function DocumentsSection({ clientId }: { clientId: string }) {
  const [documents, setDocuments] = useState<DocumentData[]>([]);
  const [isLoading, setIsLoading] = useState(true);
  const [isUploading, setIsUploading] = useState(false);
  const fileInputRef = useRef<HTMLInputElement>(null);

  useEffect(() => {
    fetchDocuments();
  }, [clientId]);

  const fetchDocuments = async () => {
    try {
      const res = await fetch(`/api/clients/${clientId}/documents`);
      if (res.ok) {
        const data = await res.json();
        setDocuments(data.documents || []);
      }
    } catch (error) {
      console.error("Failed to fetch documents", error);
    } finally {
      setIsLoading(false);
    }
  };

  const handleFileChange = async (e: React.ChangeEvent<HTMLInputElement>) => {
    if (!e.target.files || e.target.files.length === 0) return;
    
    const file = e.target.files[0];
    await uploadFile(file);
    
    // Reset input
    if (fileInputRef.current) {
      fileInputRef.current.value = "";
    }
  };

  const handleDrop = async (e: React.DragEvent) => {
    e.preventDefault();
    if (e.dataTransfer.files && e.dataTransfer.files.length > 0) {
      const file = e.dataTransfer.files[0];
      await uploadFile(file);
    }
  };

  const handleDragOver = (e: React.DragEvent) => {
    e.preventDefault();
  };

  const uploadFile = async (file: File) => {
    const allowedTypes = ['application/pdf', 'image/jpeg', 'image/png', 'image/gif', 'image/webp'];
    if (!allowedTypes.includes(file.type)) {
      alert("アップロードできるのはPDFまたは画像（JPG, PNG等）のみです。");
      return;
    }

    setIsUploading(true);
    const formData = new FormData();
    formData.append("file", file);

    try {
      const res = await fetch(`/api/clients/${clientId}/documents`, {
        method: "POST",
        body: formData,
      });

      if (res.ok) {
        await fetchDocuments();
      } else {
        const data = await res.json();
        alert(`アップロード失敗: ${data.error}`);
      }
    } catch (error) {
      console.error("Upload error", error);
      alert("アップロード中にエラーが発生しました。");
    } finally {
      setIsUploading(false);
    }
  };

  const handleDelete = async (docId: string) => {
    if (!confirm("本当にこの書類を削除しますか？この操作は元に戻せません。")) return;

    try {
      const res = await fetch(`/api/documents/${docId}`, {
        method: "DELETE",
      });

      if (res.ok) {
        setDocuments(docs => docs.filter(d => d.id !== docId));
      } else {
        alert("削除に失敗しました。");
      }
    } catch (error) {
      console.error("Delete error", error);
      alert("削除中にエラーが発生しました。");
    }
  };

  const getFileIcon = (fileType: string) => {
    if (fileType.includes('pdf')) return <FileTextIcon className="h-8 w-8 text-red-500" />;
    if (fileType.includes('image')) return <ImageIcon className="h-8 w-8 text-blue-500" />;
    return <FileIcon className="h-8 w-8 text-gray-500" />;
  };

  return (
    <Card className="shadow-sm mt-6 border-muted bg-white/50">
      <CardHeader className="bg-primary/5 border-b pb-4">
        <CardTitle className="text-xl">添付書類（履歴書・受給者証コピーなど）</CardTitle>
        <CardDescription className="text-base">
          PDFや画像ファイルをアップロードして保存できます。
        </CardDescription>
      </CardHeader>
      <CardContent className="pt-6 space-y-6">
        
        {/* アップロードエリア */}
        <div 
          className="border-2 border-dashed border-muted-foreground/25 rounded-lg p-8 text-center hover:bg-muted/30 transition-colors cursor-pointer"
          onDrop={handleDrop}
          onDragOver={handleDragOver}
          onClick={() => fileInputRef.current?.click()}
        >
          <input 
            type="file" 
            ref={fileInputRef} 
            onChange={handleFileChange} 
            className="hidden" 
            accept="application/pdf,image/jpeg,image/png,image/gif,image/webp"
          />
          <div className="flex flex-col items-center justify-center space-y-3">
            {isUploading ? (
              <>
                <Loader2Icon className="h-10 w-10 text-primary animate-spin" />
                <p className="text-lg font-medium text-muted-foreground">アップロード中...</p>
              </>
            ) : (
              <>
                <div className="p-3 bg-primary/10 rounded-full">
                  <UploadCloudIcon className="h-8 w-8 text-primary" />
                </div>
                <p className="text-lg font-medium">クリックまたはファイルをドロップしてアップロード</p>
                <p className="text-sm text-muted-foreground">対応フォーマット: PDF, JPG, PNG (最大10MB推奨)</p>
              </>
            )}
          </div>
        </div>

        {/* ファイルリスト */}
        <div className="space-y-3">
          <h4 className="font-medium text-foreground">アップロード済みファイル</h4>
          
          {isLoading ? (
            <div className="py-8 text-center text-muted-foreground">読み込み中...</div>
          ) : documents.length === 0 ? (
            <div className="py-8 text-center border rounded-md bg-muted/10 text-muted-foreground">
              添付されている書類はありません。
            </div>
          ) : (
            <div className="grid gap-3">
              {documents.map(doc => (
                <div key={doc.id} className="flex items-center justify-between p-4 border rounded-md bg-background hover:bg-muted/10 transition-colors">
                  <div className="flex items-center gap-4 overflow-hidden">
                    {getFileIcon(doc.file_type)}
                    <div className="truncate">
                      <p className="font-medium truncate" title={doc.file_name}>{doc.file_name}</p>
                      <p className="text-xs text-muted-foreground">
                        {new Date(doc.uploaded_at).toLocaleString('ja-JP')} 追加
                      </p>
                    </div>
                  </div>
                  <div className="flex items-center gap-2 pl-4">
                    <a 
                      href={`/api/documents/${doc.id}`} 
                      target="_blank" 
                      rel="noopener noreferrer"
                    >
                      <Button variant="outline" size="sm" className="h-9">
                        <DownloadIcon className="h-4 w-4 mr-1" /> 確認
                      </Button>
                    </a>
                    <Button 
                      variant="ghost" 
                      size="icon" 
                      className="h-9 w-9 text-destructive hover:text-destructive hover:bg-destructive/10"
                      onClick={() => handleDelete(doc.id)}
                    >
                      <Trash2Icon className="h-4 w-4" />
                    </Button>
                  </div>
                </div>
              ))}
            </div>
          )}
        </div>

      </CardContent>
    </Card>
  );
}

        const hourlyWage = item.hourlyWage !== undefined && !isNaN(Number(item.hourlyWage)) && Number(item.hourlyWage) > 0 ? Number(item.hourlyWage) : defaultWage;
        const weeklyHours = item.weeklyScheduledHours !== undefined && !isNaN(Number(item.weeklyScheduledHours)) && Number(item.weeklyScheduledHours) > 0 ? Number(item.weeklyScheduledHours) : 20;
        const serviceStart = item.serviceStartDate && /^\d{4}-\d{2}-\d{2}$/.test(item.serviceStartDate) ? item.serviceStartDate : todayStr;
        const birthdate = item.birthdate && /^\d{4}-\d{2}-\d{2}$/.test(item.birthdate) ? item.birthdate : null;
        const gender = item.gender ? item.gender.trim() : null;
        const address = item.address ? item.address.trim() : null;
        const phone = item.phoneNumber ? item.phoneNumber.trim() : null;
        const disabilityType = item.disabilityType ? JSON.stringify([item.disabilityType.trim()]) : null;
        const startTime = item.workStartTime && /^\d{2}:\d{2}$/.test(item.workStartTime) ? item.workStartTime : '09:00';
        const endTime = item.workEndTime && /^\d{2}:\d{2}$/.test(item.workEndTime) ? item.workEndTime : '16:00';
        const breakM = item.breakTimeMinutes !== undefined && !isNaN(Number(item.breakTimeMinutes)) ? Number(item.breakTimeMinutes) : 60;

        insertStmt.run(
          id, fullName, furigana, recipientNumber, paymentLimit,
          validFrom, validTo, null, null, allowanceDays, disabilityClass, null, null, disabilityCertificate,
          hourlyWage, weeklyHours, serviceStart, serviceStart,
          defaultPtoGranted,
          birthdate, gender, address, phone, disabilityType,
          startTime, endTime, breakM
        );

        insertedCount++;
      }
    })();

    revalidatePath('/clients');
    return {
      success: true,
      insertedCount,
      errors: errors.length > 0 ? errors : undefined,
      message: `${insertedCount}件の利用者を一括登録しました。${errors.length > 0 ? ` (エラー: ${errors.length}件)` : ''}`
    };
  } catch (error: any) {
    console.error('CSV一括インポートエラー:', error);
    return { success: false, error: `インポートに失敗しました: ${error.message}` };
  }
}

function calculateAge(birthdateStr?: string): string {
  if (!birthdateStr) return '';
  try {
    const b = new Date(birthdateStr);
    if (isNaN(b.getTime())) return '';
    const today = new Date();
    let age = today.getFullYear() - b.getFullYear();
    const m = today.getMonth() - b.getMonth();
    if (m < 0 || (m === 0 && today.getDate() < b.getDate())) {
      age--;
    }
    return `${age}歳`;
  } catch {
    return '';
  }
}

  // プレビュー・編集モーダルを開く
  const handleOpenPreview = (doc: KnowledgeDocumentRecord) => {
    setPreviewDoc(doc);
    setEditTitle(doc.title);
    setEditCategory(doc.category);
    setEditText(doc.extracted_text || '');
  };

  // プレビュー編集の保存
  const handleSaveDocEdit = async () => {
    if (!previewDoc) return;
    setIsSavingDoc(true);
    try {
      await updateKnowledgeDocumentContent(previewDoc.id, editTitle, editCategory, editText);
      setDocuments(prev =>
        prev.map(d =>
          d.id === previewDoc.id
            ? { ...d, title: editTitle, category: editCategory, extracted_text: editText }
            : d
        )
      );
      toast.success('資料の内容を更新しました');
      setPreviewDoc(null);
      router.refresh();
    } catch (e: any) {
      toast.error(e.message || '保存に失敗しました');
    } finally {
      setIsSavingDoc(false);
    }
  };

  // 資料の削除
  const handleDeleteDoc = async (doc: KnowledgeDocumentRecord) => {
    if (!confirm(`公式資料「${doc.title}」を削除しますか？`)) return;
    try {
      await deleteKnowledgeDocument(doc.id);
      setDocuments(prev => prev.filter(d => d.id !== doc.id));
      toast.success(`「${doc.title}」を削除しました`);
      router.refresh();
    } catch (e: any) {
      toast.error(e.message || '削除に失敗しました');
    }
  };

  // Q&A送信
  const handleAskQuestion = async (qText?: string) => {
    const query = (qText ?? questionInput).trim();
    if (!query) {
      toast.error('質問内容を入力してください');
      return;
    }
    setIsAsking(true);
    try {
      const selectedIds = selectedDocs.map(d => d.id);
      const res = await askNotebookQuestion(query, selectedIds);
      if (!res.success || !res.data) {
        toast.error(res.error || '回答の生成に失敗しました');
        return;
      }

      const newTurn: ChatTurn = {
        id: `turn_${Date.now()}`,
        question: query,
        timestamp: new Date().toLocaleTimeString('ja-JP', { hour: '2-digit', minute: '2-digit' }),
        result: res.data,
      };
      setChatHistory(prev => [newTurn, ...prev]);
      if (!qText) setQuestionInput('');
    } catch (e: any) {
      toast.error(e.message || 'エラーが発生しました');
    } finally {
      setIsAsking(false);
    }
  };

  // 率（%）表示変換ヘルパー
  const isRateRule = (r: { rule_key?: string; unit_label?: string }) => {
    return (
      r.unit_label === '率' ||
      r.unit_label === '%' ||
      (r.rule_key?.startsWith('treatment_') ?? false) ||
      (r.rule_key?.startsWith('deduction_') ?? false)
    );
  };

  const toPercentDisplay = (val: number) => {
    if (val > 0 && val <= 1) return parseFloat((val * 100).toFixed(2));
    return val;
  };

  const toRateDb = (val: number) => {
    if (val > 1) return parseFloat((val / 100).toFixed(4));
    return val;
  };

  // PDF直読み・法改正チェック実行
  const handleRunLawRevisionCheck = async () => {
    if (selectedDocs.length === 0) {
      toast.error('左ペインで照合に使用する公式資料を1件以上選択してください');
      return;
    }
    setIsCheckingRevision(true);
    setRevisionResult(null);
    try {
      const selectedIds = selectedDocs.map(d => d.id);
      const res = await checkLawRevisionWithNotebookDocs(selectedIds);
      if (!res.success || !res.data) {
        toast.error(res.error || '法改正照合に失敗しました');
        return;
      }

      setRevisionResult({
        has_updates: res.data.has_updates,
        revision_title: res.data.revision_title,
        summary: res.data.summary,
        used_documents: res.data.used_documents,
        proposed_rules: (res.data.proposed_rules || []).map(r => {
          const isRate = isRateRule(r);
          return {
            ...r,
            proposed_value: isRate ? toPercentDisplay(r.proposed_value) : r.proposed_value,
            unit_label: isRate ? '%' : r.unit_label,
            selected: true,
          };
        }),
      });

      if (!res.data.has_updates) {
        toast.success('選択された公式資料と現在の加算・減算マスタは完全に一致しています');
      } else {
        toast.warning(`${res.data.proposed_rules.length}件の数値差異が検出されました`);
      }
    } catch (e: any) {
      toast.error(e.message || '照合処理中にエラーが発生しました');
    } finally {
      setIsCheckingRevision(false);
    }
  };

  // 法改正差分のDB反映
  const handleApplyRevisionRules = async () => {
    if (!revisionResult) return;
    const targets = revisionResult.proposed_rules
      .filter(r => r.selected)
      .map(r => {
        const isRate = isRateRule(r);
        return {
          ...r,
          proposed_value: isRate ? toRateDb(r.proposed_value) : r.proposed_value,
          unit_label: isRate ? '率' : r.unit_label,
        };
      });

    if (targets.length === 0) {
      toast.error('適用するルールを1件以上選択してください');
      return;
    }

    setIsApplyingRules(true);
    try {
      const res = await applyBillingRuleUpdates(targets);
      if (res.success) {
        toast.success(`${res.count}件の加算・減算マスタを公式資料の根拠に基づき更新しました`);
        await handleRunLawRevisionCheck();
        router.refresh();
      } else {
        toast.error(res.error || '更新に失敗しました');
      }
    } catch (e: any) {
      toast.error(e.message || 'エラーが発生しました');
    } finally {
      setIsApplyingRules(false);
    }
  };

  return (
    <div className="space-y-4">
      {/* トップステータスバー（DB・ストレージ完全分離＆Strict Grounding表示） */}
      <div
        className={`rounded-xl border p-4 flex flex-col lg:flex-row lg:items-center justify-between gap-3 ${
          isTypeB
            ? 'bg-gradient-to-r from-emerald-50/90 via-teal-50/60 to-white border-emerald-200'
            : 'bg-gradient-to-r from-indigo-50/90 via-blue-50/60 to-white border-indigo-200'
        }`}
      >
        <div className="space-y-1">
          <div className="flex flex-wrap items-center gap-2">
            <Badge
              className={`font-bold text-xs px-2.5 py-0.5 ${
                isTypeB ? 'bg-emerald-600 text-white' : 'bg-indigo-600 text-white'
              }`}
            >
              {serviceName} 専用ナレッジベース
            </Badge>
            <Badge
              variant="outline"
              className="bg-white text-slate-700 border-slate-300 font-mono text-[11px] flex items-center gap-1"
            >
              <FolderLock className="w-3 h-3 text-slate-500" />
              storage/knowledge/{serviceMode}/ ({isTypeB ? 'local_data_b.sqlite' : 'local_data.sqlite'})
            </Badge>
            <Badge
              variant="outline"
              className="bg-amber-50 text-amber-900 border-amber-300 text-[11px] font-bold flex items-center gap-1"
            >
              <ShieldCheck className="w-3.5 h-3.5 text-amber-600" />
              Strict Grounding（事前学習・推測・他制度混同100%遮断）
            </Badge>
          </div>
          <p className="text-xs text-slate-600">
            左ペインでチェックを入れた公式資料（厚労省告示・神戸市手引き・{isTypeB ? '工賃支払規程' : '就業規則'}・公式Q&A）のみを唯一の情報源として回答・報酬マスタ照合を行います。
          </p>
        </div>

        <div className="flex items-center gap-2 shrink-0">
          <div className="text-right mr-2">
            <div className="text-[11px] text-muted-foreground">現在選択中のソース資料</div>
            <div className="text-sm font-black text-slate-900">
              {selectedDocs.length} <span className="text-xs font-normal text-slate-500">/ {documents.length} 件</span>
            </div>
          </div>
          {canManage && (
            <Button
              size="sm"
              onClick={() => setIsUploadOpen(true)}
              className={
                isTypeB
                  ? 'bg-emerald-600 hover:bg-emerald-700 text-white font-bold gap-1.5'
                  : 'bg-indigo-600 hover:bg-indigo-700 text-white font-bold gap-1.5'
              }
            >
              <Upload className="w-4 h-4" />
              新規PDF・資料を追加
            </Button>
          )}
        </div>
      </div>

      {/* 本家NotebookLMライク 2ペイン構成 */}
      <div className="grid grid-cols-1 lg:grid-cols-12 gap-6 items-start">
        {/* ============================================================
            左ペイン: ソース資料管理（col-span-5）
           ============================================================ */}
        <Card className="lg:col-span-5 border-slate-200 shadow-xs overflow-hidden">
          <CardHeader className="p-4 pb-3 bg-slate-50/80 border-b space-y-2.5">
            <div className="flex items-center justify-between">
              <CardTitle className="text-base font-bold flex items-center gap-2 text-slate-900">
                <BookOpen className="w-4 h-4 text-indigo-600" />
                ソース資料管理（{serviceName}）
              </CardTitle>
              <div className="flex items-center gap-1">
                <Button
                  type="button"
                  variant="ghost"
                  size="sm"
                  className="h-7 px-2 text-xs font-semibold text-indigo-700 hover:bg-indigo-50"
                  onClick={() => handleToggleAll(true)}
                >
                  全選択
                </Button>
                <span className="text-slate-300">|</span>
                <Button
                  type="button"
                  variant="ghost"
                  size="sm"
                  className="h-7 px-2 text-xs text-slate-600 hover:bg-slate-100"
                  onClick={() => handleToggleAll(false)}
                >
                  全解除
                </Button>
              </div>
            </div>

            {/* カテゴリフィルタ */}
            <div className="flex flex-wrap gap-1">
              {[
                { key: 'all', label: 'すべて' },
                { key: 'mhlw_notice', label: '厚労省告示' },
                { key: 'local_guideline', label: '自治体手引き' },
                { key: 'internal_rule', label: isTypeB ? '工賃規程' : '就業規則' },
                { key: 'qa', label: '公式Q&A' },
              ].map(tab => (
                <button
                  key={tab.key}
                  type="button"
                  onClick={() => setCategoryFilter(tab.key)}
                  className={`px-2.5 py-1 rounded-md text-xs font-semibold transition-colors ${
                    categoryFilter === tab.key
                      ? 'bg-slate-900 text-white'
                      : 'bg-white text-slate-600 border border-slate-200 hover:bg-slate-100'
                  }`}
                >
                  {tab.label}
                </button>
              ))}
            </div>
          </CardHeader>

          <CardContent className="p-4 space-y-4">
            {/* ドラッグ＆ドロップ アップロードゾーン */}
            {canManage && (
              <div
                onDragOver={e => {
                  e.preventDefault();
                  setIsDragging(true);
                }}
                onDragLeave={() => setIsDragging(false)}
                onDrop={handleDrop}
                onClick={() => fileInputRef.current?.click()}
                className={`border-2 border-dashed rounded-xl p-3.5 text-center cursor-pointer transition-all ${
                  isDragging
                    ? 'border-indigo-500 bg-indigo-50/70'
                    : 'border-slate-300 hover:border-indigo-400 bg-slate-50/50 hover:bg-slate-50'
                }`}
              >
                <input
                  ref={fileInputRef}
                  type="file"
                  accept=".pdf,.txt,.md"
                  className="hidden"
                  onChange={e => handleFilePicked(e.target.files?.[0] || null)}
                />
                <div className="flex items-center justify-center gap-2 text-xs font-bold text-slate-700">
                  <Upload className="w-4 h-4 text-indigo-600" />
                  <span>ここに告示PDF・手引き（.pdf / .txt / .md）をドロップ、またはクリックして追加</span>
                </div>
                <p className="text-[11px] text-muted-foreground mt-1">
                  上限20MB・アップロード時にページ別テキストを自動抽出し {serviceMode} 専用領域に安全保存します
                </p>
              </div>
            )}

            {/* 資料リスト */}
            <div className="space-y-2.5 max-h-[600px] overflow-y-auto pr-1">
              {filteredDocs.length === 0 ? (
                <div className="text-center py-8 text-sm text-muted-foreground">
                  該当するカテゴリの資料がありません
                </div>
              ) : (
                filteredDocs.map(doc => {
                  const isChecked = doc.is_active === 1;
                  const meta = CATEGORY_META[doc.category] || CATEGORY_META.mhlw_notice;
                  const charCount = (doc.extracted_text || '').length;

                  return (
                    <div
                      key={doc.id}
                      className={`p-3 rounded-xl border transition-all ${
                        isChecked
                          ? 'bg-white border-indigo-300 shadow-2xs ring-1 ring-indigo-500/10'
                          : 'bg-slate-50/60 border-slate-200 opacity-75'
                      }`}
                    >
                      <div className="flex items-start gap-2.5">
                        <button
                          type="button"
                          onClick={() => handleToggleDoc(doc)}
                          className="mt-0.5 text-indigo-600 hover:text-indigo-800 shrink-0"
                          title={isChecked ? '照合対象から外す' : '照合対象に含める'}
                        >
                          {isChecked ? (
                            <CheckSquare className="w-5 h-5 text-indigo-600" />
                          ) : (
                            <Square className="w-5 h-5 text-slate-400" />
                          )}
                        </button>

                        <div className="flex-1 min-w-0">
                          <div className="flex flex-wrap items-center gap-1.5 mb-1">
                            <span
                              className={`text-[10px] px-1.5 py-0.5 rounded border font-bold ${meta.badgeClass}`}
                            >
                              {meta.label}
                            </span>
                            <span className="text-[10px] text-slate-400 font-mono truncate">
                              {doc.file_name}
                            </span>
                          </div>

                          <div
                            onClick={() => handleToggleDoc(doc)}
                            className="text-sm font-bold text-slate-900 leading-snug cursor-pointer hover:text-indigo-700"
                          >
                            {doc.title}
                          </div>

                          <div className="flex items-center justify-between mt-2 pt-1.5 border-t border-slate-100 text-[11px] text-slate-500">
                            <span>抽出文字数: {charCount.toLocaleString()} 文字</span>
                            <div className="flex items-center gap-1">
                              <Button
                                type="button"
                                variant="ghost"
                                size="sm"
                                className="h-6 px-2 text-[11px] text-slate-600 hover:text-indigo-700"
                                onClick={() => handleOpenPreview(doc)}
                              >
                                <Eye className="w-3.5 h-3.5 mr-1" />
                                全文・編集
                              </Button>
                              {canManage && (
                                <Button
                                  type="button"
                                  variant="ghost"
                                  size="sm"
                                  className="h-6 w-6 p-0 text-slate-400 hover:text-rose-600"
                                  onClick={() => handleDeleteDoc(doc)}
                                  title="削除"
                                >
                                  <Trash2 className="w-3.5 h-3.5" />
                                </Button>
                              )}
                            </div>
                          </div>
                        </div>
                      </div>
                    </div>
                  );
                })
              )}
            </div>
          </CardContent>
        </Card>

        {/* ============================================================
            右ペイン: 根拠引用付きAIチャット ＆ 法改正照合（col-span-7）
           ============================================================ */}
        <Card className="lg:col-span-7 border-slate-200 shadow-xs overflow-hidden">
          <CardHeader className="p-4 pb-3 bg-slate-50/80 border-b">
            <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2">
              <div>
                <CardTitle className="text-base font-bold flex items-center gap-2 text-slate-900">
                  <Sparkles className="w-4 h-4 text-indigo-600" />
                  厳格グラウンディング法務アシスタント
                </CardTitle>
                <CardDescription className="text-xs mt-0.5">
                  選択中のソース資料（{selectedDocs.length}件）のみを根拠に回答・マスタ照合を実行します
                </CardDescription>
              </div>

              {/* モード切替タブ */}
              <div className="flex rounded-lg bg-slate-200/80 p-1 shrink-0">
                <button
                  type="button"
                  onClick={() => setRightMode('qa')}
                  className={`px-3 py-1.5 rounded-md text-xs font-bold transition-all flex items-center gap-1.5 ${
                    rightMode === 'qa'
                      ? 'bg-white text-indigo-950 shadow-2xs'
                      : 'text-slate-600 hover:text-slate-900'
                  }`}
                >
                  <Quote className="w-3.5 h-3.5 text-indigo-600" />
                  実地指導・運営Q&A
                </button>
                <button
                  type="button"
                  onClick={() => setRightMode('revision')}
                  className={`px-3 py-1.5 rounded-md text-xs font-bold transition-all flex items-center gap-1.5 ${
                    rightMode === 'revision'
                      ? 'bg-white text-indigo-950 shadow-2xs'
                      : 'text-slate-600 hover:text-slate-900'
                  }`}
                >
                  <Scale className="w-3.5 h-3.5 text-emerald-600" />
                  PDF直読み・法改正チェック
                </button>
              </div>
            </div>
          </CardHeader>

          <CardContent className="p-4 space-y-4">
            {rightMode === 'qa' ? (
              /* ============================================================
                 モード1: 実地指導・運営Q&Aモード（根拠引用付きAIチャット）
                 ============================================================ */
              <div className="space-y-4">
                {/* よく使う質問チップ */}
                <div className="space-y-1.5">
                  <span className="text-[11px] font-bold text-slate-500 block">
                    💡 {serviceName} よく使う法務・運営指導質問（クリックで即時照会）:
                  </span>
                  <div className="flex flex-wrap gap-1.5">
                    {sampleQuestions.map((sq, i) => (
                      <button
                        key={i}
                        type="button"
                        disabled={isAsking}
                        onClick={() => handleAskQuestion(sq)}
                        className="text-left text-xs px-2.5 py-1.5 rounded-lg border border-indigo-200/80 bg-indigo-50/40 hover:bg-indigo-100/70 text-indigo-950 font-medium transition-colors"
                      >
                        {sq}
                      </button>
                    ))}
                  </div>
                </div>

                {/* 質問入力欄 */}
                <form
                  onSubmit={e => {
                    e.preventDefault();
                    handleAskQuestion();
                  }}
                  className="space-y-2"
                >
                  <div className="flex gap-2">
                    <Textarea
                      value={questionInput}
                      onChange={e => setQuestionInput(e.target.value)}
                      placeholder={`左ペインで選択中の${serviceName}公式資料（${selectedDocs.length}件）に対して質問を入力してください...`}
                      rows={2}
                      className="flex-1 text-sm resize-none"
                    />
                    <Button
                      type="submit"
                      disabled={isAsking || !questionInput.trim()}
                      className="h-auto px-4 bg-indigo-600 hover:bg-indigo-700 text-white font-bold shrink-0"
                    >
                      {isAsking ? (
                        <Loader2 className="w-4 h-4 animate-spin" />
                      ) : (
                        <>
                          <Send className="w-4 h-4 mr-1.5" />
                          根拠照会
                        </>
                      )}
                    </Button>
                  </div>
                </form>

                {/* チャット履歴・回答表示エリア */}
                <div className="space-y-4 max-h-[560px] overflow-y-auto pr-1">
                  {chatHistory.length === 0 ? (
                    <div className="rounded-xl border border-dashed border-slate-200 p-8 text-center space-y-2 bg-slate-50/40">
                      <BookOpen className="w-8 h-8 text-indigo-400 mx-auto" />
                      <div className="text-sm font-bold text-slate-700">
                        選択された公式資料のみを根拠に回答します
                      </div>
                      <p className="text-xs text-muted-foreground max-w-md mx-auto leading-relaxed">
                        AI自身の事前学習知識や推測は一切使用しません。選択された資料内に記載がない質問（またはチェックをすべて外した状態での質問）に対しては、推測回答を行わず資料不足の旨を通知します。
                      </p>
                    </div>
                  ) : (
                    chatHistory.map(turn => (
                      <div
                        key={turn.id}
                        className="rounded-xl border border-slate-200 bg-white shadow-2xs overflow-hidden space-y-3 p-4"
                      >
                        {/* 質問ヘッダ */}
                        <div className="flex items-start justify-between gap-2 pb-2.5 border-b border-slate-100">
                          <div className="font-bold text-sm text-slate-900 flex items-start gap-2">
                            <span className="px-2 py-0.5 rounded bg-slate-900 text-white text-[11px] shrink-0 mt-0.5">
                              質問
                            </span>
                            <span>{turn.question}</span>
                          </div>
                          <span className="text-[11px] text-slate-400 shrink-0">{turn.timestamp}</span>
                        </div>

                        {/* グラウンディング判定バッジ */}
                        <div className="flex flex-wrap items-center gap-2">
                          {turn.result.has_grounding ? (
                            <Badge className="bg-emerald-50 text-emerald-800 border border-emerald-300 text-xs font-bold flex items-center gap-1">
                              <CheckCircle2 className="w-3.5 h-3.5 text-emerald-600" />
                              公式資料内に根拠確認済（参照資料: {turn.result.used_document_count}件）
                            </Badge>
                          ) : (
                            <Badge className="bg-amber-50 text-amber-900 border border-amber-300 text-xs font-bold flex items-center gap-1">
                              <AlertTriangle className="w-3.5 h-3.5 text-amber-600" />
                              該当記載なし（ハルシネーション防止のため推測回答を遮断）
                            </Badge>
                          )}
                        </div>

                        {/* AI回答本文 */}
                        <div
                          className={`p-3.5 rounded-lg text-sm whitespace-pre-wrap leading-relaxed ${
                            turn.result.has_grounding
                              ? 'bg-slate-50 text-slate-800 border border-slate-200/80'
                              : 'bg-amber-50/70 text-amber-950 border border-amber-200 font-semibold'
                          }`}
                        >
                          {turn.result.answer}
                        </div>

                        {/* 参照した資料名 ＆ 該当箇所の原文抜粋（引用ブロック） */}
                        {turn.result.citations && turn.result.citations.length > 0 && (
                          <div className="space-y-2 pt-1">
                            <div className="text-xs font-bold text-indigo-950 flex items-center gap-1.5">
                              <Quote className="w-4 h-4 text-indigo-600" />
                              参照した公式資料名および該当箇所の原文抜粋（一字一句引用）
                            </div>
                            <div className="space-y-2">
                              {turn.result.citations.map((cit, cIdx) => {
                                const catMeta =
                                  CATEGORY_META[cit.category as KnowledgeCategory] ||

/**

function ensureTableReady() {
  const mode = getServiceModeSync();
  db.exec(`
    CREATE TABLE IF NOT EXISTS knowledge_documents (
      id TEXT PRIMARY KEY,
      title TEXT NOT NULL,
      category TEXT NOT NULL,
      file_name TEXT NOT NULL,
      file_path TEXT NOT NULL,
      extracted_text TEXT,
      is_active INTEGER DEFAULT 1,
      uploaded_at DATETIME DEFAULT CURRENT_TIMESTAMP,
      deleted_at DATETIME DEFAULT NULL
    );
  `);
  ensureKnowledgeDocumentsSeeded(db, mode);
}

/**
 * 現在の事業種別（A型/B型）の公式資料一覧を取得
 */
export async function getKnowledgeDocuments(): Promise<{
  serviceMode: ServiceMode;
  documents: KnowledgeDocumentRecord[];
}> {
  try {
    const role = await getRole();
    const serviceMode = getServiceModeSync();
    if (!role) return { serviceMode, documents: [] };

    ensureTableReady();

    const documents = db
      .prepare(
        `
        SELECT id, title, category, file_name, file_path, extracted_text, is_active, uploaded_at, deleted_at
        FROM knowledge_documents
        WHERE deleted_at IS NULL
        ORDER BY uploaded_at DESC, id ASC
      `
      )
      .all() as KnowledgeDocumentRecord[];

    return {
      serviceMode,
      documents,
    };
  } catch (e) {
    console.error("Failed to get knowledge documents:", e);
    return { serviceMode: getServiceModeSync(), documents: [] };
  }
}

  serviceMode: ServiceMode
): NotebookQaResult {
  const cleanQ = question.trim();
  if (!cleanQ || docs.length === 0) {
    return {
      has_grounding: false,
      answer: NO_GROUNDING_MESSAGE,
      citations: [],
      used_document_count: docs.length,
      service_mode: serviceMode,
    };
  }

  // 他制度や対象外サービスの質問を排除するためのキーワード抽出
  const stopWords = new Set([
    'とは', 'について', '教えて', 'ください', 'ですか', 'ますか', 'どう', 'どのよう', 'なに', '何',
    '場合', 'とき', '時', '方法', '基準', '要件', 'いくら', '何単位', '何パーセント', 'ルール'
  ]);

  // 漢字・カタカナ・数字・英字の連続トークンを抽出
  const rawTokens =
    cleanQ.match(/[一-龠々ヶヵ]{2,}|[ァ-ヶー]{2,}|[0-9０-９]+(?:\.[0-9]+)?|[a-zA-Z0-9]{2,}/g) || [];
  const queryTokens = Array.from(
    new Set(
      rawTokens
        .map(t => t.trim())
        .filter(t => t.length >= 2 && !stopWords.has(t) && !t.startsWith('就労継続支援'))
    )
  );

  // もしトークンが抽出できなかった場合は2文字バイグラムを使用
  const searchTokens =
    queryTokens.length > 0
      ? queryTokens
      : [cleanQ.replace(/[？?。、\s]/g, '').slice(0, Math.min(6, cleanQ.length))].filter(Boolean);

  interface ScoredPassage {
    doc: KnowledgeDocumentRecord;
    pageLabel: string;
    paragraph: string;
    score: number;
    matchedTokens: string[];
  }

  const passages: ScoredPassage[] = [];

  for (const doc of docs) {
    const text = doc.extracted_text || '';
    if (!text.trim()) continue;

    // ページ単位に分割
    const pageBlocks = text.split(/(?=---\s*\[ページ\s*\d+\]\s*---)/g);
    for (const block of pageBlocks) {
      const pageHeaderMatch = block.match(/---\s*\[(ページ\s*\d+)\]\s*---/);
      const pageLabel = pageHeaderMatch ? pageHeaderMatch[1] : 'ページ 1';
      const bodyWithoutPageHeader = block.replace(/---\s*\[ページ\s*\d+\]\s*---/, '').trim();

      // セクションまたは段落ごとに分割
      const paragraphs = bodyWithoutPageHeader
        .split(/\n{2,}|(?=\n(?:第[0-9１-９]+|【問|\d+\.\s))/)
        .map(p => p.trim())
        .filter(p => p.length >= 15);

      for (const para of paragraphs) {
        const matched: string[] = [];
        let score = 0;
        for (const token of searchTokens) {
          if (para.includes(token)) {
            matched.push(token);
            score += token.length >= 4 ? 3 : 2;
          }
        }
        if (score > 0) {
          passages.push({
            doc,
            pageLabel,
            paragraph: para,
            score,
            matchedTokens: matched,
          });
        }
      }
    }
  }

  passages.sort((a, b) => b.score - a.score);
  const topPassages = passages.slice(0, 3);

  if (topPassages.length === 0) {
    return {
      has_grounding: false,
      answer: NO_GROUNDING_MESSAGE,
      citations: [],
      used_document_count: docs.length,
      service_mode: serviceMode,
    };
  }

  const citations: NotebookCitation[] = topPassages.map(p => {
    const firstLine = p.paragraph.split('\n')[0].replace(/^[#\-*\s]+/, '').slice(0, 40);
    return {
      document_id: p.doc.id,
      document_title: p.doc.title,
      category: p.doc.category,
      page_or_section: `${p.pageLabel} / ${firstLine}`,
      exact_quote: p.paragraph,
    };
  });

  const summaryLines = topPassages.map(
    (p, idx) =>
      `【根拠${idx + 1}: ${p.doc.title}（${p.pageLabel}）】\n${p.paragraph}`
  );

  const answer = `選択された公式資料（${docs.length}件）を照合した結果、以下の通り該当する規定・記載が確認されました。\n\n${summaryLines.join('\n\n')}`;

  return {
    has_grounding: true,
    answer,
    citations,
    used_document_count: docs.length,
    service_mode: serviceMode,
  };
}

/**
 * 【実地指導・運営Q&Aモード】
 * 左ペインで選択された資料のみを根拠に回答を生成（参照資料名＋原文一字一句抜粋付き）
 */
export async function askNotebookQuestion(
  question: string,
  selectedDocIds?: string[]
): Promise<{ success: boolean; data?: NotebookQaResult; error?: string }> {
  const role = await getRole();
  if (!role) {
    return { success: false, error: '認証されていません' };
  }

  const cleanQuestion = (question || '').trim();
  if (!cleanQuestion) {
    return { success: false, error: '質問内容を入力してください' };
  }

  const serviceMode = getServiceModeSync();
  ensureTableReady();

  // 選択された資料を取得
  let activeDocs: KnowledgeDocumentRecord[] = [];
  if (selectedDocIds && selectedDocIds.length > 0) {
    const placeholders = selectedDocIds.map(() => '?').join(', ');
    activeDocs = db
      .prepare(
        `
        SELECT id, title, category, file_name, file_path, extracted_text, is_active, uploaded_at, deleted_at
        FROM knowledge_documents
        WHERE deleted_at IS NULL AND id IN (${placeholders})
        ORDER BY uploaded_at DESC
      `
      )
      .all(...selectedDocIds) as KnowledgeDocumentRecord[];
  } else if (selectedDocIds && selectedDocIds.length === 0) {
    activeDocs = [];
  } else {
    activeDocs = db
      .prepare(
        `
        SELECT id, title, category, file_name, file_path, extracted_text, is_active, uploaded_at, deleted_at
        FROM knowledge_documents
        WHERE deleted_at IS NULL AND is_active = 1
        ORDER BY uploaded_at DESC
      `
      )
      .all() as KnowledgeDocumentRecord[];
  }

  // 選択資料が0件の場合は即座に厳格グラウンディング定型文を返却
  if (activeDocs.length === 0) {
    return {
      success: true,
      data: {
        has_grounding: false,
        answer: NO_GROUNDING_MESSAGE,
        citations: [],
        used_document_count: 0,
        service_mode: serviceMode,
      },
    };
  }

  const systemInstruction = buildStrictGroundingSystemInstruction(serviceMode);
  const serviceLabel = serviceMode === 'type_b' ? '就労継続支援B型' : '就労継続支援A型';

  const documentsContext = activeDocs
    .map(
      (doc, idx) => `========================================
【公式資料 ${idx + 1}】
- 資料ID (document_id): ${doc.id}
- 資料名 (document_title): ${doc.title}
- カテゴリ (category): ${doc.category}
- ファイル名: ${doc.file_name}
----------------------------------------
${doc.extracted_text || '（テキストなし）'}
========================================`
    )
    .join('\n\n');

  const prompt = `以下は現在ユーザーが選択している『${serviceLabel}』の【公式資料テキスト】の全文です。
この資料に記載されている内容のみを唯一の根拠として、ユーザーの質問に回答してください。

${documentsContext}

【ユーザーの質問】
${cleanQuestion}

【出力ルール（厳守）】
1. 提供された【公式資料テキスト】の中に明確な根拠がある場合は、has_grounding を true にし、answer に分かりやすい回答を記載した上で、citations に「参照した資料の document_id, document_title, category, page_or_section」および「該当箇所を一字一句そのまま抜粋した原文（exact_quote）」を必ず格納してください。
2. 提供された【公式資料テキスト】の中に該当する記載が見つからない場合は、絶対に事前知識や推測で補完せず、has_grounding を false にし、answer に「${NO_GROUNDING_MESSAGE}」と一字一句そのまま出力し、citations は空配列にしてください。`;

  try {
    // 重要: useSearchGrounding は false に固定し、外部インターネット検索による汚染を100%遮断
    const aiModel = getAiModelInstance({ useSearchGrounding: false });
    const { object } = await generateObject({
      model: aiModel,
      system: systemInstruction,
      prompt,
      schema: notebookQaSchema,
    });

    if (!object.has_grounding || !object.citations || object.citations.length === 0) {
      return {
        success: true,
        data: {
          has_grounding: false,
          answer: NO_GROUNDING_MESSAGE,
          citations: [],
          used_document_count: activeDocs.length,
          service_mode: serviceMode,
        },
      };
    }

    return {
      success: true,
      data: {
        has_grounding: true,
        answer: object.answer,
        citations: object.citations,
        used_document_count: activeDocs.length,
        service_mode: serviceMode,
      },
    };
  } catch (aiErr: any) {
    console.warn('NotebookLM AI call fallback to local strict grounding engine:', aiErr?.message || aiErr);
    // AIクォータ上限や通信エラー時でも、選択された資料テキストのみから厳格に抜粋するローカルエンジンで応答
    const fallbackResult = runLocalStrictGroundedQa(cleanQuestion, activeDocs, serviceMode);
    return {
      success: true,
      data: fallbackResult,
    };
  }
}

/**
 * 【PDF直読み・法改正チェックモード】
 * Web検索ではなく、左ペインで選択された「公式資料（報酬告示・改定資料PDF/テキスト）」と
 * 現在の billing_rules（加算・減算マスタ）を直接突き合わせ、差分と原文根拠（exact_mhlw_quote）を抽出する
 */
export async function checkLawRevisionWithNotebookDocs(
  selectedDocIds?: string[]
): Promise<{
  success: boolean;
  data?: LegalAdditionCheckResult & { used_documents: { id: string; title: string }[] };
  error?: string;
}> {
  const role = await getRole();
  if (role !== 'admin' && role !== 'service_manager') {
    return { success: false, error: '権限がありません（管理者またはサビ管のみ実行可能）' };
  }

  const serviceMode = getServiceModeSync();
  const isTypeB = serviceMode === 'type_b';
  const serviceLabel = isTypeB ? '就労継続支援B型' : '就労継続支援A型';
  ensureTableReady();

  let activeDocs: KnowledgeDocumentRecord[] = [];
  if (selectedDocIds && selectedDocIds.length > 0) {
    const placeholders = selectedDocIds.map(() => '?').join(', ');
    activeDocs = db
      .prepare(
        `
        SELECT id, title, category, file_name, file_path, extracted_text, is_active, uploaded_at, deleted_at
        FROM knowledge_documents
        WHERE deleted_at IS NULL AND id IN (${placeholders})
        ORDER BY uploaded_at DESC
      `
      )
      .all(...selectedDocIds) as KnowledgeDocumentRecord[];
  } else if (selectedDocIds && selectedDocIds.length === 0) {
    activeDocs = [];
  } else {
    activeDocs = db
      .prepare(
        `
        SELECT id, title, category, file_name, file_path, extracted_text, is_active, uploaded_at, deleted_at
        FROM knowledge_documents
        WHERE deleted_at IS NULL AND is_active = 1
        ORDER BY uploaded_at DESC
      `
      )
      .all() as KnowledgeDocumentRecord[];
  }

  if (activeDocs.length === 0) {
    return {
      success: false,
      error: NO_GROUNDING_MESSAGE,
    };
  }

  const currentRules = await getAllBillingRules();
  const systemInstruction = buildStrictGroundingSystemInstruction(serviceMode);

  const combinedDocsText = activeDocs
    .map(
      d => `【資料名: ${d.title} (${d.file_name})】\n${d.extracted_text || ''}`
    )
    .join('\n\n========================================\n\n');

  const currentRulesText = currentRules
    .map(
      r =>
        `- rule_key: "${r.rule_key}" | 名称: "${r.rule_name}" | 現在値(current_value): ${r.rule_value} | 単位: "${r.unit_label}" | 施行日: ${r.effective_from}`
    )
    .join('\n');

  // 1. 選択された資料のテキスト内に含まれる公式基準値と現在の billing_rules を厳格に照合するベースライン関数
  const computeStrictDocDiscrepancies = (): LegalAdditionItem[] => {
    const officialMaster = isTypeB ? OFFICIAL_B_TYPE_BILLING_MASTER : OFFICIAL_A_TYPE_BILLING_MASTER;
    const ruleMap = new Map(currentRules.map(r => [r.rule_key, r]));
    const items: LegalAdditionItem[] = [];

    for (const master of officialMaster) {
      const dbRule = ruleMap.get(master.rule_key);
      if (!dbRule) continue;

      // 選択された資料の中にこの加算・減算ルールに関する記述が存在するか確認
      const shortName = master.rule_name
        .replace(/（.*?）|\(.*?\)|率|単位数|就労継続支援[AB]型/g, '')
        .trim();
      const matchingDoc = activeDocs.find(d => {
        const txt = d.extracted_text || '';
        return txt.includes(shortName) || (master.exact_mhlw_quote && txt.includes(master.exact_mhlw_quote.slice(0, 15)));
      });

      if (!matchingDoc) continue;

      // 該当資料から該当行を一字一句そのまま抜粋
      const docLines = (matchingDoc.extracted_text || '').split('\n');
      const matchedLine =
        docLines.find(line => line.includes(shortName))?.trim() ||
        master.exact_mhlw_quote;

      const diff = Math.abs(Number(dbRule.rule_value) - Number(master.official_value));
      if (diff > 0.0001) {
        items.push({
          rule_key: master.rule_key,
          rule_name: master.rule_name,
          current_value: Number(dbRule.rule_value),
          proposed_value: Number(master.official_value),
          unit_label: master.unit_label,
          description: `【参照資料: ${matchingDoc.title}】${master.official_reason}`,
          effective_from: master.effective_from,
          exact_mhlw_quote: `【${matchingDoc.title} より抜粋】「${matchedLine}」`,
        });
      }
    }

    return items;
  };

  try {
    const aiModel = getAiModelInstance({ useSearchGrounding: false });
    const prompt = `以下の【選択された公式資料テキスト】のみを唯一の情報源として、現在のシステムの加算・減算マスタ（billing_rules）と直接突き合わせ、数値（単位数・加算率・減算率）に差異がある項目を抽出してください。

【選択された公式資料テキスト（${serviceLabel}）】
${combinedDocsText}

【現在のシステム登録値（billing_rules）】
${currentRulesText}

【厳格照合ルール】
1. 外部知識や推測は一切使わず、上記の【選択された公式資料テキスト】の中に明確に数値が記載されている項目のみを照合してください。
2. 現在のシステム登録値（current_value）と資料記載値（proposed_value）に差異がある場合は has_updates を true にし、proposed_rules に差分項目を出力してください。
   ※率（%）の項目（例: 9.6% や 11.2% や 3%減算）は、current_value と同じ小数形式（0.096, 0.112, 0.03）で比較・出力してください。
3. exact_mhlw_quote には、必ず【選択された公式資料テキスト】の中から該当箇所を一字一句そのまま抜粋し、「【資料名】『抜粋原文』」の形式で記載してください。
4. 差異が1件もない場合は has_updates を false、proposed_rules を空配列にし、summary に選択された公式資料の数値と完全一致している旨および確認した主な規定の抜粋を記載してください。`;

    const { object } = await generateObject({
      model: aiModel,
      system: systemInstruction,
      prompt,
      schema: legalAdditionCheckSchema,
    });

    const deterministicDiffs = computeStrictDocDiscrepancies();
    const finalProposed =
      object.proposed_rules && object.proposed_rules.length > 0
        ? object.proposed_rules.filter(
            r => Math.abs(Number(r.current_value) - Number(r.proposed_value)) > 0.0001
          )
        : deterministicDiffs;

    return {
      success: true,
      data: {
        has_updates: finalProposed.length > 0,
        is_future_revision: false,
        revision_title:
          object.revision_title ||
          `選択資料（${activeDocs.length}件）による${serviceLabel} 報酬マスタ直接照合結果`,
        summary:
          finalProposed.length > 0
            ? object.summary
            : `選択された公式資料（${activeDocs.map(d => d.title).join('、')}）の記載内容と、現在のシステム内の加算・減算マスタ（${currentRules.length}項目）を直接突き合わせた結果、すべての単位数・加算率・減算率が公式資料の基準と完全に一致していることが確認されました。`,
        proposed_rules: finalProposed,
        reference_url: '/compliance/notebook',
        used_documents: activeDocs.map(d => ({ id: d.id, title: d.title })),
      },
    };
  } catch (err: any) {
    console.warn('AI PDF revision check fallback to strict deterministic document matcher:', err?.message || err);
    const deterministicDiffs = computeStrictDocDiscrepancies();
    return {
      success: true,
      data: {
        has_updates: deterministicDiffs.length > 0,
        is_future_revision: false,
        revision_title: `選択資料（${activeDocs.length}件）による${serviceLabel} 報酬マスタ直接照合結果`,
        summary:
          deterministicDiffs.length > 0
            ? `選択された公式資料（${activeDocs.map(d => d.title).join('、')}）と現在の加算・減算マスタを直接照合し、${deterministicDiffs.length}件の数値差異と原文根拠を抽出しました。`
            : `選択された公式資料（${activeDocs.map(d => d.title).join('、')}）の記載内容と現在の加算・減算マスタ（${currentRules.length}項目）を直接突き合わせた結果、すべての単位数・加算率・減算率が公式資料の基準と完全に一致しています。`,
        proposed_rules: deterministicDiffs,
        reference_url: '/compliance/notebook',
        used_documents: activeDocs.map(d => ({ id: d.id, title: d.title })),
      },
    };
  }
}

          <div className="flex items-center gap-2 self-end sm:self-auto shrink-0">
            <Button
              size="sm"
              variant="outline"
              onClick={() => setIsOpen(!isOpen)}
              className="text-xs h-8 px-2.5 bg-white border-rose-300 text-rose-900 hover:bg-rose-50"
            >
              {isOpen ? (
                <>
                  <ChevronUp className="w-3.5 h-3.5 mr-1" />
                  内訳を閉じる
                </>
              ) : (
                <>
                  <ChevronDown className="w-3.5 h-3.5 mr-1" />
                  対象日（{alerts.length}件）を見る
                </>
              )}
            </Button>
            <Link href="/attendance/shifts">
              <Button size="sm" className="bg-rose-600 hover:bg-rose-700 text-white font-bold text-xs h-8 px-3 gap-1 shadow-xs">
                <span>シフト表で調整</span>
                <ChevronRight className="w-3.5 h-3.5" />
              </Button>
            </Link>
          </div>
        </div>
      </CardHeader>

      {isOpen && (
        <CardContent className="p-4 sm:p-6 bg-white/50">
          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-3">
            {alerts.slice(0, 6).map((alert, idx) => (
              <div key={idx} className="bg-white rounded-lg border border-rose-200 p-3 shadow-2xs hover:shadow-xs transition-shadow">
                <div className="flex items-center justify-between border-b pb-1.5 mb-2">
                  <span className="font-black text-xs text-foreground flex items-center gap-1">
                    <Calendar className="w-3.5 h-3.5 text-rose-600" />
                    {alert.dateStr} {alert.dayOfWeek}
                  </span>
                  <span className="text-[11px] font-black text-rose-700 bg-rose-100 border border-rose-300 px-1.5 py-0.5 rounded">
                    合計 +{alert.totalOverflow}名 施設外へ
                  </span>
                </div>
                <div className="space-y-1.5">
                  {alert.departments.filter((d: any) => d.isOverCapacity).map((dep: any, dIdx: number) => (
                    <div key={dIdx} className="bg-rose-50/70 p-2 rounded border border-rose-200 flex flex-col gap-0.5">
                      <div className="flex items-center justify-between">
                        <Badge variant="outline" className="text-[10px] font-bold bg-white border-rose-300">
                          {dep.facilityTypeLabel}: {dep.departmentName}
                        </Badge>
                        <span className="text-xs font-black text-rose-700">
                          内勤 {dep.inFacilityCount}名 / 上限{dep.capacity}名
                        </span>
                      </div>
                      <p className="text-[11px] text-rose-950 font-bold mt-0.5">
                        ⚠️ <strong>{dep.overflowCount}名</strong> を施設外就労に出す必要があります
                      </p>
                    </div>
                  ))}
                </div>
                <div className="mt-2 pt-1.5 border-t flex justify-end">
                  <Link href={`/attendance/shifts?month=${alert.dateStr.slice(0, 7)}`}>
                    <span className="text-[11px] font-bold text-blue-600 hover:text-blue-800 flex items-center gap-0.5">
                      この日のシフトを調整 →
                    </span>
                  </Link>
                </div>
              </div>
            ))}
          </div>
          {alerts.length > 6 && (
            <div className="mt-3 text-center">
              <Link href="/attendance/shifts">
                <span className="text-xs font-bold text-rose-700 hover:underline">
                  他 {alerts.length - 6} 件の超過予定日をシフト画面で確認する →
                </span>
              </Link>
            </div>
          )}
        </CardContent>
      )}
    </Card>
  );
}
