import React, { useState, useMemo, useCallback, memo } from "react";

interface ShiftCellProps {
  shift: any;
  actual: any;
  val: string;
  clientId: string;
  dateStr: string;
  onCellChange: (clientId: string, dateStr: string, val: string) => void;
}

const ShiftCell = memo(({ shift, actual, val, clientId, dateStr, onCellChange }: ShiftCellProps) => {
  const [isEditing, setIsEditing] = useState(false);
  const [editValue, setEditValue] = useState("");

  const isBoth = shift?.work_type === 'both' || (shift?.offsite_start_time && shift?.in_facility_start_time) || (val.includes('外') && val.includes('内'));
  const isOffsite = !isBoth && (shift?.is_offsite === 1 || shift?.work_type === 'out_facility' || val.includes('施設外'));

  const handleClick = () => {
    setEditValue(val.replace('\n(施設外)', '').replace('\n', ' '));
    setIsEditing(true);
  };

  const handleBlur = () => {
    setIsEditing(false);
    const originalEquivalent = val.replace('\n(施設外)', '').replace('\n', ' ');
    if (editValue !== originalEquivalent) {
      onCellChange(clientId, dateStr, editValue);
    }
  };

  const handleKeyDown = (e: React.KeyboardEvent<HTMLInputElement>) => {
    if (e.key === 'Enter') {
      e.preventDefault();
      (e.target as HTMLInputElement).blur();
    }
  };

  const handleQuickSelect = (quickVal: string) => {
    setIsEditing(false);
    onCellChange(clientId, dateStr, quickVal);
  };

  // 予実判定ロジック
  const hasPlan = shift?.scheduled_in || shift?.is_holiday;
  const isActualPresent = actual && actual.status === 'present';
  const isActualAbsent = actual && (actual.status === 'absent' || actual.status === 'absent_with_addition');
  
  // 予定があるのに休んだ（無断欠勤・当日公休・事前休など）
  const isWarning = shift?.scheduled_in && !isActualPresent && new Date(dateStr) < new Date();
  // 欠席時対応加算が算定されているか
  const hasAddition = actual?.has_absence_addition === 1;

  let cellBg = shift?.is_holiday ? 'bg-red-50/50 hover:bg-red-50' : 'hover:bg-black/5';
  if (isWarning && !isActualAbsent) cellBg = 'bg-yellow-50 hover:bg-yellow-100';
  if (isBoth) cellBg = 'bg-emerald-50/60 hover:bg-emerald-100/60';
  else if (isOffsite) cellBg = 'bg-cyan-50/60 hover:bg-cyan-100/60';
  
  if (isEditing) {
    return (
      <div className="relative z-20">
        <input 
          autoFocus
          type="text"
          className="w-full h-16 text-center text-[11px] bg-white border-2 border-primary focus:outline-none shadow-md"
          value={editValue}
          placeholder="09-17, 外10-12 内13-15, 施設外, 公休"
          onChange={e => setEditValue(e.target.value)}
          onBlur={handleBlur}
          onKeyDown={handleKeyDown}
        />
        <div className="absolute top-full left-1/2 -translate-x-1/2 mt-1 bg-popover border rounded-md shadow-lg p-1.5 flex gap-1 text-[10px] whitespace-nowrap z-50">
          <button 
            type="button" 
            className="px-2 py-0.5 bg-emerald-100 hover:bg-emerald-200 text-emerald-950 rounded font-semibold border border-emerald-300"
            onMouseDown={(e) => { e.preventDefault(); handleQuickSelect('外10-12 内13-15'); }}
            title="午前施設外就労・午後内勤の併用"
          >
            外10-12/内13-15
          </button>
          <button 
            type="button" 
            className="px-2 py-0.5 bg-emerald-100 hover:bg-emerald-200 text-emerald-950 rounded font-semibold border border-emerald-300"
            onMouseDown={(e) => { e.preventDefault(); handleQuickSelect('内10-12 外13-15'); }}
            title="午前内勤・午後施設外就労の併用"
          >
            内10-12/外13-15
          </button>
          <button 
            type="button" 
            className="px-2 py-0.5 bg-cyan-100 hover:bg-cyan-200 text-cyan-950 rounded font-semibold border border-cyan-300"
            onMouseDown={(e) => { e.preventDefault(); handleQuickSelect('施設外'); }}
            title="施設外就労（内勤配置の分母から自動除外）"
          >
            施設外
          </button>
          <button 
            type="button" 
            className="px-2 py-0.5 bg-red-100 hover:bg-red-200 text-red-900 rounded border border-red-300"
            onMouseDown={(e) => { e.preventDefault(); handleQuickSelect('公休'); }}
          >
            公休
          </button>
          <button 
            type="button" 
            className="px-2 py-0.5 bg-amber-100 hover:bg-amber-200 text-amber-900 rounded border border-amber-300"
            onMouseDown={(e) => { e.preventDefault(); handleQuickSelect('有休'); }}
          >
            有休
          </button>
          <button 
            type="button" 
            className="px-2 py-0.5 bg-blue-50 hover:bg-blue-100 text-blue-900 rounded border border-blue-200"
            onMouseDown={(e) => { e.preventDefault(); handleQuickSelect('09:00-17:00'); }}
          >
            09-17
          </button>
        </div>
      </div>
    );
  }

  return (
    <div 
      className={`w-full h-16 print:h-auto print:min-h-0 print:py-1 flex flex-col items-center justify-center text-center text-[11px] print:text-[8pt] leading-tight cursor-pointer transition-colors ${cellBg}`}
      onClick={handleClick}
      title={isBoth ? "施設外・内勤併用" : (isOffsite ? "施設外就労（本体内勤人員の必要人員から除外）" : undefined)}
    >
      <div className="flex-1 flex flex-col justify-center font-medium">
        {isBoth ? (
          <div className="flex flex-col items-center py-0.5 leading-tight">
            <span className="px-1.5 py-0.2 bg-emerald-200 text-emerald-950 rounded text-[9px] print:text-[7pt] font-bold leading-none shadow-xs mb-0.5">
              外内併用
            </span>
            <span className="text-[9px] print:text-[7pt] text-emerald-800 font-medium whitespace-pre-line leading-tight">
              {val}
            </span>
          </div>
        ) : isOffsite ? (
          <div className="flex flex-col items-center py-0.5">
            <span className="px-1.5 py-0.5 bg-cyan-200/80 text-cyan-950 rounded text-[9px] print:text-[7pt] font-bold leading-none shadow-xs">
              施設外
            </span>
            <span className="text-[9px] print:text-[7pt] text-cyan-800 mt-0.5 font-medium">
              {shift?.scheduled_in ? `${shift.scheduled_in.substring(0,5)}-${shift.scheduled_out?.substring(0,5)}` : '終日'}
            </span>
          </div>
        ) : val ? (
          val.split('\n').map((line: string, i: number) => <div key={i}>{line}</div>)
        ) : (
          <span className="text-muted-foreground/20">-</span>
        )}
      </div>
      
      {/* 実績バッジ（予実管理・印刷時は非表示で純粋なシフト予定のみ表示） */}
      <div className="h-4 w-full flex items-center justify-center print:hidden">
        {isActualPresent && (
          <Badge variant="outline" className="text-[8px] h-4 px-1 py-0 bg-green-50 text-green-600 border-green-200">
            出 {actual.check_in?.substring(0,5)}
          </Badge>
        )}
        {isActualAbsent && (
          <Badge variant="outline" className={`text-[8px] h-4 px-1 py-0 ${hasAddition ? 'bg-blue-50 text-blue-600 border-blue-200' : 'bg-red-50 text-red-600 border-red-200'}`}>
            {hasAddition ? '欠(加算)' : '公休・事前休'}
          </Badge>
        )}
        {isWarning && !isActualAbsent && (
           <Badge variant="outline" className="text-[8px] h-4 px-1 py-0 bg-yellow-100 text-yellow-700 border-yellow-300">
             打刻無
           </Badge>
        )}
      </div>
    </div>
  );
});
ShiftCell.displayName = "ShiftCell";

export function ClientShiftsClient({ 
  initialClients, 
  initialShifts, 
  initialAttendance,
  targetYearMonth, 
  isAdmin,
  facilitySettings,
  initialStaffingMap = {},
  departments = [],
  initialStaff = [],
  initialStaffShifts = [],
  initialCapacityCalendar = []
}: { 
  initialClients: any[], 
  initialShifts: any[], 
  initialAttendance: any[],
  targetYearMonth: string,
  isAdmin: boolean,
  facilitySettings?: any,
  initialStaffingMap?: Record<string, any>,
  departments?: any[],
  initialStaff?: any[],
  initialStaffShifts?: any[],
  initialCapacityCalendar?: any[]
}) {
  const router = useRouter();
  const [currentMonth, setCurrentMonth] = useState(targetYearMonth);
  const [selectedDeptId, setSelectedDeptId] = useState<string>("all");

  const filteredClients = useMemo(() => {
    if (selectedDeptId === "all") return initialClients;
    if (selectedDeptId === "none") return initialClients.filter(c => !c.department_id);
    return initialClients.filter(c => c.department_id === selectedDeptId);
  }, [initialClients, selectedDeptId]);

  const [shiftsData, setShiftsData] = useState<Record<string, Record<string, any>>>(() => {
    const data: Record<string, Record<string, any>> = {};
    initialShifts.forEach(s => {
      if (!data[s.client_id]) data[s.client_id] = {};
      data[s.client_id][s.target_date] = s;
    });
    return data;
  });

  const [attendanceData, setAttendanceData] = useState<Record<string, Record<string, any>>>(() => {
    const data: Record<string, Record<string, any>> = {};
    initialAttendance.forEach(a => {
      if (!data[a.client_id]) data[a.client_id] = {};
      data[a.client_id][a.date] = a;
    });
    return data;
  });

  const staffShiftsMap = useMemo(() => {
    const map: Record<string, Record<string, any>> = {};
    initialStaffShifts.forEach((s: any) => {
      if (!map[s.staff_id]) map[s.staff_id] = {};
      map[s.staff_id][s.target_date] = s;
    });
    return map;
  }, [initialStaffShifts]);

  const daysInMonth = useMemo(() => {
    const [year, month] = currentMonth.split('-').map(Number);
    return new Date(year, month, 0).getDate();
  }, [currentMonth]);

  const daysArray = useMemo(() => {
    return Array.from({ length: daysInMonth }, (_, i) => {
      const day = i + 1;
      const dateStr = `${currentMonth}-${day.toString().padStart(2, '0')}`;
      const [year, month] = currentMonth.split('-').map(Number);
      const dayOfWeek = new Date(year, month - 1, day).getDay();
      return { day, dateStr, dayOfWeek };
    });
  }, [currentMonth, daysInMonth]);

  const DAY_KANJI = ['日', '月', '火', '水', '木', '金', '土'];

  // 利用者の仮想シフト（未登録時の曜日別定例パターンまたはデフォルト）を取得
  const getClientVirtualShift = (client: any, targetDate: string, dayOfWeek: number, shift: any) => {
    if (shift) return shift;

    let matchedPattern: any = null;
    if (client.weekly_schedule_pattern) {
      try {
        const parsed = typeof client.weekly_schedule_pattern === 'string'
          ? JSON.parse(client.weekly_schedule_pattern)
          : client.weekly_schedule_pattern;
        matchedPattern = parsed[dayOfWeek.toString()];
      } catch (e) {}
    }

    if (matchedPattern) {
      if (matchedPattern.is_holiday) {
        return {
          is_holiday: 1,
          holiday_type: matchedPattern.holiday_type || '公休',
          scheduled_in: null,
          scheduled_out: null,
          work_type: 'in_facility',
          is_offsite: 0
        };
      } else {
        const isOffsite = matchedPattern.work_type === 'out_facility';
        return {
          is_holiday: 0,
          work_type: matchedPattern.work_type || 'in_facility',
          is_offsite: isOffsite ? 1 : 0,
          scheduled_in: matchedPattern.start_time || client.work_start_time || "10:00",
          scheduled_out: matchedPattern.end_time || client.work_end_time || (isOffsite ? "16:00" : "15:00"),
          offsite_workplace_id: matchedPattern.offsite_workplace_id,
          offsite_workplace_name: matchedPattern.offsite_workplace_name,
        };
      }
    }

    // 従来の regular_holidays フォールバック
    try {
      const arr = JSON.parse(client.regular_holidays || "[]");
      if (arr.includes(dayOfWeek.toString())) {
        return {
          is_holiday: 1,
          holiday_type: '公休',
          scheduled_in: null,
          scheduled_out: null,
          work_type: 'in_facility',
          is_offsite: 0
        };
      }
    } catch(e) {}

    return {
      is_holiday: 0,
      work_type: 'in_facility',
      is_offsite: 0,
      scheduled_in: client.work_start_time || "10:00",
      scheduled_out: client.work_end_time || "15:00"
    };
  };

  // 各拠点（本所・サテライト）の内勤上限・受入状況分析
  const dailyDeptCapacityAnalysis = useMemo(() => {
    const deptList = departments && departments.length > 0
      ? departments.map(d => ({
          id: d.id,
          name: d.name,
          facilityType: (d.facility_type === 'satellite' ? 'satellite' : 'main') as 'main' | 'satellite',
          facilityTypeLabel: d.facility_type === 'satellite' ? 'サテライト' : '本所',
          capacity: Number(d.capacity) || 10
        }))
      : [
          { id: 'main', name: '本所', facilityType: 'main' as const, facilityTypeLabel: '本所', capacity: 10 },
          { id: 'satellite', name: 'サテライト', facilityType: 'satellite' as const, facilityTypeLabel: 'サテライト', capacity: 10 }
        ];

    const dailyMap: Record<string, {
      hasOverflow: boolean;
      totalOverflow: number;
      depts: Record<string, {
        name: string;
        facilityType: 'main' | 'satellite';
        facilityTypeLabel: string;
        capacity: number;
        dailyLimit: number;
        inFacilityCount: number;
        offsiteCount: number;
        totalPresentCount: number;
        overflow: number;
        overflowCount: number;
        isOverCapacity: boolean;
        clients: { client: any; isOffsite: boolean; isInFacility: boolean; workType: string }[];
      }>;
    }> = {};

    daysArray.forEach(d => {
      let dayHasOverflow = false;
      let dayTotalOverflow = 0;
      const deptMap: Record<string, any> = {};

      deptList.forEach(dep => {
        const dailyLimit = dep.capacity <= 50 ? Math.floor(dep.capacity * 1.5) : dep.capacity + 25;
        deptMap[dep.id] = {
          name: dep.name,
          facilityType: dep.facilityType,
          facilityTypeLabel: dep.facilityTypeLabel,
          capacity: dep.capacity,
          dailyLimit,
          inFacilityCount: 0,
          offsiteCount: 0,
          totalPresentCount: 0,
          overflow: 0,
          overflowCount: 0,
          isOverCapacity: false,
          clients: []
        };
      });

      initialClients.forEach(c => {
        let depId = c.department_id;
        if (!depId || !deptMap[depId]) {
          depId = deptList[0]?.id;
        }
        if (!deptMap[depId]) return;

        const shift = shiftsData[c.id]?.[d.dateStr];
        const att = attendanceData[c.id]?.[d.dateStr];

        let isHoliday = false;
        let isOffsite = false;
        let isInFacility = false;
        let workType = 'in_facility';

        // シフト編成・振替操作（shift）を優先評価（施設外振替や公休設定の即時反映を保証）
        if (shift && (shift.work_type === 'out_facility' || shift.is_offsite === 1)) {
          if (shift.is_holiday === 1) {
            isHoliday = true;
          } else {
            isOffsite = true;
            workType = 'out_facility';
          }
        } else if (shift && shift.work_type === 'both') {
          isOffsite = true;
          isInFacility = true;
          workType = 'both';
        } else if (shift && shift.is_holiday === 1) {
          isHoliday = true;
        } else if (att) {
          if (att.status === 'absent' || att.status === 'holiday') {
            isHoliday = true;
          } else if (att.work_type === 'both') {
            isOffsite = true;
            isInFacility = true;
            workType = 'both';
          } else if (att.work_type === 'out_facility' || att.is_offsite === 1) {
            isOffsite = true;
            workType = 'out_facility';
          } else {
            isInFacility = true;
            workType = 'in_facility';
          }
        } else if (shift) {
          isInFacility = true;
          workType = 'in_facility';
        } else {
          // 未登録日は曜日別パターン（weekly_schedule_pattern）から判定
          const vShift = getClientVirtualShift(c, d.dateStr, d.dayOfWeek, null);
          if (vShift.is_holiday) {
            isHoliday = true;
          } else if (vShift.work_type === 'both') {
            isOffsite = true;
            isInFacility = true;
            workType = 'both';
          } else if (vShift.work_type === 'out_facility' || vShift.is_offsite === 1) {
            isOffsite = true;
            workType = 'out_facility';
          } else {
            isInFacility = true;
            workType = 'in_facility';
          }
        }

        if (!isHoliday) {
          deptMap[depId].clients.push({ client: c, isOffsite, isInFacility, workType });
          if (isInFacility) {
            deptMap[depId].inFacilityCount++;
          }
          if (isOffsite) {
            deptMap[depId].offsiteCount++;
          }
        }
      });

      deptList.forEach(dep => {
        const item = deptMap[dep.id];
        // 施設外就労は定員枠と別カウントとなるため、内勤人数（inFacilityCount）のみで定員超過を判定する
        const overflow = Math.max(0, item.inFacilityCount - item.dailyLimit);
        item.overflow = overflow;
        item.overflowCount = overflow;
        item.totalPresentCount = item.inFacilityCount + item.offsiteCount;
        item.isOverCapacity = overflow > 0;
        if (item.isOverCapacity) {
          dayHasOverflow = true;
          dayTotalOverflow += overflow;
        }
      });

      dailyMap[d.dateStr] = {
        hasOverflow: dayHasOverflow,
        totalOverflow: dayTotalOverflow,
        depts: deptMap
      };
    });

    return { deptList, dailyMap };
  }, [initialClients, departments, shiftsData, attendanceData, daysArray]);

  // 内勤上限超過日の一覧
  const overflowDaysList = useMemo(() => {
    const list: {
      dateStr: string;
      dayOfWeek: string;
      deptId: string;
      deptName: string;
      facilityTypeLabel: string;
      inFacilityCount: number;
      totalPresentCount: number;
      capacity: number;
      dailyLimit: number;
      overflow: number;
      overflowCount: number;
      clients: { client: any; isOffsite: boolean; isInFacility: boolean; workType: string }[];
    }[] = [];

    daysArray.forEach(d => {
      const dayData = dailyDeptCapacityAnalysis.dailyMap[d.dateStr];
      if (!dayData || !dayData.hasOverflow) return;

      dailyDeptCapacityAnalysis.deptList.forEach(dep => {
        const depStatus = dayData.depts[dep.id];
        if (depStatus && depStatus.isOverCapacity) {
          list.push({
            dateStr: d.dateStr,
            dayOfWeek: `(${DAY_KANJI[d.dayOfWeek]})`,
            deptId: dep.id,
            deptName: depStatus.name,
            facilityTypeLabel: depStatus.facilityTypeLabel,
            inFacilityCount: depStatus.inFacilityCount,
            totalPresentCount: depStatus.totalPresentCount,
            capacity: depStatus.capacity,
            dailyLimit: depStatus.dailyLimit,
            overflow: depStatus.overflow,
            overflowCount: depStatus.overflowCount,
            clients: depStatus.clients
          });
        }
      });
    });

    return list;
  }, [daysArray, dailyDeptCapacityAnalysis, DAY_KANJI]);

  // クイック調整モーダルState
  const [adjustingModal, setAdjustingModal] = useState<{
    dateStr: string;
    dayOfWeek: string;
    deptId: string;
    deptName: string;
    facilityTypeLabel: string;
    capacity: number;
    dailyLimit: number;
    inFacilityCount: number;
    totalPresentCount: number;
    overflow: number;
    overflowCount: number;
    clients: { client: any; isOffsite: boolean; isInFacility: boolean; workType: string }[];
  } | null>(null);

  const handleQuickSwitchToOffsite = async (clientId: string, targetDate: string, toOffsite: boolean) => {
    const newWorkType = toOffsite ? 'out_facility' : 'in_facility';
    try {
      const res = await toggleClientWorkType(clientId, targetDate, newWorkType);
      if (res.success) {
        setShiftsData(prev => {
          const cMap = { ...(prev[clientId] || {}) };
          cMap[targetDate] = {
            ...(cMap[targetDate] || {}),
            client_id: clientId,
            target_date: targetDate,
            scheduled_in: (cMap[targetDate]?.scheduled_in) || '10:00',
            scheduled_out: (cMap[targetDate]?.scheduled_out) || '15:00',
            work_type: newWorkType,
            is_offsite: toOffsite ? 1 : 0,
            is_holiday: 0
          };
          return { ...prev, [clientId]: cMap };
        });

        // 該当日のattendanceDataも即座に同期更新（出退勤実績が存在する日の再計算ズレを解消）
        setAttendanceData(prev => {
          if (!prev[clientId] || !prev[clientId][targetDate]) return prev;
          const cMap = { ...prev[clientId] };
          cMap[targetDate] = {
            ...cMap[targetDate],
            work_type: newWorkType,
            is_offsite: toOffsite ? 1 : 0
          };
          return { ...prev, [clientId]: cMap };
        });

        // モーダル内の利用者リスト状態もリアルタイム更新
        setAdjustingModal(prev => {
          if (!prev) return null;
          const updatedClients = prev.clients.map(item => {
            if (item.client.id === clientId) {
              return {
                ...item,
                isOffsite: toOffsite,
                isInFacility: !toOffsite,
                workType: newWorkType
              };
            }
            return item;
          });
          const newInFacilityCount = updatedClients.filter(item => item.isInFacility).length;
          const newOverflow = Math.max(0, newInFacilityCount - prev.capacity);
          return {
            ...prev,
            clients: updatedClients,
            inFacilityCount: newInFacilityCount,
            overflow: newOverflow
          };
        });

        toast.success(toOffsite ? "施設外就労に振り替えました" : "内勤に戻しました");
      } else {
        toast.error(res.error || "変更に失敗しました");
      }
    } catch (e: any) {
      toast.error(e.message || "通信エラーが発生しました");
    }
  };

  // 配置基準パラメータ
  const staffingClass = facilitySettings?.staffing_class || '10:1';
  const ratioDenominator = staffingClass === '7.5:1' ? 7.5 : 10;
  const regularHoursPerWeek = facilitySettings?.regular_hours_per_week ? Number(facilitySettings.regular_hours_per_week) : 40;
  const dailyStandardMinutes = Math.max(240, Math.round((regularHoursPerWeek / 5) * 60));

  // 部署別・日別の人員配置分析（内勤と施設外就労を完全分離）
  const dailyStaffingAnalysis = useMemo(() => {
    const result: Record<string, {
      totalClientCount: number;
      inFacilityClientCount: number;
      offsiteClientCount: number;
      staffCount: number;
      staffMinutes: number;
      offsiteStaffCount: number;
      requiredMinutes: number;
      status: 'none' | 'compliant' | 'shortage' | 'no_staff' | 'no_offsite_staff';
      statusText: string;
      detailText: string;
    }> = {};

    daysArray.forEach(d => {
      // 1. 当日の対象利用者の通所予定数（内勤 vs 施設外就労を分離）
      let inFacilityClientCount = 0;
      let offsiteClientCount = 0;

      filteredClients.forEach(c => {
        const shift = shiftsData[c.id]?.[d.dateStr];
        const vShift = getClientVirtualShift(c, d.dateStr, d.dayOfWeek, shift);

        if (vShift && !vShift.is_holiday) {
          if (vShift.work_type === 'both') {
            offsiteClientCount++;
            inFacilityClientCount++;
          } else if (vShift.work_type === 'out_facility' || vShift.is_offsite === 1) {
            offsiteClientCount++;
          } else {
            inFacilityClientCount++;
          }
        }
      });

      const totalClientCount = inFacilityClientCount + offsiteClientCount;

      if (totalClientCount === 0) {
        result[d.dateStr] = {
          totalClientCount: 0,
          inFacilityClientCount: 0,
          offsiteClientCount: 0,
          staffCount: 0,
          staffMinutes: 0,
          offsiteStaffCount: 0,
          requiredMinutes: 0,
          status: 'none',
          statusText: '-',
          detailText: '通所予定なし'
        };
        return;
      }

      // 対象部署または事業所全体に所属する直接処遇職員
      const targetStaff = initialStaff.filter((s: any) => {
        const inDept = selectedDeptId === 'all'
          ? true
          : (selectedDeptId === 'none' ? !s.department_id : s.department_id === selectedDeptId);
        const isDirect = (s.job_category === '職業指導員' || s.job_category === '生活支援員' || s.role === 'vocational_instructor' || s.role === 'life_support_worker');
        const isExcluded = ['管理者', 'サービス管理責任者', '管理者 兼 サービス管理責任者', '賃金向上達成指導員', '視覚・聴覚言語障害者支援指導員', '重度者支援専門員', '送迎ドライバー', '調理員', '事務員'].includes(s.job_category);
        return inDept && isDirect && !isExcluded;
      });

      let staffCount = 0;
      let staffMinutes = 0;
      let offsiteStaffCount = 0;

      targetStaff.forEach((s: any) => {
        const sShift = staffShiftsMap[s.id]?.[d.dateStr];
        let virtualShift = sShift;
        if (!sShift) {
          try {
            const arr = JSON.parse(s.regular_holidays || '[]');
            const dObj = new Date(d.dateStr);
            if (arr.includes(dObj.getDay().toString())) {
              virtualShift = { is_holiday: 1 };
            } else {
              virtualShift = { is_holiday: 0, scheduled_in: '09:00', scheduled_out: '17:00' };
            }
          } catch(e) {
            virtualShift = { is_holiday: 0, scheduled_in: '09:00', scheduled_out: '17:00' };
          }
        }

        // GH勤務等の他事業所勤務は完全除外
        const isOther = virtualShift?.is_other_facility || ['GH', 'GH勤務', 'グループホーム', '他事業所', '他事業所勤務'].includes(virtualShift?.holiday_type);
        if (isOther || virtualShift?.is_holiday) return;

        // 施設外就労引率職員判定
        const isOffsiteStaff = virtualShift?.is_offsite === 1 || ['施設外', '施設外引率', '引率'].includes(virtualShift?.holiday_type);

        if (virtualShift && (virtualShift.scheduled_in || virtualShift.scheduled_out)) {
          let sin = virtualShift.scheduled_in || '09:00';
          let sout = virtualShift.scheduled_out || '17:00';
          const [sh, sm] = sin.split(':').map(Number);
          const [eh, em] = sout.split(':').map(Number);
          let inM = sh * 60 + sm;
          let outM = eh * 60 + em;
          if (outM < inM) outM += 24 * 60;
          let dur = outM - inM;
          if (dur >= 360) dur -= 60;
          if (dur > 0) {
            const hasSplitTimes = virtualShift.work_type === 'both' || (virtualShift.offsite_start_time && virtualShift.offsite_end_time);
            if (hasSplitTimes && virtualShift.offsite_start_time && virtualShift.offsite_end_time) {
              const [osh, osm] = virtualShift.offsite_start_time.split(':').map(Number);
              const [oeh, oem] = virtualShift.offsite_end_time.split(':').map(Number);
              let offsiteM = (oeh * 60 + oem) - (osh * 60 + osm);
              if (offsiteM < 0) offsiteM += 24 * 60;
              offsiteM = Math.min(dur, Math.max(0, offsiteM));
              const inFacilityM = Math.max(0, dur - offsiteM);
              if (offsiteM > 0) offsiteStaffCount++;
              if (inFacilityM > 0) {
                staffMinutes += inFacilityM;
                staffCount++;
              }
            } else if (isOffsiteStaff) {
              offsiteStaffCount++;
            } else {
              staffMinutes += dur;
              staffCount++;
            }
          }
        }
      });

      // ※施設外利用者は内勤必要人数の分母から除外
      const requiredFte = inFacilityClientCount > 0 ? Math.round((inFacilityClientCount / ratioDenominator) * 100) / 100 : 0;
      const requiredMinutes = Math.round(requiredFte * dailyStandardMinutes);

      // 法令必須チェック：
      // 1. 内勤利用者がいるのに内勤職員不在（常時1名配置違反）
      if (inFacilityClientCount > 0 && (staffCount === 0 || staffMinutes === 0)) {
        result[d.dateStr] = {
          totalClientCount,
          inFacilityClientCount,
          offsiteClientCount,
          staffCount: 0,
          staffMinutes: 0,
          offsiteStaffCount,
          requiredMinutes,
          status: 'no_staff',
          statusText: selectedDeptId !== 'all' ? '⚠️拠点不在' : '⚠️本体不在',
          detailText: `【常時配置違反】内勤利用者${inFacilityClientCount}名に対し、${selectedDeptId !== 'all' ? 'この拠点に' : '事業所に'}直接処遇職員が配置されていません（開所中常時1名以上必須）`
        };
      } else if (offsiteClientCount > 0 && offsiteStaffCount === 0) {
        // 2. 施設外就労利用者がいるのに引率職員不在
        result[d.dateStr] = {
          totalClientCount,
          inFacilityClientCount,
          offsiteClientCount,
          staffCount,
          staffMinutes,
          offsiteStaffCount: 0,
          requiredMinutes,
          status: 'no_offsite_staff',
          statusText: '⚠️引率不在',
          detailText: `【引率未配置】施設外就労利用者${offsiteClientCount}名に対し、専任の引率職員が配置されていません（現場専従必須）`
        };
      } else if (selectedDeptId === 'all' && inFacilityClientCount > 0 && staffMinutes < requiredMinutes) {
        // 3. 事業所全体の比率不足（※7.5:1/10:1は事業所全体で満たす法令要件）
        const shortageH = Math.round(((requiredMinutes - staffMinutes) / 60) * 10) / 10;
        result[d.dateStr] = {
          totalClientCount,
          inFacilityClientCount,
          offsiteClientCount,
          staffCount,
          staffMinutes,
          offsiteStaffCount,
          requiredMinutes,
          status: 'shortage',
          statusText: `⚠️不足(-${shortageH}h)`,
          detailText: `【全体配置不足】内勤必要:${Math.round(requiredMinutes/60*10)/10}h / 内勤実配置:${Math.round(staffMinutes/60*10)/10}h (${staffCount}名)`
        };
      } else {
        // 4. 適合
        const isDeptView = selectedDeptId !== 'all';
        result[d.dateStr] = {
          totalClientCount,
          inFacilityClientCount,
          offsiteClientCount,
          staffCount,
          staffMinutes,
          offsiteStaffCount,
          requiredMinutes,
          status: 'compliant',
          statusText: isDeptView ? '✓常時配置OK' : '✓充足',
          detailText: isDeptView
            ? `【拠点常時配置OK】直接処遇職員${staffCount}名配置（開所中常時1名以上充足）。※法定比率(${staffingClass})は事業所全体で合算判定されます`
            : `【基準充足】内勤利用者:${inFacilityClientCount}名(職員:${staffCount}名), 施設外利用者:${offsiteClientCount}名(引率:${offsiteStaffCount}名)`
        };
      }
    });

    return result;
  }, [daysArray, filteredClients, shiftsData, selectedDeptId, initialStaffingMap, initialStaff, staffShiftsMap, ratioDenominator, dailyStandardMinutes]);

  const getDayColor = (dayOfWeek: number) => {
    if (dayOfWeek === 0) return "text-red-500 bg-red-50";
    if (dayOfWeek === 6) return "text-blue-500 bg-blue-50";
    return "";
  };

  const handleCellChange = useCallback(async (clientId: string, dateStr: string, value: string) => {
    let is_holiday = false;
    let holiday_type = null;
    let scheduled_in = null;
    let scheduled_out = null;
    let work_type: 'in_facility' | 'out_facility' | 'both' = 'in_facility';
    let is_offsite = 0;
    let offsite_start_time: string | null = null;
    let offsite_end_time: string | null = null;
    let in_facility_start_time: string | null = null;
    let in_facility_end_time: string | null = null;
    let displayValue = value.trim();

    const parseTime = (t: string) => {
      t = t.replace('時', ':').replace('分', '').trim();
      if (!t.includes(':')) {
        if (t.length <= 2) t = t + '00';
        if (t.length === 3 || t.length === 4) {
          const padded = t.padStart(4, '0');
          t = `${padded.substring(0,2)}:${padded.substring(2,4)}`;
        }
      }
      if (t.includes(':')) {
        const [h, m] = t.split(':');
        return `${h.padStart(2, '0')}:${(m || '00').padEnd(2, '0')}`;
      }
      return t;
    };

    const offsiteKeywords = ['施設外', '施設外就労', '外', '外勤', '施設外利用'];
    const isOffsiteOnly = offsiteKeywords.includes(displayValue);

    // 外勤・内勤併用パターン（例: 外10-12 内13-15, 併用）
    const isBothPattern = (displayValue.includes('外') && displayValue.includes('内')) || 
                          displayValue === '併用' || displayValue === '両方';

    if (isBothPattern) {
      is_holiday = false;
      is_offsite = 1;
      work_type = 'both';
      holiday_type = '併用';

      const offMatch = displayValue.match(/(?:外|施設外|引率)[:\s]*([0-9:]{1,5})[-~～]([0-9:]{1,5})/);
      const inMatch = displayValue.match(/(?:内|施設内|内勤)[:\s]*([0-9:]{1,5})[-~～]([0-9:]{1,5})/);

      if (offMatch && inMatch) {
        offsite_start_time = parseTime(offMatch[1]);
        offsite_end_time = parseTime(offMatch[2]);
        in_facility_start_time = parseTime(inMatch[1]);
        in_facility_end_time = parseTime(inMatch[2]);
        scheduled_in = offsite_start_time < in_facility_start_time ? offsite_start_time : in_facility_start_time;
        scheduled_out = offsite_end_time > in_facility_end_time ? offsite_end_time : in_facility_end_time;
      } else if (displayValue === '併用' || displayValue === '両方') {
        offsite_start_time = '10:00';
        offsite_end_time = '12:00';
        in_facility_start_time = '13:00';
        in_facility_end_time = '15:00';
        scheduled_in = '10:00';
        scheduled_out = '15:00';
      } else {
        toast.error('入力形式が正しくありません (例: 外10-12 内13-15)');
        return;
      }
    } else if (isOffsiteOnly) {
      is_holiday = false;
      work_type = 'out_facility';
      is_offsite = 1;
      scheduled_in = '09:00';
      scheduled_out = '17:00';
      holiday_type = '施設外就労';
    } else if (displayValue === '休' || displayValue === '公休' || displayValue === '有給休暇' || displayValue === '有休' || displayValue === '特休' || displayValue === '年休' || displayValue === '時間休') {
      is_holiday = true;
      holiday_type = (displayValue === '有休' || displayValue === '年休') ? '有給休暇' : displayValue;
    } else if (displayValue.includes('-') || displayValue.includes('\n') || displayValue.includes('~') || displayValue.includes('～')) {
      let cleanVal = displayValue;
      if (cleanVal.includes('施設外') || cleanVal.includes('外勤') || cleanVal.includes('外')) {
        work_type = 'out_facility';
        is_offsite = 1;
        holiday_type = '施設外就労';
        cleanVal = cleanVal.replace(/施設外就労|施設外|外勤|外/g, '').trim();
      }
      let parts = cleanVal.split(/[-~～\n]/).filter(p => p.trim() !== '');
      if (parts.length === 2) {
        scheduled_in = parseTime(parts[0]);
        scheduled_out = parseTime(parts[1]);
        if (!is_offsite) {
          work_type = 'in_facility';
        }
      } else if (parts.length === 0 && is_offsite) {
        scheduled_in = '09:00';
        scheduled_out = '17:00';
      }
    } else if (displayValue === '') {
      is_holiday = false;
      scheduled_in = null;
      scheduled_out = null;
      work_type = 'in_facility';
      is_offsite = 0;
      offsite_start_time = null;
      offsite_end_time = null;
      in_facility_start_time = null;
      in_facility_end_time = null;
    } else {
      toast.error('入力形式が正しくありません (例: 0900-1700, 外10-12 内13-15, 施設外, 公休, 有休)');
      return;
    }

    setShiftsData(prev => {
      const next = { ...prev };
      if (!next[clientId]) next[clientId] = {};
      next[clientId] = {
        ...next[clientId],
        [dateStr]: { 
          client_id: clientId, 
          target_date: dateStr, 
          is_holiday, 
          holiday_type, 
          scheduled_in, 
          scheduled_out,
          work_type,
          is_offsite,
          offsite_start_time,
          offsite_end_time,
          in_facility_start_time,
          in_facility_end_time
        }
      };
      return next;
    });

    const res = await saveClientShift(clientId, dateStr, { 
      scheduled_in, 
      scheduled_out, 
      is_holiday, 
      holiday_type, 
      work_type, 
      is_offsite,
      offsite_start_time,
      offsite_end_time,
      in_facility_start_time,
      in_facility_end_time
    });
    if (!res.success) {
      toast.error(`保存に失敗しました: ${dateStr}`);
    } else {
      if (work_type === 'both') {
        toast.info(`${dateStr}: 施設外(${offsite_start_time}~${offsite_end_time})＋内勤(${in_facility_start_time}~${in_facility_end_time})として保存しました`);
      } else if (is_offsite) {
        toast.info(`${dateStr}: 施設外就労として保存しました（内勤配置から除外）`);
      }
    }
  }, []);

  const formatCellValue = (shift: any) => {
    if (!shift) return "";
    const isBoth = shift.work_type === 'both' || (shift.offsite_start_time && shift.in_facility_start_time);
    if (isBoth) {
      const formatTime = (t: string) => {
        if (!t) return '';
        const [h, m] = t.split(':');
        return `${parseInt(h, 10)}:${m}`;
      };
      const offStr = shift.offsite_start_time ? `外${formatTime(shift.offsite_start_time)}-${formatTime(shift.offsite_end_time)}` : '';
      const inStr = shift.in_facility_start_time ? `内${formatTime(shift.in_facility_start_time)}-${formatTime(shift.in_facility_end_time)}` : '';
      return `${offStr}\n${inStr}`.trim();
    }
    const isOffsite = shift.is_offsite === 1 || shift.work_type === 'out_facility';
    if (shift.is_holiday) return shift.holiday_type || "公休";
    if (shift.scheduled_in && shift.scheduled_out) {
      const formatTime = (t: string) => {
        if (!t) return '';
        const [h, m] = t.split(':');
        return `${parseInt(h, 10)}:${m}`;
      };
      const timeStr = `${formatTime(shift.scheduled_in)}\n${formatTime(shift.scheduled_out)}`;
      return isOffsite ? `${timeStr}\n(施設外)` : timeStr;
    }
    if (isOffsite) return "施設外";
    return "";
  };

  const handlePrevMonth = () => {
    const [y, m] = currentMonth.split('-').map(Number);
    const prev = new Date(y, m - 2, 1);
    const newYm = `${prev.getFullYear()}-${(prev.getMonth() + 1).toString().padStart(2, '0')}`;
    window.location.href = `/attendance/shifts?month=${newYm}`;
  };

  const handleNextMonth = () => {
    const [y, m] = currentMonth.split('-').map(Number);
    const next = new Date(y, m, 1);
    const newYm = `${next.getFullYear()}-${(next.getMonth() + 1).toString().padStart(2, '0')}`;
    window.location.href = `/attendance/shifts?month=${newYm}`;
  };

  return (
    <div className="space-y-6 print:space-y-0 print:m-0 print:p-0">
      <style dangerouslySetInnerHTML={{ __html: `
        @page {
          size: A4 landscape;
          margin: 6mm 6mm;
        }
        @media print {
          body {
            background: white !important;
            color: black !important;
            -webkit-print-color-adjust: exact;
            print-color-adjust: exact;
          }
          /* 印刷不要な全UI要素・超過アラート・フッター診断を徹底非表示 */
          header, nav, aside, .print\\:hidden, [class*="print:hidden"], tfoot, .screen-only-element, .holiday-alert-badge {
            display: none !important;
            visibility: hidden !important;
            height: 0 !important;
            overflow: hidden !important;
          }
          .print\\:block {
            display: block !important;
          }
          table {
            width: 100% !important;
            min-width: 0 !important;
            font-size: 7pt !important;
            line-height: 1.05 !important;
            table-layout: fixed !important;
            page-break-inside: auto;
          }
          tr {
            page-break-inside: avoid !important;
            break-inside: avoid !important;
          }
          th, td {
            min-width: 0 !important;
            padding: 1px 0.5px !important;
            overflow: hidden !important;
          }
        }
      `}} />

      {/* 印刷用公式ヘッダー（通常時は非表示） */}
      <div className="hidden print:block mb-2 border-b-2 border-slate-900 pb-1">
        <div className="flex justify-between items-end">
          <div className="flex-1 mr-4">
            <span className="text-[8px] font-mono text-slate-500">就労継続支援Ａ型 月間通所予定</span>
            <h1 className="text-base font-bold text-slate-900 leading-tight">利用者通所予定表</h1>
            <div className="flex items-center gap-1 text-[9px] text-slate-700">
              <span className="shrink-0">事業所名:</span>
              <div className="max-w-[240px]">
                <AutoFitText text={facilitySettings?.facility_name || '就労継続支援A型事業所'} maxFontSizePt={9} minFontSizePt={6.5} className="font-bold" />
              </div>
              <span className="shrink-0 text-slate-500">（番号: {facilitySettings?.facility_number || '2810099999'}）</span>
            </div>
          </div>
          <div className="flex items-center gap-3 shrink-0">
            <div className="text-right">
              <div className="text-xs font-bold text-slate-900 font-mono">
                {currentMonth.replace('-', '年')}月分
              </div>
              <div className="text-[8px] text-slate-500">
                登録利用者数: {initialClients?.length || 0}名
              </div>
            </div>
            <div className="flex gap-1">
              <div className="text-center">
                <div className="border border-slate-400 bg-slate-50 px-1.5 py-0 text-[7px] font-bold">管理者</div>
                <div className="h-8 w-8 border-b border-l border-r border-slate-400 flex items-center justify-center text-[7px] text-slate-400">印</div>
              </div>
              <div className="text-center">
                <div className="border border-slate-400 bg-slate-50 px-1.5 py-0 text-[7px] font-bold">作成者</div>
                <div className="h-8 w-8 border-b border-l border-r border-slate-400 flex items-center justify-center text-[7px] text-slate-400">印</div>
              </div>
            </div>
          </div>
        </div>
      </div>

      <div className="flex justify-between items-center bg-white p-4 rounded-md border shadow-sm flex-wrap gap-4 print:hidden">
        <h2 className="text-xl font-bold flex items-center gap-4">
          <Button variant="outline" size="icon" onClick={handlePrevMonth}>
            <ChevronLeft className="h-4 w-4" />
          </Button>
          {currentMonth.replace('-', '年')}月 利用者シフト・予実管理
          <Button variant="outline" size="icon" onClick={handleNextMonth}>
            <ChevronRight className="h-4 w-4" />
          </Button>
        </h2>
        <div className="flex items-center gap-4 flex-wrap">
          <div className="text-sm flex items-center gap-4">
            <div className="flex items-center gap-1 text-muted-foreground text-xs">
              <span className="inline-block w-3 h-3 bg-green-50 border border-green-200 rounded-sm"></span> 実績あり
            </div>
            <div className="flex items-center gap-1 text-muted-foreground text-xs">
              <span className="inline-block w-3 h-3 bg-red-50 border border-red-200 rounded-sm"></span> 公休・事前休
            </div>
            <div className="flex items-center gap-1 text-muted-foreground text-xs">
              <span className="inline-block w-3 h-3 bg-yellow-100 border border-yellow-300 rounded-sm"></span> 予定あり・打刻なし
            </div>
          </div>
          <Button 
            onClick={() => window.print()}
            variant="outline" 
            className="gap-1.5 border-slate-300 bg-white hover:bg-slate-50 text-slate-800 shadow-xs shrink-0 font-bold"
          >
            <Printer className="h-4 w-4 text-slate-600" />
            通所予定表を印刷 (A4横)
          </Button>
        </div>
      </div>

      {/* 1日利用定員超過アラート（150% または 定員+25人） */}

export async function getClientList() {
  try {
    const role = await getRole();
    if (!role) return [];

    return db.prepare(`
      SELECT c.id, c.full_name as name, c.regular_holidays, c.weekly_schedule_pattern, c.work_start_time, c.work_end_time, c.department_id, d.name as department_name
      FROM clients c
      LEFT JOIN departments d ON c.department_id = d.id
      WHERE c.deleted_at IS NULL AND (c.status IS NULL OR c.status = 'active')
      ORDER BY c.created_at ASC
    `).all();
  } catch (error) {
    console.error("Failed to fetch clients:", error);
    return [];
  }
}

export async function getClientShifts(yearMonth: string) {
  try {
    const role = await getRole();
    if (!role) return [];

    return db.prepare(`
      SELECT * FROM client_shifts 
      WHERE target_date LIKE ?
    `).all(`${yearMonth}-%`);
  } catch (error) {
    console.error("Failed to fetch client shifts:", error);
    return [];
  }
}

export async function toggleClientWorkType(
  clientId: string, 
  targetDate: string, 
  targetWorkType: 'in_facility' | 'out_facility' | 'both',
  offsiteWorkplaceId?: string,
  offsiteWorkplaceName?: string
) {
  try {
    const role = await getRole();
    if (!role) return { success: false, error: "認証されていません" };
    await assertCanAccessClient(clientId);

    const existing = db.prepare('SELECT * FROM client_shifts WHERE client_id = ? AND target_date = ?').get(clientId, targetDate) as any;
    const isOffsite = targetWorkType === 'out_facility' || targetWorkType === 'both' ? 1 : 0;
    const safeOffsiteWorkplaceId = sanitizeForeignKeyId('offsite_workplaces', offsiteWorkplaceId);

    if (existing) {
      db.prepare(`
        UPDATE client_shifts 
        SET work_type = ?, is_offsite = ?, 
            offsite_workplace_id = COALESCE(?, offsite_workplace_id),
            offsite_workplace_name = COALESCE(?, offsite_workplace_name),
            is_holiday = 0
        WHERE id = ?
      `).run(targetWorkType, isOffsite, safeOffsiteWorkplaceId, offsiteWorkplaceName || null, existing.id);
    } else {
      db.prepare(`
        INSERT INTO client_shifts (
          id, client_id, target_date, scheduled_in, scheduled_out, is_holiday, 
          work_type, is_offsite, offsite_workplace_id, offsite_workplace_name
        )
        VALUES (?, ?, ?, '10:00', '15:00', 0, ?, ?, ?, ?)
      `).run(
        randomUUID(),
        clientId,
        targetDate,
        targetWorkType,
        isOffsite,
        safeOffsiteWorkplaceId,
        offsiteWorkplaceName || null
      );
    }
    
    // 該当日に出勤実績（attendance_records）が存在する場合も同期して更新（定員判定・実績の不整合防止）
    try {
      db.prepare(`
        UPDATE attendance_records
        SET work_type = ?, is_offsite = ?,
            offsite_workplace_id = COALESCE(?, offsite_workplace_id)
        WHERE client_id = ? AND record_date = ? AND deleted_at IS NULL
      `).run(targetWorkType, isOffsite, safeOffsiteWorkplaceId, clientId, targetDate);
    } catch (attErr) {
      console.warn("attendance_records sync skipped:", attErr);
    }

    const { revalidatePath } = await import("next/cache");
    revalidatePath('/attendance/shifts');
    revalidatePath('/attendance');
    revalidatePath('/offsite');
    revalidatePath('/');

    return { success: true };
  } catch (error: any) {
    console.error("Failed to toggle work type:", error);
    return { success: false, error: error.message || "勤務形態の変更に失敗しました" };
  }
}

/**
 * 厚生労働省告示・留意事項通知準拠:
 * 個別支援計画未作成減算の未作成継続月数を計算（内部用同期関数）
 * 戻り値:
 *  0: 有効な計画あり（減算なし）
 *  1〜2: 未作成1〜2ヶ月目（所定単位数の100分の70＝30%減算）
 *  3以上: 未作成3ヶ月以上継続（所定単位数の100分の50＝50%減算）
 */
function getPlanUncreatedMonthsSync(clientId: string, targetMonth: string): number {
  const activePlans = db.prepare(`
    SELECT start_date, end_date, consent_date, delivery_date FROM support_plans 
    WHERE client_id = ? AND status = 'active' AND deleted_at IS NULL
    ORDER BY end_date DESC, start_date DESC
  `).all(clientId) as any[];

  // 有効期間内にある計画が存在するか判定（実地指導遵守：同意日および交付日が存在し適法に作成された計画のみ有効）
  const hasValidPlan = activePlans.some(p => {
    // 同意日と交付日の両方が完了していることが厚労省指定基準上の必須要件
    if (!p.consent_date || !p.delivery_date) return false;

    const pStart = p.start_date ? p.start_date.substring(0, 7) : null;
    const pEnd = p.end_date ? p.end_date.substring(0, 7) : null;
    if (!pStart) return true;
    if (pEnd) return targetMonth >= pStart && targetMonth <= pEnd;
    return targetMonth >= pStart;
  });

  if (hasValidPlan) {
    return 0; // 減算なし
  }

  // 直近で終了した計画の終了月を取得
  let lastEndMonth: string | null = null;
  for (const p of activePlans) {
    if (p.end_date) {
      const em = p.end_date.substring(0, 7);
      if (em < targetMonth) {
        if (!lastEndMonth || em > lastEndMonth) {
          lastEndMonth = em;
        }
      }
    }
  }

  if (lastEndMonth) {
    // 終了月の翌月が未作成1ヶ月目
    const [tY, tM] = targetMonth.split('-').map(Number);
    const [eY, eM] = lastEndMonth.split('-').map(Number);
    const monthDiff = (tY - eY) * 12 + (tM - eM);
    return Math.max(1, monthDiff);
  }

  // 過去に計画が存在しない場合、利用開始日（service_start_date）から計算
  const client = db.prepare('SELECT service_start_date, contract_date FROM clients WHERE id = ?').get(clientId) as any;
  const startStr = client?.service_start_date || client?.contract_date;
  if (startStr) {
    const startMonth = startStr.substring(0, 7);
    const [tY, tM] = targetMonth.split('-').map(Number);
    const [sY, sM] = startMonth.split('-').map(Number);
    const monthDiff = (tY - sY) * 12 + (tM - sM);
    return Math.max(1, monthDiff + 1);
  }

  return 1;
}

        const severeUnit = facilitySetting.severe_disability_addition === 'severe_2' 
          ? (billingRules['severe_disability_2'] ?? 23) 
          : (billingRules['severe_disability_1'] ?? 47);
        const severeCode = facilitySetting.severe_disability_addition === 'severe_2' ? CODE_SEVERE_2 : CODE_SEVERE_1;
        const severeTotal = actualDays * severeUnit;
        clientTotalUnits += severeTotal;
        itemDetails.push(`${recordTypeItemDetail},${facilityNumber},${recipientNumber},${serviceTypeCode},${severeCode},${severeUnit},${actualDays},${severeTotal}`);
      }

      // 10. 食事提供体制加算（DB動的取得）
      const mealUnit = billingRules['meal_addition'] ?? 30;
      if (facilitySetting.meal_provision_addition === 1 && record.meal_days > 0) {
        const mealUnits = record.meal_days * mealUnit;
        clientTotalUnits += mealUnits;
        itemDetails.push(`${recordTypeItemDetail},${facilityNumber},${recipientNumber},${serviceTypeCode},${CODE_MEAL},${mealUnit},${record.meal_days},${mealUnits}`);
      }

      // 11. 送迎加算（DB動的取得）
      const transUnit = billingRules['transportation_addition'] ?? 21;
      const totalTransTrips = record.transportation_trips || 0;
      if (facilitySetting.transportation_addition && facilitySetting.transportation_addition !== 'none' && totalTransTrips > 0) {
        const transUnits = totalTransTrips * transUnit;
        clientTotalUnits += transUnits;
        itemDetails.push(`${recordTypeItemDetail},${facilityNumber},${recipientNumber},${serviceTypeCode},${CODE_TRANSPORT},${transUnit},${totalTransTrips},${transUnits}`);
      }

      // 12. 処遇改善新加算（DB動的取得: Ⅰ〜Ⅴ率 & 最新サービスコード動的解決）
      if (facilitySetting.treatment_addition_class && facilitySetting.treatment_addition_class !== 'なし') {
        const tClass = facilitySetting.treatment_addition_class;
        let rate = 0;
        if (tClass.includes('Ⅰ')) rate = billingRules['treatment_rate_new_1'] ?? 0.096;
        else if (tClass.includes('Ⅱ')) rate = billingRules['treatment_rate_new_2'] ?? 0.094;
        else if (tClass.includes('Ⅲ')) rate = billingRules['treatment_rate_new_3'] ?? 0.079;
        else if (tClass.includes('Ⅳ')) rate = billingRules['treatment_rate_new_4'] ?? 0.063;
        else if (tClass.includes('Ⅴ')) rate = billingRules['treatment_rate_new_5'] ?? 0;

        const resolvedCode = resolveTreatmentAdditionCode(facilitySetting);
        const tCode = resolvedCode ? resolvedCode.service_item_code : '6111';

        if (rate > 0 && clientTotalUnits > 0) {
          const treatUnits = Math.round(clientTotalUnits * rate);
          clientTotalUnits += treatUnits;
          itemDetails.push(`${recordTypeItemDetail},${facilityNumber},${recipientNumber},${serviceTypeCode},${tCode},${treatUnits},1,${treatUnits}`);
        }
      }

      // 費用計算
      const totalCost = Math.floor(clientTotalUnits * AREA_UNIT_COST);
      const userBurden = clientLimit?.calculated_our_burden ?? 0;
      const billingAmount = totalCost - userBurden;

      totalClientCount++;
      totalFacilityUnits += clientTotalUnits;
      totalBillingAmount += billingAmount;
      totalUserBurdenAmount += userBurden;

      // 明細基本情報レコード（21 / 24）
      detailLines.push(
        `${recordTypeDetailHeader},${facilityNumber},${recipientNumber},${serviceTypeCode},${cityCode},${monthStr},${actualDays},${clientTotalUnits},${clientTotalUnits},${totalCost},${billingAmount},${userBurden}`
      );
      // 明細項目レコード（22 / 25）を追加
      detailLines.push(...itemDetails);

      // サービス提供実績記録票（71: 基本情報）
      provisionLines.push(`${recordTypeProvisionHeader},${facilityNumber},${recipientNumber},${serviceTypeCode},${monthStr},${actualDays}`);

      // 日別明細レコード（72）の生成
      const dailyRecords = db.prepare(`
        SELECT record_date, status, is_absence_addition, is_offsite, work_type, meal_provided, 
               transportation_used, pickup_used, dropoff_used,
               transportation_service_used, pickup_service, dropoff_service
        FROM attendance_records
        WHERE client_id = ? AND record_date LIKE ? AND deleted_at IS NULL
        ORDER BY record_date ASC
      `).all(record.client_id, `${targetMonth}-%`) as any[];

      let absenceCountOutput = 0;
      for (const d of dailyRecords) {
        const dayNum = parseInt(d.record_date.slice(8, 10), 10);
        const formattedDay = String(dayNum).padStart(2, '0');
        let serviceStatus = '0'; // 0: なし, 1: 通所, 2: 施設外, 4: 欠席時対応
        if (d.is_offsite === 1 || d.status === 'out_facility' || d.work_type === 'out_facility') {
          serviceStatus = '2'; // 施設外就労
        } else if (d.status === 'present') {
          serviceStatus = '1'; // 通所
        } else if (d.is_absence_addition === 1) {
          // S-5: 請求明細（22レコード）の月最大回数と完全一致させ、国保連の突合エラーを未然防止
          // 国保連「実績記録票作成の手引き」: 欠席時対応加算の提供状況コードは '4'
          if (absenceCountOutput < maxAbsenceDays) {
            serviceStatus = '4';
            absenceCountOutput++;
          } else {
            serviceStatus = '0';
          }
        }

        if (serviceStatus !== '0') {
          // 送迎区分: 0: なし, 1: 往復, 2: 往路のみ, 3: 復路のみ
          const hasPickup = d.pickup_used === 1 || d.pickup_service === 1;
          const hasDropoff = d.dropoff_used === 1 || d.dropoff_service === 1;
          const hasTransport = d.transportation_used === 1 || d.transportation_service_used === 1;

          let transportCode = 0;
          if (hasPickup && hasDropoff) {
            transportCode = 1;
          } else if (hasPickup) {
            transportCode = 2;
          } else if (hasDropoff) {
            transportCode = 3;
          } else if (hasTransport) {
            transportCode = 1;
          }

          // 72,事業所番号,受給者証番号,サービス種類コード,対象年月YYYYMM,日(01-31),提供状況,送迎,食事
          provisionLines.push(`${recordTypeProvisionDetail},${facilityNumber},${recipientNumber},${serviceTypeCode},${monthStr},${formattedDay},${serviceStatus},${transportCode},${d.meal_provided ? 1 : 0}`);
        }
      }

      // S-03: 上限額管理結果票（73レコード）
      // B-6: 0円が Falsy として totalCost に置き換わらないよう ?? を使用
      if (clientLimit && clientLimit.is_manager === 1) {
        burdenResultLines.push(`${recordTypeBurdenResult},${facilityNumber},${recipientNumber},${monthStr},${clientLimit.our_total_cost ?? totalCost},${clientLimit.other_total_cost ?? 0},${clientLimit.calculated_our_burden ?? userBurden},1`);
      }
    }

    // レコード01: 交換情報（ヘッダー）
    // 01,交換情報識別番号(01),事業所番号,請求年月YYYYMM,作成年月日YYYYMMDD,作成者名称
    const headerLine = `01,01,${facilityNumber},${monthStr},${createdDateStr},"${facilityName}"`;

    // レコード11: 請求書レコード（総括集計）
    // 11,事業所番号,サービス種類コード,請求年月YYYYMM,件数,合計単位数,総費用額,自己負担額,給付費請求額
    const totalCostAll = totalBillingAmount + totalUserBurdenAmount;
    const invoiceLine = `${recordTypeInvoice},${facilityNumber},${serviceTypeCode},${monthStr},${totalClientCount},${totalFacilityUnits},${totalCostAll},${totalUserBurdenAmount},${totalBillingAmount}`;

    // レコード99: トレーラー（終了行）
    const totalLinesCount = detailLines.length + provisionLines.length + burdenResultLines.length + 3;
    const trailerLine = `99,${totalLinesCount}`;

    const fullCsv = [
      headerLine,
      invoiceLine,
      ...detailLines,
      ...provisionLines,
      ...burdenResultLines,
      trailerLine
    ].join('\r\n') + '\r\n';

    return fullCsv;
  } catch (error: any) {
    console.error("Failed to generate Kokuho CSV:", error);
    throw new Error(error.message || '国保連伝送用データの生成に失敗しました');
  }
}

  } catch (e) {
    return false;
  }
}

/**
 * 対象月の月次ロック状況を取得
 */
export async function getBillingMonthLockStatus(targetMonth: string) {
  try {
    const row = db.prepare(`SELECT * FROM billing_monthly_locks WHERE target_month = ?`).get(targetMonth) as any;
    if (!row) {
      return { isLocked: false, lockedAt: null, lockedBy: null, unlockReason: null };
    }
    return {
      isLocked: !!row.is_locked,
      lockedAt: row.locked_at || null,
      lockedBy: row.locked_by || null,
      unlockReason: row.unlock_reason || null
    };
  } catch (e) {
    return { isLocked: false, lockedAt: null, lockedBy: null, unlockReason: null };
  }
}

/**
 * 対象月の月次請求確定ロックを実行
 */
export async function lockBillingMonth(targetMonth: string) {
  try {
    const role = await getRole();
    if (role !== 'admin' && role !== 'service_manager') {
      return { success: false, error: '管理者またはサビ管権限が必要です', message: '管理者またはサビ管権限が必要です' };
    }

    const readiness = await validateBillingReadiness(targetMonth);
    if (!readiness.isReady) {
      const msg = `請求準備が完了していません:\n・${readiness.errors.join('\n・')}`;
      return { success: false, error: msg, message: msg };
    }

    const now = new Date().toISOString();
    db.prepare(`
      INSERT INTO billing_monthly_locks (id, target_month, is_locked, locked_at, locked_by)
      VALUES (lower(hex(randomblob(16))), ?, 1, ?, '管理者')
      ON CONFLICT(target_month) DO UPDATE SET
        is_locked = 1,
        locked_at = excluded.locked_at,
        locked_by = excluded.locked_by
    `).run(targetMonth, now);

    try {
      revalidatePath('/billing/kokuho');
      revalidatePath('/attendance');
    } catch (e) {}
    return { success: true, message: `${targetMonth} の請求データを確定し、月次ロックしました。` };
  } catch (e: any) {
    console.error("Failed to lock billing month:", e);
    const msg = e.message || '月次ロックに失敗しました';
    return { success: false, error: msg, message: msg };
  }
}

/**
 * 対象月の月次ロックを解除
 */
export async function unlockBillingMonth(targetMonth: string, reason: string) {
  try {
    const role = await getRole();
    if (role !== 'admin' && role !== 'service_manager') {
      return { success: false, error: '管理者またはサビ管権限が必要です', message: '管理者またはサビ管権限が必要です' };
    }

    if (!reason || reason.trim() === '') {
      return { success: false, error: '解除理由を入力してください。', message: '解除理由を入力してください。' };
    }

    db.prepare(`
      UPDATE billing_monthly_locks
      SET is_locked = 0, unlock_reason = ?
      WHERE target_month = ?
    `).run(reason, targetMonth);

    try {
      revalidatePath('/billing/kokuho');
      revalidatePath('/attendance');
    } catch (e) {}
    return { success: true, message: `${targetMonth} の月次確定ロックを解除しました。` };
  } catch (e: any) {
    console.error("Failed to unlock billing month:", e);
    const msg = e.message || '月次ロック解除に失敗しました';
    return { success: false, error: msg, message: msg };
  }
}

/**
 * 請求データの妥当性・整合性を事前検証（国保連レセプト事前審査・返戻ゼロ防御）
 */
export async function validateBillingReadiness(targetMonth: string): Promise<BillingReadinessResult> {
  const issues: { type: 'error' | 'warning'; message: string; category?: string }[] = [];
  const errors: string[] = [];
  const warnings: string[] = [];

  const addIssue = (type: 'error' | 'warning', message: string, category: string) => {
    issues.push({ type, message, category });
    if (type === 'error') errors.push(message);
    else warnings.push(message);
  };

  // 0. 事業所設定
  const facility = db.prepare(`SELECT * FROM facility_settings WHERE id = 'default'`).get() as any || {};
  const defaultCityCode = facility.city_code;
  const serviceMode = getServiceModeSync();

  // 1. 基本単位数の確定チェック（B型で facility_settings.base_unit_score が設定済みの場合は有効とみなす）
  const facilityUnit = db.prepare(`SELECT base_unit FROM monthly_facility_units WHERE target_month = ?`).get(targetMonth);
  const hasTypeBDefaultUnit = serviceMode === 'type_b' && Number(facility.base_unit_score || 0) > 0;
  if (!facilityUnit && !hasTypeBDefaultUnit) {
    addIssue('error', '基本報酬単位数が確定されていません。「スコア・単位数シミュレーション」で確定してください。', '基本報酬');
  }

  // 2. 勤怠実績チェック
  const attendance = db.prepare(`
    SELECT client_id, 
           COUNT(*) as total_days,
           SUM(CASE WHEN status = 'present' AND COALESCE(is_offsite, 0) = 0 THEN 1 ELSE 0 END) as present_days,
           SUM(CASE WHEN status = 'out_facility' OR (status = 'present' AND is_offsite = 1) THEN 1 ELSE 0 END) as out_facility_days,
           SUM(CASE WHEN is_absence_addition = 1 THEN 1 ELSE 0 END) as absence_days
    FROM attendance_records 
    WHERE record_date LIKE ? AND deleted_at IS NULL
    GROUP BY client_id
  `).all(`${targetMonth}%`) as any[];

  if (attendance.length === 0) {
    addIssue('error', '対象月の勤怠実績が1件も登録されていません。勤怠管理画面で実績を記録してください。', '実績データ');
    return {
      isReady: false,
      hasErrors: true,
      hasWarnings: false,
      errors,
      warnings,
      issues,
      clientCount: 0,
      totalDays: 0
    };
  }

  let totalFacilityAllDays = 0;
  let totalFacilityOffsiteDays = 0;

  // 3. 各利用者の詳細審査（受給者証、有効期限、支給量、計画、上限管理等）
  const allClients = db.prepare(`SELECT * FROM clients WHERE deleted_at IS NULL`).all() as any[];
  const clientMap = new Map<string, any>();
  for (const c of allClients) {
    clientMap.set(c.id, c);
  }

  const limits = db.prepare(`SELECT * FROM monthly_burden_limits WHERE target_month = ?`).all(targetMonth) as any[];
  const limitMap = new Map<string, any>();
  for (const l of limits) {
    limitMap.set(l.client_id, l);
  }

  const activePlans = db.prepare(`
    SELECT client_id, start_date, end_date FROM support_plans 
    WHERE status = 'active' AND deleted_at IS NULL
  `).all() as any[];

  for (const att of attendance) {
    const client = clientMap.get(att.client_id);
    const clientName = client?.full_name || '未登録利用者';
    const actualDays = att.present_days + att.out_facility_days;
    totalFacilityAllDays += actualDays;
    totalFacilityOffsiteDays += att.out_facility_days;

    if (!client) {
      addIssue('error', `勤怠データに紐付く利用者情報が見つかりません (ID: ${att.client_id})。`, 'マスタ整合性');
      continue;
    }

    // A. 受給者証番号チェック（半角数字10桁必須: 国保連K002/K003返戻防止）
    const recipNum = client.recipient_number ? client.recipient_number.trim() : '';
    if (!recipNum || !/^\d{10}$/.test(recipNum)) {
      addIssue('error', `${clientName} 様の受給者証番号（${recipNum || '未入力'}）が半角数字10桁ではありません。国保連伝送で即時返戻となります。`, '受給者証');
    }

    // B. 受給者証の有効期限切れチェック（国保連K001返戻防止）
    if (client.valid_to) {
      const monthStart = `${targetMonth}-01`;
      if (client.valid_to < monthStart) {
        addIssue('error', `${clientName} 様の受給者証有効期限（${client.valid_to}）が請求対象月（${targetMonth}）より前に切れています。市町村での更新確認が必要です。`, '有効期限');
      }
    } else {
      addIssue('warning', `${clientName} 様の受給者証有効期限（終了日）が未登録です。`, '有効期限');
    }

    // C. 契約支給量（月間利用日数上限）オーバーチェック（国保連超過返戻防止）
    if (client.allowance_days && client.allowance_days > 0) {
      if (actualDays > client.allowance_days) {
        addIssue('error', `${clientName} 様の利用日数（${actualDays}日）が契約支給量（${client.allowance_days}日/月）を超過しています。自治体への増量申請または日数の調整が必要です。`, '支給量超過');
      }
    }

    // D. 個別支援計画の未作成・期限切れ警告（未作成減算 告示: 1〜2ヶ月目30%減、3ヶ月目以降50%減の事前告知）
    const uncreatedMonths = getPlanUncreatedMonthsSync(client.id, targetMonth);
    if (uncreatedMonths > 0 && actualDays > 0) {
      const reductionText = uncreatedMonths >= 3 
        ? `未作成3ヶ月目以降（基本報酬50%減）` 
        : `未作成1〜2ヶ月目（基本報酬30%減）`;
      addIssue('warning', `${clientName} 様の有効な個別支援計画がありません。当月は「個別支援計画未作成減算（${reductionText}）」が適用されます。`, '個別支援計画');
    }

    // E. 負担上限額管理の未済チェック
    if (client.payment_limit && client.payment_limit > 0) {
      const limitRecord = limitMap.get(client.id);
      if (!limitRecord) {
        addIssue('warning', `${clientName} 様（負担上限月額: ${client.payment_limit.toLocaleString()}円）の「上限額管理」の計算・保存が完了していません。`, '上限額管理');
      }
    }

    // F. 市区町村コードの未設定チェック
    if (!client.city_code && !defaultCityCode) {
      addIssue('error', `${clientName} 様の市区町村コードが未登録です。事業所設定または利用者情報で市区町村コードを設定してください。`, '市区町村コード');
    }

    // G. 欠席時対応加算の月4回上限チェック
    if (att.absence_days > 4) {
      addIssue('warning', `${clientName} 様の欠席時対応加算が${att.absence_days}回記録されています（国保連算定上限は月4回のため、4回に丸めて請求されます）。`, '加算算定');
    }
  }

  // 4. 施設外就労の7割（70%）上限規制チェック（厚労省指定基準第191条の2・指定就労継続支援A型）
  // 基準: 「1日あたりの施設外就労利用者数」が「事業所の利用定員」の100分の70（70%）以下であること
  const capacity = parseCapacityFromClass(facility.capacity_class);
  const dailyOffsiteRecords = db.prepare(`
    SELECT record_date, COUNT(DISTINCT client_id) as offsite_count
    FROM attendance_records
    WHERE record_date LIKE ? AND (status = 'out_facility' OR (status = 'present' AND is_offsite = 1)) AND deleted_at IS NULL
    GROUP BY record_date
  `).all(`${targetMonth}%`) as { record_date: string; offsite_count: number }[];

  const maxDailyOffsite = dailyOffsiteRecords.length > 0 
    ? Math.max(...dailyOffsiteRecords.map(r => r.offsite_count)) 
    : 0;

  if (capacity > 0 && maxDailyOffsite > 0) {
    const maxAllowed = Math.floor(capacity * 0.7);
    if (maxDailyOffsite > maxAllowed) {
      const exceededList = dailyOffsiteRecords
        .filter(r => r.offsite_count > maxAllowed)
        .map(r => `${r.record_date} (${r.offsite_count}名)`)
        .join(', ');
      const offsiteRatio = Math.round((maxDailyOffsite / capacity) * 100);
      addIssue('warning', `【実地指導・法令遵守警告】1日あたりの施設外就労利用者数が利用定員（${capacity}名）の70%上限（${maxAllowed}名）を超過している日があります: ${exceededList}（最大定員比: ${offsiteRatio}%）。指定基準第191条の2違反による実地指導での指摘・返還リスクがあるため速やかに是正してください。`, '施設外就労');
    }
  }

  return {
    isReady: errors.length === 0,
    hasErrors: errors.length > 0,
    hasWarnings: warnings.length > 0,
    errors,
    warnings,
    issues,
    clientCount: attendance.length,
    totalDays: totalFacilityAllDays
  };
}

export async function getBillingSummaryForPrint(targetMonth: string): Promise<PrintBillingSummary> {
  const facility = db.prepare(`SELECT * FROM facility_settings WHERE id = 'default'`).get() as any || {};
  const facilityNumber = facility.facility_number || '2810099999';
  const facilityName = facility.facility_name || '就労継続支援A型事業所';
  const defaultCityCode = facility.city_code || '28100';
  const representativeName = facility.representative_name || facility.facility_manager || '管理者';
  const postalCode = facility.postal_code || '';
  const address = facility.address || '';

          initialDays++;
        }

        const u = billingRules['initial_addition'] ?? 30;

        clientTotalUnits += t;
        additions.push({ name: '初期加算', code: '6101', unit: u, count: initialDays, total: t });

    }

      const u = billingRules['visual_auditory_addition'] ?? 41;
      const t = actualDays * u;
      clientTotalUnits += t;

        ? (billingRules['severe_disability_2'] ?? 23) 
        : (billingRules['severe_disability_1'] ?? 47);
      const severeCode = facility.severe_disability_addition === 'severe_2' ? '6801' : '6800';
      const severeName = facility.severe_disability_addition === 'severe_2' ? '重度障害者支援加算(Ⅱ)' : '重度障害者支援加算(Ⅰ)';
      const t = actualDays * severeUnit;

      additions.push({ name: severeName, code: severeCode, unit: severeUnit, count: actualDays, total: t });
    }

    if (facility.meal_provision_addition === 1 && record.meal_days > 0) {
      const u = billingRules['meal_addition'] ?? 30;
      const t = record.meal_days * u;
      clientTotalUnits += t;
      additions.push({ name: '食事提供体制加算', code: '5441', unit: u, count: record.meal_days, total: t });
    }

    const totalTransTrips = record.transportation_trips || 0;
    if (facility.transportation_addition && facility.transportation_addition !== 'none' && totalTransTrips > 0) {

    }

    if (facility.treatment_addition_class && facility.treatment_addition_class !== 'なし') {
      const tClass = facility.treatment_addition_class;
      let rate = 0;
      let tCode = '6111';
      let tName = '福祉・介護職員処遇改善新加算(Ⅰ)';
      if (tClass.includes('Ⅰ')) { rate = billingRules['treatment_rate_new_1'] ?? 0.096; tCode = '6111'; tName = '処遇改善新加算(Ⅰ)'; }
      else if (tClass.includes('Ⅱ')) { rate = billingRules['treatment_rate_new_2'] ?? 0.094; tCode = '6112'; tName = '処遇改善新加算(Ⅱ)'; }
      else if (tClass.includes('Ⅲ')) { rate = billingRules['treatment_rate_new_3'] ?? 0.079; tCode = '6113'; tName = '処遇改善新加算(Ⅲ)'; }
      else if (tClass.includes('Ⅳ')) { rate = billingRules['treatment_rate_new_4'] ?? 0.063; tCode = '6114'; tName = '処遇改善新加算(Ⅳ)'; }
      else if (tClass.includes('Ⅴ')) { rate = billingRules['treatment_rate_new_5'] ?? 0; tCode = '6115'; tName = '処遇改善新加算(Ⅴ)'; }

      if (rate > 0 && clientTotalUnits > 0) {
        const treatUnits = Math.round(clientTotalUnits * rate);
        clientTotalUnits += treatUnits;
        additions.push({ name: tName, code: tCode, unit: treatUnits, count: 1, total: treatUnits });
      }
    }

    const totalCost = Math.floor(clientTotalUnits * AREA_UNIT_COST);
    const userBurden = clientLimit?.calculated_our_burden ?? 0;
    const billingAmount = totalCost - userBurden;

    totalClientCount++;
    totalDays += actualDays;
    totalFacilityUnits += clientTotalUnits;
    totalCostSum += totalCost;
    totalBillingAmount += billingAmount;
    totalUserBurdenAmount += userBurden;

    clientsResult.push({
      clientId: record.client_id,
      clientName: clientProfile?.full_name || '利用者名未設定',
      recipientNumber,
      cityCode,
      actualDays,
      absenceDays: safeAbsenceDays,
      effectiveBaseUnit,
      baseUnitsTotal,
      hasValidPlan,
      uncreatedMonths,
      additions,
      totalUnits: clientTotalUnits,
      totalCost,
      userBurden,
      billingAmount
    });
  }

  return {
    targetMonth,
    facility: {
      facilityNumber,
      facilityName,
      representativeName,
      postalCode,
      address,
      phone,
      cityCode: defaultCityCode,
      regionClass,
      areaUnitCost: AREA_UNIT_COST,
      standardBaseUnit
    },
    clientCount: totalClientCount,
    totalDays,
    totalUnits: totalFacilityUnits,
    totalCost: totalCostSum,
    totalUserBurden: totalUserBurdenAmount,
    totalBillingAmount: totalBillingAmount,
    clients: clientsResult
  };
}

              <Download className="w-3.5 h-3.5 mr-1.5 shrink-0" />

            </Link>
            <Button

              size="sm"
              onClick={handleToggleLock}

  );
}

import { Input } from "@/components/ui/input";

import { Badge } from "@/components/ui/badge";
import Link from "next/link";
import { ChevronLeft, Download, Upload, FileSpreadsheet, AlertCircle, Lock, Unlock, ShieldCheck, ShieldAlert, CheckCircle2, Printer } from "lucide-react";
import { cn } from "@/lib/utils";
import { parseKokuhoErrorCsv, KokuhoErrorRecord } from "@/lib/billing/kokuhoCsvParser";
import { 
  generateKokuhoCsvSafe, 
  getBillingMonthLockStatus, 
  lockBillingMonth, 
  unlockBillingMonth, 
  getMonthlyStaffingForBilling,
  validateBillingReadiness,
  type BillingReadinessResult
} from "../actions";

const getDefaultTargetMonth = () => {
  const d = new Date();
  d.setDate(1);
  d.setMonth(d.getMonth() - 1);
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
};

export function KokuhoClient() {
  const [targetMonth, setTargetMonth] = useState(getDefaultTargetMonth);
  const [errorRecords, setErrorRecords] = useState<KokuhoErrorRecord[] | null>(null);
  const [isGenerating, setIsGenerating] = useState(false);
  const [isOverpaymentClaim, setIsOverpaymentClaim] = useState(false);
  const [lockStatus, setLockStatus] = useState<{ isLocked: boolean; lockedAt: string | null; lockedBy: string | null; unlockReason?: string | null } | null>(null);
  const [staffingStatus, setStaffingStatus] = useState<any>(null);
  const [readinessStatus, setReadinessStatus] = useState<BillingReadinessResult | null>(null);
  const [isLocking, setIsLocking] = useState(false);

  const fetchMonthStatus = async (month: string) => {
    try {
      const [status, staffing, readiness] = await Promise.all([
        getBillingMonthLockStatus(month),
        getMonthlyStaffingForBilling(month),
        validateBillingReadiness(month)
      ]);
      setLockStatus(status);
      setStaffingStatus(staffing);
      setReadinessStatus(readiness);
    } catch (e) {
      console.error(e);
    }
  };

  useEffect(() => {
    fetchMonthStatus(targetMonth);
  }, [targetMonth]);

  const handleLock = async () => {
    if (!confirm(`${targetMonth} の請求データを確定し、月次ロックしますか？\n\n【重要】ロック後は該当月の勤怠・支援記録の修正や追加が完全にブロックされ、請求データの整合性が保証されます。`)) return;
    setIsLocking(true);
    try {
      const res = await lockBillingMonth(targetMonth);
      if (!res.success) {
        alert("ロックに失敗しました:\n" + ((res as any).error || res.message));
        return;
      }
      alert(res.message);
      await fetchMonthStatus(targetMonth);
    } catch (e: any) {
      alert("ロックに失敗しました: " + e.message);
    } finally {
      setIsLocking(false);
    }
  };

  const handleUnlock = async () => {
    const reason = prompt(`${targetMonth} の月次ロックを解除する理由を入力してください（実地指導対策として監査ログに恒久記録されます）:`);
    if (!reason || reason.trim() === "") {
      alert("解除理由は必須です。");
      return;
    }
    setIsLocking(true);
    try {
      const res = await unlockBillingMonth(targetMonth, reason);
      if (!res.success) {
        alert("解除に失敗しました: " + ((res as any).error || res.message));
        return;
      }
      alert(res.message);
      await fetchMonthStatus(targetMonth);
    } catch (e: any) {
      alert("解除に失敗しました: " + e.message);
    } finally {
      setIsLocking(false);
    }
  };

  const handleDownloadCsv = async () => {
    setIsGenerating(true);
    try {
      const res = await generateKokuhoCsvSafe(targetMonth, isOverpaymentClaim);
      
      if (!res || !res.success || !res.csvData) {
        alert(res?.error || "指定された月の計算済みデータが見つかりません。「上限額管理」モジュールで計算・保存を行ってから再度お試しください。");
        return;
      }

      const { encodeToShiftJis } = await import('../actions');
      const uint8Array = await encodeToShiftJis(res.csvData);

      const blob = new Blob([uint8Array as any], { type: 'text/csv' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `DATA01_${targetMonth.replace('-', '')}.CSV`;
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      URL.revokeObjectURL(url);
    } catch (err: any) {
      alert("CSVの生成中にエラーが発生しました: " + err.message);
    } finally {
      setIsGenerating(false);
    }
  };

  const handleFileUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;

    const reader = new FileReader();
    reader.onload = (event) => {
      const text = event.target?.result as string;
      const parsed = parseKokuhoErrorCsv(text);
      setErrorRecords(parsed);
    };
    reader.readAsText(file, 'Shift_JIS');
  };

  return (
    <main className="flex-1 p-3.5 sm:p-6 md:p-12">
      <div className="mx-auto max-w-5xl space-y-6">
        <div className="flex items-center gap-4">
          <Link href="/billing" className={cn(buttonVariants({ variant: "outline", size: "icon" }))}>
            <ChevronLeft className="h-4 w-4" />
          </Link>
          <div>
            <h1 className="text-2xl sm:text-3xl font-bold tracking-tight text-foreground">国保連への請求</h1>
            <p className="mt-1 sm:mt-2 text-sm sm:text-lg text-muted-foreground">
              国保連に提出する請求データを作成したり、エラー（返戻）の内容を確認します。
            </p>
          </div>
        </div>

        {/* 人員配置基準警告（月間FTE不足リスク） */}
        {staffingStatus && !staffingStatus.isCompliant && (
          <div className="bg-rose-50 border-l-4 border-rose-500 p-4 rounded-r-md shadow-sm">
            <div className="flex items-start">
              <AlertCircle className="h-6 w-6 text-rose-600 mt-0.5 mr-3 flex-shrink-0" />
              <div>
                <h3 className="text-base font-bold text-rose-950">
                  【人員配置警告】対象月（{targetMonth}）の常勤換算（FTE）が人員基準を下回っています
                </h3>
                <p className="mt-1 text-sm text-rose-900">
                  当月の平均利用者数: <strong>{staffingStatus.averageClientsPerDay}名</strong> に対し、必要常勤換算数は <strong>{staffingStatus.requiredFte}人</strong> ですが、現在のシフト実績は <strong className="text-rose-600">{staffingStatus.actualFte}人</strong> です。
                  人員基準を満たさない場合、3割の人員欠如減算またはレセプト返戻の対象となるおそれがあります。
                </p>
                <div className="mt-2 flex gap-4 text-xs font-bold">
                  <Link href="/staff/shifts" className="underline text-rose-950 hover:text-rose-700">
                    → 職員シフトを確認・調整する
                  </Link>
                  <Link href="/settings" className="underline text-rose-950 hover:text-rose-700">
                    → 事業所体制設定の人員欠如減算フラグを確認する
                  </Link>
                </div>
              </div>
            </div>
          </div>
        )}

        {/* 月次確定・データロック（改ざん防止・実地指導対策） */}
        <Card className={cn(
          "border-2 transition-all shadow-sm",
          lockStatus?.isLocked ? "border-emerald-500 bg-emerald-50/30" : "border-amber-300 bg-amber-50/20"
        )}>
          <CardContent className="p-6">
            <div className="flex flex-col md:flex-row md:items-center md:justify-between gap-4">
              <div className="space-y-1">
                <div className="flex items-center gap-2 flex-wrap">
                  <ShieldCheck className={cn("h-6 w-6", lockStatus?.isLocked ? "text-emerald-600" : "text-amber-600")} />
                  <h2 className="text-xl font-bold">
                    {targetMonth}月 請求データ月次確定・ロック管理
                  </h2>
                  {lockStatus?.isLocked ? (
                    <Badge className="bg-emerald-600 text-white hover:bg-emerald-700 px-3 py-1 text-sm">
                      <Lock className="h-3.5 w-3.5 mr-1" /> 確定ロック中（改変防止）
                    </Badge>
                  ) : (
                    <Badge variant="outline" className="text-amber-700 border-amber-400 bg-amber-50 px-3 py-1 text-sm">
                      <Unlock className="h-3.5 w-3.5 mr-1" /> 未確定（編集可能）
                    </Badge>
                  )}
                </div>
                <p className="text-sm text-muted-foreground mt-1">
                  {lockStatus?.isLocked ? (
                    <span>
                      {lockStatus.lockedAt?.substring(0, 10)} に {lockStatus.lockedBy || '管理者'} により確定ロックされました。当月の勤怠・支援記録の修正はブロックされています。
                    </span>
                  ) : (
                    <span>
                      ※国保連への伝送データ作成後は、過去データの不整合や実地指導での指摘を防ぐため、必ず「月次ロック」を実行してください。
                    </span>
                  )}
                </p>
              </div>

              <div className="flex flex-wrap items-center gap-2 shrink-0">
                {lockStatus?.isLocked ? (
                  <Button 
                    variant="outline" 
                    size="sm"
                    className="border-destructive text-destructive hover:bg-destructive/10"
                    onClick={handleUnlock}
                    disabled={isLocking}
                  >
                    <Unlock className="mr-1 h-4 w-4" /> ロック解除（過誤修正用）
                  </Button>
                ) : (
                  <Button 
                    variant="default"
                    className="bg-emerald-600 hover:bg-emerald-700 text-white shadow-sm font-semibold"
                    onClick={handleLock}
                    disabled={isLocking}

                    返戻リスク検知: {readinessStatus.issues.filter(i => i.type === 'error').length}件

                ) : readinessStatus.hasWarnings ? (

              <Button

      )}

  const facilityRules = getFacilityEffectiveRules();
  const defaultFormat = facilityRules.reports.offsite_work_format || 'standard';
  const activeFormat = params.format || defaultFormat;
  
  // 施設外就労の打刻データを取得
  const offsiteRecords = db.prepare(`
    SELECT a.*, c.full_name, c.recipient_number,
           w.name as workplace_name, w.company_name,
           st.name as staff_name
    FROM attendance_records a
    JOIN clients c ON a.client_id = c.id
    LEFT JOIN offsite_workplaces w ON a.offsite_workplace_id = w.id
    LEFT JOIN staff st ON a.offsite_staff_id = st.id
    WHERE a.record_date LIKE ? AND a.status = 'present' AND a.is_offsite = 1
    ORDER BY a.record_date ASC, a.clock_in ASC
  `).all(`${currentMonth}-%`) as any[];

  // 請負契約情報を取得
  const contractorAgreements = db.prepare(`
    SELECT *
    FROM contractor_agreements
    WHERE contract_start_date <= ? AND (contract_end_date IS NULL OR contract_end_date >= ?)
  `).all(`${currentMonth}-31`, `${currentMonth}-01`) as any[];

  // 1日あたりの施設外就労人数と定員比率の集計
  const dailyCounts: Record<string, number> = {};
  for (const r of offsiteRecords) {
    dailyCounts[r.record_date] = (dailyCounts[r.record_date] || 0) + 1;
  }
  const maxDailyCount = Object.values(dailyCounts).length > 0 ? Math.max(...Object.values(dailyCounts)) : 0;
  const capacity = parseCapacityFromClass(facility.capacity_class);
  const ratio = capacity > 0 ? Math.round((maxDailyCount / capacity) * 100) : 0;
  const capacityRatioData = {
    capacity,
    maxDailyCount,
    ratio,
    isCompliant: ratio <= 70
  };

  const offsiteClientIds = Array.from(new Set(offsiteRecords.map((r: any) => r.client_id)));

    clientName: string; 
    hasPlan: boolean; 
    hasOffsiteMention: boolean; 
    planId?: string;
    planDate?: string;

'use client';

import { useState, useEffect } from "react";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Button, buttonVariants } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from "@/components/ui/table";
import { PlusCircle, Printer, Save, Trash2, FileSpreadsheet } from "lucide-react";
import Link from "next/link";
import { useRouter } from "next/navigation";
import { cn } from "@/lib/utils";
import { savePayslips, exportAccountingJournalCsv } from "./actions";
import { toast } from "sonner";

export function PayrollClient({ targetMonth, initialData, employmentInsuranceRate, minWage = 951, isTypeB = false }: any) {
  const router = useRouter();
  const [data, setData] = useState<any[]>(initialData);
  const [isSaving, setIsSaving] = useState(false);
  const [isExporting, setIsExporting] = useState(false);

  useEffect(() => {
    setData(initialData);
  }, [initialData]);

  const handleExportJournalCsv = async () => {
    try {
      setIsExporting(true);
      const res = await exportAccountingJournalCsv(targetMonth);
      if (!res || !res.success || !res.csv) {
        toast.error(res?.error || '仕訳CSVのエクスポートに失敗しました');
        return;
      }
      const blob = new Blob([new Uint8Array([0xEF, 0xBB, 0xBF]), res.csv], { type: 'text/csv;charset=utf-8;' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `会計仕訳_${targetMonth}.csv`;
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      URL.revokeObjectURL(url);
      toast.success('会計仕訳CSVを出力しました');
    } catch (err: any) {
      toast.error(err.message || '仕訳CSVのエクスポートに失敗しました');
    } finally {
      setIsExporting(false);
    }
  };

  const handleMonthChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    router.push(`/billing/payroll?month=${e.target.value}`);
  };

  const addCustomDeduction = (clientId: string) => {
    setData(prev => prev.map(p => {
      if (p.client_id === clientId) {
        return {
          ...p,
          custom_deductions: [...p.custom_deductions, { name: '昼食代', amount: 0 }]
        };
      }
      return p;
    }));
  };

  const updateCustomDeduction = (clientId: string, index: number, field: string, value: string) => {
    setData(prev => prev.map(p => {
      if (p.client_id === clientId) {
        const newDeds = [...p.custom_deductions];
        newDeds[index] = { ...newDeds[index], [field]: field === 'amount' ? parseInt(value) || 0 : value };
        return { ...p, custom_deductions: newDeds };
      }
      return p;
    }));
  };

  const removeCustomDeduction = (clientId: string, index: number) => {
    setData(prev => prev.map(p => {
      if (p.client_id === clientId) {
        return {
          ...p,
          custom_deductions: p.custom_deductions.filter((_: any, i: number) => i !== index)
        };
      }
      return p;
    }));
  };

  const calculateNet = (row: any) => {
    const customSum = (row.custom_deductions || []).reduce((sum: number, d: any) => sum + (d.amount || 0), 0);
    const totalPay = (row.base_salary || 0) + (row.pto_allowance || 0) + (row.overtime_allowance || 0) + (row.night_shift_allowance || 0) + (row.holiday_allowance || 0) + (row.commute_allowance || 0) + (row.ability_allowance || 0);
    const totalDed = (row.employment_insurance_deduction || 0) + (row.health_insurance_deduction || 0) + (row.welfare_pension_deduction || 0) + (row.income_tax || 0) + customSum;
    return totalPay - totalDed;
  };

  const handleSave = async () => {
    if (!isTypeB) {
      const hasMinWageViolation = data.some(r => r.hourly_wage > 0 && r.hourly_wage < minWage);
      if (hasMinWageViolation) {
        const confirmSave = window.confirm(
          `【警告】地域別最低賃金（${minWage.toLocaleString()}円）を下回っている利用者が存在します。\nこのまま確定すると労使間トラブルや労働基準法違反となる可能性があります。\n本当に給与データを確定・保存しますか？`
        );
        if (!confirmSave) return;
      }
    }

    setIsSaving(true);
    try {
      const payload = data.map(r => ({
        ...r,
        net_payment: calculateNet(r),
        custom_deductions: JSON.stringify(r.custom_deductions || [])
      }));
      const res = await savePayslips(targetMonth, payload);
      if (res && !res.success) {
        toast.error(res.error || (isTypeB ? '工賃データの保存に失敗しました' : '給与データの保存に失敗しました'));
        return;
      }
      toast.success(isTypeB ? '工賃データを保存・確定しました' : '給与データを保存・確定しました');
    } catch (err: any) {
      console.error(isTypeB ? '工賃保存エラー:' : '給与保存エラー:', err);
      toast.error((isTypeB ? '工賃データの保存に失敗しました: ' : '給与データの保存に失敗しました: ') + (err.message || '予期せぬエラー'));
    } finally {
      setIsSaving(false);
    }
  };

  return (
    <div className="space-y-6">
      <div className="flex flex-col md:flex-row justify-between items-start md:items-end gap-4">
        <div>
          <h1 className="text-2xl sm:text-3xl font-bold tracking-tight text-foreground">
            {isTypeB ? '工賃計算・工賃支払台帳' : '給与計算・賃金台帳'}
          </h1>
          <p className="text-muted-foreground mt-1 text-xs sm:text-sm">
            {isTypeB 
              ? '作業時間・工賃単価（時給・日給・出来高・能力手当）および実費控除に基づき工賃を計算・確定します。'
              : '実働時間と時給、控除設定に基づき給与を計算・確定します。'}
          </p>
        </div>
        <div className="flex flex-wrap items-center gap-2 w-full md:w-auto">

                          <div className="font-semibold">基本給: {(row.base_salary || 0).toLocaleString()}円</div>
                          {row.pto_allowance > 0 && (
                            <div className="text-xs text-blue-600 font-semibold mt-0.5">有休: {(row.pto_allowance || 0).toLocaleString()}円 ({row.pto_days}日)</div>
                          )}
                          <div className="text-xs text-muted-foreground mt-0.5">時給: {(row.hourly_wage || 0).toLocaleString()}円</div>
                          {row.hourly_wage > 0 && row.hourly_wage < minWage && (
                            <span className="inline-block mt-1 px-1.5 py-0.5 rounded bg-red-100 text-red-700 text-[10px] font-bold">
                              ⚠ 最賃割れ
                            </span>
                          )}
                        </div>
                      )}
                    </TableCell>
                    
                    <TableCell className={cn("p-2 text-right text-xs", isTypeB ? "bg-emerald-50/30 text-emerald-900" : "bg-green-50/50 text-green-800")}>
                      {isTypeB ? (
                        <div className="space-y-0.5">
                          <div>能力: {(row.ability_allowance || 0).toLocaleString()}円</div>
                          <div>通勤: {(row.commute_allowance || 0).toLocaleString()}円</div>
                        </div>
                      ) : (
                        <div className="space-y-0.5">
                          <div>残(25%): {(row.overtime_normal_allowance || (row.overtime_allowance - (row.overtime_excess_allowance || 0)) || 0).toLocaleString()}</div>
                          {(row.overtime_excess_allowance > 0 || (row.overtime_excess_hours && row.overtime_excess_hours > 0)) && (
                            <div className="font-bold text-amber-700">残(50%): {(row.overtime_excess_allowance || 0).toLocaleString()}</div>
                          )}
                          <div>深(25%): {(row.night_shift_allowance||0).toLocaleString()}</div>
                          <div>休(35%): {(row.holiday_allowance||0).toLocaleString()}</div>
                          <div>通: {(row.commute_allowance || 0).toLocaleString()}</div>
                        </div>
                      )}
                    </TableCell>

                    <TableCell className="bg-red-50/50 p-2 text-right text-red-800 text-xs">
                      {isTypeB ? (
                        <div className="space-y-0.5">
                          <div>所得税: {(row.income_tax || 0).toLocaleString()}円</div>
                          <div className="text-[10px] text-muted-foreground">（社会保険・雇用保険非適用）</div>
                        </div>
                      ) : (
                        <div className="space-y-0.5">
                          <div>雇: {(row.employment_insurance_deduction || 0).toLocaleString()}</div>
                          <div>健: {(row.health_insurance_deduction || 0).toLocaleString()}</div>
                          <div>厚: {(row.welfare_pension_deduction || 0).toLocaleString()}</div>
                          <div>税: {(row.income_tax||0).toLocaleString()}</div>
                        </div>
                      )}
                    </TableCell>

                    <TableCell className="bg-red-50/50 p-2 align-top">
                      <div className="space-y-2">
                        {row.custom_deductions.map((ded: any, idx: number) => (
                          <div key={idx} className="flex items-center gap-2">
                            <Input 
                              value={ded.name} 
                              onChange={(e) => updateCustomDeduction(row.client_id, idx, 'name', e.target.value)} 
                              className="h-7 text-xs w-24"
                            />
                            <Input 
                              type="number"
                              value={ded.amount} 
                              onChange={(e) => updateCustomDeduction(row.client_id, idx, 'amount', e.target.value)} 
                              className="h-7 text-xs w-20 text-right"
                            />
                            <button onClick={() => removeCustomDeduction(row.client_id, idx)} className="text-red-500 hover:text-red-700">
                              <Trash2 className="w-4 h-4" />
                            </button>
                          </div>
                        ))}
                        <Button variant="ghost" size="sm" className="h-6 text-xs w-full" onClick={() => addCustomDeduction(row.client_id)}>
                          <PlusCircle className="w-3 h-3 mr-1" /> 追加
                        </Button>
                      </div>
                    </TableCell>

                    <TableCell className={cn("text-right font-bold text-lg", isTypeB ? "bg-emerald-50/50 text-emerald-800" : "bg-blue-50/50 text-blue-800")}>
                      {net.toLocaleString()}円
                    </TableCell>
                  </TableRow>
                );
              })}
              {data.length === 0 && (
                <TableRow>
                  <TableCell colSpan={8} className="text-center py-8 text-muted-foreground">
                    利用者がいません
                  </TableCell>
                </TableRow>
              )}
            </TableBody>
          </Table>
        </CardContent>
      </Card>
    </div>
  );
}
