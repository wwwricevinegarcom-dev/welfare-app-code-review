'use client';

import { useState, useMemo } from 'react';
import Link from 'next/link';
import { Card, CardContent, CardDescription, CardHeader, CardTitle } from "@/components/ui/card";
import { buttonVariants, Button } from "@/components/ui/button";
import { Badge } from "@/components/ui/badge";
import { Input } from "@/components/ui/input";
import {
  FileText,
  User,
  PlusCircle,
  Calendar,
  AlertTriangle,
  CheckCircle2,
  Clock,
  ChevronDown,
  ChevronUp,
  History,
  LayoutGrid,
  List,
  Search,
  Filter,
  ArrowRight,
  ExternalLink,
  Users,
  AlertCircle
} from "lucide-react";
import { cn } from "@/lib/utils";
import { PlanActionButtons } from "./components/PlanActionButtons";

export interface PlanRecord {
  id: string;
  client_id: string;
  client_full_name?: string;
  plan_date: string;
  start_date?: string | null;
  end_date?: string | null;
  status: string;
  monitoring_period?: string | number | null;
  latest_monitoring_date?: string | null;
  consent_date?: string | null;
  delivery_date?: string | null;
  domains_data?: string | null;
  created_at?: string;
}

export interface ClientRecord {
  id: string;
  full_name: string;
  recipient_number?: string | null;
  status?: string | null;
}

interface PlansClientProps {
  clients: ClientRecord[];
  plans: PlanRecord[];
  initialClientId?: string;
  initialStatus?: string;
}

type FilterStatus = 'all' | 'overdue' | 'warning' | 'active' | 'draft' | 'no_plan';
type ViewMode = 'cards' | 'table';

export function PlansClient({
  clients,
  plans,
  initialClientId = '',
  initialStatus = 'all'
}: PlansClientProps) {
  const [selectedFilter, setSelectedFilter] = useState<FilterStatus>(
    initialStatus === 'active' ? 'active' : initialStatus === 'draft' ? 'draft' : 'all'
  );
  const [searchQuery, setSearchQuery] = useState(initialClientId ? '' : '');
  const [selectedClientFilter, setSelectedClientFilter] = useState(initialClientId);
  const [viewMode, setViewMode] = useState<ViewMode>('cards');
  const [expandedClients, setExpandedClients] = useState<Record<string, boolean>>({});

  const today = useMemo(() => new Date(), []);

  // 利用者ごとに計画書をグルーピング＆ステータス判定
  const groupedData = useMemo(() => {
    return clients.map(client => {
      // 当該利用者の全計画書（作成日降順）
      const clientPlans = plans.filter(p => p.client_id === client.id);

      // 有効な現行計画書（status === 'active' の中で最新）
      const activePlan = clientPlans.find(p => p.status === 'active') || null;

      // 最新の下書き計画書（status === 'draft'）
      const draftPlan = clientPlans.find(p => p.status === 'draft') || null;

      // 過去履歴（activePlan 以外のもの、または過去ステータス）
      const historyPlans = clientPlans.filter(p => p.id !== activePlan?.id && p.status !== 'draft');

      // メインで前面表示する計画書（現行優先、なければ下書き、それもなければ最新の過去計画）
      const primaryPlan = activePlan || draftPlan || (clientPlans.length > 0 ? clientPlans[0] : null);

      // モニタリング期日計算
      let dueDate: Date | null = null;
      let dueDateStr = '';
      let diffDays: number | null = null;
      let isOverdue = false;
      let isWarning = false;

      if (primaryPlan && primaryPlan.status === 'active') {
        const baseDateStr = primaryPlan.latest_monitoring_date || primaryPlan.start_date || primaryPlan.plan_date;
        const baseDate = new Date(baseDateStr);
        const months = parseInt(String(primaryPlan.monitoring_period)) || 6;
        dueDate = new Date(baseDate);
        const targetMonth = dueDate.getMonth() + months;
        dueDate.setMonth(targetMonth);
        if (dueDate.getMonth() !== (targetMonth % 12)) {
          dueDate.setDate(0);
        }
        dueDateStr = dueDate.toISOString().slice(0, 10);
        diffDays = Math.ceil((dueDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
        isOverdue = diffDays < 0;
        isWarning = diffDays >= 0 && diffDays <= 30;
      }

      // カテゴリ分類
      let category: FilterStatus = 'no_plan';
      if (!primaryPlan) {
        category = 'no_plan';
      } else if (isOverdue) {
        category = 'overdue';
      } else if (isWarning) {
        category = 'warning';
      } else if (primaryPlan.status === 'draft') {
        category = 'draft';
      } else if (primaryPlan.status === 'active') {
        category = 'active';
      }

      return {
        client,
        allPlans: clientPlans,
        activePlan,
        draftPlan,
        historyPlans,
        primaryPlan,
        dueDateStr,
        diffDays,
        isOverdue,
        isWarning,
        category,
        totalPlansCount: clientPlans.length
      };
    });
  }, [clients, plans, today]);

  // サマリー集計
  const counts = useMemo(() => {
    let overdue = 0;
    let warning = 0;
    let active = 0;
    let draft = 0;
    let noPlan = 0;

    groupedData.forEach(item => {
      if (item.category === 'overdue') overdue++;
      else if (item.category === 'warning') warning++;
      else if (item.category === 'draft') draft++;
      else if (item.category === 'active') active++;
      else if (item.category === 'no_plan') noPlan++;
    });

    return {
      all: groupedData.length,
      overdue,
      warning,
      active,
      draft,
      noPlan,
      needsAttention: overdue + warning + noPlan
    };
  }, [groupedData]);

  // フィルタリング処理
  const filteredData = useMemo(() => {
    return groupedData.filter(item => {
      // 利用者セレクトボックスでの絞り込み
      if (selectedClientFilter && item.client.id !== selectedClientFilter) {
        return false;
      }

      // テキスト検索（氏名）
      if (searchQuery.trim()) {
        const q = searchQuery.trim().toLowerCase();
        if (!item.client.full_name.toLowerCase().includes(q)) {
          return false;
        }
      }

      // ステータスフィルター
      if (selectedFilter === 'all') return true;
      if (selectedFilter === 'overdue') return item.isOverdue;
      if (selectedFilter === 'warning') return item.isWarning;
      if (selectedFilter === 'active') return item.primaryPlan?.status === 'active';
      if (selectedFilter === 'draft') return item.primaryPlan?.status === 'draft';
      if (selectedFilter === 'no_plan') return item.category === 'no_plan';

      return true;
    }).sort((a, b) => {
      // 要対応（期限超過・期日接近）を優先して上位に表示
      const order = { overdue: 1, warning: 2, no_plan: 3, draft: 4, active: 5 };
      const scoreA = order[a.category] || 99;
      const scoreB = order[b.category] || 99;
      if (scoreA !== scoreB) return scoreA - scoreB;
      return a.client.full_name.localeCompare(b.client.full_name, 'ja');
    });
  }, [groupedData, selectedFilter, searchQuery, selectedClientFilter]);

  // アコーディオン開閉トグル
  const toggleExpand = (clientId: string) => {
    setExpandedClients(prev => ({
      ...prev,
      [clientId]: !prev[clientId]
    }));
  };

  const expandAll = () => {
    const next: Record<string, boolean> = {};
    groupedData.forEach(item => {
      if (item.historyPlans.length > 0) {
        next[item.client.id] = true;
      }
    });
    setExpandedClients(next);
  };

  const collapseAll = () => {
    setExpandedClients({});
  };

  return (
    <div className="space-y-6">
      {/* 1. 上部KPIサマリーカード（ワンタップで絞り込み可能） */}
      <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-6 gap-3">
        <button
          type="button"
          onClick={() => setSelectedFilter('all')}
          className={cn(
            "p-3.5 rounded-xl border text-left transition-all flex flex-col justify-between",
            selectedFilter === 'all'
              ? "bg-primary/10 border-primary ring-2 ring-primary/20 shadow-sm"
              : "bg-card hover:bg-muted/40 border-border"
          )}
        >
          <div className="flex items-center justify-between text-xs font-bold text-muted-foreground">
            <span>全利用者</span>
            <Users className="h-4 w-4 text-primary" />
          </div>
          <div className="mt-2 flex items-baseline gap-1">
            <span className="text-2xl font-extrabold text-foreground">{counts.all}</span>
            <span className="text-xs text-muted-foreground">名</span>
          </div>
        </button>

        <button
          type="button"
          onClick={() => setSelectedFilter('overdue')}
          className={cn(
            "p-3.5 rounded-xl border text-left transition-all flex flex-col justify-between",
            counts.overdue > 0 ? "border-red-300 bg-red-50/40" : "bg-card border-border",
            selectedFilter === 'overdue' && "ring-2 ring-red-400 bg-red-100/60 border-red-400"
          )}
        >
          <div className="flex items-center justify-between text-xs font-bold text-red-700">
            <span>期日超過</span>
            <AlertTriangle className="h-4 w-4 text-red-600" />
          </div>
          <div className="mt-2 flex items-baseline gap-1">
            <span className={cn("text-2xl font-extrabold", counts.overdue > 0 ? "text-red-700" : "text-muted-foreground")}>
              {counts.overdue}
            </span>
            <span className="text-xs text-red-600">名</span>
          </div>
        </button>

        <button
          type="button"
          onClick={() => setSelectedFilter('warning')}
          className={cn(
            "p-3.5 rounded-xl border text-left transition-all flex flex-col justify-between",
            counts.warning > 0 ? "border-amber-300 bg-amber-50/40" : "bg-card border-border",
            selectedFilter === 'warning' && "ring-2 ring-amber-400 bg-amber-100/60 border-amber-400"
          )}
        >
          <div className="flex items-center justify-between text-xs font-bold text-amber-700">
            <span>期日接近 (30日)</span>
            <Clock className="h-4 w-4 text-amber-600" />
          </div>
          <div className="mt-2 flex items-baseline gap-1">
            <span className={cn("text-2xl font-extrabold", counts.warning > 0 ? "text-amber-700" : "text-muted-foreground")}>
              {counts.warning}
            </span>
            <span className="text-xs text-amber-600">名</span>
          </div>
        </button>

        <button
          type="button"
          onClick={() => setSelectedFilter('active')}
          className={cn(
            "p-3.5 rounded-xl border text-left transition-all flex flex-col justify-between",
            selectedFilter === 'active'
              ? "bg-emerald-50 border-emerald-400 ring-2 ring-emerald-300"
              : "bg-card hover:bg-muted/40 border-border"
          )}
        >
          <div className="flex items-center justify-between text-xs font-bold text-emerald-700">
            <span>現行運用中</span>
            <CheckCircle2 className="h-4 w-4 text-emerald-600" />
          </div>
          <div className="mt-2 flex items-baseline gap-1">
            <span className="text-2xl font-extrabold text-emerald-800">{counts.active}</span>
            <span className="text-xs text-muted-foreground">名</span>
          </div>
        </button>

        <button
          type="button"
          onClick={() => setSelectedFilter('draft')}
          className={cn(
            "p-3.5 rounded-xl border text-left transition-all flex flex-col justify-between",
            selectedFilter === 'draft'
              ? "bg-blue-50 border-blue-400 ring-2 ring-blue-300"
              : "bg-card hover:bg-muted/40 border-border"
          )}
        >
          <div className="flex items-center justify-between text-xs font-bold text-blue-700">
            <span>下書き作成中</span>
            <FileText className="h-4 w-4 text-blue-600" />
          </div>
          <div className="mt-2 flex items-baseline gap-1">
            <span className="text-2xl font-extrabold text-blue-800">{counts.draft}</span>
            <span className="text-xs text-muted-foreground">名</span>
          </div>
        </button>

        <button
          type="button"
          onClick={() => setSelectedFilter('no_plan')}
          className={cn(
            "p-3.5 rounded-xl border text-left transition-all flex flex-col justify-between",
            counts.noPlan > 0 ? "border-slate-300 bg-slate-50" : "bg-card border-border",
            selectedFilter === 'no_plan' && "ring-2 ring-slate-400 bg-slate-100 border-slate-400"
          )}
        >
          <div className="flex items-center justify-between text-xs font-bold text-slate-700">
            <span>未作成 (要策定)</span>
            <AlertCircle className="h-4 w-4 text-slate-500" />
          </div>
          <div className="mt-2 flex items-baseline gap-1">
            <span className={cn("text-2xl font-extrabold", counts.noPlan > 0 ? "text-slate-800" : "text-muted-foreground")}>
              {counts.noPlan}
            </span>
            <span className="text-xs text-slate-600">名</span>
          </div>
        </button>
      </div>

      {/* 2. 検索・操作ツールバー */}
      <Card className="shadow-xs border-border/80 bg-card">
        <CardContent className="p-3.5 sm:p-4">
          <div className="flex flex-col md:flex-row gap-3 items-stretch md:items-center justify-between">
            <div className="flex flex-wrap items-center gap-2.5 flex-1">
              {/* 名前検索 */}
              <div className="relative min-w-[200px] flex-1 sm:flex-initial">
                <Search className="absolute left-3 top-2.5 h-4 w-4 text-muted-foreground" />
                <Input
                  type="text"
                  placeholder="利用者を検索..."
                  value={searchQuery}
                  onChange={e => setSearchQuery(e.target.value)}
                  className="pl-9 h-9 text-sm"
                />
              </div>

              {/* 利用者セレクト */}
              <select
                value={selectedClientFilter}
                onChange={e => setSelectedClientFilter(e.target.value)}
                className="h-9 rounded-md border border-input bg-background px-3 text-sm shadow-xs focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring"
              >
                <option value="">全利用者を表示</option>
                {clients.map(c => (
                  <option key={c.id} value={c.id}>{c.full_name}</option>
                ))}
              </select>

              {(searchQuery || selectedClientFilter || selectedFilter !== 'all') && (
                <Button
                  variant="ghost"
                  size="sm"
                  onClick={() => {
                    setSearchQuery('');
                    setSelectedClientFilter('');
                    setSelectedFilter('all');
                  }}
                  className="h-9 px-2.5 text-xs text-muted-foreground hover:text-foreground"
                >
                  条件クリア
                </Button>
              )}
            </div>

            {/* 表示モード＆展開切替 */}
            <div className="flex items-center gap-2 justify-end shrink-0 pt-2 md:pt-0 border-t md:border-t-0 border-border/60">
              <Button
                variant="outline"
                size="sm"
                onClick={Object.keys(expandedClients).length > 0 ? collapseAll : expandAll}
                className="h-9 text-xs text-muted-foreground"
              >
                <History className="mr-1.5 h-3.5 w-3.5" />
                {Object.keys(expandedClients).length > 0 ? '全履歴を閉じる' : '全履歴を展開'}
              </Button>

              <div className="flex items-center rounded-lg border border-input bg-muted/30 p-0.5">
                <Button
                  variant={viewMode === 'cards' ? 'secondary' : 'ghost'}
                  size="sm"
                  onClick={() => setViewMode('cards')}
                  className={cn("h-7 px-2.5 text-xs", viewMode === 'cards' && "font-bold shadow-xs")}
                  title="カード表示"
                >
                  <LayoutGrid className="h-3.5 w-3.5 mr-1" />
                  カード
                </Button>
                <Button
                  variant={viewMode === 'table' ? 'secondary' : 'ghost'}
                  size="sm"
                  onClick={() => setViewMode('table')}
                  className={cn("h-7 px-2.5 text-xs", viewMode === 'table' && "font-bold shadow-xs")}
                  title="コンパクト一覧表示"
                >
                  <List className="h-3.5 w-3.5 mr-1" />
                  一覧表
                </Button>
              </div>
            </div>
          </div>
        </CardContent>
      </Card>

      {/* 3. メイン表示エリア */}
      {filteredData.length === 0 ? (
        <Card className="shadow-xs p-12 text-center text-muted-foreground">
          <p className="text-base font-medium">条件に一致する計画書・利用者は見つかりませんでした。</p>
          <Button
            variant="link"
            size="sm"
            onClick={() => {
              setSelectedFilter('all');
              setSearchQuery('');
              setSelectedClientFilter('');
            }}
            className="mt-2 text-primary"
          >
            すべての利用者を表示する
          </Button>
        </Card>
      ) : viewMode === 'cards' ? (
        /* ======================== カードビュー（利用者軸カード） ======================== */
        <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
          {filteredData.map(item => {
            const { client, primaryPlan, activePlan, draftPlan, historyPlans, isOverdue, isWarning, diffDays, dueDateStr } = item;
            const isExpanded = !!expandedClients[client.id];

            let domainsData = {};
            if (primaryPlan?.domains_data) {
              try {
                domainsData = JSON.parse(primaryPlan.domains_data);
              } catch (e) {}
            }

            return (
              <Card
                key={client.id}
                className={cn(
                  "shadow-xs hover:shadow-md transition-all flex flex-col justify-between border",
                  isOverdue && "border-red-300 bg-red-50/15",
                  isWarning && "border-amber-300 bg-amber-50/15",
                  !primaryPlan && "border-dashed border-slate-300 bg-slate-50/30"
                )}
              >
                <div>
                  {/* カードヘッダー */}
                  <CardHeader className="p-4 pb-3 border-b border-border/50 bg-muted/10">
                    <div className="flex items-start justify-between gap-2">
                      <div className="flex items-center gap-2 min-w-0">
                        <div className="h-8 w-8 rounded-full bg-primary/10 flex items-center justify-center shrink-0">
                          <User className="h-4 w-4 text-primary" />
                        </div>
                        <div className="min-w-0">
                          <h3 className="font-bold text-base text-foreground truncate flex items-center gap-1.5">
                            {client.full_name}
                            <span className="text-xs font-normal text-muted-foreground">様</span>
                          </h3>
                          <div className="flex items-center gap-1.5 text-[11px] text-muted-foreground">
                            <span>計画書: 全{item.totalPlansCount}件</span>
                          </div>
                        </div>
                      </div>

                      {/* 状態バッジ */}
                      <div className="flex flex-col items-end gap-1 shrink-0">
                        {primaryPlan ? (
                          <Badge
                            variant={primaryPlan.status === 'draft' ? 'outline' : 'default'}
                            className={cn(
                              "text-[11px] px-2 py-0.5 font-bold",
                              primaryPlan.status === 'draft'
                                ? "bg-amber-100 text-amber-800 border-amber-300"
                                : "bg-emerald-600 text-white"
                            )}
                          >
                            {primaryPlan.status === 'draft' ? '下書き作成中' : '現行運用中'}
                          </Badge>
                        ) : (
                          <Badge variant="outline" className="text-[11px] bg-slate-100 text-slate-700 border-slate-300">
                            未作成
                          </Badge>
                        )}

                        {isOverdue && (
                          <span className="text-[10px] px-1.5 py-0.5 rounded bg-red-100 text-red-700 font-bold border border-red-200">
                            ⚠ モニタリング{Math.abs(diffDays || 0)}日超過
                          </span>
                        )}
                        {isWarning && (
                          <span className="text-[10px] px-1.5 py-0.5 rounded bg-amber-100 text-amber-800 font-bold border border-amber-200">
                            期日接近 (残り{diffDays}日)
                          </span>
                        )}
                      </div>
                    </div>
                  </CardHeader>

                  {/* カードボディ（現行・最新計画書） */}
                  <CardContent className="p-4 space-y-3">
                    {primaryPlan ? (
                      <>
                        <div className="space-y-1.5 text-xs">
                          <div className="flex items-center justify-between text-muted-foreground">
                            <span>計画期間:</span>
                            <span className="font-semibold text-foreground">
                              {primaryPlan.start_date || primaryPlan.plan_date} 〜 {primaryPlan.end_date || '未設定'}
                            </span>
                          </div>

                          <div className="flex items-center justify-between text-muted-foreground">
                            <span>作成日: {primaryPlan.plan_date}</span>
                            {primaryPlan.delivery_date ? (
                              <span className="text-emerald-700 font-medium">交付済 ({primaryPlan.delivery_date})</span>
                            ) : primaryPlan.consent_date ? (
                              <span className="text-blue-700 font-medium">同意済 ({primaryPlan.consent_date})</span>
                            ) : (
                              <span className="text-amber-700 font-medium">未同意・未交付</span>
                            )}
                          </div>

                          {dueDateStr && (
                            <div className="flex items-center justify-between pt-1 border-t border-border/40 text-[11px]">
                              <span className="text-muted-foreground">次回モニタリング予定:</span>
                              <span className={cn("font-bold", isOverdue ? "text-red-600" : isWarning ? "text-amber-700" : "text-foreground")}>
                                {dueDateStr}
                              </span>
                            </div>
                          )}
                        </div>

                        {/* 課題・目標プレビュー */}
                        <div className="rounded-lg bg-muted/30 p-2.5 text-xs text-muted-foreground line-clamp-2">
                          <span className="font-bold text-foreground/80 mr-1">課題:</span>
                          {(domainsData as any)['健康・生活']?.issues || (domainsData as any)['本人・家族の希望']?.wishes || '設定された課題を入力中...'}
                        </div>

                        {/* アクションボタン */}
                        <div className="flex items-center gap-2 pt-1">
                          <Link
                            href={`/plans/${primaryPlan.id}`}
                            className={cn(buttonVariants({ size: "sm", variant: "default" }), "flex-1 h-8 text-xs font-bold")}
                          >
                            計画書を開く
                            <ArrowRight className="ml-1 h-3.5 w-3.5" />
                          </Link>
                          <Link
                            href={`/plans/${primaryPlan.id}/edit`}
                            className={cn(buttonVariants({ size: "sm", variant: "outline" }), "h-8 text-xs")}
                          >
                            編集
                          </Link>
                          {primaryPlan.status === 'draft' && (
                            <PlanActionButtons
                              planId={primaryPlan.id}
                              clientId={client.id}
                              status={primaryPlan.status}
                              consentDate={primaryPlan.consent_date}
                              compact={true}
                            />
                          )}
                        </div>
                      </>
                    ) : (
                      /* 計画書未作成時の案内 */
                      <div className="py-4 text-center space-y-3">
                        <p className="text-xs text-muted-foreground">
                          有効な個別支援計画書がまだありません。
                        </p>
                        <Link
                          href={`/plans/new?clientId=${client.id}`}
                          className={cn(buttonVariants({ size: "sm", className: "w-full font-bold" }))}
                        >
                          <PlusCircle className="mr-1.5 h-4 w-4" />

// 和暦変換ユーティリティ
function formatJapaneseEra(dateStr?: string): { era: string; year: string; month: string; day: string } {
  if (!dateStr) return { era: '', year: '', month: '', day: '' };
  try {
    const d = new Date(dateStr);
    if (isNaN(d.getTime())) return { era: '', year: '', month: '', day: '' };
    const year = d.getFullYear();
    const month = String(d.getMonth() + 1);
    const day = String(d.getDate());

    if (year >= 2019) {
      const reiwaYear = year === 2019 ? '元' : String(year - 2018);
      return { era: '令和', year: reiwaYear, month, day };
    } else if (year >= 1989) {
      const heiseiYear = year === 1989 ? '元' : String(year - 1988);
      return { era: '平成', year: heiseiYear, month, day };
    } else {
      const showaYear = year === 1926 ? '元' : String(year - 1925);
      return { era: '昭和', year: showaYear, month, day };
    }
  } catch {
    return { era: '', year: '', month: '', day: '' };
  }
}

// 生年月日のフォーマットと年齢計算
function formatBirthdateAndAge(birthdateStr?: string): { eraText: string; ageText: string } {
  if (!birthdateStr) return { eraText: '昭和 / 平成　　年　　月　　日生', ageText: '　　歳' };
  try {
    const d = new Date(birthdateStr);
    if (isNaN(d.getTime())) return { eraText: birthdateStr, ageText: '' };
    const year = d.getFullYear();
    const month = d.getMonth() + 1;
    const day = d.getDate();

    let eraLabel = '西暦';
    let eraYear: string | number = year;
    if (year >= 2019) {
      eraLabel = '令和';
      eraYear = year === 2019 ? '元' : year - 2018;
    } else if (year >= 1989) {
      eraLabel = '平成';
      eraYear = year === 1989 ? '元' : year - 1988;
    } else if (year >= 1926) {
      eraLabel = '昭和';
      eraYear = year === 1926 ? '元' : year - 1925;
    }

    const today = new Date();
    let age = today.getFullYear() - year;
    const m = today.getMonth() + 1 - month;
    if (m < 0 || (m === 0 && today.getDate() < day)) {
      age--;
    }

    return {
      eraText: `${eraLabel} ${eraYear} 年 ${month} 月 ${day} 日生`,
      ageText: `${age} 歳`
    };
  } catch {
    return { eraText: birthdateStr, ageText: '' };
  }
}

// 文字数に応じた枠内自動縮小（Auto-fit）ヘルパー
function getAutoFitClass(
  text: string | undefined | null,
  thresholds: { normal: number; small: number; tiny?: number } = { normal: 60, small: 110, tiny: 160 }
): string {
  const len = (text || '').trim().length;
  if (thresholds.tiny && len > thresholds.tiny) {
    return 'text-[6pt] leading-[1.05] tracking-tight';
  }
  if (len > thresholds.small) {
    return 'text-[6.5pt] leading-[1.1] tracking-tight';
  }
  if (len > thresholds.normal) {
    return 'text-[7.5pt] leading-[1.15]';
  }
  return 'text-[8pt] leading-snug';
}

export async function generateSupportPlan(
  supportRecordsText: string,
  clientData: ClientMaskingData,
  clientId?: string,
  monitoringText?: string,
  previousPlanContext?: string
) {
  try {
    const role = await getRole();
    if (!role) return { success: false, error: '認証されていません' };
    if (clientId) {
      await assertCanAccessClient(clientId);
    }
  } catch (err: any) {
    return { success: false, error: err?.message || 'アクセス権限がありません' };
  }

  const aiModel = getAiModelInstance();

  // clientData が不足している場合に備えて補完
  if (clientId) {
    try {
      const c = db.prepare('SELECT full_name, furigana, recipient_number, address, phone_number, disability_certificate, primary_doctor, emergency_contact_name, emergency_contact_phone FROM clients WHERE id = ?').get(clientId) as any;
      if (c) {
        clientData.full_name = clientData.full_name || c.full_name;
        clientData.furigana = clientData.furigana || c.furigana;
        clientData.recipient_number = clientData.recipient_number || c.recipient_number;
        clientData.address = clientData.address || c.address;
        clientData.phone_number = clientData.phone_number || c.phone_number;
        clientData.primary_doctor = clientData.primary_doctor || c.primary_doctor;
        clientData.emergency_contacts = clientData.emergency_contacts || (c.emergency_contact_name ? `${c.emergency_contact_name} ${c.emergency_contact_phone || ''}` : null);
      }
    } catch (e) {}
  }

  const maskedText = maskPII(supportRecordsText, clientData);

  let facesheetInfo = '未入力';
  let selfPlanInfo = '未入力';
  let assessmentInfo = '（アセスメント未作成またはデータなし）';
  
  if (clientId) {
    let facesheet = null;
    try {
      facesheet = db.prepare('SELECT * FROM client_facesheets WHERE client_id = ?').get(clientId) as any;
    } catch (e) {
      // テーブル未作成の場合はスキップ
      console.warn('client_facesheets テーブルが見つかりません:', e);
    }
    
    if (facesheet) {
      facesheetInfo = `
- 家族構成: ${facesheet.family_structure || '未入力'}
- 医療・服薬: ${facesheet.medical_history || '未入力'}
- 障害特性・配慮: ${facesheet.disability_details || '未入力'}
      `.trim();
    }

    let selfPlan = null;
    try {
      selfPlan = db.prepare('SELECT * FROM self_plans WHERE client_id = ? ORDER BY creation_date DESC, created_at DESC LIMIT 1').get(clientId) as any;
    } catch (e) {
      console.warn('self_plans テーブルの取得に失敗:', e);
    }

    if (selfPlan) {
      selfPlanInfo = `
- 本人の希望・将来の夢: ${selfPlan.hopes_and_dreams || '未入力'}
- 本人が感じる現在の課題・困りごと: ${selfPlan.current_challenges || '未入力'}
- 週間スケジュール・通所希望: ${selfPlan.weekly_schedule || '未入力'}
      `.trim();
    }

    try {
      const latestAssessment = db.prepare('SELECT * FROM assessments WHERE client_id = ? ORDER BY assessment_date DESC, created_at DESC LIMIT 1').get(clientId) as any;
      if (latestAssessment) {
        assessmentInfo = `
- 本人の希望・就労に対する意向: ${latestAssessment.user_wishes || latestAssessment.client_hopes || '未入力'}
- 基本的生活・体調面: ${latestAssessment.health_and_life || '未入力'}
- 作業遂行能力: ${latestAssessment.working_capabilities || '未入力'}
- 対人関係・コミュニケーション: ${latestAssessment.communication_skills || '未入力'}
- 現在の課題・留意点: ${latestAssessment.current_challenges || '未入力'}
- 支援方針（作業指示の工夫）: ${latestAssessment.support_policy_instructions || '未入力'}
- 支援方針（相談体制）: ${latestAssessment.support_policy_consultation || '未入力'}
- 支援方針（環境の配慮）: ${latestAssessment.support_policy_environment || '未入力'}
- 運用上の留意点: ${latestAssessment.operational_notes || '未入力'}
        `.trim();
      }
    } catch (e) {
      console.warn('assessments テーブルの取得に失敗:', e);
    }
  }

  // 要配慮個人情報（PII）の徹底マスキング
  const maskedFacesheet = facesheetInfo !== '未入力' ? maskPII(facesheetInfo, clientData) : '未入力';
  const maskedSelfPlan = selfPlanInfo !== '未入力' ? maskPII(selfPlanInfo, clientData) : '未入力';
  const maskedAssessment = assessmentInfo.includes('未作成') ? assessmentInfo : maskPII(assessmentInfo, clientData);
  const maskedPrevPlan = previousPlanContext ? maskPII(previousPlanContext, clientData) : '（初回作成、またはデータなし）';
  const maskedMonitoring = monitoringText ? maskPII(monitoringText, clientData) : '（初回作成、またはデータなし）';

  const serviceMode = getServiceModeSync();
  const isTypeB = serviceMode === 'type_b';
  const serviceName = isTypeB ? '就労継続支援B型' : '就労継続支援A型';

  const systemInstruction = `あなたは${serviceName}の経験豊富なサービス管理責任者（サビ管）です。利用者の支援記録、前回のモニタリング結果から、個別支援計画書の原案を作成してください。

【絶対遵守の禁止事項】本システムは障害者総合支援法に基づく『${serviceName}』専用です。介護保険法（デイサービス等）や児童福祉法（放課後等デイサービス等）、生活介護・自立訓練、${isTypeB ? '就労A型（雇用契約・労働基準法等）' : '就労B型'}など別サービスの用語・基準・目標設定を絶対に混同しないでください。必ず『${serviceName}』の基準・目標に合致した個別支援計画を立案してください。`;

  const prompt = `
【本人の基本情報】
${maskedFacesheet}

【本人の意向・セルフプラン（サービス等利用計画案）】
${maskedSelfPlan}

【アセスメントシート（課題分析票・心身の特性・配慮事項）】
${maskedAssessment}

【前回の計画内容（抜粋）】
${maskedPrevPlan}

【直近のモニタリング結果・次期の課題】
${maskedMonitoring}

【その他の支援記録・日々の様子】
${maskedText}

【絶対遵守の禁止事項】
本システムは障害者総合支援法に基づく『${serviceName}』専用です。介護保険法や児童福祉法、生活介護・自立訓練など他制度の基準を絶対に混同しないでください。

[指示事項]
1. 「モニタリング結果」で達成できている目標は、次のステップ（より高度な目標や一般就労への準備など）に引き上げてください。
2. 達成できていない課題がある場合は、スモールステップ化するなど、支援方法（プロセス）を見直した内容を提案してください。
3. 支援記録や日々の様子に「施設外就労」や企業先での請負作業が含まれている場合、実地指導（指定基準）に準拠するため、就労課題および具体的な支援内容に「施設外就労における作業内容・目標・安全配慮」を必ず明確に位置付けてください。
4. 指定のJSONフォーマットで出力してください。以下の項目について具体的な内容を生成してください。
- 本人の希望・ニーズ（意向）
- 生活に関する課題
- 就労に関する課題
- 一般就労への移行に向けた課題と支援
- 長期目標
- 短期目標
- 短期目標の達成予定時期（例: 「3ヶ月後」「令和7年3月」）
- 具体的な支援内容（作業内容、就労上の目標、${isTypeB ? '工賃向上に向けた作業目標やステップ' : '賃金の目安'}など）
- 具体的な支援を行う担当者の職種（例: 「職業指導員」「生活支援員」「サービス管理責任者」）
- 総合的な支援の方針
- 一般就労移行フェーズ（以下のいずれかから1つ選択: 「準備フェーズ（基礎的な労働習慣の定着）」「求職活動フェーズ（ハローワーク登録・面接練習）」「移行フェーズ（企業実習・トライアル雇用）」「定着支援フェーズ（一般就労後のフォロー）」）
`;

  try {
    let resultObject;
    try {
      const { object } = await generateObject({
        model: aiModel,
        system: systemInstruction,
        prompt: prompt,
        schema: planDataSchema,
      });
      resultObject = object;
    } catch (primaryErr: any) {
      console.warn('プライマリAIモデルでの生成に失敗しました。安定モデルへのフォールバックを試みます:', primaryErr.message);
      
      // 一時的な過負荷（High Demand）やモデルエラー時、安定版 gemini-3.5-flash に自動フォールバック
      try {
        const fallbackModel = getAiModelInstance({ overrideModelString: 'google:gemini-3.5-flash' });
        const { object } = await generateObject({
          model: fallbackModel,
          system: systemInstruction,
          prompt: prompt,
          schema: planDataSchema,
        });
        resultObject = object;
      } catch (fallbackErr: any) {
        // さらに軽量版の gemini-3.5-flash-lite にフォールバック
        console.warn('第2フォールバックを試みます:', fallbackErr.message);
        try {
          const secondFallbackModel = getAiModelInstance({ overrideModelString: 'google:gemini-3.5-flash-lite' });
          const { object } = await generateObject({
            model: secondFallbackModel,
            system: systemInstruction,
            prompt: prompt,
            schema: planDataSchema,
          });
          resultObject = object;
        } catch (liteErr: any) {
          console.warn('第2フォールバックも失敗しました:', liteErr.message);
        }
      }
    }

    // 全てのモデル呼び出しが失敗した（またはクォータ超過した）場合のスマートフォールバック
    if (!resultObject) {
      console.warn('AIモデルでの生成が利用できないため、標準テンプレートから個別支援計画原案を安全に生成します。');
      resultObject = createFallbackPlanData(clientData, supportRecordsText, monitoringText, previousPlanContext);
      return {
        success: true,
        originalText: supportRecordsText,
        maskedText: maskedText,
        plan: resultObject,
        isFallback: true,
        fallbackNotice: 'AIサービスが制限中または混雑しているため、事業所の標準テンプレートから個別支援計画原案を安全に自動生成しました。内容を確認し、必要に応じて編集してください。'
      };
    }

    return {
      success: true,
      originalText: supportRecordsText,
      maskedText: maskedText,
      plan: resultObject
    };
  } catch (error: any) {
    console.error('AI API Error (Fallback to template):', error);
    // 外側エラーでもテンプレートを返して業務を継続できるようにする
    const fallbackPlan = createFallbackPlanData(clientData, supportRecordsText, monitoringText, previousPlanContext);
    return {
      success: true,
      originalText: supportRecordsText,
      maskedText: maskedText,
      plan: fallbackPlan,
      isFallback: true,
      fallbackNotice: 'AIサービスの利用上限または通信エラーのため、標準テンプレートに基づいて個別支援計画原案を作成しました。内容をご確認のうえ編集してください。'
    };
  }
}

import { getFacilityEffectiveRules } from '@/lib/rules/ruleEngine';

export async function validatePlanDates(dates: {
  assessmentDate?: string | null;
  planDate?: string | null;
  conferenceDate?: string | null;
  consentDate?: string | null;
  deliveryDate?: string | null;
  startDate?: string | null;
  endDate?: string | null;
}): Promise<{ valid: boolean; error?: string }> {
  const rules = getFacilityEffectiveRules();
  const { assessmentDate, planDate, conferenceDate, consentDate, deliveryDate, startDate, endDate } = dates;

  // 終了日が開始日より前の明らかな論理破綻のみブロック
  if (startDate && endDate && startDate > endDate) {
    return { valid: false, error: `【支援プロセス不整合】計画の適用終了日（${endDate}）は開始日（${startDate}）以降の日付である必要があります。` };
  }

  if (rules.plans.strict_date_sequence) {
    if (assessmentDate && planDate && assessmentDate > planDate) {
      return { valid: false, error: `【支援プロセス不整合】アセスメント日（${assessmentDate}）が計画原案作成日（${planDate}）より後になっています。法定順序（アセスメント→原案）を遵守してください。` };
    }
    if (planDate && conferenceDate && planDate > conferenceDate) {
      return { valid: false, error: `【支援プロセス不整合】計画原案作成日（${planDate}）がサービス担当者会議日（${conferenceDate}）より後になっています。法定順序（原案→会議）を遵守してください。` };
    }
    if (conferenceDate && consentDate && conferenceDate > consentDate) {
      return { valid: false, error: `【支援プロセス不整合】サービス担当者会議日（${conferenceDate}）が利用者同意日（${consentDate}）より後になっています。法定順序（会議→同意）を遵守してください。` };
    }
    if (consentDate && deliveryDate && consentDate > deliveryDate) {
      return { valid: false, error: `【支援プロセス不整合】利用者交付日（${deliveryDate}）が利用者同意日（${consentDate}）より前になっています。法定順序（同意→交付）を遵守してください。` };
    }
  }

  return { valid: true };
}

export async function saveSupportPlan(clientId: string, planData: z.infer<typeof planDataSchema>, metaData?: any) {
  try {
    const role = await getRole();
    if (!role) return { success: false, error: '認証されていません。ログインしてください。' };
    await assertCanAccessClient(clientId);

    const validatedPlanData = planDataSchema.parse(planData);

    const planDate = metaData?.plan_date || toLocalISODate(new Date());
    const startDate = metaData?.start_date || null;
    const endDate = metaData?.end_date || null;
    const consentDate = metaData?.consent_date || null;
    const deliveryDate = metaData?.delivery_date || null;

    const valResult = await validatePlanDates({
      planDate,
      startDate,
      endDate,
      consentDate,
      deliveryDate
    });

    if (!valResult.valid) {
      return { success: false, error: valResult.error };
    }

    const isSelfPlan = metaData?.is_self_plan ? 1 : 0;

    const stmt = db.prepare(`
      INSERT INTO support_plans (
        id, client_id, plan_date, status, domains_data, 
        start_date, end_date, consent_date, delivery_date, creator_id, creator_name, monitoring_result, is_self_plan
      )
      VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
    `);

    const newId = uuidv4();
    const validCreatorId = sanitizeForeignKeyId('staff', metaData?.creator_id);
    stmt.run(
      newId,
      clientId,
      planDate,
      'draft',
      JSON.stringify(validatedPlanData),
      startDate,
      endDate,
      consentDate,
      deliveryDate,
      validCreatorId,
      metaData?.creator_name || null,
      metaData?.monitoring_result || null,
      isSelfPlan
    );

    try {
      await logAudit({
        action: 'create',
        targetTable: 'support_plans',
        targetId: newId,
        details: { clientId, planDate, creatorName: metaData?.creator_name }
      });
    } catch (e) {}

    try {
      revalidatePath('/plans');
      revalidatePath('/');
    } catch (e) {}

    return { success: true, id: newId };
  } catch (error: any) {
    console.error('Failed to save plan:', error);
    return { success: false, error: `保存に失敗しました: ${error.message || '入力内容をご確認ください'}` };
  }
}

export async function activateSupportPlan(planId: string) {
  try {
    const role = await getRole();
    if (!role) return { success: false, error: '認証されていません。ログインしてください。' };

    const validPlanId = uuidSchema.parse(planId);
    resolveServiceModeForPlan(validPlanId);
    const plan = db.prepare('SELECT * FROM support_plans WHERE id = ?').get(validPlanId) as any;
    if (!plan) return { success: false, error: '対象の個別支援計画が見つかりません' };
    await assertCanAccessClient(plan.client_id);

    if (!plan.consent_date || plan.consent_date.trim() === '') {
      return {
        success: false,
        error: '利用者の説明・同意日（署名日）が未入力です。実地指導（指定基準第58条第6項）において本人の同意がない計画は無効となり「個別支援計画未作成減算」の対象となります。先に計画詳細より同意日をご入力ください。'
      };
    }

    if (!plan.delivery_date || plan.delivery_date.trim() === '') {
      return {
        success: false,
        error: '計画書の交付日が未入力です。実地指導において、同意を得た上で計画書を本人に交付することが指定基準で義務付けられています。先に計画詳細より交付日をご入力ください。'
      };
    }
    
    if (new Date(plan.delivery_date) < new Date(plan.consent_date)) {
      return {
        success: false,
        error: '交付日は同意日以降の日付である必要があります。'
      };
    }

    // 関連するアセスメントおよび担当者会議の日時整合性を検証
    const latestAssessment = db.prepare(`
      SELECT assessment_date FROM assessments 
      WHERE (support_plan_id = ? OR client_id = ?) AND deleted_at IS NULL
      ORDER BY assessment_date DESC LIMIT 1
    `).get(validPlanId, plan.client_id) as any;

    const latestConference = db.prepare(`
      SELECT conference_date FROM service_conferences 
      WHERE (support_plan_id = ? OR client_id = ?) AND deleted_at IS NULL
      ORDER BY conference_date DESC LIMIT 1
    `).get(validPlanId, plan.client_id) as any;

    if (!latestAssessment) {
      return {
        success: false,
        error: '【法定プロセス不備】アセスメント記録が登録されていません。指定基準第58条に基づき、個別支援計画の作成には事前にアセスメントの実施・記録が必須です。'
      };
    }

    if (!latestConference) {
      return {
        success: false,
        error: '【法定プロセス不備】サービス担当者会議録が登録されていません。指定基準第58条に基づき、原案作成後に担当者会議（ケース会議）を開催・記録することが必須です。'
      };
    }

    const valResult = await validatePlanDates({
      assessmentDate: latestAssessment?.assessment_date,
      planDate: plan.plan_date,
      conferenceDate: latestConference?.conference_date,
      consentDate: plan.consent_date,
      deliveryDate: plan.delivery_date,
      startDate: plan.start_date,
      endDate: plan.end_date
    });

    if (!valResult.valid) {
      return { success: false, error: valResult.error };
    }

    db.transaction(() => {
      // 同一利用者の既存のアクティブ計画を過去計画（archived）に変更
      db.prepare(`UPDATE support_plans SET status = 'archived' WHERE client_id = ? AND status = 'active'`).run(plan.client_id);
      // 対象の計画を有効化
      db.prepare(`UPDATE support_plans SET status = 'active' WHERE id = ?`).run(validPlanId);
    })();

    try {
      await logAudit({
        action: 'update',
        targetTable: 'support_plans',
        targetId: validPlanId,
        details: { status: 'active', consentDate: plan.consent_date }
      });
    } catch (e) {}

    try {
      revalidatePath('/plans');
      revalidatePath(`/plans/${planId}`);
      revalidatePath('/');
    } catch (e) {}
    return { success: true };
  } catch (error: any) {
    console.error('Failed to activate plan:', error);
    return { success: false, error: `有効化に失敗しました: ${error.message}` };
  }
}

/**
 * 【Step 4: 利用者カルテ連動「過去記録ノート（モニタリング自動要約）」】
 * 対象利用者の「直近6ヶ月間の支援記録（support_records）および勤怠実績（attendance_records）」のみを
 * ソース資料としてAIに渡し、PIIを[利用者A]等に完全マスキングした上で、
 * 記録された事実のみから6ヶ月間の通所安定度・作業達成度・体調変化を要約する。
 */
export async function generateGroundedMonitoringDraft(
  clientId: string,
  planId?: string,
  referenceDate?: string
) {
  try {
    const role = await getRole();
    if (!role) return { success: false, error: '認証されていません' };
    await assertCanAccessClient(clientId);

    const serviceMode = getServiceModeSync();
    const isTypeB = serviceMode === 'type_b';
    const serviceName = isTypeB ? '就労継続支援B型' : '就労継続支援A型';

    // 1. 利用者情報取得（PIIマスキング辞書用）
    const client = db
      .prepare(
        `
        SELECT id, full_name, furigana, recipient_number, address, phone_number,
               disability_certificate, primary_doctor, emergency_contact_name, emergency_contact_phone
        FROM clients
        WHERE id = ?
      `
      )
      .get(clientId) as any;

    if (!client) {
      return { success: false, error: '対象の利用者が見つかりません' };
    }

    const maskData: ClientMaskingData = {
      full_name: client.full_name,
      furigana: client.furigana,
      recipient_number: client.recipient_number,
      address: client.address,
      phone_number: client.phone_number,
      primary_doctor: client.primary_doctor,
      emergency_contacts: client.emergency_contact_name
        ? `${client.emergency_contact_name} ${client.emergency_contact_phone || ''}`
        : null,
    };

    const maskToClientA = (raw: string | null | undefined): string => {
      if (!raw) return '';
      let text = raw;
      if (client.full_name) {
        const nameNoSpace = client.full_name.replace(/[\s　]+/g, '');
        text = text.split(client.full_name).join('[利用者A]');
        if (nameNoSpace.length >= 2) {
          text = text.split(nameNoSpace).join('[利用者A]');
        }
      }
      if (client.recipient_number) {
        text = text.split(client.recipient_number).join('[受給者証番号]');
      }
      return maskPII(text, maskData).replace(/\[利用者氏名\]/g, '[利用者A]');
    };

    // 2. 直近6ヶ月間の期間算出
    const endObj = referenceDate ? new Date(referenceDate) : new Date();
    const startObj = new Date(endObj);
    startObj.setMonth(startObj.getMonth() - 6);
    const startDateStr = toLocalISODate(startObj);
    const endDateStr = toLocalISODate(endObj);

    // 3. 直近6ヶ月間の勤怠実績（attendance_records）取得
    const attendanceRows = db
      .prepare(
        `
        SELECT record_date, status, clock_in, clock_out, break_time_minutes, work_type,
               absence_reason, absence_condition
        FROM attendance_records
        WHERE client_id = ?
          AND record_date >= ?
          AND record_date <= ?
          AND deleted_at IS NULL
        ORDER BY record_date ASC
      `
      )
      .all(clientId, startDateStr, endDateStr) as any[];

    // 4. 直近6ヶ月間の支援記録（support_records）取得
    const supportRows = db
      .prepare(
        `
        SELECT record_date, vitals, task_content, staff_comment, service_location
        FROM support_records
        WHERE client_id = ?
          AND record_date >= ?
          AND record_date <= ?
          AND deleted_at IS NULL
        ORDER BY record_date DESC
      `
      )
      .all(clientId, startDateStr, endDateStr) as any[];

    let shortGoal = '';
    let longGoal = '';
    try {
      const planRow = planId
        ? (db.prepare(`SELECT domains_data FROM support_plans WHERE id = ?`).get(planId) as any)
        : (db
            .prepare(
              `SELECT domains_data FROM support_plans WHERE client_id = ? AND deleted_at IS NULL ORDER BY plan_date DESC LIMIT 1`
            )
            .get(clientId) as any);
      if (planRow?.domains_data) {
        const parsed = JSON.parse(planRow.domains_data);
        shortGoal = maskToClientA(parsed.short_term_goal || '');
        longGoal = maskToClientA(parsed.long_term_goal || '');
      }
    } catch (e) {}

    // 6. 勤怠の客観的集計
    const presentRows = attendanceRows.filter(r => r.status === 'present');
    const absentRows = attendanceRows.filter(r => r.status === 'absent');
    const ptoRows = attendanceRows.filter(r => r.status === 'pto' || r.status === 'paid_leave');
    const offsiteDays = presentRows.filter(
      r => r.work_type === 'out_facility' || r.work_type === 'both'
    ).length;
    const inFacilityDays = presentRows.length - offsiteDays;
    const totalScheduledDays = presentRows.length + absentRows.length + ptoRows.length;
    const attendanceRate =
      totalScheduledDays > 0
        ? Math.round(((presentRows.length + ptoRows.length) / totalScheduledDays) * 1000) / 10
        : 0;

    const maskedSupportLogs = supportRows.slice(0, 60).map(r => {
      const v = maskToClientA(r.vitals || '特記なし');
      const t = maskToClientA(r.task_content || '通常作業');
      const c = maskToClientA(r.staff_comment || '');
      return `- [${r.record_date}] 体調・バイタル: ${v} | 作業内容: ${t} | 支援記録・様子: ${c}`;
    });

    const maskedAbsenceLogs = absentRows
      .filter(r => r.absence_reason || r.absence_condition)
      .slice(0, 20)
      .map(
        r =>
          `- [${r.record_date}] 欠席理由: ${maskToClientA(r.absence_reason || '未記載')} / 体調状況: ${maskToClientA(r.absence_condition || '未記載')}`
      );

    const sourceNotebookText = `
【対象期間】${startDateStr} 〜 ${endDateStr}（直近6ヶ月間）
【対象者識別子】[利用者A]（※個人情報保護のため氏名・受給者証番号等のPIIはすべてマスキング済）
【事業種別】${serviceName}
${shortGoal ? `【現行の短期目標】${shortGoal}` : ''}
${longGoal ? `【現行の長期目標】${longGoal}` : ''}

==================================================
■ ソース資料1: 直近6ヶ月間の勤怠実績（attendance_records 集計: 全${attendanceRows.length}件）
- 通所（出勤）日数: ${presentRows.length}日（うち施設内: ${inFacilityDays}日 / 施設外就労: ${offsiteDays}日）
- 欠席日数: ${absentRows.length}日
${!isTypeB ? `- 有給休暇取得日数: ${ptoRows.length}日\n` : ''}- 通所安定率（出勤率）: ${attendanceRate}%
${maskedAbsenceLogs.length > 0 ? `- 記録された欠席時の理由・体調:\n${maskedAbsenceLogs.join('\n')}` : '- 欠席時の特記理由記録: なし'}

==================================================

${maskedSupportLogs.length > 0 ? maskedSupportLogs.join('\n') : '（該当期間の日次支援記録は0件です）'}

    // 8. 事実限定フォールバック要約生成関数（AI未接続・制限時でも記録事実のみから100%正確に構成）

    try {

        system: systemInstruction,
        prompt: `以下の【PIIマスキング済み 直近6ヶ月間 過去記録ノート】のみを根拠として、一切の創作を行わずにモニタリング評価の下書きを作成してください。\n\n${sourceNotebookText}`,
        schema: groundedMonitoringSchema,

        success: true,
        data: {
          ...object,
          period_start: startDateStr,
          period_end: endDateStr,
          attendance_count: attendanceRows.length,
          support_record_count: supportRows.length,
          masked_source_preview: sourceNotebookText,
        },
      };
    } catch (aiErr: any) {
      console.warn('Grounded monitoring AI fallback to strict fact-only summarizer:', aiErr?.message || aiErr);
      const fallbackDraft = buildFactOnlyFallback();
      return {
        success: true,
        data: {
          ...fallbackDraft,
          period_start: startDateStr,
          period_end: endDateStr,
          attendance_count: attendanceRows.length,
          support_record_count: supportRows.length,
          masked_source_preview: sourceNotebookText,
        },
      };
    }
  } catch (error: any) {
    console.error('Failed to generate grounded monitoring draft:', error);
    return {
      success: false,
      error: '過去記録ノートからのモニタリング要約生成に失敗しました: ' + (error.message || String(error)),
    };
  }
}

'use client';

import { useState } from 'react';
import { Button } from '@/components/ui/button';
import { FileCheck2, Trash2, Loader2, AlertCircle } from 'lucide-react';
import { activateSupportPlan, deleteSupportPlan } from '../actions';
import { useRouter } from 'next/navigation';

interface PlanActionButtonsProps {
  planId: string;
  clientId: string;
  status: string;
  consentDate?: string | null;
  redirectTo?: string;
  compact?: boolean;
}

export function PlanActionButtons({
  planId,
  clientId,
  status,
  consentDate,
  redirectTo = '/plans',
  compact = false
}: PlanActionButtonsProps) {
  const router = useRouter();
  const [isActivating, setIsActivating] = useState(false);
  const [isDeleting, setIsDeleting] = useState(false);

  const handleActivate = async () => {
    if (!consentDate || consentDate.trim() === '') {
      alert(
        '【利用者の同意日（署名日）が必要です】\n\n' +
        '障害者総合支援法の指定基準（第58条第6項）および実地指導ルールにより、' +
        '本人の同意がない個別支援計画は無効となり「個別支援計画未作成減算」の対象となります。\n\n' +
        '「編集する」ボタンを押し、利用者・ご家族への説明・同意日を入力して保存してから有効化を行ってください。'
      );
      return;
    }

    if (!confirm('この個別支援計画書を有効化（現行の運用計画として確定）しますか？\n同一利用者の既存の有効計画は過去履歴に移行されます。')) {
      return;
    }

    setIsActivating(true);
    try {
      const res = await activateSupportPlan(planId);
      if (!res.success) {
        alert(res.error || '有効化に失敗しました');
        return;
      }
      router.refresh();
    } catch (e: any) {
      alert(e.message || '有効化に失敗しました');
    } finally {
      setIsActivating(false);
    }
  };

  const handleDelete = async () => {
    const msg = status === 'draft' 
      ? 'この下書き計画書を削除してもよろしいですか？'
      : 'この計画書を削除しますか？（※法令保存義務の対象となる場合があります）';

    if (!confirm(msg)) {
      return;
    }

    setIsDeleting(true);
    try {
      const res = await deleteSupportPlan(planId);
      if (!res.success) {
        alert(res.error || '削除に失敗しました');
        setIsDeleting(false);
        return;
      }
      if (redirectTo) {
        router.push(redirectTo);
      } else {
        router.refresh();
      }
    } catch (e: any) {
      alert(e.message || '削除に失敗しました');
      setIsDeleting(false);
    }
  };

  if (compact) {
    return (
      <div className="flex items-center gap-1">
        {status === 'draft' && (
          <Button
            size="sm"
            variant="ghost"
            className="h-8 w-8 p-0 text-destructive hover:bg-destructive/10"
            onClick={handleDelete}
            disabled={isDeleting}
            title="下書きを削除"
          >
            {isDeleting ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Trash2 className="h-3.5 w-3.5" />}
          </Button>
        )}
      </div>
    );
  }

  return (
    <div className="flex items-center gap-2">
      {status === 'draft' && (
        <>
          <Button
            type="button"
            variant="default"
            onClick={handleActivate}
            disabled={isActivating || isDeleting}
            className="bg-emerald-600 hover:bg-emerald-700 text-white shadow-xs font-bold"
          >
            {isActivating ? (
              <Loader2 className="mr-2 h-4 w-4 animate-spin" />
            ) : (
              <FileCheck2 className="mr-2 h-4 w-4" />
            )}
            有効化する
          </Button>

          <Button
            type="button"
            variant="outline"
            onClick={handleDelete}
            disabled={isActivating || isDeleting}
            className="text-destructive hover:bg-destructive/10 hover:text-destructive border-destructive/30"
          >
            {isDeleting ? (
              <Loader2 className="mr-2 h-4 w-4 animate-spin" />
            ) : (
              <Trash2 className="mr-2 h-4 w-4" />
            )}
            下書きを削除
          </Button>
        </>
      )}
    </div>
  );
}

export interface ComplianceItemStatus {
  category: string;
  label: string;
  period: string;
  lastHeldDate: string | null;
  deadlineDate: string;
  daysRemaining: number;
  status: 'ok' | 'notice' | 'warning' | 'overdue';
  statusText: string;
}

export async function checkMandatoryComplianceStatus(baseDateStr?: string): Promise<{
  missingCommittees: { label: string; period: string; category: string }[];
  items: ComplianceItemStatus[];
  urgentCount: number;
}> {
  const role = await getRole();
  if (!role) return { missingCommittees: [], items: [], urgentCount: 0 };

  const baseDate = baseDateStr ? new Date(baseDateStr) : new Date();
  const todayStr = baseDate.toISOString().slice(0, 10);

  const mandatoryItems = [
    { category: 'abuse_prevention', label: '虐待防止委員会・職員研修', period: '1年に1回以上', intervalMonths: 12 },
    { category: 'restraint_reduction', label: '身体拘束適正化検討委員会・研修', period: '6ヶ月に1回以上', intervalMonths: 6 },
    { category: 'infection_control', label: '感染症対策委員会・研修・訓練', period: '6ヶ月に1回以上', intervalMonths: 6 },
    { category: 'bcp', label: 'BCP（業務継続計画）研修・訓練', period: '1年に1回以上', intervalMonths: 12 },
  ];

  const missingCommittees: { label: string; period: string; category: string }[] = [];
  const items: ComplianceItemStatus[] = [];
  let urgentCount = 0;

  for (const item of mandatoryItems) {
    const latest = db.prepare(`
      SELECT id, held_date FROM committee_records
      WHERE category = ? AND deleted_at IS NULL
      ORDER BY held_date DESC
      LIMIT 1
    `).get(item.category) as { id: string; held_date: string } | undefined;

    let deadlineDateStr: string;
    let daysRemaining: number;
    let status: 'ok' | 'notice' | 'warning' | 'overdue';
    let statusText: string;

    if (!latest || !latest.held_date) {
      // 過去に一度も開催記録がない
      deadlineDateStr = todayStr;
      daysRemaining = -999;
      status = 'overdue';
      statusText = '未実施（減算対象）';
      missingCommittees.push({ label: item.label, period: item.period, category: item.category });
      urgentCount++;
    } else {
      const heldD = new Date(latest.held_date);
      const deadline = new Date(heldD);
      deadline.setMonth(deadline.getMonth() + item.intervalMonths);
      deadlineDateStr = deadline.toISOString().slice(0, 10);

      const diffTime = deadline.getTime() - baseDate.getTime();
      daysRemaining = Math.ceil(diffTime / (1000 * 60 * 60 * 24));

      if (daysRemaining < 0) {
        status = 'overdue';
        statusText = `期限超過 (${Math.abs(daysRemaining)}日超過)`;
        missingCommittees.push({ label: item.label, period: item.period, category: item.category });
        urgentCount++;
      } else if (daysRemaining <= 30) {
        status = 'warning';
        statusText = `あと${daysRemaining}日（期限切迫）`;
        urgentCount++;
      } else if (daysRemaining <= 60) {
        status = 'notice';
        statusText = `あと${daysRemaining}日（準備期間）`;
      } else {
        status = 'ok';
        statusText = `充足（あと${daysRemaining}日）`;
      }
    }

    items.push({
      category: item.category,
      label: item.label,
      period: item.period,
      lastHeldDate: latest?.held_date || null,
      deadlineDate: deadlineDateStr,
      daysRemaining,
      status,
      statusText
    });
  }

  return { missingCommittees, items, urgentCount };
}

'use server'

import { db, resolveServiceModeForTableRecord } from "@/lib/db"
import { randomUUID } from "crypto"
import { redirect } from "next/navigation"
import { getRole } from "@/lib/auth"

export async function saveComplaintRecord(formData: FormData): Promise<void> {
  const role = await getRole();
  if (!role) {
    redirect('/login');
  }

  try {
    const receivedDate = formData.get('received_date') as string;
    const complaintContent = formData.get('complaint_content') as string;

    if (!receivedDate || !receivedDate.trim()) {
      redirect('/records/complaints?error=' + encodeURIComponent('受付日は必須入力です。'));
    }
    if (!complaintContent || !complaintContent.trim()) {
      redirect('/records/complaints?error=' + encodeURIComponent('苦情内容は必須入力です。'));
    }

    const isThirdPartyReported = formData.get('third_party_reported') === 'on';
    const thirdPartyReportDate = formData.get('third_party_report_date') as string;
    if (isThirdPartyReported && (!thirdPartyReportDate || thirdPartyReportDate.trim() === '')) {
      redirect('/records/complaints?error=' + encodeURIComponent('「第三者委員への報告」を行う場合は、報告日を必ず入力してください（実地指導での客観的記録要件）。'));
    }

    const existingId = (formData.get('id') as string)?.trim() || '';
    if (existingId) {
      resolveServiceModeForTableRecord('complaint_records', existingId);
    }
    const isEdit = !!existingId;
    const id = existingId || randomUUID();

    if (isEdit) {
      db.prepare(`
        UPDATE complaint_records SET
          received_date = ?, complainant_name = ?, complainant_relation = ?,
          complaint_content = ?, investigation_content = ?, response_content = ?, response_date = ?, improvement_measures = ?,
          handled_by = ?, third_party_reported = ?, third_party_report_date = ?, status = ?,
          updated_at = CURRENT_TIMESTAMP
        WHERE id = ?
      `).run(
        formData.get('received_date') || null,
        formData.get('complainant_name') || null,
        formData.get('complainant_relation') || null,
        formData.get('complaint_content') || null,
        formData.get('investigation_content') || null,
        formData.get('response_content') || null,
        formData.get('response_date') || null,
        formData.get('improvement_measures') || null,
        formData.get('handled_by') || null,
        formData.get('third_party_reported') === 'on' ? 1 : 0,
        formData.get('third_party_report_date') || null,
        formData.get('status') || '対応中',
        id
      );
    } else {
      const stmt = db.prepare(`
        INSERT INTO complaint_records (
          id, received_date, complainant_name, complainant_relation,
          complaint_content, investigation_content, response_content, response_date, improvement_measures,
          handled_by, third_party_reported, third_party_report_date, status
        ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
      `);

      stmt.run(
        id,
        formData.get('received_date') || null,
        formData.get('complainant_name') || null,
        formData.get('complainant_relation') || null,
        formData.get('complaint_content') || null,
        formData.get('investigation_content') || null,
        formData.get('response_content') || null,
        formData.get('response_date') || null,
        formData.get('improvement_measures') || null,
        formData.get('handled_by') || null,
        formData.get('third_party_reported') === 'on' ? 1 : 0,
        formData.get('third_party_report_date') || null,
        formData.get('status') || '対応中'
      );
    }

    try {
      const { logAudit } = await import('@/lib/audit');
      await logAudit({
        action: 'create',
        targetTable: 'complaint_records',
        targetId: id,
        details: { received_date: formData.get('received_date'), complainant_name: formData.get('complainant_name') }
      });
    } catch (e) {}
  } catch (error: any) {
    if (error?.digest?.startsWith('NEXT_REDIRECT')) throw error;
    console.error("Failed to save complaint record:", error);
    redirect('/records/complaints?error=' + encodeURIComponent('登録に失敗しました: ' + error.message));
  }
  redirect('/records/complaints?msg=' + encodeURIComponent('苦情対応記録を登録しました'));
}

export async function deleteComplaintRecord(id: string): Promise<void> {
  const role = await getRole();
  if (role !== 'admin' && role !== 'service_manager') {
    redirect('/records/complaints?error=' + encodeURIComponent('削除には管理者権限が必要です'));
  }

  try {
    resolveServiceModeForTableRecord('complaint_records', id);
    db.prepare('UPDATE complaint_records SET deleted_at = CURRENT_TIMESTAMP WHERE id = ?').run(id);
  } catch (error: any) {
    if (error?.digest?.startsWith('NEXT_REDIRECT')) throw error;
    console.error("Failed to delete complaint record:", error);
    redirect('/records/complaints?error=' + encodeURIComponent('削除に失敗しました'));
  }
  redirect('/records/complaints?msg=' + encodeURIComponent('苦情対応記録を削除しました'));
}

                          <span className="text-muted-foreground">未報告</span>
                        )}
                      </td>
                      <td className="px-4 py-3">
                        <div className="flex items-center gap-1">
                          <Link
                            href={`/records/complaints?editId=${record.id}`}
                            className="inline-flex items-center justify-center text-primary hover:text-primary/80 text-xs h-7 px-2 border rounded bg-background hover:bg-muted"
                          >
                            編集
                          </Link>
                          <DeleteComplaintButton recordId={record.id} />
                        </div>
                      </td>
                    </tr>
                  ))}
                  {records.length === 0 && (
                    <tr>
                      <td colSpan={8} className="px-4 py-8 text-center text-muted-foreground">記録がありません</td>
                    </tr>
                  )}
                </tbody>
              </table>
            </div>
          </CardContent>
        </Card>
      </main>
    </div>
  );
}
