function FactCheckLinksBox({ 
  links, 
  defaultUrl,
  title = "厚生労働省 公式ファクトチェック（多重照会リンク）" 
}: { 
  links?: FactCheckLink[]; 
  defaultUrl?: string;
  title?: string;
}) {
  if (!links || links.length === 0) {
    if (!defaultUrl) return null;
    return (
      <div className="text-xs">
        <a 
          href={defaultUrl} 
          target="_blank" 
          rel="noopener noreferrer"
          className="inline-flex items-center gap-1 text-blue-600 hover:text-blue-800 hover:underline font-medium"
        >
          <ExternalLink className="h-3.5 w-3.5" />
          厚生労働省の公式情報を確認する
        </a>
      </div>
    );
  }

  return (
    <div className="rounded-lg border border-slate-200/90 bg-slate-50/80 p-3 space-y-2">
      <div className="flex flex-wrap items-center justify-between gap-1">
        <span className="text-xs font-bold text-slate-800 flex items-center gap-1.5">
          <ShieldCheck className="h-4 w-4 text-indigo-600" />
          {title}
        </span>
        <span className="text-[10px] text-muted-foreground">
          ※ 万が一の国のURL変更時でも確実に繋がる恒久ポータル・公式内検索を併記
        </span>
      </div>

      <div className="grid grid-cols-1 sm:grid-cols-2 gap-2 pt-0.5">
        {links.map((link, idx) => (
          <a 
            key={idx}
            href={link.url}
            target="_blank"
            rel="noopener noreferrer"
            className="flex items-start gap-2 p-2 rounded-md bg-white border border-slate-200/80 hover:border-indigo-300 hover:shadow-xs transition-all group"
          >
            <span className={`text-[10px] px-1.5 py-0.5 rounded font-semibold shrink-0 mt-0.5 border ${
              link.category === 'primary' 
                ? 'bg-amber-50 text-amber-800 border-amber-200' 
                : link.category === 'permanent_portal'
                ? 'bg-blue-50 text-blue-800 border-blue-200'
                : 'bg-indigo-50 text-indigo-800 border-indigo-200'
            }`}>
              {link.badgeLabel}
            </span>
            <div className="min-w-0 flex-1">
              <div className="text-xs font-semibold text-slate-800 group-hover:text-indigo-600 flex items-center gap-1">
                <span className="truncate">{link.title}</span>
                <ExternalLink className="h-3 w-3 shrink-0 text-slate-400 group-hover:text-indigo-600" />
              </div>
              <p className="text-[10px] text-muted-foreground line-clamp-1 mt-0.5">
                {link.description}
              </p>
            </div>
          </a>
        ))}
      </div>
    </div>
  );
}

export function ComplianceAssistant({ 

  currentAddress, 
  currentWage 
}: { 
  currentAddress: string, 
  currentWage: number 
}) {
  const router = useRouter();
  
  // 最低賃金チェック状態
  const [isCheckingWage, setIsCheckingWage] = useState(false);
  const [wageResult, setWageResult] = useState<ComplianceCheckResult | null>(null);
  const [editableWage, setEditableWage] = useState<number>(currentWage);
  const [syncClients, setSyncClients] = useState<boolean>(true);
  const [isUpdatingWage, setIsUpdatingWage] = useState(false);
  const [wageError, setWageError] = useState<string | null>(null);
  const [isWageApprovalOpen, setIsWageApprovalOpen] = useState(false);

  // 加算・報酬改定チェック状態
  const [isCheckingAdditions, setIsCheckingAdditions] = useState(false);
  const [additionResult, setAdditionResult] = useState<LegalAdditionCheckResult | null>(null);
  const [editableAdditionRules, setEditableAdditionRules] = useState<(LegalAdditionItem & { selected: boolean })[]>([]);
  const [isUpdatingAdditions, setIsUpdatingAdditions] = useState(false);
  const [additionError, setAdditionError] = useState<string | null>(null);
  const [isAdditionApprovalOpen, setIsAdditionApprovalOpen] = useState(false);

  const [isRulesModalOpen, setIsRulesModalOpen] = useState(false);
  const [rules, setRules] = useState<BillingRuleRecord[]>([]);
  const [isRulesLoading, setIsRulesLoading] = useState(false);
  const [editingKey, setEditingKey] = useState<string | null>(null);
  const [editValue, setEditValue] = useState<number>(0);
  const [isSavingRule, setIsSavingRule] = useState(false);

  // ルールが「率（%）」を扱う項目かどうかを判定
  const isRateRule = (rule: { rule_key?: string; unit_label?: string; category?: string }) => {
    return rule.unit_label === '率' || rule.unit_label === '%' ||
           rule.category === 'treatment_addition' || rule.category === 'deduction' ||
           (rule.rule_key?.startsWith('treatment_') ?? false) || 
           (rule.rule_key?.startsWith('deduction_') ?? false);
  };

  // 小数率（例: 0.094 や 0.03）をパーセント表示用数値（例: 9.4 や 3.0）に変換
  const toPercentDisplay = (val: number) => {
    if (val > 0 && val <= 1) {
      return parseFloat((val * 100).toFixed(2));
    }
    return val;
  };

  // ユーザー入力・画面のパーセント数値（例: 9.4 や 3）をDB保存用の小数（0.094 や 0.03）に変換
  const toRateDb = (val: number) => {
    if (val > 1) {
      return parseFloat((val / 100).toFixed(4));
    }
    return val;
  };

  // モーダル表示時にルールを読み込む
  const loadRules = async () => {
    setIsRulesLoading(true);
    try {
      const data = await getAllBillingRules();
      setRules(data);
    } catch (e) {
      console.error(e);
    } finally {
      setIsRulesLoading(false);
    }
  };

  // 1. 最低賃金チェック実行
  const handleWageCheck = async () => {
    setIsCheckingWage(true);
    setWageError(null);
    setWageResult(null);
    try {
      const res = await checkLegalUpdates(currentAddress, currentWage);
      if (res.success && res.data) {
        setWageResult(res.data);
        if (res.data.predicted_minimum_wage) {
          setEditableWage(res.data.predicted_minimum_wage);
        }
        if (!res.data.has_minimum_wage_update) {
          toast.success("現在の最低賃金設定は最新の法規制に準拠しています。");
        }
      } else {
        setWageError(res.error || "チェックに失敗しました");
      }
    } catch (err: any) {
      setWageError(err.message || "予期せぬエラーが発生しました");
    } finally {
      setIsCheckingWage(false);
    }
  };

  // 最低賃金適用（手動修正された値 editableWage を適用 ＋ 利用者時給一括同期）
  const handleWageUpdate = async () => {
    if (!editableWage || editableWage <= 0) {
      toast.error("有効な最低賃金額を入力してください");
      return;
    }
    setIsUpdatingWage(true);
    try {
      const res = await updateComplianceSettings(editableWage, syncClients);
      if (res.success) {
        if (res.updatedClientCount && res.updatedClientCount > 0) {
          toast.success(`地域別最低賃金を ${editableWage}円 に更新し、対象利用者 ${res.updatedClientCount}名の時給を一括改定しました。`);
        } else {
          toast.success(`地域別最低賃金を ${editableWage}円 に更新しました。`);
        }
        setWageResult(null);
        router.refresh();
      } else {
        toast.error(res.error || "設定の更新に失敗しました");
      }
    } catch (err: any) {
      toast.error(err.message || "予期せぬエラーが発生しました");
    } finally {
      setIsUpdatingWage(false);
    }
  };

  // 2. 加算・報酬改定AIチェック実行
  const handleAdditionCheck = async () => {
    setIsCheckingAdditions(true);
    setAdditionError(null);
    setAdditionResult(null);
    setEditableAdditionRules([]);
    try {
      const res = await checkAdditionLegalUpdates();
      if (res.success && res.data) {
        setAdditionResult(res.data);
        if (res.data.proposed_rules && res.data.proposed_rules.length > 0) {
          setEditableAdditionRules(
            res.data.proposed_rules.map(r => {
              const isRate = isRateRule(r);
              return {
                ...r,
                // 率の項目の場合はパーセント数値（例: 9.4 や 3）および '%' 表記に正規化
                proposed_value: isRate ? toPercentDisplay(r.proposed_value) : r.proposed_value,
                unit_label: isRate ? '%' : r.unit_label,
                selected: true
              };
            })
          );
        }
        if (!res.data.has_updates) {
          toast.success("加算・減算ルールは厚生労働省の最新告示に準拠しています。");
        }
      } else {
        setAdditionError(res.error || "加算改定チェックに失敗しました");
      }
    } catch (err: any) {
      setAdditionError(err.message || "予期せぬエラーが発生しました");
    } finally {
      setIsCheckingAdditions(false);
    }
  };

  // 加算ルール差分の適用（スタッフが手動修正・選択した項目のみ適用）
  const handleAdditionUpdate = async () => {
    const targets = editableAdditionRules.filter(r => r.selected).map(r => {
      const isRate = isRateRule(r);
      return {
        ...r,
        // 保存時はDB保存仕様の小数（0.094 や 0.03）に安全変換
        proposed_value: isRate ? toRateDb(r.proposed_value) : r.proposed_value,
        unit_label: isRate ? '率' : r.unit_label
      };
    });
    if (targets.length === 0) {
      toast.error("適用対象として選択されたルールがありません");
      return;
    }
    setIsUpdatingAdditions(true);
    try {
      const res = await applyBillingRuleUpdates(targets);
      if (res.success) {
        toast.success(`${res.count} 件の加算・減算ルールをシステムに適用しました！`);
        setAdditionResult(null);
        setEditableAdditionRules([]);
        loadRules();
        router.refresh();
      } else {
        toast.error(res.error || "更新に失敗しました");
      }
    } catch (err: any) {
      toast.error(err.message || "予期せぬエラーが発生しました");
    } finally {
      setIsUpdatingAdditions(false);
    }
  };


  // 単一ルールの手動保存
  const handleSaveSingleRule = async (rule_key: string) => {
    setIsSavingRule(true);
    try {
      const targetRule = rules.find(r => r.rule_key === rule_key);
      const isRate = targetRule ? isRateRule(targetRule) : false;
      const valToSave = isRate ? toRateDb(editValue) : editValue;
      const res = await updateSingleBillingRule(rule_key, valToSave);
      if (res.success) {
        toast.success("加算ルールを更新しました");
        setEditingKey(null);
        await loadRules();
        router.refresh();
      } else {
        toast.error(res.error || "更新に失敗しました");
      }
    } catch (err: any) {
      toast.error(err.message || "予期せぬエラーが発生しました");
    } finally {
      setIsSavingRule(false);
    }
  };

  return (
    <Card className="shadow-sm border-indigo-200/70 bg-gradient-to-br from-white via-indigo-50/20 to-blue-50/30 overflow-hidden mb-6">
      <CardHeader className="p-5 pb-3 border-b border-indigo-100/80 bg-white/60">
        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
          <div className="flex items-start sm:items-center gap-2.5">
            <div className="p-2 rounded-lg bg-indigo-600 text-white shadow-sm shrink-0 mt-0.5 sm:mt-0">
              <ShieldCheck className="h-5 w-5" />
            </div>
            <div className="min-w-0">
              <CardTitle className="text-base sm:text-lg font-bold text-indigo-950 flex flex-wrap items-center gap-1.5 leading-snug">
                <span>AI法令・報酬改定コンプライアンス管理</span>
                <Badge variant="outline" className="bg-emerald-50 text-emerald-800 border-emerald-300 text-[10px] sm:text-xs font-semibold py-0.5 shrink-0">
                  安全承認（Human-in-the-Loop）仕様
                </Badge>
              </CardTitle>
              <CardDescription className="text-xs text-muted-foreground mt-1 leading-relaxed">
                厚生労働省の最新告示・最低賃金をリアルタイム照合。完全自動更新は行わず、<strong>変更点・根拠・影響範囲を提示して管理者様の承認（許可）を求めてから適用</strong>します。
              </CardDescription>
            </div>
          </div>

          <div className="flex flex-wrap items-center gap-2 shrink-0 mt-1 sm:mt-0">
            <Button
              variant="default"
              size="sm"
              onClick={() => router.push('/compliance/notebook')}
              className="w-full sm:w-auto h-9 gap-1.5 bg-indigo-600 hover:bg-indigo-700 text-white font-bold shadow-xs"
            >
              <BookOpen className="h-3.5 w-3.5" />
              内蔵NotebookLM（公式資料Q&A・PDF照合）
            </Button>

            {/* 現行ルール確認・微調整ボタン & モーダル */}
            <Button 
              variant="outline" 
              size="sm" 
              onClick={() => {
                setIsRulesModalOpen(true);
                loadRules();
              }}
              className="w-full sm:w-auto h-9 gap-1.5 border-indigo-200 text-indigo-900 hover:bg-indigo-50 font-bold"
            >
              <Sliders className="h-3.5 w-3.5 text-indigo-600" />
              現行の加算・減算ルール一覧
            </Button>
          </div>

          <Dialog open={isRulesModalOpen} onOpenChange={setIsRulesModalOpen}>
            <DialogContent className="sm:max-w-4xl max-h-[85vh] overflow-y-auto">

              <DialogHeader>
                <DialogTitle className="text-xl font-bold flex items-center gap-2">
                  <Layers className="h-5 w-5 text-indigo-600" />
                  システム稼働中の加算・減算ルールマスター
                </DialogTitle>
                <DialogDescription>
                  国保連請求計算およびシミュレーションで動的に参照されているルールマスター値です。exe化後もDBを直接更新して反映されます。
                </DialogDescription>
              </DialogHeader>

              {isRulesLoading ? (
                <div className="py-12 flex justify-center items-center gap-3 text-muted-foreground">
                  <Loader2 className="h-6 w-6 animate-spin text-indigo-600" />
                  <span>ルールマスターを読み込み中...</span>
                </div>
              ) : (
                <div className="space-y-4 pt-2">
                  <div className="overflow-x-auto rounded-md border text-sm">
                    <div className="min-w-[560px] divide-y">
                      <div className="bg-muted/50 p-3 grid grid-cols-12 font-bold text-xs text-muted-foreground uppercase">
                        <div className="col-span-5">項目名 / キー</div>
                        <div className="col-span-2">区分</div>
                        <div className="col-span-3 text-right">設定値</div>
                        <div className="col-span-2 text-center">操作</div>
                      </div>
                      {rules.map((r) => (
                        <div key={r.rule_key} className="p-3 grid grid-cols-12 items-center hover:bg-slate-50/80 transition-colors gap-2">
                          <div className="col-span-5">
                            <div className="font-semibold text-slate-900">{r.rule_name}</div>
                            <div className="text-xs text-muted-foreground font-mono">{r.rule_key}</div>
                            {r.description && <div className="text-xs text-slate-500 mt-0.5 line-clamp-1">{r.description}</div>}
                          </div>
                          <div className="col-span-2">
                            <Badge variant="secondary" className="text-xs font-normal">
                              {r.category === 'treatment_addition' ? '処遇改善' : r.category === 'deduction' ? '減算要件' : '基本加算'}
                            </Badge>
                          </div>
                          <div className="col-span-3 text-right">
                            {editingKey === r.rule_key ? (
                              <div className="flex items-center justify-end gap-1">
                                <input 
                                  type="number" 
                                  step={isRateRule(r) ? "0.1" : "any"}
                                  value={editValue} 
                                  onChange={(e) => setEditValue(parseFloat(e.target.value) || 0)}
                                  className="w-20 px-2 py-1 text-sm border rounded font-mono text-right"
                                />
                                <span className="text-xs text-muted-foreground font-semibold">{isRateRule(r) ? '%' : r.unit_label}</span>
                              </div>
                            ) : (
                              <span className="font-mono font-bold text-base text-slate-800">
                                {isRateRule(r)
                                  ? `${toPercentDisplay(r.rule_value)}%`
                                  : `${r.rule_value} ${r.unit_label}`}
                              </span>
                            )}
                          </div>
                          <div className="col-span-2 text-center">
                            {editingKey === r.rule_key ? (
                              <div className="flex items-center justify-center gap-1">
                                <Button 
                                  size="sm" 
                                  className="h-7 px-2 text-xs bg-green-600 hover:bg-green-700" 
                                  onClick={() => handleSaveSingleRule(r.rule_key)}
                                  disabled={isSavingRule}
                                >
                                  保存
                                </Button>
                                <Button 
                                  size="sm" 
                                  variant="ghost" 
                                  className="h-7 px-2 text-xs" 
                                  onClick={() => setEditingKey(null)}
                                >
                                  取消
                                </Button>
                              </div>
                            ) : (
                              <Button 
                                size="sm" 
                                variant="ghost" 
                                className="h-7 text-xs text-indigo-600 hover:text-indigo-800" 
                                onClick={() => {
                                  setEditingKey(r.rule_key);
                                  setEditValue(isRateRule(r) ? toPercentDisplay(r.rule_value) : r.rule_value);
                                }}
                              >
                                変更
                              </Button>
                            )}
                          </div>
                        </div>
                      ))}
                    </div>
                  </div>
                </div>
              )}
            </DialogContent>
          </Dialog>
        </div>
      </CardHeader>

      <CardContent className="p-3.5 sm:p-5">
        <Tabs defaultValue="additions" className="w-full">
          <TabsList className="grid grid-cols-2 mb-4 bg-indigo-100/70 p-1 rounded-xl h-auto gap-1">
            <TabsTrigger 
              value="additions" 
              className="font-bold text-xs sm:text-sm py-2 px-1.5 sm:px-3 data-[state=active]:bg-white data-[state=active]:text-indigo-900 data-[state=active]:shadow-xs rounded-lg flex items-center justify-center gap-1 sm:gap-2 text-center min-h-[40px]"
            >
              <Sparkles className="h-3.5 w-3.5 sm:h-4 sm:w-4 text-indigo-600 shrink-0" />
              <span className="sm:hidden">加算・告示改定</span>
              <span className="hidden sm:inline">加算・報酬告示 改定チェック</span>
            </TabsTrigger>
            <TabsTrigger 
              value="wage" 
              className="font-bold text-xs sm:text-sm py-2 px-1.5 sm:px-3 data-[state=active]:bg-white data-[state=active]:text-indigo-900 data-[state=active]:shadow-xs rounded-lg flex items-center justify-center gap-1 sm:gap-2 text-center min-h-[40px]"
            >
              <BookOpen className="h-3.5 w-3.5 sm:h-4 sm:w-4 text-indigo-600 shrink-0" />
              <span className="sm:hidden">最低賃金</span>
              <span className="hidden sm:inline">地域別 最低賃金チェック</span>
            </TabsTrigger>
          </TabsList>

          {/* ======================================================== */}
          {/* TAB 1: 加算・報酬改定チェック（大規模法改正対応） */}
          {/* ======================================================== */}
          <TabsContent value="additions" className="space-y-4 m-0">
            {!additionResult && !isCheckingAdditions && (
              <div className="flex flex-col sm:flex-row items-center justify-between gap-4 p-4 rounded-xl bg-indigo-50/50 border border-indigo-100">
                <div className="space-y-1">
                  <p className="font-bold text-indigo-950 text-sm">
                    処遇改善一本化新加算・各種加算・減算ルールの法改正チェック
                  </p>
                  <p className="text-xs text-indigo-700/80">
                    厚生労働省の公式報酬告示・通知と照合し、新設加算や加算率の変更があれば検知します。
                  </p>
                  <p className="text-[11px] text-muted-foreground mt-1">
                    ※ AIの推論や厚労省のWeb構造に関わらず、右上の<strong>【現行の加算・減算ルール一覧】</strong>から全ルールをいつでも直接手動入力・保存できます（AIの再照合を繰り返す必要はありません）。
                  </p>
                </div>
                <Button 
                  onClick={handleAdditionCheck} 
                  className="w-full sm:w-auto h-11 px-5 text-sm font-bold bg-indigo-600 hover:bg-indigo-700 shadow-sm shrink-0"
                >
                  <Sparkles className="mr-2 h-4 w-4 text-yellow-300" />
                  最新の報酬告示・加算をAIで照合
                </Button>
              </div>
            )}


            {isCheckingAdditions && (
              <div className="p-6 rounded-xl border border-indigo-200 bg-indigo-50/40 flex items-center gap-4 animate-pulse">
                <Loader2 className="h-6 w-6 text-indigo-600 animate-spin shrink-0" />
                <div className="space-y-1">
                  <p className="text-base font-bold text-indigo-950">
                    🤖 AIが厚生労働省の報酬改定告示・処遇改善一本化通知を精査しています...
                  </p>
                  <p className="text-xs text-indigo-700">
                    現在のシステム内全22ルールと最新の報酬基準・減算要件を照合中（所要時間: 数秒）
                  </p>
                </div>
              </div>
            )}

            {additionError && (
              <div className="p-4 bg-destructive/10 text-destructive text-sm font-bold rounded-lg border border-destructive/20">
                {additionError}
              </div>
            )}

            {/* 改定差分が検知された場合 */}
            {additionResult && additionResult.has_updates && (
              <Card className="border-amber-300 bg-amber-50/60 shadow-md relative overflow-hidden">
                <div className="absolute top-0 left-0 w-1.5 h-full bg-amber-500" />
                <CardContent className="p-5 space-y-4">
                  <div className="flex items-start gap-3">
                    <AlertTriangle className="h-6 w-6 text-amber-600 shrink-0 mt-0.5" />
                    <div className="space-y-1 w-full">
                      <div className="flex flex-wrap items-center justify-between gap-2">
                        <div className="flex flex-wrap items-center gap-1.5">
                          <h3 className="text-sm sm:text-base font-bold text-amber-950 leading-snug">
                            【報酬改定検知・承認要求】{additionResult.revision_title || '厚生労働省の加算・報酬基準に更新があります'}
                          </h3>
                          {additionResult.is_future_revision && (
                            <Badge className="bg-purple-600 text-white font-bold text-xs shrink-0">
                              🚀 次期・未来法改正検知
                            </Badge>
                          )}
                          <Badge className="bg-amber-600 text-white font-bold text-xs shrink-0">
                            ⚠️ 未承認（管理者の許可待ち）
                          </Badge>
                        </div>
                        <Badge variant="outline" className="bg-amber-100 text-amber-900 border-amber-300 font-bold shrink-0">
                          {editableAdditionRules.filter(r => r.selected).length} / {editableAdditionRules.length} 項目を選択中
                        </Badge>
                      </div>
                      <p className="text-xs text-amber-900">
                        {additionResult.summary}
                      </p>
                      <p className="text-[11px] text-amber-800 font-medium mt-1">
                        ※ <strong>完全自動更新は行われません</strong>。以下の変更点を確認・必要に応じて数値を修正し、承認（許可）してください。

                    </div>
                  </div>

                  {/* 手動調整可能なヒントバナー */}
                  <div className="p-2.5 rounded-md bg-amber-100/70 border border-amber-200 text-xs text-amber-900 flex items-center justify-between gap-2">
                    <span>
                      💡 <strong>現場スタッフ様による手動修正が可能</strong>です。AIが提示した数値に誤りがある場合は直接打ち替え、不要な項目はチェックを外してください。
                    </span>
                    <Button 
                      variant="ghost" 
                      size="sm" 
                      className="h-6 text-[11px] px-2 text-amber-950 hover:bg-amber-200"

                      {editableAdditionRules.every(r => r.selected) ? "すべて解除" : "すべて選択"}

                    <div className="min-w-[580px] divide-y divide-amber-100">
                    <div className="p-2.5 bg-amber-100/50 font-bold text-amber-900 grid grid-cols-12 items-center">
                      <div className="col-span-1 text-center">選択</div>
                      <div className="col-span-4">加算・減算項目名</div>
                      <div className="col-span-4 text-center">現在値 → 改定後（編集可）</div>
                      <div className="col-span-3">適用時期・改定理由</div>
                    </div>
                    {editableAdditionRules.map((rule, idx) => (
                      <div key={idx} className={`p-2.5 grid grid-cols-12 items-center gap-2 transition-colors ${rule.selected ? 'hover:bg-amber-50/40 bg-white' : 'bg-slate-50 opacity-60'}`}>
                        <div className="col-span-1 text-center">
                          <input 
                            type="checkbox"
                            className="h-4 w-4 rounded border-gray-300 text-amber-600 focus:ring-amber-500 cursor-pointer"
                            checked={rule.selected}
                            onChange={(e) => {
                              const updated = [...editableAdditionRules];
                              updated[idx] = { ...rule, selected: e.target.checked };
                              setEditableAdditionRules(updated);
                            }}
                          />
                        </div>
                        <div className="col-span-4 font-semibold text-slate-900">
                          {rule.rule_name}
                          <div className="text-[10px] text-muted-foreground font-mono">{rule.rule_key}</div>
                        </div>
                        <div className="col-span-4 flex items-center justify-center gap-1.5">
                          <span className="text-muted-foreground line-through text-[11px]">
                            {isRateRule(rule) ? `${toPercentDisplay(rule.current_value)}%` : `${rule.current_value}${rule.unit_label}`}
                          </span>
                          <ArrowRight className="h-3 w-3 text-amber-600 shrink-0" />
                          <div className="flex items-center gap-1">
                            <input 
                              type="number"
                              step={isRateRule(rule) ? "0.1" : "any"}
                              value={rule.proposed_value}
                              onChange={(e) => {
                                const val = parseFloat(e.target.value) || 0;
                                const updated = [...editableAdditionRules];
                                updated[idx] = { ...rule, proposed_value: val };
                                setEditableAdditionRules(updated);
                              }}
                              className="w-20 px-1.5 py-0.5 border rounded font-mono font-bold text-amber-900 text-right text-xs bg-amber-50/60 focus:bg-white"
                            />
                            <span className="text-[11px] text-muted-foreground font-semibold">
                              {isRateRule(rule) ? '%' : rule.unit_label}
                            </span>
                          </div>
                        </div>
                        <div className="col-span-3 space-y-1">
                          {rule.effective_from && (
                            <Badge variant="outline" className="text-[10px] py-0 px-1 border-amber-300 text-amber-900 bg-amber-100/50">
                              <Calendar className="mr-1 h-2.5 w-2.5" />
                              {rule.effective_from}
                            </Badge>
                          )}
                          <div className="text-[11px] text-slate-700 leading-snug">{rule.description}</div>
                          {rule.exact_mhlw_quote && (
                            <div className="mt-1 p-1.5 rounded bg-amber-100/70 border border-amber-200 text-[10px] text-amber-950 font-sans leading-tight">
                              <span className="font-bold text-amber-900 block mb-0.5">📜 厚労省告示原文:</span>
                              「{rule.exact_mhlw_quote}」
                            </div>
                          )}
                        </div>
                      </div>
                    ))}
                    </div>
                  </div>

                  {/* 厚生労働省公式ファクトチェックリンク集（多重照会リンク） */}
                  <FactCheckLinksBox 
                    links={additionResult.fact_check_links} 
                    defaultUrl={additionResult.reference_url} 
                    title="厚生労働省 報酬告示・加算ファクトチェックリンク集"
                  />

                  {/* 管理者承認・許可アクションフッター */}
                  <div className="flex flex-col sm:flex-row items-center justify-between gap-3 pt-3 border-t border-amber-200">
                    <div className="text-[11px] text-amber-900 flex items-center gap-1.5">
                      <ShieldCheck className="h-4 w-4 text-emerald-600 shrink-0" />
                      <span>
                        <strong>完全自動適用は行われません。</strong>内容を確認し、右のボタンから承認して適用してください。
                      </span>
                    </div>
                    <div className="flex items-center gap-2 w-full sm:w-auto justify-end">
                      <Button 
                        variant="outline" 
                        size="sm"
                        onClick={() => setAdditionResult(null)}
                        disabled={isUpdatingAdditions}
                        className="text-xs"
                      >
                        却下・閉じる
                      </Button>
                      <Button 
                        size="sm"
                        onClick={() => setIsAdditionApprovalOpen(true)} 
                        disabled={isUpdatingAdditions || editableAdditionRules.filter(r => r.selected).length === 0}
                        className="bg-emerald-600 hover:bg-emerald-700 text-white font-bold h-9 px-4 text-xs gap-1.5 shadow-xs"
                      >
                        <ShieldCheck className="h-4 w-4" />
                        選択項目の改定を承認・適用する
                      </Button>
                    </div>
                  </div>
                </CardContent>
              </Card>
            )}

            {/* 加算ルール 承認確認モーダル */}
            <Dialog open={isAdditionApprovalOpen} onOpenChange={setIsAdditionApprovalOpen}>
              <DialogContent className="sm:max-w-lg">
                <DialogHeader>
                  <DialogTitle className="text-base font-bold flex items-center gap-2 text-slate-900">
                    <ShieldCheck className="h-5 w-5 text-emerald-600" />
                    加算・報酬ルールの適用承認（最終確認）
                  </DialogTitle>
                  <DialogDescription className="text-xs text-muted-foreground">
                    完全自動更新は行われません。以下の選択された {editableAdditionRules.filter(r => r.selected).length} 件のルール改定をシステムマスターへ反映することを承認（許可）しますか？
                  </DialogDescription>
                </DialogHeader>

                <div className="space-y-3 py-2 text-xs">
                  <div className="max-h-[45vh] overflow-y-auto rounded-md border divide-y bg-slate-50">
                    {editableAdditionRules.filter(r => r.selected).map((r, idx) => (
                      <div key={idx} className="p-2.5 flex items-center justify-between gap-2">
                        <div>
                          <div className="font-semibold text-slate-900">{r.rule_name}</div>
                          <div className="text-[10px] text-muted-foreground">{r.effective_from ? `適用: ${r.effective_from}` : ''}</div>
                          {r.exact_mhlw_quote && (
                            <div className="text-[10px] text-amber-900 bg-amber-100/60 rounded px-1.5 py-0.5 mt-0.5 font-sans">
                              📜 告示原文: 「{r.exact_mhlw_quote}」
                            </div>
                          )}
                        </div>
                        <div className="font-mono font-bold text-slate-800 text-right shrink-0">
                          <span className="text-slate-400 line-through mr-1.5 text-[11px]">
                            {isRateRule(r) ? `${toPercentDisplay(r.current_value)}%` : `${r.current_value}${r.unit_label}`}
                          </span>
                          → <span className="text-emerald-700 text-xs font-bold">
                            {isRateRule(r) ? `${toPercentDisplay(r.proposed_value)}%` : `${r.proposed_value}${r.unit_label}`}
                          </span>
                        </div>
                      </div>
                    ))}
                  </div>

                  <div className="p-2.5 rounded bg-amber-50/70 border border-amber-200 text-[11px] text-amber-900 flex items-start gap-2">
                    <AlertTriangle className="h-4 w-4 text-amber-600 shrink-0 mt-0.5" />
                    <span>
                      承認ボタンを押すと、システム内の加算・減算ルールマスターが更新され、今後の請求計算およびシミュレーションに即時反映されます。
                    </span>
                  </div>
                </div>

                <DialogFooter className="flex sm:justify-between gap-2 pt-2">
                  <Button 
                    variant="outline" 
                    size="sm" 
                    onClick={() => setIsAdditionApprovalOpen(false)}
                    disabled={isUpdatingAdditions}
                    className="text-xs"
                  >
                    キャンセル（保留）
                  </Button>
                  <Button 
                    size="sm" 
                    onClick={() => {
                      setIsAdditionApprovalOpen(false);
                      handleAdditionUpdate();
                    }}
                    disabled={isUpdatingAdditions}
                    className="bg-emerald-600 hover:bg-emerald-700 text-white font-bold text-xs gap-1.5"
                  >
                    {isUpdatingAdditions ? (
                      <>
                        <Loader2 className="h-4 w-4 animate-spin" />
                        適用中...
                      </>
                    ) : (
                      <>
                        <Check className="h-4 w-4" />
                        承認してシステムに適用
                      </>
                    )}
                  </Button>
                </DialogFooter>
              </DialogContent>
            </Dialog>

            {/* 改定差分がない（最新準拠）場合 */}
            {additionResult && !additionResult.has_updates && (
              <div className="p-4 rounded-xl border border-green-200 bg-green-50/50 space-y-3">
                <div className="flex items-center justify-between gap-4">
                  <div className="flex items-center gap-3">
                    <CheckCircle className="h-6 w-6 text-green-600 shrink-0" />
                    <div>
                      <h4 className="text-sm font-bold text-green-950">厚生労働省の最新報酬告示に完全準拠しています</h4>
                      <p className="text-xs text-green-800 mt-0.5">
                        {additionResult.summary || "処遇改善新加算Ⅰ〜Ⅳ、各種加算、減算要件はすべて最新の法定基準を満たしています。"}
                      </p>
                    </div>
                  </div>
                  <Button variant="outline" size="sm" onClick={() => setAdditionResult(null)}>
                    閉じる
                  </Button>
                </div>
                <FactCheckLinksBox 
                  links={additionResult.fact_check_links} 
                  defaultUrl={additionResult.reference_url} 
                  title="厚生労働省 報酬告示・加算ファクトチェックリンク集"
                />
              </div>
            )}

          </TabsContent>

          {/* ======================================================== */}
          {/* TAB 2: 最低賃金チェック */}
          {/* ======================================================== */}
          <TabsContent value="wage" className="space-y-4 m-0">
            {!wageResult && !isCheckingWage && (
              <div className="flex flex-col sm:flex-row items-center justify-between gap-4 p-4 rounded-xl bg-blue-50/50 border border-blue-100">
                <div className="space-y-1">
                  <div className="flex items-center gap-2">
                    <span className="text-xs text-muted-foreground font-medium">現在の事業所設定:</span>
                    <Badge variant="outline" className="bg-white font-mono text-xs font-bold">
                      {currentWage} 円/時
                    </Badge>
                    <span className="text-xs text-muted-foreground font-medium">（{currentAddress || '所在地未設定'}）</span>
                  </div>
                  <p className="text-xs text-blue-700/80">
                    事業所所在地の最新地域別最低賃金（厚労省告示）と即時照合します。
                  </p>
                  <p className="text-[11px] text-muted-foreground mt-1">
                    ※ AI照合を使わず直接設定したい場合は、画面下の事業所基本情報フォームからいつでもキーボード入力で保存できます。
                  </p>
                </div>

                <Button 
                  onClick={handleWageCheck} 
                  className="w-full sm:w-auto h-11 px-5 text-sm font-bold bg-blue-600 hover:bg-blue-700 shadow-sm shrink-0"
                >
                  <Sparkles className="mr-2 h-4 w-4 text-yellow-300" />
                  最低賃金をAI照合
                </Button>
              </div>
            )}

            {isCheckingWage && (
              <div className="p-6 rounded-xl border border-blue-200 bg-blue-50/40 flex items-center gap-4 animate-pulse">
                <Loader2 className="h-6 w-6 text-blue-600 animate-spin shrink-0" />
                <span className="text-sm font-medium text-blue-950">
                  🤖 厚生労働省の最新地域別最低賃金マスターと照会しています...
                </span>
              </div>
            )}

            {wageError && (
              <div className="p-4 bg-destructive/10 text-destructive text-sm font-bold rounded-lg border border-destructive/20">
                {wageError}
              </div>
            )}

            {wageResult && wageResult.has_minimum_wage_update && (
              <Card className="border-amber-300 bg-amber-50/60 shadow-md relative overflow-hidden">
                <div className="absolute top-0 left-0 w-1.5 h-full bg-amber-500" />
                <CardContent className="p-5 space-y-4">
                  <div className="flex items-start justify-between gap-3">
                    <div className="flex items-start gap-3">
                      <AlertTriangle className="h-6 w-6 text-amber-600 shrink-0 mt-0.5" />
                      <div className="space-y-1">
                        <div className="flex flex-wrap items-center gap-1.5">
                          <h3 className="text-sm sm:text-base font-bold text-amber-950 leading-snug">
                            【法改正検知・承認要求】地域別最低賃金の改定が公表されています
                          </h3>
                          <Badge className="bg-amber-600 text-white font-bold text-xs shrink-0">
                            ⚠️ 未承認（管理者の許可待ち）
                          </Badge>
                        </div>
                        <p className="text-xs text-amber-900">
                          {currentAddress} の最低賃金が改定されました。<strong>完全自動更新は行われません</strong>。以下の変更点を確認し、承認（許可）してください。
                        </p>
                      </div>
                    </div>
                  </div>

                  {/* 変更点のビジュアル対比カード */}
                  <div className="rounded-xl border border-amber-200 bg-white p-4 shadow-xs space-y-3">
                    <div className="text-xs font-bold text-slate-800 flex items-center justify-between border-b pb-2">
                      <span className="flex items-center gap-1.5">
                        <Sliders className="h-4 w-4 text-amber-600" />
                        最低賃金改定に伴う変更点サマリー
                      </span>
                      {wageResult.effective_date && (
                        <Badge variant="outline" className="bg-amber-50 text-amber-800 border-amber-300 text-[11px] font-bold">
                          <Calendar className="mr-1 h-3 w-3" />
                          改定発効日: {wageResult.effective_date}
                        </Badge>
                      )}
                    </div>

                    <div className="grid grid-cols-1 sm:grid-cols-3 gap-3 pt-1">
                      {/* 変更前 */}
                      <div className="p-3 rounded-lg bg-slate-50 border border-slate-200">
                        <span className="text-[11px] font-medium text-slate-500 block">現在の事業所設定</span>
                        <div className="mt-1 flex items-baseline gap-1">
                          <span className="text-2xl font-bold font-mono text-slate-700">{currentWage}</span>
                          <span className="text-xs text-slate-500">円/時</span>
                        </div>
                        <span className="text-[10px] text-slate-400 mt-1 block">（現行DB登録値）</span>
                      </div>

                      {/* 変更後（公表値・手動編集可） */}
                      <div className="p-3 rounded-lg bg-amber-50/70 border border-amber-300 ring-1 ring-amber-300">
                        <div className="flex items-center justify-between">
                          <span className="text-[11px] font-bold text-amber-900 block">厚生労働省公表値</span>
                          <span className="text-[10px] text-amber-700 bg-amber-100 px-1.5 py-0.5 rounded font-medium">手動修正可</span>
                        </div>
                        <div className="mt-1 flex items-center gap-1.5">
                          <input 
                            type="number"
                            value={editableWage}
                            onChange={(e) => setEditableWage(parseInt(e.target.value) || 0)}
                            className="w-24 px-2 py-0.5 border rounded text-2xl font-bold font-mono text-amber-800 bg-white focus:ring-2 focus:ring-amber-500"
                          />
                          <span className="text-xs font-bold text-amber-900">円/時</span>
                        </div>
                        <div className="mt-1 flex items-center gap-1">
                          {editableWage > currentWage ? (
                            <span className="text-[11px] font-bold text-emerald-600">
                              +{editableWage - currentWage}円 引き上げ
                            </span>
                          ) : editableWage === currentWage ? (
                            <span className="text-[11px] text-slate-500">同額</span>
                          ) : (
                            <span className="text-[11px] font-bold text-red-600">
                              {editableWage - currentWage}円 不足
                            </span>
                          )}
                        </div>
                      </div>

                      {/* 影響範囲（対象人数） */}
                      <div className="p-3 rounded-lg bg-blue-50/60 border border-blue-200">
                        <span className="text-[11px] font-medium text-blue-900 block">影響を受ける対象利用者</span>
                        <div className="mt-1 flex items-baseline gap-1">
                          <span className="text-2xl font-bold font-mono text-blue-800">
                            {wageResult.affected_client_count ?? 0}
                          </span>
                          <span className="text-xs text-blue-800">名</span>
                        </div>
                        <span className="text-[10px] text-blue-600 mt-1 block">
                          （時給が新基準未満の利用者）
                        </span>
                      </div>
                    </div>
                  </div>

                  {/* 理由・詳細 */}
                  <div className="p-3 rounded-lg bg-amber-100/40 border border-amber-200/80 text-xs text-amber-950">
                    <p className="font-semibold mb-1">【改定の背景・法令根拠】</p>
                    <p className="leading-relaxed">{wageResult.reasoning}</p>
                  </div>

                  {/* 厚生労働省公式ファクトチェックリンク集 */}
                  <FactCheckLinksBox 
                    links={wageResult.fact_check_links} 
                    defaultUrl={wageResult.reference_url} 
                    title="厚生労働省 最低賃金ファクトチェック（多重照会リンク）"
                  />

                  {/* 最低賃金割れ利用者の時給一括更新チェックボックス */}
                  <div className="bg-white p-3 rounded-lg border border-amber-300 flex items-center justify-between">
                    <label className="flex items-start sm:items-center gap-2.5 text-xs font-bold text-amber-950 cursor-pointer select-none">
                      <input 
                        type="checkbox"
                        checked={syncClients}
                        onChange={(e) => setSyncClients(e.target.checked)}
                        className="h-4 w-4 mt-0.5 sm:mt-0 accent-amber-600 rounded cursor-pointer shrink-0"
                      />
                      <span className="leading-relaxed">
                        【労基法違反防止】事業所設定と同時に、時給が {editableWage}円 未満の全利用者の時給も {editableWage}円 に一括改定する（推奨）
                      </span>
                    </label>
                  </div>

                  {/* 管理者承認・許可アクションフッター */}
                  <div className="flex flex-col sm:flex-row items-center justify-between gap-3 pt-3 border-t border-amber-200">
                    <div className="text-[11px] text-amber-900 flex items-center gap-1.5">
                      <ShieldCheck className="h-4 w-4 text-emerald-600 shrink-0" />
                      <span>
                        <strong>完全自動適用は行われません。</strong>内容を確認し、右のボタンから承認して適用してください。
                      </span>
                    </div>
                    <div className="flex items-center gap-2 w-full sm:w-auto justify-end">
                      <Button 
                        variant="outline" 
                        size="sm"
                        onClick={() => setWageResult(null)}
                        disabled={isUpdatingWage}
                        className="text-xs"
                      >
                        却下・保留（適用しない）
                      </Button>
                      <Button 
                        size="sm"
                        onClick={() => setIsWageApprovalOpen(true)} 
                        disabled={isUpdatingWage}
                        className="bg-emerald-600 hover:bg-emerald-700 text-white font-bold h-9 px-4 text-xs gap-1.5 shadow-xs"
                      >
                        <ShieldCheck className="h-4 w-4" />
                        変更内容を確認して承認・適用する
                      </Button>
                    </div>
                  </div>
                </CardContent>
              </Card>
            )}

            {/* 最低賃金 承認確認モーダル */}
            <Dialog open={isWageApprovalOpen} onOpenChange={setIsWageApprovalOpen}>
              <DialogContent className="sm:max-w-md">
                <DialogHeader>
                  <DialogTitle className="text-base font-bold flex items-center gap-2 text-slate-900">
                    <ShieldCheck className="h-5 w-5 text-emerald-600" />

                />
              </div>
            )}

          </TabsContent>

      </CardContent>
    </Card>
  );
}

  if (!isAdmin && !isServiceManager) {
    return null;

  const supportAlertCount = 
    expiringSupportPlans.length + 
    monitoringAlerts.length + 
    expiringClients.length + 
    uninterviewedClients.length + 
    unevaluatedOffsiteClients.length;

  // 2. 減算・法令順守（管理者＋サビ管の一部）
  const legalAlertCount = 
    complianceIssues.length + 
    urgentComplianceCount + 
    (monthlyStaffing && !monthlyStaffing.isCompliant ? 1 : 0);

  // 3. 経営・労務・契約（管理者専用）
  const managementAlertCount = 
    (deficitAlert ? 1 : 0) + 
    (offsiteAlert ? 1 : 0) + 
    expiringContractors.length + 
    expiringClientContracts.length + 
    expiringPtoRecords.length +
    mandatoryPtoAlerts.length;

    (monitoringAlerts.filter(m => m.isOverdue).length) +

    (mandatoryPtoAlerts.filter((m: any) => m.isCritical).length);

  // 通知がゼロ件ならスッキリ完了カードを表示
  if (totalCount === 0) {
    return (
      <div className="flex flex-col sm:flex-row sm:items-center justify-between p-3.5 bg-indigo-50/80 border border-indigo-200 rounded-lg text-indigo-950 text-xs shadow-2xs gap-2.5">

'use server';

import { db, resolveServiceModeForContractor } from "@/lib/db";
import { getRole } from "@/lib/auth";
import { revalidatePath } from "next/cache";
import { v4 as uuidv4 } from "uuid";
import { contractorSchema, getFirstZodErrorMessage } from "@/lib/validations";

export async function getContractors() {
  try {
    const role = await getRole();
    if (!role) return [];
    return db.prepare('SELECT * FROM contractor_agreements WHERE deleted_at IS NULL ORDER BY contract_end_date ASC').all();
  } catch (error) {
    console.error("Failed to get contractors:", error);
    return [];
  }
}

export async function getContractor(id: string) {
  try {
    const role = await getRole();
    if (!role) return null;
    await resolveServiceModeForContractor(id);
    return db.prepare('SELECT * FROM contractor_agreements WHERE id = ? AND deleted_at IS NULL').get(id);
  } catch (error) {
    console.error("Failed to get contractor:", error);
    return null;
  }
}

export async function saveContractor(data: any) {
  try {
    const role = await getRole();
    if (!role) return { success: false, error: '認証されていません。ログインしてください。' };

    const parseResult = contractorSchema.safeParse(data);
    if (!parseResult.success) {
      return { success: false, error: getFirstZodErrorMessage(parseResult.error) };
    }
    const parsedData = parseResult.data;

    const id = parsedData.id && parsedData.id !== "new" ? parsedData.id : uuidv4();
    
    if (parsedData.id && parsedData.id !== "new") {
      await resolveServiceModeForContractor(id);
      db.prepare(`
        UPDATE contractor_agreements SET
          company_name = ?, work_description = ?, contract_start_date = ?, contract_end_date = ?, contract_amount = ?, 
          unit_price = ?, expected_monthly_revenue = ?, assigned_clients_count = ?, notes = ?
        WHERE id = ?
      `).run(
        parsedData.company_name, parsedData.work_description, parsedData.contract_start_date, parsedData.contract_end_date, parsedData.contract_amount,
        parsedData.unit_price, parsedData.expected_monthly_revenue, parsedData.assigned_clients_count, parsedData.notes, id
      );
    } else {
      db.prepare(`
        INSERT INTO contractor_agreements (id, company_name, work_description, contract_start_date, contract_end_date, contract_amount, unit_price, expected_monthly_revenue, assigned_clients_count, notes)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
      `).run(
        id, parsedData.company_name, parsedData.work_description, parsedData.contract_start_date, parsedData.contract_end_date, parsedData.contract_amount,
        parsedData.unit_price, parsedData.expected_monthly_revenue, parsedData.assigned_clients_count, parsedData.notes
      );
    }

    try {
      revalidatePath('/contractors');
      revalidatePath('/');
    } catch (e) {}
    return { success: true, id };
  } catch (error: any) {
    console.error("Failed to save contractor:", error);
    return { success: false, error: error.message || '受託先の保存に失敗しました' };
  }
}

export async function deleteContractor(id: string) {
  try {
    const role = await getRole();
    if (role !== 'admin') return { success: false, error: '管理者権限が必要です' };

    await resolveServiceModeForContractor(id);
    db.prepare('UPDATE contractor_agreements SET deleted_at = CURRENT_TIMESTAMP WHERE id = ?').run(id);
    try {
      revalidatePath('/contractors');
      revalidatePath('/');
    } catch (e) {}
    return { success: true };
  } catch (error: any) {
    console.error("Failed to delete contractor:", error);
    return { success: false, error: error.message || '受託先の削除に失敗しました' };
  }
}

'use server'

import { revalidatePath } from 'next/cache'
import { db, sanitizeForeignKeyId } from '@/lib/db'
import { v4 as uuidv4 } from 'uuid'
import { getRole } from '@/lib/auth'
import { dailyLogSchema, getFirstZodErrorMessage } from '@/lib/validations'
import { getActiveDepartments } from '@/lib/location'

export async function saveDailyLog(formData: FormData) {
  try {
    const role = await getRole()
    if (!role) return { success: false, error: '認証されていません。ログインしてください。' }

    const parseResult = dailyLogSchema.safeParse({
      recordDate: formData.get('recordDate'),
      departmentId: formData.get('departmentId') as string || undefined,
      weather: formData.get('weather') as string || '',
      staffOnDuty: formData.get('staffOnDuty') as string || '',
      activitiesSummary: formData.get('activitiesSummary') as string || '',
      notes: formData.get('notes') as string || '',
    })

    if (!parseResult.success) {
      return { success: false, error: getFirstZodErrorMessage(parseResult.error) }
    }
    const parsedData = parseResult.data

    const activeDepartments = getActiveDepartments();
    const rawDepartmentId = parsedData.departmentId || activeDepartments[0]?.id || null;
    const departmentId = sanitizeForeignKeyId('departments', rawDepartmentId) || activeDepartments[0]?.id || null;

    // Calculate total present clients for this department automatically
    const presentCountRow = db.prepare(`
      SELECT COUNT(*) as cnt 
      FROM attendance_records a
      JOIN clients c ON a.client_id = c.id
      WHERE a.record_date = ? AND a.status = 'present' AND a.deleted_at IS NULL
        AND (c.department_id = ? OR (? IS NULL AND c.department_id IS NULL))
    `).get(parsedData.recordDate, departmentId, departmentId) as any;

    const totalPresent = presentCountRow ? presentCountRow.cnt : 0;

    const stmt = db.prepare(`
      INSERT INTO daily_logs (
        id, record_date, department_id, weather, total_present, staff_on_duty, activities_summary, notes
      ) VALUES (
        ?, ?, ?, ?, ?, ?, ?, ?
      )
      ON CONFLICT(record_date, department_id) DO UPDATE SET
        weather = excluded.weather,
        total_present = excluded.total_present,
        staff_on_duty = excluded.staff_on_duty,
        activities_summary = excluded.activities_summary,
        notes = excluded.notes,
        updated_at = CURRENT_TIMESTAMP
    `);

    stmt.run(
      uuidv4(),
      parsedData.recordDate,
      departmentId,
      parsedData.weather,
      totalPresent,
      parsedData.staffOnDuty,
      parsedData.activitiesSummary,
      parsedData.notes
    );

    try {
      revalidatePath('/daily-logs')
    } catch (e) {}
    return { success: true, recordDate: parsedData.recordDate, departmentId }
  } catch (error) {
    console.error('業務日誌保存エラー:', error)
    return { success: false, error: '業務日誌の保存に失敗しました' }
  }
}

'use server'

import { redirect } from 'next/navigation'
import { revalidatePath } from 'next/cache'
import { setRole, logout as clearCookie } from '@/lib/auth'
import { setServiceMode, getServiceMode, ServiceMode } from '@/lib/serviceModeServer'

function parseServiceMode(formData?: FormData): ServiceMode | undefined {
  if (!formData) return undefined;
  const mode = formData.get('serviceMode') as string;
  if (mode === 'type_a' || mode === 'type_b' || mode === 'gh' || mode === 'houday' || mode === 'corporate') {
    return mode;
  }
  return undefined;
}

function getRedirectUrl(mode?: ServiceMode): string {
  switch (mode) {
    case 'type_a':
      return '/';
    case 'type_b':
      return '/';
    case 'gh':
      return '/gh/dashboard';
    case 'houday':
      return '/houday/dashboard';
    case 'corporate':
      return '/corporate/dashboard';
    default:
      return '/';
  }
}

export async function loginAsAdmin(formData?: FormData) {
  const mode = parseServiceMode(formData) || 'type_a';
  await setServiceMode(mode);
  await setRole('admin');
  revalidatePath('/', 'layout');
  redirect(getRedirectUrl(mode));
}

export async function loginAsAdminToSettings(formData?: FormData) {
  const mode = parseServiceMode(formData) || 'type_a';
  await setServiceMode(mode);
  await setRole('admin');
  revalidatePath('/', 'layout');
  redirect('/settings');
}

export async function loginAsServiceManager(formData?: FormData) {
  const mode = parseServiceMode(formData) || 'type_a';
  await setServiceMode(mode);
  await setRole('service_manager');
  revalidatePath('/', 'layout');
  redirect(getRedirectUrl(mode));
}

export async function loginAsStaff(formData?: FormData) {
  const mode = parseServiceMode(formData) || 'type_a';
  await setServiceMode(mode);
  await setRole('staff');
  revalidatePath('/', 'layout');
  redirect(getRedirectUrl(mode));
}

export async function switchServiceMode(targetMode: ServiceMode, returnUrl?: string) {
  await setServiceMode(targetMode);
  revalidatePath('/', 'layout');
  const target = returnUrl || getRedirectUrl(targetMode);
  redirect(target);
}

export async function logout() {
  await clearCookie();
  revalidatePath('/', 'layout');
  redirect('/login');
}

        ?, ?, ?, ?,
        ?, ?, ?, ?,
        ?, ?, ?,
        ?, ?, ?,
        ?, ?, ?,
        ?, ?, ?, ?,
        CURRENT_TIMESTAMP, CURRENT_TIMESTAMP
      )
    `);

    const updateStmt = db.prepare(`
      UPDATE offsite_evaluations SET
        workplace_id = ?,
        contractor_id = ?,
        work_tasks = ?,

        work_performance_score = ?,
        interpersonal_score = ?,
        safety_score = ?,
        readiness_score = ?,

        challenges_and_support = ?,

        evaluator_name = ?,
        supervisor_name = ?,

    `);

    const checkStmt = db.prepare(`
      SELECT id FROM offsite_evaluations
      WHERE client_id = ? AND evaluation_date = ?
    `);

    const smRow = db.prepare(`SELECT name, job_category FROM staff WHERE (role IN ('service_manager', 'admin_and_service_manager') OR job_category LIKE '%サービス管理責任者%' OR job_category LIKE '%サビ管%') AND deleted_at IS NULL ORDER BY created_at ASC LIMIT 1`).get() as any;

          );
        } else {

            safeClientId,
            safeWorkplaceId,
            safeContractorId,
            item.evaluationDate,
            item.periodStart,
            item.periodEnd,
            item.workTasks || '施設外就労作業',

           st.name as staff_name
    FROM attendance_records a
    JOIN clients c ON a.client_id = c.id
    LEFT JOIN offsite_workplaces w ON a.offsite_workplace_id = w.id
    LEFT JOIN staff st ON a.offsite_staff_id = st.id
    WHERE a.record_date LIKE ? AND a.status = 'present' AND a.is_offsite = 1 AND a.deleted_at IS NULL

  const dailyCounts: Record<string, number> = {};

  const capacity = parseCapacityFromClass(facility.capacity_class);

    capacity,
    maxDailyCount,
    ratio,
    isCompliant: ratio <= 70
  };

    clientName: string;
    recipientNumber?: string;

              if (outM < inM) outM += 24 * 60;
              dayPtoMinutes = outM - inM;
              // 休憩時間の控除（労基法第34条: 6時間超45分、8時間超60分）
              if (dayPtoMinutes > 480) {
                dayPtoMinutes -= 60;
              } else if (dayPtoMinutes > 360) {
                dayPtoMinutes -= 45;
              }
            } else if (client.weekly_scheduled_hours && client.weekly_scheduled_hours > 0) {
              const holidaysCount = regularHolidays.length > 0 ? regularHolidays.length : 2;
              const scheduledDaysPerWeek = Math.max(1, Math.min(6, 7 - holidaysCount));
              dayPtoMinutes = Math.round((client.weekly_scheduled_hours / scheduledDaysPerWeek) * 60);
            } else {
              dayPtoMinutes = 240; // デフォルト4時間
            }
            ptoMinutes += dayPtoMinutes;
          } else if (record.status === 'present') {
            workingDays++;
            if (record.clock_in && record.clock_out) {
              const [inH, inM] = record.clock_in.split(':').map(Number);
              const [outH, outM] = record.clock_out.split(':').map(Number);
              
              let outTotalM = outH * 60 + outM;
              const inTotalM = inH * 60 + inM;
              if (outTotalM < inTotalM) {
                outTotalM += 24 * 60; // 日またぎ
              }
              
              let breakM = record.break_time_minutes || 0;
              let rawMins = outTotalM - inTotalM;
              // 休憩時間の控除（労基法第24条・全額払いの原則の遵守）
              // 実績打刻を優先。事業所で自動控除が明示的に設定されている場合のみ自動控除、未設定時は実打刻に従う
              if (breakM === 0 && facility.auto_deduct_break_time === 1) {
                if (rawMins > 480) breakM = 60;
                else if (rawMins > 360) breakM = 45;
              }
              const mins = Math.max(0, rawMins - breakM);
              if (mins > 0) {
                totalMinutes += mins;
                
                const dObj = new Date(record.record_date);
                const dayOfWeekStr = dObj.getDay().toString();
                const isLegalHoliday = dayOfWeekStr === legalHoliday;

                if (isLegalHoliday) {
                  // 労基法第35条・第37条: 法定休日労働（35%割増、週40時間枠外）
                  holidayWorkingMinutes += mins;
                } else {
                  // 平日または法定外休日労働: 1日8時間超の時間外労働および週40時間枠への積算
                  if (mins > 480) {
                    dailyOvertimeMinutes += (mins - 480);
                    const regularPart = 480;
                    const weekKey = getWeekStart(record.record_date);
                    weeklyRegularMinutes.set(weekKey, (weeklyRegularMinutes.get(weekKey) || 0) + regularPart);
                  } else {
                    const weekKey = getWeekStart(record.record_date);
                    weeklyRegularMinutes.set(weekKey, (weeklyRegularMinutes.get(weekKey) || 0) + mins);
                  }
                }

                // 深夜労働計算 (22:00 - 05:00)
                let rawNightMins = 0;
                for (let m = inTotalM; m < outTotalM; m++) {
                   const hourOfDay = Math.floor(m / 60) % 24;
                   if (hourOfDay >= 22 || hourOfDay < 5) {
                     rawNightMins++;
                   }
                }
                // 深夜帯の休憩控除考慮: 日中拘束時間で休憩が消化されたとみなし、日中帯で不足する場合のみ深夜から控除
                const totalSpan = outTotalM - inTotalM;
                const daySpan = totalSpan - rawNightMins;
                const remainingBreakToDeductFromNight = Math.max(0, breakM - daySpan);
                const netNightMins = Math.max(0, rawNightMins - remainingBreakToDeductFromNight);
                nightShiftMinutes += netNightMins;
              }
            }
          }
        }

        // 週40時間を超える法定時間外労働の集計
        let weeklyOvertimeMinutes = 0;
        for (const mins of weeklyRegularMinutes.values()) {
          if (mins > 2400) { // 40時間 * 60分
            weeklyOvertimeMinutes += (mins - 2400);
          }
        }
        const totalOvertimeMinutes = dailyOvertimeMinutes + weeklyOvertimeMinutes;

        const hourlyWage = client.hourly_wage || 0;
        // 労基法第24条・通達昭63.3.14基発150号準拠: 50銭未満切捨、50銭以上切上（四捨五入）
        // 浮動小数点誤差防止のため、分×時給を先に計算してから60で割る
        const baseSalary = Math.round((totalMinutes * hourlyWage) / 60);
        const ptoAllowance = Math.round((ptoMinutes * hourlyWage) / 60);

        // 法定端数処理（通達昭63.3.14基発150号準拠: 1ヶ月の総時間外合計に対して30分未満切捨、30分以上切上）
        // 合計を丸めた上で、60時間以内と60時間超に按分する（二重切捨てによる未払い防止）
        const totalOvertimeHours = roundStatutoryHours(totalOvertimeMinutes);
        const normalOvertimeHours = Math.min(totalOvertimeHours, 60);
        const excessOvertimeHours = Math.max(0, totalOvertimeHours - 60);
        const normalOvertimeAllowance = Math.round(normalOvertimeHours * hourlyWage * overtimeRate);
        const excessOvertimeAllowance = Math.round(excessOvertimeHours * hourlyWage * overtimeOver60Rate);
        const overtimeAllowance = normalOvertimeAllowance + excessOvertimeAllowance;
        const overtimeExcessMinutes = Math.max(0, totalOvertimeMinutes - 3600);

        const roundedNightHours = roundStatutoryHours(nightShiftMinutes);
        const nightShiftAllowance = Math.round(roundedNightHours * hourlyWage * nightRate);

        const roundedHolidayHours = roundStatutoryHours(holidayWorkingMinutes);
        const holidayAllowance = Math.round(roundedHolidayHours * hourlyWage * holidayRate);
        
        const totalPayment = baseSalary + ptoAllowance + overtimeAllowance + nightShiftAllowance + holidayAllowance;

        if (totalMinutes > 0 || ptoMinutes > 0 || totalPayment > 0) {
          runInsert.run(
            randomUUID(),
            client.id,
            targetMonth,
            totalMinutes,
            workingDays,
            ptoDays,
            ptoAllowance,
            hourlyWage,
            baseSalary,
            totalPayment,
            totalOvertimeMinutes,
            nightShiftMinutes,
            holidayWorkingMinutes,
            overtimeAllowance,
            nightShiftAllowance,
            holidayAllowance,
            overtimeExcessMinutes,
            excessOvertimeAllowance
          );
          totalWagesPaid += totalPayment;
        }
      }

      // 連動して production_activities の total_wages_paid も更新する
      db.prepare(`
        INSERT INTO production_activities (
          id, target_month, total_wages_paid
        ) VALUES (
          ?, ?, ?
        ) ON CONFLICT(target_month) DO UPDATE SET
          total_wages_paid = excluded.total_wages_paid,
          updated_at = CURRENT_TIMESTAMP
      `).run(randomUUID(), targetMonth, totalWagesPaid);

    })();

    try {
      revalidatePath('/payroll');
      revalidatePath('/billing/production'); // 収支データも更新されるため
    } catch (e) {}
    
    return { success: true, count: clients.length, totalWagesPaid };
  } catch (e: any) {
    console.error("Failed to calculate payroll for month:", e);
    return { success: false, error: e.message || "給与一括計算に失敗しました", count: 0, totalWagesPaid: 0 };
  }
}

export async function getPayslipsForMonth(targetMonth: string) {
  try {
    const role = await getRole();
    if (!role) return [];

    return db.prepare(`
      SELECT p.*, c.full_name 
      FROM monthly_payslips p
      JOIN clients c ON p.client_id = c.id
      WHERE p.target_month = ?
      ORDER BY c.full_name ASC
    `).all(targetMonth) as any[];
  } catch (e) {
    console.error("Failed to get payslips for month:", e);
    return [];
  }
}

import { redirect } from "next/navigation";

export default async function PrintPayslipPage({ 
  searchParams 
}: { 
  searchParams: Promise<{ month?: string; clientId?: string }> 
}) {
  const resolvedParams = await searchParams;
  const { month, clientId } = resolvedParams;

  if (!month) {
    redirect("/billing/payroll");
  }

  // 法定必須全項目（基本給・有休手当・割増・各種社保控除・税額等）が完全に揃った正式な給与明細ページへリダイレクト
  const redirectUrl = clientId 
    ? `/billing/payroll/payslips?month=${month}&clientId=${clientId}`
    : `/billing/payroll/payslips?month=${month}`;

  redirect(redirectUrl);
}
