'use client';

import { useState, useTransition } from "react";
import { useRouter } from "next/navigation";
import { Card, CardContent, CardHeader, CardTitle, CardDescription } from "@/components/ui/card";
import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from "@/components/ui/table";
import { Badge } from "@/components/ui/badge";
import { Button, buttonVariants } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Textarea } from "@/components/ui/textarea";
import { 
  Clock, 
  CheckCircle2, 
  AlertCircle, 
  Edit3, 
  PlusCircle, 
  Calendar as CalendarIcon,
  X,
  Save,
  Loader2,
  CalendarDays
} from "lucide-react";
import Link from "next/link";
import { PrintButton } from "@/components/PrintButton";
import { updateSupportRecordInline } from "../actions";

export interface StaffMember {
  id: string;
  full_name: string;
  job_category?: string;
  role?: string;
}

export interface LedgerRecord {
  id: string;
  client_id: string;
  client_full_name: string;
  client_status: string;
  record_date: string;
  service_start_time: string | null;
  service_end_time: string | null;
  service_location: string | null;
  vitals: string | null;
  task_content: string | null;
  staff_comment: string | null;
  staff_id: string | null;
  staff_full_name: string | null;
}

export interface MissingClient {
  client_id: string;
  client_full_name: string;
  client_status: string;
  record_date: string;
  attendance_status: string;
  clock_in: string | null;
  clock_out: string | null;
  is_absence_addition: number;
}

interface ModalTarget {
  isNew: boolean;
  clientId: string;
  clientName: string;
  recordDate: string;
  serviceStartTime: string;
  serviceEndTime: string;
  serviceLocation: string;
  temperature: string;
  bloodPressure: string;
  sleep: string;
  mood: string;
  taskContent: string;
  staffComment: string;
  staffId: string;
}

export function SupportLedgerClient({
  records,
  missingClients,
  staffMembers,
  selectedMonth,
  selectedDate,
  selectedClientId,
  selectedClientName,
}: {
  records: LedgerRecord[];
  missingClients: MissingClient[];
  staffMembers: StaffMember[];
  selectedMonth: string;
  selectedDate: string | null;
  selectedClientId: string;
  selectedClientName: string;
}) {
  const router = useRouter();
  const [modalTarget, setModalTarget] = useState<ModalTarget | null>(null);
  const [errorMsg, setErrorMsg] = useState<string | null>(null);
  const [isSaving, startSavingTransition] = useTransition();

  // 既存レコードの編集モーダルを開く
  const handleOpenEdit = (record: LedgerRecord) => {
    let parsedVitals: any = {};
    if (record.vitals) {
      try {
        if (record.vitals.startsWith('{')) {
          parsedVitals = JSON.parse(record.vitals);
        }
      } catch {}
    }

    setModalTarget({
      isNew: false,
      clientId: record.client_id,
      clientName: record.client_full_name,
      recordDate: record.record_date,
      serviceStartTime: record.service_start_time || '10:00',
      serviceEndTime: record.service_end_time || '15:00',
      serviceLocation: record.service_location || '施設内',
      temperature: parsedVitals.temperature || '',
      bloodPressure: parsedVitals.bloodPressure || '',
      sleep: parsedVitals.sleep || '',
      mood: parsedVitals.mood || '',
      taskContent: record.task_content || '',
      staffComment: record.staff_comment || '',
      staffId: record.staff_id || (staffMembers[0]?.id || ''),
    });
    setErrorMsg(null);
  };

  // 未作成者の新規作成モーダルを開く
  const handleOpenCreateForMissing = (missing: MissingClient) => {
    const isOutFacility = missing.attendance_status === 'out_facility';
    const isAbsence = missing.is_absence_addition === 1;

    setModalTarget({
      isNew: true,
      clientId: missing.client_id,
      clientName: missing.client_full_name,
      recordDate: missing.record_date,
      serviceStartTime: missing.clock_in || (isAbsence ? '' : '10:00'),
      serviceEndTime: missing.clock_out || (isAbsence ? '' : '15:00'),
      serviceLocation: isOutFacility ? '施設外' : '施設内',
      temperature: '',
      bloodPressure: '',
      sleep: '',
      mood: '',
      taskContent: isAbsence ? '欠席時対応（体調不良等の電話連絡・状況確認）' : '施設内作業',
      staffComment: '',
      staffId: staffMembers[0]?.id || '',
    });
    setErrorMsg(null);
  };

  // 保存処理
  const handleSave = () => {
    if (!modalTarget) return;

    if (!modalTarget.taskContent || modalTarget.taskContent.trim() === '') {
      setErrorMsg('作業内容を入力してください');
      return;
    }

    startSavingTransition(async () => {
      const res = await updateSupportRecordInline({
        clientId: modalTarget.clientId,
        recordDate: modalTarget.recordDate,
        serviceStartTime: modalTarget.serviceStartTime || null,
        serviceEndTime: modalTarget.serviceEndTime || null,
        serviceLocation: modalTarget.serviceLocation || null,
        temperature: modalTarget.temperature,
        bloodPressure: modalTarget.bloodPressure,
        sleep: modalTarget.sleep,
        mood: modalTarget.mood,
        taskContent: modalTarget.taskContent,
        staffComment: modalTarget.staffComment,
        staffId: modalTarget.staffId || null,
      });

      if (!res.success) {
        setErrorMsg(res.error || '保存に失敗しました');
      } else {
        setModalTarget(null);
        router.refresh();
      }
    });
  };

  return (
    <div className="space-y-4">
      {/* ⚠️ 未作成のアラートバナー（対象日に未作成の出勤者がいる場合） */}
      {missingClients.length > 0 && (
        <Card className="border-amber-300 bg-amber-50/60 print:hidden shadow-xs">
          <CardHeader className="py-3 px-4 pb-2">
            <CardTitle className="text-sm font-bold text-amber-900 flex items-center gap-2">
              <AlertCircle className="w-4 h-4 text-amber-600 shrink-0" />
              <span>
                {selectedDate ? `【${selectedDate}】` : '対象期間内'}に支援記録が未作成の利用者が {missingClients.length} 名います
              </span>
            </CardTitle>
            <CardDescription className="text-xs text-amber-800">
              出勤または欠席時対応加算の対象ですが、支援記録が登録されていません。「作成」からその場で入力できます。
            </CardDescription>
          </CardHeader>
          <CardContent className="px-4 py-2 pt-0">
            <div className="flex flex-wrap gap-2 pt-1">
              {missingClients.map(mc => (
                <div 
                  key={`${mc.client_id}-${mc.record_date}`}
                  className="flex items-center gap-2 bg-white px-2.5 py-1 rounded-md border border-amber-200 text-xs shadow-2xs"
                >
                  <span className="font-semibold text-slate-800">{mc.client_full_name}</span>
                  <span className="text-[11px] text-muted-foreground font-mono">({mc.record_date.slice(5)})</span>
                  <Badge variant="outline" className="text-[10px] py-0 px-1 border-amber-400 text-amber-800 bg-amber-50">
                    {mc.attendance_status === 'out_facility' ? '施設外' : mc.is_absence_addition === 1 ? '欠席加算' : '出勤'}
                  </Badge>
                  <Button
                    size="sm"
                    variant="outline"
                    className="h-6 px-2 text-[11px] bg-amber-500 hover:bg-amber-600 text-white font-semibold border-amber-600 cursor-pointer"
                    onClick={() => handleOpenCreateForMissing(mc)}
                  >
                    <PlusCircle className="w-3 h-3 mr-0.5" />
                    作成
                  </Button>
                </div>
              ))}
            </div>
          </CardContent>
        </Card>
      )}

      {/* 📑 月間支援記録台帳 テーブルカード */}
      <Card className="shadow-sm print:shadow-none print:border-none border-teal-200/80">
        <CardHeader className="bg-primary/5 border-b pb-4 print:hidden flex flex-col sm:flex-row sm:items-center justify-between gap-3">
          <div>
            <div className="flex items-center gap-2">
              <CardTitle className="text-xl text-foreground flex items-center gap-2">
                <CalendarDays className="w-5 h-5 text-teal-600" />
                <span>
                  {selectedDate ? (
                    <>【{selectedDate}】の支援記録</>
                  ) : selectedMonth === 'all' ? (
                    <>全期間の支援記録台帳</>
                  ) : (
                    <>{selectedMonth} の月間支援記録台帳</>
                  )}
                  {selectedClientId && `（${selectedClientName}）`}
                </span>
              </CardTitle>
              <Badge variant="outline" className="bg-white font-mono text-xs border-teal-300">
                {records.length} 件
              </Badge>
            </div>
            <CardDescription className="mt-1 text-xs">
              {selectedDate ? (
                <span>カレンダーで選択した日付の記録です。右端の「編集」からその場で内容を修正・保存できます。</span>
              ) : (
                <span>各行の「編集」から内容を確認・直接修正できます。実地指導用の5年間保存帳票として印刷可能です。</span>
              )}
            </CardDescription>
          </div>

          <div className="flex items-center gap-2 shrink-0">
            {selectedDate && (
              <Link
                href={`/support-records/ledger?month=${selectedMonth}${selectedClientId ? `&clientId=${selectedClientId}` : ''}`}
                className={buttonVariants({ variant: "outline", size: "sm", className: "h-8 px-2.5 text-xs bg-white border-teal-300" })}
              >
                <CalendarIcon className="w-3.5 h-3.5 mr-1 text-teal-600" />
                この月の全件表示に戻す
              </Link>
            )}
            <PrintButton label="支援記録簿を印刷" />
          </div>
        </CardHeader>

        <CardContent className="p-0 overflow-x-auto">
          {records && records.length > 0 ? (
            <Table className="min-w-[750px]">
              <TableHeader className="bg-muted/50">
                <TableRow>
                  <TableHead className="py-3 w-[110px]">日付</TableHead>
                  <TableHead className="py-3 w-[150px]">利用者名</TableHead>
                  <TableHead className="py-3 w-[160px]">時間 / 場所</TableHead>
                  <TableHead className="py-3 w-[140px]">バイタル</TableHead>
                  <TableHead className="py-3">作業内容 / 支援員コメント / 担当者</TableHead>
                  <TableHead className="py-3 w-[90px] text-center print:hidden">操作</TableHead>
                </TableRow>
              </TableHeader>
              <TableBody>
                {records.map(record => {
                  let parsedVitals: any = null;
                  let vitalsFallbackStr: string | null = null;
                  try {
                    if (record.vitals && record.vitals.startsWith('{')) {
                      parsedVitals = JSON.parse(record.vitals);
                    } else if (record.vitals) {
                      vitalsFallbackStr = record.vitals;
                    }
                  } catch (e) {
                    vitalsFallbackStr = "（データ読取エラー）";
                  }

                  return (
                    <TableRow key={record.id} className="hover:bg-muted/20">
                      {/* 日付 */}
                      <TableCell className="py-3 font-medium font-mono text-sm whitespace-nowrap">
                        <Link 
                          href={`/support-records/ledger?date=${record.record_date}${selectedClientId ? `&clientId=${selectedClientId}` : ''}`}
                          className="text-primary hover:underline font-bold"
                          title="この日のみに絞り込む"
                        >
                          {record.record_date}
                        </Link>
                      </TableCell>

                      {/* 利用者名 */}
                      <TableCell className="py-3">
                        <div className="font-medium text-sm flex items-center gap-1.5">
                          <Link 
                            href={`/clients/${record.client_id}/documents`}
                            className="text-primary hover:underline font-semibold"
                          >
                            {record.client_full_name}
                          </Link>
                          {record.client_status === 'terminated' && (
                            <span className="text-[10px] px-1.5 py-0.2 rounded bg-gray-100 text-gray-600 border border-gray-200">
                              退職
                            </span>
                          )}
                        </div>
                      </TableCell>

                      {/* 時間・場所 */}
                      <TableCell className="py-3 text-xs text-muted-foreground">
                        {record.service_start_time || record.service_end_time ? (
                          <div className="font-mono font-medium text-foreground">
                            {record.service_start_time || '--:--'} 〜 {record.service_end_time || '--:--'}
                          </div>
                        ) : null}
                        {record.service_location && (
                          <div className="mt-0.5 inline-block px-1.5 py-0.2 rounded bg-muted text-[11px]">
                            {record.service_location}
                          </div>
                        )}
                      </TableCell>

                      {/* バイタル */}
                      <TableCell className="py-3 text-xs">
                        {parsedVitals ? (
                          <div className="space-y-0.5">
                            {parsedVitals.temperature && <div>体温: {parsedVitals.temperature}℃</div>}
                            {parsedVitals.bloodPressure && <div>血圧: {parsedVitals.bloodPressure}</div>}
                            {parsedVitals.sleep && <div>睡眠: {parsedVitals.sleep}</div>}
                            {parsedVitals.mood && <div>気分: {parsedVitals.mood}</div>}
                          </div>
                        ) : (
                          <span className="text-muted-foreground">{vitalsFallbackStr || '-'}</span>
                        )}
                      </TableCell>

                      {/* 作業内容・支援員コメント・担当者 */}
                      <TableCell className="py-3">
                        <div className="text-sm space-y-1">
                          <div>
                            <span className="text-xs font-semibold text-muted-foreground">【作業内容】</span>
                            <div className="text-foreground whitespace-pre-wrap">{record.task_content}</div>
                          </div>
                          {record.staff_comment && (
                            <div className="mt-1 pt-1 border-t border-dashed">
                              <span className="text-xs font-semibold text-muted-foreground">【支援員コメント】</span>
                              <div className="text-muted-foreground whitespace-pre-wrap">{record.staff_comment}</div>
                            </div>
                          )}
                          <div className="text-xs text-muted-foreground flex items-center justify-between gap-1 mt-1">
                            <div>
                              <span>担当支援員:</span>
                              <span className="font-medium text-foreground ml-1">{record.staff_full_name || '記録なし'}</span>
                            </div>
                          </div>
                        </div>
                      </TableCell>

                      {/* 操作（確認・直接編集） */}
                      <TableCell className="py-3 text-center print:hidden">
                        <Button
                          variant="outline"
                          size="sm"
                          className="h-8 px-2.5 text-xs font-semibold hover:bg-teal-50 hover:text-teal-700 hover:border-teal-400 cursor-pointer"
                          onClick={() => handleOpenEdit(record)}
                        >
                          <Edit3 className="w-3.5 h-3.5 mr-1 text-teal-600" />
                          編集
                        </Button>
                      </TableCell>
                    </TableRow>
                  );
                })}
              </TableBody>
            </Table>
          ) : (
            <div className="p-12 text-center text-muted-foreground">
              指定された条件に一致する支援記録は見つかりませんでした。
            </div>
          )}
        </CardContent>
      </Card>

      {/* 📝 確認・直接編集モーダルダイアログ */}
      {modalTarget && (
        <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/50 p-4">
          <div className="w-full max-w-xl rounded-lg bg-background p-6 shadow-xl border max-h-[90vh] overflow-y-auto">
            <div className="flex items-center justify-between border-b pb-3 mb-4">
              <div>
                <h3 className="text-lg font-bold text-foreground flex items-center gap-2">
                  <Edit3 className="w-5 h-5 text-teal-600" />
                  <span>支援記録の{modalTarget.isNew ? '新規作成' : '確認・直接編集'}</span>
                </h3>
                <p className="text-xs text-muted-foreground mt-0.5">
                  利用者: <strong className="text-foreground">{modalTarget.clientName}</strong>　
                  日付: <strong className="text-foreground font-mono">{modalTarget.recordDate}</strong>
                </p>
              </div>
              <Button
                variant="ghost"
                size="sm"
                className="h-8 w-8 p-0"
                onClick={() => setModalTarget(null)}
              >
                <X className="w-4 h-4" />
              </Button>
            </div>

            {errorMsg && (
              <div className="mb-4 rounded-md bg-destructive/15 p-3 text-sm text-destructive flex items-center gap-2">
                <AlertCircle className="w-4 h-4 shrink-0" />
                <span>{errorMsg}</span>
              </div>
            )}

            <div className="space-y-4">
              {/* 時間・場所 */}
              <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
                <div>
                  <Label className="text-xs font-semibold">提供開始時間</Label>
                  <Input
                    type="time"
                    value={modalTarget.serviceStartTime}
                    onChange={(e) => setModalTarget({ ...modalTarget, serviceStartTime: e.target.value })}
                    className="h-9 text-xs font-mono mt-1"
                  />
                </div>
                <div>
                  <Label className="text-xs font-semibold">提供終了時間</Label>
                  <Input
                    type="time"
                    value={modalTarget.serviceEndTime}
                    onChange={(e) => setModalTarget({ ...modalTarget, serviceEndTime: e.target.value })}
                    className="h-9 text-xs font-mono mt-1"
                  />
                </div>
                <div>
                  <Label className="text-xs font-semibold">提供場所</Label>
                  <select
                    value={modalTarget.serviceLocation}
                    onChange={(e) => setModalTarget({ ...modalTarget, serviceLocation: e.target.value })}
                    className="h-9 w-full rounded-md border border-input bg-background px-3 py-1 text-xs shadow-xs focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring mt-1"
                  >
                    <option value="施設内">施設内</option>
                    <option value="施設外">施設外（施設外就労）</option>
                    <option value="在宅">在宅</option>
                  </select>
                </div>
              </div>

              {/* バイタル情報 */}
              <div className="rounded-md border p-3 bg-muted/20 space-y-2">
                <Label className="text-xs font-semibold text-muted-foreground block">
                  バイタル・健康状態
                </Label>
                <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
                  <div>
                    <span className="text-[11px] text-muted-foreground block">体温 (℃)</span>
                    <Input
                      placeholder="36.5"
                      value={modalTarget.temperature}
                      onChange={(e) => setModalTarget({ ...modalTarget, temperature: e.target.value })}
                      className="h-8 text-xs mt-0.5"
                    />
                  </div>
                  <div>
                    <span className="text-[11px] text-muted-foreground block">血圧</span>
                    <Input
                      placeholder="120/80"
                      value={modalTarget.bloodPressure}
                      onChange={(e) => setModalTarget({ ...modalTarget, bloodPressure: e.target.value })}
                      className="h-8 text-xs mt-0.5"
                    />
                  </div>
                  <div>
                    <span className="text-[11px] text-muted-foreground block">睡眠</span>
                    <Input
                      placeholder="良好・7時間"
                      value={modalTarget.sleep}
                      onChange={(e) => setModalTarget({ ...modalTarget, sleep: e.target.value })}
                      className="h-8 text-xs mt-0.5"
                    />
                  </div>
                  <div>
                    <span className="text-[11px] text-muted-foreground block">気分</span>
                    <Input
                      placeholder="安定・良好"
                      value={modalTarget.mood}
                      onChange={(e) => setModalTarget({ ...modalTarget, mood: e.target.value })}
                      className="h-8 text-xs mt-0.5"
                    />
                  </div>
                </div>
              </div>

              {/* 作業内容 */}
              <div>
                <Label className="text-xs font-semibold flex items-center justify-between">
                  <span>作業内容 <span className="text-destructive">*</span></span>
                  <span className="text-[10px] text-muted-foreground font-normal">実地指導での必須記載項目</span>
                </Label>
                <Textarea
                  value={modalTarget.taskContent}
                  onChange={(e) => setModalTarget({ ...modalTarget, taskContent: e.target.value })}
                  placeholder="例: パソコン入力作業（データ入力50件）、ラベル貼り作業。集中して作業に取り組まれた。"
                  rows={3}
                  className="mt-1 text-xs resize-none"
                />
              </div>

              {/* 支援員コメント */}
              <div>
                <Label className="text-xs font-semibold flex items-center justify-between">
                  <span>支援員コメント・特記事項</span>
                  <span className="text-[10px] text-muted-foreground font-normal">様子や声かけ、気づきなど</span>
                </Label>
                <Textarea
                  value={modalTarget.staffComment}
                  onChange={(e) => setModalTarget({ ...modalTarget, staffComment: e.target.value })}
                  placeholder="例: 午前中は集中して作業を継続。午後少し疲労が見られたため小休憩を促した。"
                  rows={2}
                  className="mt-1 text-xs resize-none"
                />
              </div>

              {/* 担当支援スタッフ */}
              <div>
                <Label className="text-xs font-semibold">担当支援スタッフ</Label>
                <select
                  value={modalTarget.staffId}
                  onChange={(e) => setModalTarget({ ...modalTarget, staffId: e.target.value })}
                  className="h-9 w-full rounded-md border border-input bg-background px-3 py-1 text-xs shadow-xs focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring mt-1"
                >
                  <option value="">未指定</option>
                  {staffMembers.map(s => (
                    <option key={s.id} value={s.id}>
                      {s.full_name} {s.job_category ? `(${s.job_category})` : ''}
                    </option>
                  ))}
                </select>
              </div>
            </div>

            {/* モーダル下部ボタン */}
            <div className="mt-6 flex items-center justify-end gap-2 border-t pt-4">
              <Button
                variant="outline"
                size="sm"
                onClick={() => setModalTarget(null)}
                disabled={isSaving}
              >
                キャンセル
              </Button>
              <Button
                size="sm"
                className="bg-teal-600 hover:bg-teal-700 text-white font-semibold flex items-center gap-1.5"
                onClick={handleSave}
                disabled={isSaving}
              >
                {isSaving ? (
                  <>
                    <Loader2 className="w-4 h-4 animate-spin" />
                    保存中...
                  </>
                ) : (
                  <>
                    <Save className="w-4 h-4" />
                    保存する
                  </>
                )}
              </Button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}

import { Header } from "@/components/Header"
import { SupportSubNav } from "@/components/NavigationTabs"
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card"
import { buttonVariants } from "@/components/ui/button"
import { Label } from "@/components/ui/label"
import { toLocalISODate } from "@/lib/utils"
import { db } from "@/lib/db"
import { 
  Filter, 
  Calendar as CalendarIcon,
  Edit3,
  CalendarClock
} from "lucide-react"
import Link from "next/link"
import { SupportCalendarMonthPicker } from "../SupportCalendarMonthPicker"
import { SupportLedgerClient } from "./SupportLedgerClient"
import { getRole } from "@/lib/auth"
import { getActiveDepartments } from "@/lib/location"

export default async function SupportRecordsLedgerPage(props: { 
  searchParams?: Promise<{ 
    month?: string; 
    clientId?: string; 
    date?: string; 
    departmentId?: string;
  }> 
}) {
  const role = await getRole();
  const searchParams = await props.searchParams;

  const isGeneralStaff = role === 'staff';
  const rawDepartmentId = searchParams?.departmentId;
  const activeDepartments = getActiveDepartments();
  const activeDepartmentId = isGeneralStaff
    ? (rawDepartmentId && activeDepartments.some(d => d.id === rawDepartmentId) ? rawDepartmentId : activeDepartments[0]?.id)
    : (rawDepartmentId && activeDepartments.some(d => d.id === rawDepartmentId) ? rawDepartmentId : 'all');

  const now = new Date();
  const todayStr = toLocalISODate(now);

  const targetDateParam = searchParams?.date;
  const targetMonthParam = searchParams?.month;
  const targetClientParam = searchParams?.clientId || "";

  // カレンダー基準日
  let calendarBaseDate = targetDateParam;
  if (!calendarBaseDate) {
    if (targetMonthParam && targetMonthParam !== 'all') {
      calendarBaseDate = `${targetMonthParam}-01`;
    } else {
      calendarBaseDate = todayStr;
    }
  }

  // カレンダー表示用の年月
  const targetDate = new Date(calendarBaseDate);
  const currentYear = targetDate.getFullYear();
  const currentMonth = targetDate.getMonth(); // 0-11
  const currentMonthStr = `${currentYear}-${String(currentMonth + 1).padStart(2, '0')}`;
  const currentMonthStart = `${currentMonthStr}-01`;
  const nextMonthObj = new Date(currentYear, currentMonth + 1, 1);
  const nextMonthStart = toLocalISODate(nextMonthObj);

  // 選択中の月（パラメータまたはカレンダー年月）
  const selectedMonth = targetMonthParam || currentMonthStr;

  // 全利用者（退職者含む、拠点フィルタ対応）
  const allClientsRaw = db.prepare('SELECT id, full_name, status, department_id FROM clients WHERE deleted_at IS NULL ORDER BY full_name').all() as any[];
  const allClients = activeDepartmentId === 'all'
    ? allClientsRaw
    : allClientsRaw.filter(c => c.department_id === activeDepartmentId);

  const selectedClientName = targetClientParam 
    ? allClients.find(c => c.id === targetClientParam)?.full_name 
    : "全利用者";

  // 直接支援スタッフ（生活支援員・職業指導員）
  const staffMembers = db.prepare(`
    SELECT id, name as full_name, job_category, role 
    FROM staff 
    WHERE deleted_at IS NULL 
      AND (
        job_category IN ('職業指導員', '生活支援員')
        OR role IN ('vocational_instructor', 'life_support_worker')
      )
      AND (
        job_category NOT IN (
          '管理者', 'サービス管理責任者', '管理者 兼 サービス管理責任者', 
          '賃金向上達成指導員', '視覚・聴覚言語障害者支援指導員', '重度者支援専門員', 
          '送迎ドライバー', '調理員', '事務員'
        )
        OR job_category IS NULL
      )
    ORDER BY name
  `).all() as any[];

  // 過去に記録が存在する年月の一覧（最大72ヶ月）
  const dbMonths = db.prepare(`
    SELECT DISTINCT substr(record_date, 1, 7) as ym 
    FROM support_records 
    WHERE deleted_at IS NULL
    ORDER BY ym DESC 
    LIMIT 72
  `).all() as { ym: string }[];

  const availableMonths = dbMonths.map(m => m.ym);
  if (!availableMonths.includes(currentMonthStr)) {
    availableMonths.unshift(currentMonthStr);
  }

  // --- カレンダー用の月間データ集計 ---
  let monthSupportQuery = `
    SELECT s.id, s.client_id, s.record_date
    FROM support_records s
    JOIN clients c ON s.client_id = c.id
    WHERE s.record_date >= ? AND s.record_date < ? AND s.deleted_at IS NULL
  `;
  const monthSupportParams: any[] = [currentMonthStart, nextMonthStart];
  if (activeDepartmentId !== 'all') {
    monthSupportQuery += ` AND c.department_id = ?`;
    monthSupportParams.push(activeDepartmentId);
  }
  const monthSupportRecords = db.prepare(monthSupportQuery).all(...monthSupportParams) as any[];

  let monthAttendanceQuery = `
    SELECT a.id, a.client_id, a.record_date, a.status, a.is_absence_addition
    FROM attendance_records a
    JOIN clients c ON a.client_id = c.id
    WHERE a.record_date >= ? AND a.record_date < ? AND a.deleted_at IS NULL
  `;
  const monthAttendanceParams: any[] = [currentMonthStart, nextMonthStart];
  if (activeDepartmentId !== 'all') {
    monthAttendanceQuery += ` AND c.department_id = ?`;
    monthAttendanceParams.push(activeDepartmentId);
  }
  const monthAttendanceRecords = db.prepare(monthAttendanceQuery).all(...monthAttendanceParams) as any[];

  // 月間カレンダーグリッドの作成
  const daysInMonth = new Date(currentYear, currentMonth + 1, 0).getDate();
  const firstDayOfMonth = new Date(currentYear, currentMonth, 1).getDay();

  const calendarCells = [];
  for (let i = 0; i < firstDayOfMonth; i++) {
    calendarCells.push(null);
  }
  for (let i = 1; i <= daysInMonth; i++) {
    const dStr = `${currentYear}-${String(currentMonth + 1).padStart(2, '0')}-${String(i).padStart(2, '0')}`;
    const daySupportCount = monthSupportRecords.filter(r => r.record_date === dStr).length;
    
    // 出勤（または欠席時対応加算）の対象人数
    const dayAtt = monthAttendanceRecords.filter(r => r.record_date === dStr);
    const activeAttCount = dayAtt.filter(r => ['present', 'out_facility'].includes(r.status) || r.is_absence_addition === 1).length;

    calendarCells.push({
      day: i,
      dateStr: dStr,
      supportCount: daySupportCount,
      attendanceCount: activeAttCount,
      isComplete: activeAttCount > 0 && daySupportCount >= activeAttCount,
      hasMissing: activeAttCount > daySupportCount,
      isToday: dStr === todayStr,
      isSelected: dStr === targetDateParam
    });
  }

  // 月間統計サマリー
  const monthlyStats = {
    totalRecords: monthSupportRecords.length,
    totalAttendance: monthAttendanceRecords.filter(r => ['present', 'out_facility'].includes(r.status)).length,
    totalAbsenceAddition: monthAttendanceRecords.filter(r => r.is_absence_addition === 1).length,
  };

  // --- 台帳一覧クエリ ---
  let ledgerQuery = `
    SELECT s.*, c.full_name as client_full_name, c.status as client_status, c.department_id as client_department_id, st.name as staff_full_name
    FROM support_records s
    JOIN clients c ON s.client_id = c.id
    LEFT JOIN staff st ON s.staff_id = st.id
    WHERE s.deleted_at IS NULL
  `;
  const ledgerParams: any[] = [];

  if (activeDepartmentId !== 'all') {
    ledgerQuery += ` AND c.department_id = ?`;
    ledgerParams.push(activeDepartmentId);
  }

  if (targetDateParam) {
    ledgerQuery += ` AND s.record_date = ?`;
    ledgerParams.push(targetDateParam);
  } else if (selectedMonth && selectedMonth !== 'all') {
    const [yStr, mStr] = selectedMonth.split('-');
    const y = parseInt(yStr, 10);
    const m = parseInt(mStr, 10);
    const startOfMonth = `${selectedMonth}-01`;
    const nextM = new Date(y, m, 1);
    const endOfMonth = `${nextM.getFullYear()}-${String(nextM.getMonth() + 1).padStart(2, '0')}-01`;

    ledgerQuery += ` AND s.record_date >= ? AND s.record_date < ?`;
    ledgerParams.push(startOfMonth, endOfMonth);
  }

  if (targetClientParam) {
    ledgerQuery += ` AND s.client_id = ?`;
    ledgerParams.push(targetClientParam);
  }

  ledgerQuery += ` ORDER BY s.record_date DESC, c.full_name ASC`;
  if (!targetDateParam && selectedMonth === 'all' && !targetClientParam) {
    ledgerQuery += ` LIMIT 500`;
  }

  const ledgerRecords = db.prepare(ledgerQuery).all(...ledgerParams) as any[];

  // --- 未作成者（特定日選択時） ---
  let missingClients: any[] = [];
  if (targetDateParam) {
    let attendedQuery = `
      SELECT a.client_id, a.record_date, a.status as attendance_status, a.clock_in, a.clock_out, a.is_absence_addition,
             c.full_name as client_full_name, c.status as client_status, c.department_id as client_department_id
      FROM attendance_records a
      JOIN clients c ON a.client_id = c.id
      WHERE a.record_date = ? 
        AND a.deleted_at IS NULL
        AND (a.status IN ('present', 'out_facility') OR a.is_absence_addition = 1)
    `;
    const attendedParams: any[] = [targetDateParam];
    if (activeDepartmentId !== 'all') {
      attendedQuery += ` AND c.department_id = ?`;
      attendedParams.push(activeDepartmentId);
    }
    const attendedClientsOnDate = db.prepare(attendedQuery).all(...attendedParams) as any[];

    const recordedClientIds = new Set(ledgerRecords.map(r => r.client_id));
    missingClients = attendedClientsOnDate.filter(a => !recordedClientIds.has(a.client_id));
    if (targetClientParam) {
      missingClients = missingClients.filter(m => m.client_id === targetClientParam);
    }
  }

  return (
    <div className="flex min-h-screen flex-col bg-background/50">
      <div className="print:hidden">
        <Header activePage="support-records" />
        <SupportSubNav />
      </div>

      <main className="flex-1 p-3.5 sm:p-6 md:p-12 print:p-0">
        <div className="mx-auto max-w-6xl space-y-6">
          {/* ページ上部タイトル */}
          <div className="print:hidden flex flex-col sm:flex-row sm:items-center justify-between gap-4 border-b pb-3">
            <div>
              <h1 className="text-2xl sm:text-3xl font-bold tracking-tight text-foreground flex items-center gap-2.5">
                <CalendarClock className="h-7 w-7 sm:h-8 sm:w-8 text-teal-600 shrink-0" />
                <span>月間支援記録台帳・過去データ検索（実地指導・印刷用）</span>
              </h1>
              <p className="mt-1 text-sm text-muted-foreground">
                カレンダーや検索条件から過去・現在の記録を瞬時に確認・直接編集できます。実地指導用の帳票印刷にも対応しています。
              </p>
            </div>
            <div className="flex items-center gap-2 shrink-0">
              <Link
                href={`/support-records?departmentId=${activeDepartmentId}`}
                className={buttonVariants({ variant: "outline", size: "sm", className: "h-9 px-3 text-xs font-semibold" })}
              >
                <Edit3 className="w-3.5 h-3.5 mr-1 text-emerald-600" />
                日々の支援を記録する（入力画面へ）
              </Link>
            </div>
          </div>

          {/* 🏢 拠点切り替えタブ */}
          <div className="flex items-center gap-2 p-2 bg-muted/30 border rounded-xl shadow-2xs print:hidden overflow-x-auto whitespace-nowrap">
            <span className="text-xs font-bold text-muted-foreground px-2 shrink-0">台帳対象拠点:</span>
            {!isGeneralStaff && (
              <Link
                href={`/support-records/ledger?month=${selectedMonth}${targetDateParam ? `&date=${targetDateParam}` : ''}&departmentId=all`}
                className={buttonVariants({
                  variant: activeDepartmentId === 'all' ? 'default' : 'outline',
                  size: 'sm',
                  className: `h-8 px-3.5 text-xs font-bold shrink-0 ${activeDepartmentId === 'all' ? 'shadow-xs' : ''}`
                })}
              >
                すべて表示
              </Link>
            )}
            {activeDepartments.map(dep => (
              <Link
                key={dep.id}
                href={`/support-records/ledger?month=${selectedMonth}${targetDateParam ? `&date=${targetDateParam}` : ''}&departmentId=${dep.id}`}
                className={buttonVariants({
                  variant: activeDepartmentId === dep.id ? 'default' : 'outline',
                  size: 'sm',
                  className: `h-8 px-3.5 text-xs font-bold shrink-0 ${
                    activeDepartmentId === dep.id
                      ? 'bg-sky-600 hover:bg-sky-700 text-white shadow-xs'
                      : 'border-sky-200 text-sky-800 hover:bg-sky-50'
                  }`
                })}
              >
                {dep.facility_type === 'satellite' ? '🛰️' : '🏢'} {dep.name}{isGeneralStaff ? '担当' : ''}
              </Link>
            ))}
            <span className="text-[11px] text-muted-foreground ml-auto hidden sm:inline shrink-0">
              現在: <strong className="text-foreground">{activeDepartmentId === 'all' ? '全拠点' : activeDepartments.find(d => d.id === activeDepartmentId)?.name || ''}</strong> の台帳データを表示中
            </span>
          </div>

          {/* 印刷用ヘッダー（ブラウザ印刷時のみ表示） */}
          <div className="hidden print:block mb-6 border-b pb-4">
            <h1 className="text-2xl font-bold text-center">
              {(() => {
                const fac = db.prepare("SELECT service_type, facility_name FROM facility_settings LIMIT 1").get() as any;
                const isB = fac?.service_type === 'type_b' || fac?.service_type === 'B' || fac?.facility_name?.includes('B型');
                return isB ? '就労継続支援B型 日次支援記録台帳' : '就労継続支援A型 日次支援記録台帳';
              })()}
            </h1>
            <div className="flex justify-between items-end mt-4 text-xs">
              <div>
                <div>対象期間: {targetDateParam ? targetDateParam : selectedMonth === 'all' ? '全期間' : `${selectedMonth}月分`}</div>
                <div>対象者: {selectedClientName}</div>
              </div>
              <div className="text-right">
                <div>印刷日: {todayStr}</div>
                <div>抽出件数: {ledgerRecords.length} 件</div>
              </div>
            </div>
          </div>

          {/* 1. 📅 支援記録カレンダー ＆ 検索バー（一体型カード） */}
          <Card className="shadow-sm print:hidden border-teal-200/60">
            <CardHeader className="bg-teal-50/30 border-b pb-4 space-y-3">
              <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
                <div>
                  <CardTitle className="text-xl flex items-center gap-2 text-teal-950">
                    <CalendarIcon className="w-5 h-5 text-teal-600" />
                    <span>{currentYear}年{currentMonth + 1}月 の支援記録カレンダー</span>
                  </CardTitle>
                  <p className="text-xs text-muted-foreground mt-0.5">
                    日付をクリックすると、下の台帳がその日の記録に瞬時に切り替わり、その場で確認・直接編集できます。
                  </p>
                </div>
                <div className="flex items-center gap-3 text-xs shrink-0">
                  <span className="flex items-center gap-1 font-medium text-emerald-800">
                    <span className="w-2.5 h-2.5 rounded-full bg-emerald-500 inline-block"></span> 記録完了
                  </span>
                  <span className="flex items-center gap-1 font-medium text-amber-800">
                    <span className="w-2.5 h-2.5 rounded-full bg-amber-500 inline-block"></span> 記録未作成あり
                  </span>
                </div>
              </div>

              {/* 年月ピッカー・前後月ショートカット */}
              <SupportCalendarMonthPicker currentDateStr={calendarBaseDate} todayStr={todayStr} basePath="/support-records/ledger" />

              {/* 月間実績サマリーバナー */}
              <div className="grid grid-cols-3 gap-2 p-2.5 bg-white dark:bg-slate-900 border rounded-lg text-center text-xs shadow-2xs">
                <div className="border-r border-slate-100">
                  <span className="text-[10px] text-muted-foreground block">月間 支援記録数</span>
                  <span className="text-sm font-bold text-teal-700">{monthlyStats.totalRecords} 件</span>
                </div>
                <div className="border-r border-slate-100">
                  <span className="text-[10px] text-muted-foreground block">延べ 出勤実績</span>
                  <span className="text-sm font-bold text-emerald-700">{monthlyStats.totalAttendance} 名</span>
                </div>
                <div>
                  <span className="text-[10px] text-muted-foreground block">欠席時対応加算 算定</span>
                  <span className="text-sm font-bold text-amber-700">{monthlyStats.totalAbsenceAddition} 件</span>
                </div>
              </div>
            </CardHeader>

            <CardContent className="p-4 sm:p-6 space-y-4">
              {/* カレンダー曜日ヘッダー */}
              <div className="grid grid-cols-7 gap-1 text-center mb-2 font-medium text-sm text-muted-foreground">
                <div className="text-red-500">日</div>
                <div>月</div>
                <div>火</div>
                <div>水</div>
                <div>木</div>
                <div>金</div>
                <div className="text-blue-500">土</div>
              </div>

              {/* カレンダーグリッド */}
              <div className="grid grid-cols-7 gap-1 sm:gap-2">
                {calendarCells.map((cell, index) => {
                  if (!cell) {
                    return <div key={`empty-${index}`} className="h-16 sm:h-20 bg-muted/10 rounded-md border border-transparent" />
                  }

                  const dayOfWeek = new Date(cell.dateStr).getDay();
                  const isSun = dayOfWeek === 0;
                  const isSat = dayOfWeek === 6;

                  return (
                    <Link
                      key={cell.dateStr}
                      href={`/support-records/ledger?date=${cell.dateStr}${targetClientParam ? `&clientId=${targetClientParam}` : ''}`}
                      className={`h-16 sm:h-20 p-1 sm:p-1.5 rounded-md border flex flex-col justify-between transition-all hover:shadow-md cursor-pointer ${
                        cell.isSelected
                          ? "ring-2 ring-teal-600 border-teal-600 bg-teal-50 shadow-sm"
                          : cell.isToday
                          ? "border-teal-400 bg-teal-50/40"
                          : "border-border/60 hover:border-border bg-card"
                      }`}
                    >
                      <div className="flex items-center justify-between">
                        <span className={`text-xs sm:text-sm font-bold ${
                          cell.isSelected ? "text-teal-700" : isSun ? "text-red-500" : isSat ? "text-blue-500" : "text-foreground"
                        }`}>
                          {cell.day}
                        </span>
                        {cell.isToday && (
                          <span className="text-[9px] px-1 py-0.2 rounded bg-teal-600 text-white font-semibold">
                            今日
                          </span>
                        )}
                      </div>

                      <div className="flex flex-col gap-0.5 mt-auto">
                        {cell.attendanceCount > 0 && (
                          <span className="text-[9px] text-muted-foreground hidden sm:block truncate">
                            出勤:{cell.attendanceCount}
                          </span>
                        )}
                        {cell.supportCount > 0 ? (
                          cell.isComplete ? (
                            <span className="text-[9px] sm:text-[10px] font-semibold bg-emerald-100 text-emerald-800 border border-emerald-200 px-1 rounded-sm text-center">
                              ✓ 記録:{cell.supportCount}
                            </span>
                          ) : (
                            <span className="text-[9px] sm:text-[10px] font-semibold bg-amber-100 text-amber-900 border border-amber-300 px-1 rounded-sm text-center">
                              記:{cell.supportCount}/{cell.attendanceCount}
                            </span>
                          )
                        ) : cell.attendanceCount > 0 ? (
                          <span className="text-[8px] sm:text-[9px] font-bold bg-rose-100 text-rose-800 border border-rose-300 px-0.5 rounded-sm text-center animate-pulse">
                            未作:{cell.attendanceCount}
                          </span>
                        ) : null}
                      </div>
                    </Link>
                  )
                })}
              </div>

              {/* カレンダー下部の検索・絞り込みフォーム */}
              <div className="pt-3 border-t">
                <form method="GET" className="flex flex-wrap items-center justify-between gap-3">
                  <div className="flex flex-wrap items-center gap-3">
                    <div className="flex items-center gap-2">
                      <CalendarIcon className="h-4 w-4 text-muted-foreground" />
                      <Label htmlFor="filterMonth" className="text-xs font-semibold">対象年月:</Label>
                      <select
                        id="filterMonth"
                        name="month"
                        defaultValue={selectedMonth}
                        className="h-8 rounded-md border border-input bg-background px-2.5 py-1 text-xs shadow-xs focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring font-mono"
                      >
                        <option value="all">全期間（最新500件）</option>
                        {availableMonths.map(ym => (
                          <option key={ym} value={ym}>
                            {ym} {ym === currentMonthStr ? '(当月)' : ''}
                          </option>
                        ))}
                      </select>
                    </div>

                    <div className="flex items-center gap-2">
                      <Filter className="h-4 w-4 text-muted-foreground" />
                      <Label htmlFor="filterClient" className="text-xs font-semibold">対象利用者:</Label>
                      <select
                        id="filterClient"
                        name="clientId"
                        defaultValue={targetClientParam}
                        className="h-8 rounded-md border border-input bg-background px-2.5 py-1 text-xs shadow-xs focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring"
                      >
                        <option value="">全利用者</option>
                        {allClients.map(c => (
                          <option key={c.id} value={c.id}>
                            {c.full_name} {c.status === 'terminated' ? '(退職済)' : ''}
                          </option>
                        ))}
                      </select>
                    </div>

                    <button
                      type="submit"
                      className="inline-flex items-center justify-center rounded-md text-xs font-medium transition-colors focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring bg-teal-600 text-white shadow-xs hover:bg-teal-700 h-8 px-3 cursor-pointer"
                    >
                      台帳を絞り込み
                    </button>

                    {(targetDateParam || selectedMonth !== currentMonthStr || targetClientParam) && (
                      <Link
                        href={`/support-records/ledger?month=${currentMonthStr}`}
                        className="text-xs text-muted-foreground hover:text-foreground underline ml-1"
                      >
                        当月・全件に戻す
                      </Link>
                    )}
                  </div>
                </form>
              </div>
            </CardContent>
          </Card>

          {/* 2. 📑 月間支援記録台帳（確認・直接編集モーダル対応） */}
          <SupportLedgerClient
            records={ledgerRecords}
            missingClients={missingClients}
            staffMembers={staffMembers}
            selectedMonth={selectedMonth}
            selectedDate={targetDateParam || null}
            selectedClientId={targetClientParam}
            selectedClientName={selectedClientName}
          />
        </div>
      </main>
    </div>
  )
}

'use client';

import { useEffect } from 'react';
import { useSearchParams, useRouter, usePathname } from 'next/navigation';
import { toast } from 'sonner';

export default function ActionToast() {
  const searchParams = useSearchParams();
  const router = useRouter();
  const pathname = usePathname();

  useEffect(() => {
    if (!searchParams) return;
    
    const success = searchParams.get('success');
    const error = searchParams.get('error');
    const msg = searchParams.get('msg');
    
    if (success === 'true') {
      toast.success(msg || '保存しました');
      // Remove query params
      const newParams = new URLSearchParams(searchParams.toString());
      newParams.delete('success');
      newParams.delete('msg');
      router.replace(`${pathname}?${newParams.toString()}`);
    }

    if (error) {
      toast.error(error);
      const newParams = new URLSearchParams(searchParams.toString());
      newParams.delete('error');
      router.replace(`${pathname}?${newParams.toString()}`);
    }
  }, [searchParams, pathname, router]);

  return null;
}

'use client';

import React, { useRef, useLayoutEffect, useState, useEffect, ElementType } from 'react';
import { cn } from '@/lib/utils';

export interface AutoFitTextProps {
  text?: string | null;
  children?: React.ReactNode;
  defaultText?: string;
  className?: string;
  minFontSize?: number; // 最小フォントサイズ（デフォルト: 5.5）
  maxFontSize?: number; // 最大フォントサイズ（デフォルト: 8.5）
  minFontSizePt?: number; // エイリアス
  maxFontSizePt?: number; // エイリアス
  unit?: 'pt' | 'px'; // 単位（デフォルト: 'pt'）
  multiLine?: boolean; // 複数行の高さフィット判定を行うか（デフォルト: false）
  mode?: 'single-line' | 'multi-line'; // エイリアス
  as?: ElementType; // レンダリング要素（デフォルト: 'span'）
  title?: string;
}

/**
 * 枠（コンテナ）の幅や高さに合わせて、テキストが絶対に溢れず・途切れないように
 * 二分探索（Binary Search）を用いてフォントサイズをミリ単位で自動調整する高精度コンポーネント。
 * 全国の自治体・事業所のあらゆる長文データ（事業所名、住所、氏名、支援目標等）に対応。
 */
export function AutoFitText({
  text,
  children,
  defaultText = '',
  className,
  minFontSize: minFontSizeProp,
  maxFontSize: maxFontSizeProp,
  minFontSizePt,
  maxFontSizePt,
  unit = 'pt',
  multiLine: multiLineProp = false,
  mode,
  as: Component = 'span',
  title,
}: AutoFitTextProps) {
  const containerRef = useRef<HTMLDivElement>(null);
  const textRef = useRef<HTMLElement>(null);
  
  const minFontSize = minFontSizePt ?? minFontSizeProp ?? 5.5;
  const maxFontSize = maxFontSizePt ?? maxFontSizeProp ?? 8.5;
  const multiLine = mode ? mode === 'multi-line' : multiLineProp;

  // text または children から文字列コンテンツを抽出
  const rawContent = (text !== undefined && text !== null && text !== '') 
    ? text 
    : (children !== undefined && children !== null && children !== '') 
      ? (typeof children === 'string' ? children : String(children))
      : defaultText;
  const content = rawContent || '';

  // SSR時の初期推定サイズ（文字長ベースの高速フォールバック）
  const getInitialFontSize = () => {
    const len = content.length;
    if (multiLine) return maxFontSize;
    if (len > 35) return Math.max(minFontSize, minFontSize + 0.5);
    if (len > 25) return Math.max(minFontSize, (minFontSize + maxFontSize) / 2);
    if (len > 18) return Math.max(minFontSize, maxFontSize - 1);
    return maxFontSize;
  };

  const [currentFontSize, setCurrentFontSize] = useState<number>(getInitialFontSize);

  const calculateFit = () => {
    const container = containerRef.current;
    const textEl = textRef.current;
    if (!container || !textEl || !content) return;

    // コンテナ幅・高さがゼロ（非表示時など）はスキップ
    if (container.clientWidth === 0 || (multiLine && container.clientHeight === 0)) return;

    let low = minFontSize;
    let high = maxFontSize;
    let optimal = minFontSize;

    // 二分探索（最大8ステップで 0.1pt 精度で瞬時に最適サイズを算出）
    for (let i = 0; i < 8; i++) {
      const mid = Math.round(((low + high) / 2) * 10) / 10;
      textEl.style.fontSize = `${mid}${unit}`;

      const overflows = multiLine
        ? textEl.scrollHeight > container.clientHeight || textEl.scrollWidth > container.clientWidth
        : textEl.scrollWidth > container.clientWidth;

      if (overflows) {
        high = mid - 0.1;
      } else {
        optimal = mid;
        low = mid + 0.1;
      }

      if (high - low < 0.1) break;
    }

    textEl.style.fontSize = `${optimal}${unit}`;
    setCurrentFontSize(optimal);
  };

  useLayoutEffect(() => {
    calculateFit();
  }, [content, minFontSize, maxFontSize, unit, multiLine]);

  // リサイズおよび印刷ダイアログ（beforeprint）の監視
  useEffect(() => {
    const container = containerRef.current;
    if (!container) return;

    let resizeObserver: ResizeObserver | null = null;
    if (typeof ResizeObserver !== 'undefined') {
      resizeObserver = new ResizeObserver(() => {
        calculateFit();
      });
      resizeObserver.observe(container);
    }

    const handleBeforePrint = () => {
      calculateFit();
    };

    window.addEventListener('beforeprint', handleBeforePrint);
    window.addEventListener('resize', calculateFit);

    return () => {
      if (resizeObserver) resizeObserver.disconnect();
      window.removeEventListener('beforeprint', handleBeforePrint);
      window.removeEventListener('resize', calculateFit);
    };
  }, [content]);

  return (
    <div
      ref={containerRef}
      className={cn(
        "w-full h-full flex items-center overflow-hidden",
        multiLine ? "block" : "whitespace-nowrap"
      )}
    >
      <Component
        ref={textRef as any}
        style={{ fontSize: `${currentFontSize}${unit}` }}
        className={cn(
          "inline-block leading-tight transition-none",
          !multiLine && "whitespace-nowrap",
          className
        )}
        title={title || (typeof content === 'string' ? content : undefined)}
      >
        {content}
      </Component>
    </div>
  );
}

export default AutoFitText;

"use client";

import React, { createContext, useContext, useEffect, useState } from 'react';

type FontSize = 'xsmall' | 'small' | 'normal' | 'large' | 'xlarge';

interface FontSizeContextType {
  fontSize: FontSize;
  setFontSize: (size: FontSize) => void;
}

const FontSizeContext = createContext<FontSizeContextType | undefined>(undefined);

export function FontSizeProvider({ children }: { children: React.ReactNode }) {
  const [fontSize, setFontSize] = useState<FontSize>('normal');
  const [mounted, setMounted] = useState(false);

  useEffect(() => {
    const saved = localStorage.getItem('app-font-size') as FontSize | null;
    if (saved && ['xsmall', 'small', 'normal', 'large', 'xlarge'].includes(saved)) {
      setFontSize(saved);
    }
    setMounted(true);
  }, []);

  useEffect(() => {
    if (!mounted) return;
    localStorage.setItem('app-font-size', fontSize);
    
    // root html element
    const html = document.documentElement;
    if (fontSize === 'xsmall') {
      html.style.fontSize = '80%';
    } else if (fontSize === 'small') {
      html.style.fontSize = '90%';
    } else if (fontSize === 'normal') {
      html.style.fontSize = '100%';
    } else if (fontSize === 'large') {
      html.style.fontSize = '115%';
    } else if (fontSize === 'xlarge') {
      html.style.fontSize = '130%';
    }
  }, [fontSize, mounted]);

  return (
    <FontSizeContext.Provider value={{ fontSize, setFontSize }}>
      {children}
    </FontSizeContext.Provider>
  );
}

export function useFontSize() {
  const context = useContext(FontSizeContext);
  if (context === undefined) {
    throw new Error('useFontSize must be used within a FontSizeProvider');
  }
  return context;
}

export function UIModeProvider({ children }: { children: React.ReactNode }) {
  // SSRハイドレーション不一致を防ぐため、初期値はサーバー・クライアント共通の定数とする
  const [modePreference, setModePreferenceState] = useState<UIModePreference>("auto");
  const [windowWidth, setWindowWidth] = useState<number>(1200);
  const [mounted, setMounted] = useState(false);

  useEffect(() => {
    setMounted(true);

    try {
      const saved = localStorage.getItem("app-ui-mode") as UIModePreference;
      if (saved && ["auto", "mobile", "tablet", "desktop"].includes(saved)) {
        setModePreferenceState(saved);
      }
    } catch {}

    setWindowWidth(window.innerWidth);

    const handleResize = () => {
      setWindowWidth(window.innerWidth);
    };
    window.addEventListener("resize", handleResize);
    return () => window.removeEventListener("resize", handleResize);
  }, []);

  const setModePreference = useCallback((mode: UIModePreference) => {
    setModePreferenceState(mode);
    try {
      if (typeof window !== "undefined") {
        localStorage.setItem("app-ui-mode", mode);
      }
    } catch {}
  }, []);

  // 実効モードの判定（未マウント時はサーバーと完全に一致させる）
  const activeMode: ActiveUIMode = React.useMemo(() => {
    if (!mounted) return "desktop";

    if (modePreference === "mobile") return "mobile";
    if (modePreference === "tablet") return "tablet";
    if (modePreference === "desktop") return "desktop";

    // "auto" の場合は画面幅で判定
    if (windowWidth < 768) return "mobile";
    if (windowWidth < 1024) return "tablet";
    return "desktop";
  }, [mounted, modePreference, windowWidth]);

  useEffect(() => {
    if (!mounted) return;
    const root = document.documentElement;
    root.setAttribute("data-ui-mode", activeMode);
    root.setAttribute("data-ui-pref", modePreference);

    // スマホ・タブレット等のブラウザでPC表示が選択された場合、viewportメタタグを動的に切り替える
    try {
      let viewportMeta = document.querySelector('meta[name="viewport"]') as HTMLMetaElement | null;
      if (!viewportMeta) {
        viewportMeta = document.createElement("meta");
        viewportMeta.name = "viewport";
        document.head.appendChild(viewportMeta);
      }

      if (activeMode === "desktop") {
        // PC通常モード: 標準スケール（1.0倍・ピンチ拡大縮小可能）
        viewportMeta.content = "width=device-width, initial-scale=1.0, minimum-scale=0.5, maximum-scale=5.0, user-scalable=yes";
      } else if (activeMode === "tablet") {
        // タブレット面談モード
        viewportMeta.content = "width=device-width, initial-scale=1.0, minimum-scale=0.5, maximum-scale=5.0, user-scalable=yes";
      } else {
        // 現場スマホモード: 通常のスマホ最適化幅（親指操作・縦スクロール）
        viewportMeta.content = "width=device-width, initial-scale=1, maximum-scale=5, user-scalable=yes, viewport-fit=cover";
      }

      // ブラウザにリサイズイベントを通知してCSS/JSコンポーネントを即時再計算
      setTimeout(() => {
        window.dispatchEvent(new Event("resize"));
      }, 50);
    } catch (e) {
      console.error("viewport update error:", e);
    }
  }, [activeMode, modePreference, mounted]);

  const value = React.useMemo(
    () => ({
      modePreference,
      setModePreference,
      activeMode,
      isMobile: activeMode === "mobile",
      isTablet: activeMode === "tablet",
      isDesktop: activeMode === "desktop",
      mounted,
    }),
    [modePreference, setModePreference, activeMode, mounted]
  );

  return <UIModeContext.Provider value={value}>{children}</UIModeContext.Provider>;
}

export function useUIMode() {
  const context = useContext(UIModeContext);
  if (!context) {
    throw new Error("useUIMode must be used within a UIModeProvider");
  }
  return context;
}

"use client";

import React, { useRef, useState, useEffect, useCallback } from "react";
import { Button } from "@/components/ui/button";
import { Eraser, Check, RotateCcw } from "lucide-react";

interface SignaturePadProps {
  onSave?: (dataUrl: string) => void;
  onClear?: () => void;
  width?: number;
  height?: number;
  className?: string;
  initialImage?: string;
}

export function SignaturePad({
  onSave,
  onClear,
  width = 500,
  height = 180,
  className = "",
  initialImage,
}: SignaturePadProps) {
  const canvasRef = useRef<HTMLCanvasElement | null>(null);
  const [isDrawing, setIsDrawing] = useState(false);
  const [hasSignature, setHasSignature] = useState(false);

  // キャンバスの初期化
  const initCanvas = useCallback(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");
    if (!ctx) return;

    // 高解像度ディスプレイ対応（Retina / タブレット対応）
    const ratio = Math.max(window.devicePixelRatio || 1, 1);
    canvas.width = width * ratio;
    canvas.height = height * ratio;
    canvas.style.width = `${width}px`;
    canvas.style.height = `${height}px`;

    ctx.scale(ratio, ratio);
    ctx.lineCap = "round";
    ctx.lineJoin = "round";
    ctx.strokeStyle = "#0f172a"; // slate-900 (濃い墨色)
    ctx.lineWidth = 2.5;

    // 白背景で塗りつぶし（透過PNGではなく白背景付きにすると帳票印刷で綺麗に映る）
    ctx.fillStyle = "#ffffff";
    ctx.fillRect(0, 0, width, height);

    if (initialImage) {
      const img = new Image();
      img.onload = () => {
        ctx.drawImage(img, 0, 0, width, height);
        setHasSignature(true);
      };
      img.src = initialImage;
    }
  }, [width, height, initialImage]);

  useEffect(() => {
    initCanvas();
  }, [initCanvas]);

  // 描画座標の取得
  const getCoordinates = (e: React.MouseEvent | React.TouchEvent) => {
    const canvas = canvasRef.current;
    if (!canvas) return { x: 0, y: 0 };
    const rect = canvas.getBoundingClientRect();

    if ("touches" in e) {
      const touch = e.touches[0];
      return {
        x: touch.clientX - rect.left,
        y: touch.clientY - rect.top,
      };
    } else {
      return {
        x: (e as React.MouseEvent).clientX - rect.left,
        y: (e as React.MouseEvent).clientY - rect.top,
      };
    }
  };

  const startDrawing = (e: React.MouseEvent | React.TouchEvent) => {
    e.preventDefault();
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");
    if (!ctx) return;

    const { x, y } = getCoordinates(e);
    ctx.beginPath();
    ctx.moveTo(x, y);
    setIsDrawing(true);
  };

  const draw = (e: React.MouseEvent | React.TouchEvent) => {
    if (!isDrawing) return;
    e.preventDefault();
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");
    if (!ctx) return;

    const { x, y } = getCoordinates(e);
    ctx.lineTo(x, y);
    ctx.stroke();
    setHasSignature(true);
  };

  const stopDrawing = (e: React.MouseEvent | React.TouchEvent) => {
    if (!isDrawing) return;
    e.preventDefault();
    setIsDrawing(false);
  };

  const handleClear = () => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");
    if (!ctx) return;

    ctx.fillStyle = "#ffffff";
    ctx.fillRect(0, 0, width, height);
    setHasSignature(false);
    if (onClear) onClear();
  };

  const handleSave = () => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const dataUrl = canvas.toDataURL("image/png");
    if (onSave) {
      onSave(dataUrl);
    }
  };

  return (
    <div className={`flex flex-col items-center gap-3 ${className}`}>
      <div className="relative border-2 border-slate-300 rounded-xl overflow-hidden bg-white shadow-inner touch-none select-none">
        <canvas
          ref={canvasRef}
          onMouseDown={startDrawing}
          onMouseMove={draw}
          onMouseUp={stopDrawing}
          onMouseLeave={stopDrawing}
          onTouchStart={startDrawing}
          onTouchMove={draw}
          onTouchEnd={stopDrawing}
          className="cursor-crosshair block"
          style={{ width: `${width}px`, height: `${height}px` }}
        />
        {!hasSignature && (
          <div className="absolute inset-0 flex items-center justify-center pointer-events-none text-slate-300 font-bold text-sm tracking-widest">
            ✍️ 指またはスタイラスペンでここに署名してください
          </div>
        )}
      </div>

      <div className="flex items-center gap-2 w-full justify-between max-w-[500px]">
        <Button
          type="button"
          variant="outline"
          size="sm"
          onClick={handleClear}
          className="gap-1.5 text-xs text-slate-600 border-slate-300 hover:bg-slate-100"
        >
          <RotateCcw className="w-3.5 h-3.5" />
          書き直す（クリア）
        </Button>

        {onSave && (
          <Button
            type="button"
            size="sm"
            onClick={handleSave}
            disabled={!hasSignature}
            className="gap-1.5 text-xs bg-primary hover:bg-primary/90 text-white font-bold px-4 shadow-sm"
          >
            <Check className="w-3.5 h-3.5" />

}

  // nativeButton must be false, otherwise Base UI throws an accessibility violation

    finalRender

      : true
  )

    <ButtonPrimitive
      data-slot="button"
      className={cn(buttonVariants({ variant, size, className }))}
      render={finalRender}
      nativeButton={isNative}
      {...props}
    >
      {finalChildren}
    </ButtonPrimitive>
  )
}

/**
 * モデル文字列（例: "google:gemini-3.5-flash"）をプロバイダとモデルIDに分解
 */
export function parseModelString(modelStr: string | null | undefined): { provider: AiProvider; modelId: string } {
  const fallback = { provider: 'google' as AiProvider, modelId: 'gemini-3.5-flash' };
  if (!modelStr || typeof modelStr !== 'string') return fallback;

  const trimmed = modelStr.trim();
  if (trimmed.includes(':')) {
    const parts = trimmed.split(':');
    const provider = parts[0].toLowerCase() as AiProvider;
    const modelId = parts.slice(1).join(':');
    if (['google', 'anthropic', 'openai'].includes(provider)) {
      return { provider, modelId };
    }
  }

  // プレフィックスなしのレガシーモデル判定
  if (trimmed.startsWith('gemini')) return { provider: 'google', modelId: trimmed };
  if (trimmed.startsWith('claude')) return { provider: 'anthropic', modelId: trimmed };
  if (trimmed.startsWith('gpt') || trimmed.startsWith('o1') || trimmed.startsWith('o3')) return { provider: 'openai', modelId: trimmed };

  return fallback;
}

export async function getSessionData(): Promise<SessionData | null> {
  try {
    const cookieStore = await cookies();
    const session = await getIronSession<SessionData>(cookieStore, sessionOptions);
    if (session.role) {
      return session;
    }
    const appRole = cookieStore.get('app_role')?.value as UserRole;
    if (appRole === 'admin' || appRole === 'service_manager' || appRole === 'staff') {
      return { role: appRole, departmentId: session.departmentId };
    }
    return null;
  } catch (e) {
    return null;
  }
}

/**
 * IDORガード ＆ DBモード自動解決:
 * 実行ユーザーが対象クライアントの拠点にアクセス権を持つか検証すると同時に、
 * 複数タブ操作等で service_mode Cookie と対象 clientId の所属DB（A型/B型）がズレていた場合に
 * 自動的に対象クライアントが所属するDBへリクエストコンテキストとCookieを同期する
 */
export async function assertCanAccessClient(clientId: string): Promise<boolean> {
  const { db, resolveServiceModeForClient, SERVICE_MODE_COOKIE } = await import('@/lib/db');
  const resolvedMode = resolveServiceModeForClient(clientId);

  try {
    const cookieStore = await cookies();
    const currentCookieMode = cookieStore.get(SERVICE_MODE_COOKIE)?.value;
    if (currentCookieMode && currentCookieMode !== resolvedMode) {
      cookieStore.set(SERVICE_MODE_COOKIE, resolvedMode, {
        path: '/',
        maxAge: 60 * 60 * 24 * 365,
        sameSite: 'lax',
      });
    }
  } catch (e) {
    // Server Component からの呼び出し等で cookie.set が不可の場合はリクエストスコープのオーバーライドのみで動作
  }

  const session = await getSessionData();
  const role = session?.role || await getRole();
  if (!role) {
    throw new Error('認証されていません。ログインしてください。');
  }
  if (role === 'admin') {
    return true;
  }
  if (!session?.departmentId || session.departmentId === 'all') {
    return true;
  }
  
  const client = db.prepare('SELECT department_id FROM clients WHERE id = ?').get(clientId) as any;
  if (client && client.department_id && client.department_id !== session.departmentId) {
    throw new Error('【アクセス権限エラー】他拠点の利用者データへのアクセス権限がありません。');
  }
  return true;
}

/**
 * データベースの整合性スナップショットを取得し、AES-256-GCMで暗号化したバッファを生成します。
 */
export async function generateEncryptedBackup(passphrase: string): Promise<{
  buffer: Buffer;
  fileName: string;
  sizeBytes: number;
}> {
  if (!passphrase || passphrase.length < 8) {
    throw new Error('暗号化パスフレーズは8文字以上である必要があります。');
  }

  // 一時ファイルのパス
  const tempSnapshotPath = path.join(process.cwd(), `temp_snapshot_${Date.now()}_${Math.random().toString(36).slice(2, 7)}.sqlite`);

  try {
    // WAL整合性を考慮したフラッシュとSQLiteオンラインバックアップAPI
    try {
      db.pragma('wal_checkpoint(TRUNCATE)');
    } catch (e) {}

    await db.backup(tempSnapshotPath);

    // スナップショットをバッファとして読み込み
    const rawBuffer = fs.readFileSync(tempSnapshotPath);

    // AES-256-GCM で暗号化
    const encryptedBuffer = encryptBuffer(rawBuffer, passphrase);

    // ファイル名生成: welfare_backup_[type_a|type_b]_YYYYMMDD_HHmmss.enc
    const currentMode = getServiceModeSync();
    const timestampStr = format(new Date(), 'yyyyMMdd_HHmmss');
    const fileName = `welfare_backup_${currentMode}_${timestampStr}.enc`;

    return {
      buffer: encryptedBuffer,
      fileName,
      sizeBytes: encryptedBuffer.length
    };
  } finally {
    // 一時ファイルの後始末
    if (fs.existsSync(tempSnapshotPath)) {
      try {
        fs.unlinkSync(tempSnapshotPath);
      } catch (e) {
        console.warn('一時バックアップファイルの削除に失敗しました:', e);
      }
    }
  }
}

/**
 * クラウドストレージ（ローカル同期フォルダまたはS3）へバックアップを保存します。
 */
export async function syncEncryptedBackupToCloud(
  encryptedBuffer: Buffer,
  fileName: string,
  backupType: 'manual_cloud' | 'auto_cloud'
): Promise<{ success: boolean; message: string; destination: string }> {
  const settings = getBackupSettings();

  if (settings.cloud_provider === 'none') {
    throw new Error('クラウドバックアップ先が設定されていません。[設定]から同期先（Google DriveフォルダまたはS3）を選択してください。');
  }

  const logId = uuidv4();
  let destination = '';

  try {
    if (settings.cloud_provider === 'google_drive_folder') {
      destination = 'google_drive_folder';
      const syncPath = settings.local_sync_path;
      if (!syncPath || syncPath.trim() === '') {
        throw new Error('Google Drive / OneDrive の同期先フォルダパスが指定されていません。');
      }

      // パストラバーサル防止 & パス正規化
      const resolvedSyncPath = path.resolve(syncPath);
      const safeFileName = path.basename(fileName);
      if (!safeFileName || safeFileName.includes('..') || !safeFileName.endsWith('.enc')) {
        throw new Error('不正なバックアップファイル名です。');
      }

      // フォルダが存在しない場合は作成
      if (!fs.existsSync(resolvedSyncPath)) {
        fs.mkdirSync(resolvedSyncPath, { recursive: true });
      }

      const syncPathWithSep = resolvedSyncPath.endsWith(path.sep) ? resolvedSyncPath : resolvedSyncPath + path.sep;
      const targetFilePath = path.join(resolvedSyncPath, safeFileName);
      if (!targetFilePath.startsWith(syncPathWithSep)) {
        throw new Error('不正なファイルパス（パストラバーサル）が検出されました。');
      }
      fs.writeFileSync(targetFilePath, encryptedBuffer);

      // 自動クリーンアップ: 同期先フォルダ内の古いバックアップファイル（.enc）を検索し、最新30世代のみ保持
      try {
        const files = fs.readdirSync(resolvedSyncPath)
          .filter(f => f.startsWith('welfare_backup_') && f.endsWith('.enc'))
          .map(f => ({
            name: f,
            time: fs.statSync(path.join(resolvedSyncPath, f)).mtime.getTime()
          }))
          .sort((a, b) => b.time - a.time);

        // 30世代を超えたファイルを削除
        if (files.length > 30) {
          const filesToDelete = files.slice(30);
          for (const f of filesToDelete) {
            try {
              fs.unlinkSync(path.join(resolvedSyncPath, f.name));
            } catch (err) {
              console.warn(`古いバックアップファイルの削除失敗: ${f.name}`, err);
            }
          }
        }
      } catch (cleanErr) {
        console.warn('バックアップ世代整理中の軽微な警告:', cleanErr);
      }

      const msg = `Google Drive / クラウド同期フォルダへの保存が完了しました (${(encryptedBuffer.length / 1024).toFixed(1)} KB)`;

      // 設定およびログ更新
      db.prepare(`
        UPDATE backup_settings 
        SET last_sync_at = datetime('now', 'localtime'), last_sync_status = 'success', last_sync_message = ?
        WHERE id = 'default'
      `).run(msg);

      db.prepare(`
        INSERT INTO backup_logs (id, backup_type, destination, file_name, file_size_bytes, status, message)
        VALUES (?, ?, ?, ?, ?, 'success', ?)
      `).run(logId, backupType, 'google_drive', fileName, encryptedBuffer.length, msg);

      return { success: true, message: msg, destination };

    } else if (settings.cloud_provider === 's3') {
      destination = 's3';
      if (!settings.s3_endpoint || !settings.s3_bucket || !settings.s3_access_key_id || !settings.s3_secret_access_key) {
        throw new Error('S3ストレージの接続情報（エンドポイント、バケット名、アクセスキー）が未設定です。');
      }

      const s3Result = await uploadToS3({
        endpoint: settings.s3_endpoint,
        bucket: settings.s3_bucket,
        region: settings.s3_region || 'auto',
        accessKeyId: settings.s3_access_key_id,
        secretAccessKey: settings.s3_secret_access_key,
        key: `backups/${fileName}`,
        body: encryptedBuffer,
        contentType: 'application/octet-stream'
      });

      if (!s3Result.success) {
        throw new Error(s3Result.message);
      }

      const msg = `S3互換クラウドストレージへの暗号化バックアップ転送が完了しました (${(encryptedBuffer.length / 1024).toFixed(1)} KB)`;

      db.prepare(`
        UPDATE backup_settings 
        SET last_sync_at = datetime('now', 'localtime'), last_sync_status = 'success', last_sync_message = ?
        WHERE id = 'default'
      `).run(msg);

      db.prepare(`
        INSERT INTO backup_logs (id, backup_type, destination, file_name, file_size_bytes, status, message)
        VALUES (?, ?, ?, ?, ?, 'success', ?)
      `).run(logId, backupType, 's3', fileName, encryptedBuffer.length, msg);

      return { success: true, message: msg, destination };
    }

    throw new Error('未対応のクラウドプロバイダーです。');
  } catch (error: any) {
    const errorMsg = error.message || 'クラウド同期に失敗しました。';

    try {
      db.prepare(`
        UPDATE backup_settings 
        SET last_sync_at = datetime('now', 'localtime'), last_sync_status = 'failed', last_sync_message = ?
        WHERE id = 'default'
      `).run(errorMsg);

      db.prepare(`
        INSERT INTO backup_logs (id, backup_type, destination, file_name, file_size_bytes, status, message)
        VALUES (?, ?, ?, ?, ?, 'failed', ?)
      `).run(logId, backupType, destination, fileName, encryptedBuffer.length, errorMsg);
    } catch (e) {}

    return { success: false, message: errorMsg, destination };
  }
}

/**
 * クラウド接続テスト
 */
export async function testCloudConnection(targetSettings: Partial<BackupSettingsRecord>): Promise<{ success: boolean; message: string }> {
  if (targetSettings.cloud_provider === 'google_drive_folder') {
    const syncPath = targetSettings.local_sync_path;
    if (!syncPath || syncPath.trim() === '') {
      return { success: false, message: 'フォルダパスを入力してください。' };
    }

    try {
      const resolvedSyncPath = path.resolve(syncPath);
      const syncPathWithSep = resolvedSyncPath.endsWith(path.sep) ? resolvedSyncPath : resolvedSyncPath + path.sep;
      if (!fs.existsSync(resolvedSyncPath)) {
        fs.mkdirSync(resolvedSyncPath, { recursive: true });
      }
      const testFile = path.join(resolvedSyncPath, `.test_write_${Date.now()}.tmp`);
      if (!testFile.startsWith(syncPathWithSep)) {
        return { success: false, message: '不正なフォルダパスです。' };
      }
      fs.writeFileSync(testFile, 'write_test_ok');
      fs.unlinkSync(testFile);
      return { success: true, message: `フォルダへの書き込み・アクセス確認に成功しました: ${resolvedSyncPath}` };
    } catch (err: any) {
      return { success: false, message: `フォルダへのアクセスに失敗しました: ${err.message}` };
    }
  } else if (targetSettings.cloud_provider === 's3') {
    if (!targetSettings.s3_endpoint || !targetSettings.s3_bucket || !targetSettings.s3_access_key_id || !targetSettings.s3_secret_access_key) {
      return { success: false, message: 'S3の全項目（エンドポイント、バケット、アクセスキー、シークレットキー）を入力してください。' };
    }

    try {
      const testResult = await uploadToS3({
        endpoint: targetSettings.s3_endpoint,
        bucket: targetSettings.s3_bucket,
        region: targetSettings.s3_region || 'auto',
        accessKeyId: targetSettings.s3_access_key_id,
        secretAccessKey: targetSettings.s3_secret_access_key,
        key: 'backups/.connection_test.txt',
        body: Buffer.from('welfare-app connection test ' + new Date().toISOString(), 'utf-8'),
        contentType: 'text/plain'
      });

      if (testResult.success) {
        return { success: true, message: 'S3バケットへのテスト書き込み・認証に成功しました。' };
      } else {
        return { success: false, message: testResult.message };
      }
    } catch (err: any) {
      return { success: false, message: `S3接続テスト失敗: ${err.message}` };
    }
  }

  return { success: false, message: '同期先プロバイダーを選択してください。' };
}

/**
 * 暗号化バックアップデータからのデータベース復元（インポート・リストア）
 * フェイルセーフ設計:
 * 1. ヘッダー・フォーマットチェック
 * 2. 復号テスト＆破損・改ざん検証 (AES-256-GCM AuthTag)
 * 3. SQLiteヘッダーチェック
 * 4. 一時DBファイル展開による主要テーブル構造検証
 * 5. 現行DBの自動退避スナップショット作成 (ロールバック用)
 * 6. データベースファイルの置換と再接続
 */
export async function restoreFromEncryptedBackup(
  encryptedBuffer: Buffer,
  passphrase: string
): Promise<{
  success: boolean;
  message: string;
  details?: {
    clientCount: number;
    staffCount: number;
    facilityName: string;
    backupSavedTime?: string;
  };
}> {
  if (!passphrase) {
    throw new Error('復号パスフレーズを入力してください。');
  }

  // 1. ヘッダー検証
  if (!validateBackupHeader(encryptedBuffer)) {
    throw new Error('選択されたファイルは当システムの暗号化バックアップ形式ではありません。');
  }

  // 2. 復号処理 (認証タグ検証)
  let decryptedBuffer: Buffer;
  try {
    decryptedBuffer = decryptBuffer(encryptedBuffer, passphrase);
  } catch (err: any) {
    throw new Error(err.message || '復号に失敗しました。パスフレーズが一致しません。');
  }

  // 3. SQLiteファイル検証
  if (!isSqliteBuffer(decryptedBuffer)) {
    throw new Error('復号されたデータは有効なSQLiteデータベースではありません。');
  }

  // 4. 一時ファイルでの整合性・テーブル存在検証
  const tempVerifyPath = path.join(process.cwd(), `temp_restore_verify_${Date.now()}.sqlite`);
  let clientCount = 0;
  let staffCount = 0;
  let facilityName = '事業所';
  let targetTableNames: string[] = [];

  let verifyDb: Database.Database | null = null;
  const cleanupTempVerifyFiles = () => {
    if (verifyDb) {
      try { verifyDb.close(); } catch (e) {}
      verifyDb = null;
    }
    for (const suffix of ['', '-shm', '-wal']) {
      const p = `${tempVerifyPath}${suffix}`;
      if (fs.existsSync(p)) {
        try { fs.unlinkSync(p); } catch (e) {}
      }
    }
  };

  try {
    fs.writeFileSync(tempVerifyPath, decryptedBuffer);
    verifyDb = new Database(tempVerifyPath, { readonly: true });

    // 主要テーブルの存在確認
    const tables = verifyDb.prepare("SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%'").all() as { name: string }[];
    targetTableNames = tables.map(t => t.name);
    const tableNames = new Set(targetTableNames);

    const requiredTables = ['clients', 'facility_settings', 'staff', 'attendance_records'];
    const missing = requiredTables.filter(t => !tableNames.has(t));
    if (missing.length > 0) {
      throw new Error(`バックアップデータ内に必須テーブル（${missing.join(', ')}）が見つかりません。整合性に問題があります。`);
    }

    // 件数や事業所名の確認
    try {
      const cRow = verifyDb.prepare('SELECT count(*) as cnt FROM clients').get() as { cnt: number };
      clientCount = cRow ? cRow.cnt : 0;

      const sRow = verifyDb.prepare('SELECT count(*) as cnt FROM staff').get() as { cnt: number };
      staffCount = sRow ? sRow.cnt : 0;

      const fRow = verifyDb.prepare('SELECT facility_name FROM facility_settings LIMIT 1').get() as { facility_name: string } | undefined;
      if (fRow && fRow.facility_name) {
        facilityName = fRow.facility_name;
      }
    } catch (e) {
      console.warn('復元データプレビュー読み取り中の軽微な警告:', e);
    }

    // 事業種別（A型 / B型）の厳格なクロスチェック
    const activeServiceMode = getServiceModeSync(); // 現在のセッションの事業種別
    let backupServiceMode: string | null = null;
    try {
      const metaRow = verifyDb.prepare("SELECT meta_value FROM app_system_meta WHERE meta_key = 'service_mode'").get() as { meta_value: string } | undefined;
      if (metaRow && metaRow.meta_value) {
        backupServiceMode = metaRow.meta_value;
      } else {
        const fRow = verifyDb.prepare("SELECT service_type, facility_name FROM facility_settings LIMIT 1").get() as any;
        if (fRow && fRow.service_type) {
          backupServiceMode = fRow.service_type;
        } else if (fRow && fRow.facility_name && (fRow.facility_name.includes('B型') || fRow.facility_name.includes('就労Ｂ'))) {
          backupServiceMode = 'type_b';
        } else {
          backupServiceMode = 'type_a';
        }
      }
    } catch (e) {
      backupServiceMode = 'type_a';
    }

    if (backupServiceMode && backupServiceMode !== activeServiceMode) {
      const currentName = activeServiceMode === 'type_b' ? '就労継続支援B型' : '就労継続支援A型';
      const targetName = backupServiceMode === 'type_b' ? '就労継続支援B型' : '就労継続支援A型';
      const activeDbPath = getActiveDbPath();
      throw new Error(
        `【事業種別不一致エラー】\n` +
        `選択されたバックアップファイルは「${targetName}」専用データです。\n` +
        `現在操作中の「${currentName}」データベース（${path.basename(activeDbPath)}）への復元は、データの混線・破壊を防止するため固く禁止されています。\n` +
        `ヘッダーまたはログイン画面から「${targetName}」に切り替えてから復元を実行してください。`
      );
    }

    verifyDb.close();
    verifyDb = null;
  } catch (err: any) {
    cleanupTempVerifyFiles();
    throw err;
  }

  // 5. 現行DBの自動退避（フェイルセーフスナップショット）
  const activeDbPath = getActiveDbPath();
  const dbBaseName = path.basename(activeDbPath);
  const backupTimestamp = format(new Date(), 'yyyyMMdd_HHmmss');
  const rollbackSnapshotPath = path.join(process.cwd(), `${dbBaseName}.before_restore_${backupTimestamp}`);

  try {
    // 現行DBファイルを退避（安全バックアップ）
    if (fs.existsSync(activeDbPath)) {
      try {
        db.pragma('wal_checkpoint(TRUNCATE)');
        fs.copyFileSync(activeDbPath, rollbackSnapshotPath);
      } catch (e) {}
    }

    // 6. ATTACH DATABASE を用いた完全オンライン復元処理
    // 稼働中プロセスのファイルハンドルを維持したまま、トランザクション内で全テーブルを安全に同期
    const safeTempPath = tempVerifyPath.replace(/\\/g, '/').replace(/'/g, "''");
    db.exec(`ATTACH DATABASE '${safeTempPath}' AS restore_src;`);

    // テーブル名の安全なホワイトリスト検証（SQLインジェクション対策）
    const validTablePattern = /^[a-zA-Z0-9_]+$/;
    const existingMainTables = new Set(
      (db.prepare("SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%'").all() as { name: string }[]).map(t => t.name)
    );

    db.pragma('foreign_keys = OFF');
    try {
      db.transaction(() => {
        for (const tableName of targetTableNames) {
          // メインDBに実在し、英数字・アンダースコアのみの正規テーブル名のみ許可
          if (!validTablePattern.test(tableName) || !existingMainTables.has(tableName)) {
            console.warn(`許可されていないテーブル名の同期をスキップ: ${tableName}`);
            continue;
          }
          try {
            db.prepare(`DELETE FROM main.${tableName}`).run();
            db.prepare(`INSERT INTO main.${tableName} SELECT * FROM restore_src.${tableName}`).run();
          } catch (tblErr) {
            console.warn(`テーブル ${tableName} の同期中に警告:`, tblErr);
          }
        }
      })();
    } finally {
      db.pragma('foreign_keys = ON');
      try {
        db.exec("DETACH DATABASE restore_src;");
      } catch (e) {}
    }

    // WALフラッシュ
    try {
      db.pragma('wal_checkpoint(TRUNCATE)');
    } catch (e) {}

    const msg = `データベースの復元が正常に完了しました（事業所: ${facilityName}、利用者数: ${clientCount}名、職員数: ${staffCount}名）。万が一のため直前のデータは「${path.basename(rollbackSnapshotPath)}」として自動保存されています。`;

    // 復元ログを記録
    try {
      db.prepare(`
        INSERT INTO backup_logs (id, backup_type, destination, file_name, file_size_bytes, status, message)
        VALUES (?, 'restore', 'local_restore', ?, ?, 'success', ?)
      `).run(uuidv4(), `restore_${backupTimestamp}.enc`, encryptedBuffer.length, msg);
    } catch (e) {}

    return {
      success: true,
      message: msg,
      details: {
        clientCount,
        staffCount,
        facilityName
      }
    };
  } catch (restoreErr: any) {
    // 置換中に致命的エラーが起きた場合は自動ロールバックを試行
    if (fs.existsSync(rollbackSnapshotPath)) {
      try {
        fs.copyFileSync(rollbackSnapshotPath, activeDbPath);
      } catch (rbErr) {}
    }
    throw new Error(`復元処理中にエラーが発生したため、変更をロールバックしました: ${restoreErr.message}`);
  } finally {
    cleanupTempVerifyFiles();
  }
}
