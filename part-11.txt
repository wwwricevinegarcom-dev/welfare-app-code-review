"use server"

import { revalidatePath } from "next/cache";
import { redirect } from "next/navigation";
import { getRole } from "@/lib/auth";
import { getCorporateDb } from "@/app/(corporate)/corporate/lib/db";

export async function updateCorporateSettings(formData: FormData) {
  const role = await getRole();
  if (role !== 'admin') {
    redirect('/settings?error=' + encodeURIComponent('管理者権限が必要です'));
  }

  const corporation_name = (formData.get('corporation_name') as string)?.trim() || '社会福祉法人 かがやき会';
  const facility_name = (formData.get('facility_name') as string)?.trim() || '法人統括本部';
  const representative_name = (formData.get('representative_name') as string)?.trim() || '';
  const postal_code = (formData.get('postal_code') as string)?.trim() || '';
  const address = (formData.get('address') as string)?.trim() || '';
  const phone_number = (formData.get('phone_number') as string)?.trim() || '';
  const fax_number = (formData.get('fax_number') as string)?.trim() || '';
  
  const zengin_bank_code = (formData.get('zengin_bank_code') as string)?.trim() || '0001';
  const zengin_branch_code = (formData.get('zengin_branch_code') as string)?.trim() || '100';
  const zengin_account_type = (formData.get('zengin_account_type') as string)?.trim() || '1';
  const zengin_account_number = (formData.get('zengin_account_number') as string)?.trim() || '1234567';
  const zengin_client_name = (formData.get('zengin_client_name') as string)?.trim() || '';
  const zengin_client_code = (formData.get('zengin_client_code') as string)?.trim() || '';

  try {
    const db = getCorporateDb();
    db.prepare(`
      UPDATE corporate_settings
      SET 
        corporation_name = ?,
        facility_name = ?,
        representative_name = ?,
        postal_code = ?,
        address = ?,
        phone_number = ?,
        fax_number = ?,
        zengin_bank_code = ?,
        zengin_branch_code = ?,
        zengin_account_type = ?,
        zengin_account_number = ?,
        zengin_client_name = ?,
        zengin_client_code = ?,
        updated_at = CURRENT_TIMESTAMP
      WHERE id = 'default'
    `).run(
      corporation_name,
      facility_name,
      representative_name,
      postal_code,
      address,
      phone_number,
      fax_number,
      zengin_bank_code,
      zengin_branch_code,
      zengin_account_type,
      zengin_account_number,
      zengin_client_name,
      zengin_client_code
    );

    revalidatePath('/settings');
    revalidatePath('/corporate/dashboard');
    revalidatePath('/corporate/billing');
    revalidatePath('/');
  } catch (error) {
    console.error('Failed to update corporate settings:', error);
    redirect('/settings?error=' + encodeURIComponent('法人統括設定の保存に失敗しました'));
  }

  redirect('/settings?msg=' + encodeURIComponent('法人統括の基本設定を正常に更新しました'));
}

'use server';

import { db, resolveServiceModeForDepartment } from "@/lib/db";
import { getRole } from "@/lib/auth";
import { randomUUID } from "crypto";
import { revalidatePath } from "next/cache";

export async function getDepartments() {
  try {
    const departments = db.prepare(`
      SELECT d.*,
             (SELECT COUNT(*) FROM clients WHERE department_id = d.id AND deleted_at IS NULL) as client_count,
             (SELECT COUNT(*) FROM staff WHERE department_id = d.id AND deleted_at IS NULL) as staff_count
      FROM departments d
      WHERE d.deleted_at IS NULL
      ORDER BY d.sort_order ASC, d.created_at ASC
    `).all() as any[];
    return departments;
  } catch (error) {
    console.error("Failed to get departments:", error);
    return [];
  }
}

export async function createDepartment(formData: FormData) {
  try {
    const role = await getRole();
    if (role !== 'admin') return { success: false, error: "管理者権限が必要です" };

    const name = (formData.get('name') as string || '').trim();
    const description = (formData.get('description') as string || '').trim();
    const facilityType = (formData.get('facility_type') as string || 'main').trim();
    const capacity = parseInt(formData.get('capacity') as string || '10', 10) || 10;

    if (!name) {
      return { success: false, error: "部署・拠点名は必須です" };
    }

    const id = randomUUID();
    db.prepare(`
      INSERT INTO departments (id, name, description, facility_type, capacity, sort_order)
      VALUES (?, ?, ?, ?, ?, (SELECT COALESCE(MAX(sort_order), 0) + 1 FROM departments))
    `).run(id, name, description || null, facilityType, capacity);

    revalidatePath('/settings');
    revalidatePath('/settings/departments');
    revalidatePath('/clients');
    revalidatePath('/staff');
    revalidatePath('/attendance');
    revalidatePath('/attendance/shifts');
    revalidatePath('/daily-logs');
    revalidatePath('/support-records');
    revalidatePath('/support-records/ledger');
    revalidatePath('/');

    return { success: true, id };
  } catch (error: any) {
    console.error("Failed to create department:", error);
    return { success: false, error: error.message || "部署・拠点の登録に失敗しました" };
  }
}

export async function updateDepartment(id: string, name: string, description?: string, facilityType: string = 'main', capacity: number = 10) {
  try {
    const role = await getRole();
    if (role !== 'admin') return { success: false, error: "管理者権限が必要です" };

    const trimmedName = name.trim();
    if (!trimmedName) return { success: false, error: "部署・拠点名は必須です" };

    await resolveServiceModeForDepartment(id);
    db.prepare(`
      UPDATE departments 
      SET name = ?, description = ?, facility_type = ?, capacity = ?
      WHERE id = ?
    `).run(trimmedName, description?.trim() || null, facilityType, capacity, id);

    revalidatePath('/settings');
    revalidatePath('/settings/departments');
    revalidatePath('/clients');
    revalidatePath('/staff');
    revalidatePath('/attendance');
    revalidatePath('/attendance/shifts');
    revalidatePath('/daily-logs');
    revalidatePath('/support-records');
    revalidatePath('/support-records/ledger');
    revalidatePath('/');

    return { success: true };
  } catch (error: any) {
    console.error("Failed to update department:", error);
    return { success: false, error: error.message || "部署・拠点の更新に失敗しました" };
  }
}

export async function deleteDepartment(id: string) {
  try {
    const role = await getRole();
    if (role !== 'admin') return { success: false, error: "管理者権限が必要です" };

    await resolveServiceModeForDepartment(id);
    // 所属している利用者や職員のdepartment_idをNULLに更新
    db.transaction(() => {
      db.prepare('UPDATE clients SET department_id = NULL WHERE department_id = ?').run(id);
      db.prepare('UPDATE staff SET department_id = NULL WHERE department_id = ?').run(id);
      db.prepare('UPDATE departments SET deleted_at = CURRENT_TIMESTAMP WHERE id = ?').run(id);
    })();

    revalidatePath('/settings');
    revalidatePath('/settings/departments');
    revalidatePath('/clients');
    revalidatePath('/staff');
    revalidatePath('/attendance');
    revalidatePath('/attendance/shifts');
    revalidatePath('/daily-logs');
    revalidatePath('/support-records');
    revalidatePath('/support-records/ledger');
    revalidatePath('/');

    return { success: true };
  } catch (error: any) {
    console.error("Failed to delete department:", error);
    return { success: false, error: error.message || "部署・拠点の削除に失敗しました" };
  }
}

import { db } from '@/lib/db';
import { getRole } from '@/lib/auth';

export interface JurisdictionRecord {
  id: string;
  jurisdiction_type: 'national' | 'prefecture' | 'city';
  jurisdiction_code: string;
  name: string;
  local_rules_json: string;
  draft_rules_json: string | null;
  draft_source_url: string | null;
  draft_summary: string | null;
  draft_updated_at: string | null;
  created_at: string;
  updated_at: string;
}

/**
 * 自治体マスター一覧と現在の有効ルールを取得
 */
export async function getJurisdictionsData(): Promise<{
  jurisdictions: JurisdictionRecord[];
  facilityRules: EffectiveLocalRules;
  facilityLocation: { prefCode?: string; cityCode?: string; facilityName?: string };
}> {
  try {
    const role = await getRole();
    if (!role) {
      return {
        jurisdictions: [],
        facilityRules: DEFAULT_NATIONAL_RULES,
        facilityLocation: {},
      };
    }

    const jurisdictions = db.prepare(`
      SELECT * FROM jurisdictions 
      ORDER BY 
        CASE jurisdiction_type 
          WHEN 'national' THEN 1 
          WHEN 'prefecture' THEN 2 
          WHEN 'city' THEN 3 
          ELSE 4 
        END,
        jurisdiction_code ASC
    `).all() as JurisdictionRecord[];

    const facility = db.prepare(`
      SELECT pref_code, city_code, facility_name FROM facility_settings LIMIT 1
    `).get() as { pref_code?: string; city_code?: string; facility_name?: string } | undefined;

    const facilityRules = getFacilityEffectiveRules();

    return {
      jurisdictions,
      facilityRules,
      facilityLocation: {
        prefCode: facility?.pref_code,
        cityCode: facility?.city_code,
        facilityName: facility?.facility_name,
      },
    };
  } catch (e) {
    console.error('Failed to get jurisdictions data:', e);
    return {
      jurisdictions: [],
      facilityRules: DEFAULT_NATIONAL_RULES,
      facilityLocation: {},
    };
  }
}

/**

export async function updateJurisdictionRules(id: string, rulesJson: string) {
  const role = await getRole();
  if (role !== 'admin') return { success: false, error: '管理者権限が必要です' };

    JSON.parse(rulesJson);
  } catch (e) {

  }

  try {
    db.prepare(`
      UPDATE jurisdictions 
      SET local_rules_json = ?, updated_at = CURRENT_TIMESTAMP 
      WHERE id = ?
    `).run(rulesJson, id);

    revalidatePath('/settings/jurisdictions');
    revalidatePath('/settings');
    return { success: true };
  } catch (e: any) {
    return { success: false, error: e.message };
  }
}

/**
 * AIによるルール変更案（Draft）の作成
 * ※【絶対ルール】この処理では local_rules_json を直接更新しません。
 * draft_rules_json, draft_source_url に保存し、ユーザーの承認（Approve）を待ちます。
 */
export async function proposeAiRuleDraft(params: {
  jurisdictionId: string;
  sourceUrl: string;
  noticeText: string;
}) {
  const role = await getRole();
  if (role !== 'admin') return { success: false, error: '管理者権限が必要です' };

  const jur = db.prepare('SELECT * FROM jurisdictions WHERE id = ?').get(params.jurisdictionId) as JurisdictionRecord | undefined;
  if (!jur) {
    return { success: false, error: '指定された自治体が見つかりません' };
  }

  try {
    const aiModel = getAiModelInstance({ useSearchGrounding: true });

    const prompt = `
あなたは厚生労働省の指定基準、報酬告示、自治体の福祉指導監査基準の専門家です。
以下の自治体向けに、提供された一次情報（自治体通達テキストまたはURL情報）からルール変更案を抽出し、JSON形式で提案してください。

【対象自治体】
名称: ${jur.name} (${jur.jurisdiction_code})
種別: ${jur.jurisdiction_type}
現在の設定JSON:
${jur.local_rules_json}

【参照一次情報URL】
${params.sourceUrl}

【自治体通達テキスト・情報】
${params.noticeText}

【抽出・適用可能なキーの仕様】
- plans:
  - strict_date_sequence: boolean (アセスメント日 <= 原案日 <= 会議日 <= 同意日 <= 交付日の時系列順序の厳格チェック)
  - monitoring_frequency_months: number (モニタリング周期月数。通常6ヶ月)
  - require_user_consent_before_start: boolean (事前同意の厳格必須)
  - require_delivery_record: boolean (交付記録の必須)
- billing:
  - absence_addition_max_days_per_month: number (欠席時対応加算の月上限回数。通常4)
  - telework_requires_daily_report: boolean (在宅勤務利用者の日報提出・確認義務)
  - allow_facility_outreach: boolean (施設外支援の算定可否)
- payroll:
  - strict_minimum_wage_enforcement: boolean (最低賃金の厳格強制)
- reports:
  - offsite_work_format: string ("standard", "kobe_custom", "hyogo_standard", "tokyo_custom" 等)

提供されたテキストや根拠URLから変更が必要な部分のみを更新した「新しい設定JSON全体」と、変更サマリーを作成してください。
`.trim();

    const { object } = await generateObject({
      model: aiModel,
      schema: z.object({
        summary: z.string().describe('改定内容と根拠の日本語サマリー'),
        proposedRulesJson: z.string().describe('改定後の完全なJSON文字列'),
        verifiedSourceUrl: z.string().describe('参照した公的一次情報URL'),
      }),
      prompt,
    });

    // JSON構文の検証
    JSON.parse(object.proposedRulesJson);

    // Draftとして保存（本番ルールは書き換えない！）
    db.prepare(`
      UPDATE jurisdictions 
      SET 
        draft_rules_json = ?, 
        draft_source_url = ?, 
        draft_summary = ?, 
        draft_updated_at = CURRENT_TIMESTAMP
      WHERE id = ?
    `).run(object.proposedRulesJson, object.verifiedSourceUrl || params.sourceUrl, object.summary, params.jurisdictionId);

    revalidatePath('/settings/jurisdictions');
    return { 
      success: true, 
      summary: object.summary,
      draftJson: object.proposedRulesJson
    };
  } catch (e: any) {
    console.error('Error generating rule draft:', e);
    return { success: false, error: 'AIによるルール案作成に失敗しました: ' + e.message };
  }
}

/**
 * ユーザーによる変更案の承認（Human-in-the-loop Approve）
 * ユーザーが明示的にクリックした場合にのみ、draft を本番 local_rules_json に反映します。
 */
export async function approveDraftRules(jurisdictionId: string) {
  const role = await getRole();
  if (role !== 'admin') return { success: false, error: '管理者権限が必要です' };

  const jur = db.prepare('SELECT draft_rules_json FROM jurisdictions WHERE id = ?').get(jurisdictionId) as { draft_rules_json: string | null } | undefined;
  if (!jur || !jur.draft_rules_json) {
    return { success: false, error: '適用可能な変更案（Draft）が存在しません' };
  }

  try {
    JSON.parse(jur.draft_rules_json);
  } catch (e) {
    return { success: false, error: 'DraftのJSON構文が無効です' };
  }

  try {
    db.prepare(`
      UPDATE jurisdictions 
      SET 
        local_rules_json = draft_rules_json,
        draft_rules_json = NULL,
        draft_source_url = NULL,
        draft_summary = NULL,
        draft_updated_at = NULL,
        updated_at = CURRENT_TIMESTAMP
      WHERE id = ?
    `).run(jurisdictionId);

    revalidatePath('/settings/jurisdictions');
    revalidatePath('/settings');
    return { success: true };
  } catch (e: any) {
    return { success: false, error: e.message };
  }
}

/**
 * ユーザーによる変更案の破棄（Human-in-the-loop Reject）
 */
export async function rejectDraftRules(jurisdictionId: string) {
  const role = await getRole();
  if (role !== 'admin') return { success: false, error: '管理者権限が必要です' };

  try {
    db.prepare(`
      UPDATE jurisdictions 
      SET 
        draft_rules_json = NULL,
        draft_source_url = NULL,
        draft_summary = NULL,
        draft_updated_at = NULL
      WHERE id = ?
    `).run(jurisdictionId);

    revalidatePath('/settings/jurisdictions');
    return { success: true };
  } catch (e: any) {
    return { success: false, error: e.message };
  }
}

'use server';

import { db } from "@/lib/db";
import { getRole } from "@/lib/auth";
import iconv from "iconv-lite";
import { parse } from "csv-parse/sync";
import { revalidatePath } from "next/cache";

export async function uploadMasterCsv(formData: FormData) {
  try {
    const role = await getRole();
    if (role !== 'admin') return { success: false, error: "管理者権限が必要です", count: 0 };

    const file = formData.get("file") as File;
    if (!file) return { success: false, error: "ファイルが選択されていません", count: 0 };

    // File size check (50MB)
    if (file.size > 50 * 1024 * 1024) {
      return { success: false, error: "ファイルサイズが上限（50MB）を超えています", count: 0 };
    }

    // File extension check (.csv only)
    if (!file.name.toLowerCase().endsWith('.csv')) {
      return { success: false, error: "CSVファイル（.csv）のみアップロード可能です", count: 0 };
    }

    const buffer = Buffer.from(await file.arrayBuffer());
    
    // WAM NET CSV is typically Shift_JIS
    const decodedStr = iconv.decode(buffer, 'Shift_JIS');

    // Parse CSV
    const records = parse(decodedStr, {
      skip_empty_lines: true,
      relax_column_count: true,
    });

    // Bulk insert
    const stmt = db.prepare(`
      INSERT INTO service_code_master (service_type_code, service_item_code, service_name, unit_score)
      VALUES (?, ?, ?, ?)
      ON CONFLICT(service_type_code, service_item_code) 
      DO UPDATE SET service_name = excluded.service_name, unit_score = excluded.unit_score
    `);

    const insertMany = db.transaction((rows: any[]) => {
      let importedCount = 0;
      for (const row of rows) {
        const serviceType = row[1]?.trim();
        const serviceItem = row[2]?.trim();
        const serviceName = row[6]?.trim() || row[7]?.trim(); // Fallback
        
        // Service Type 43 (就労継続支援A型) and 45/37 (就労継続支援B型)
        const validServiceTypes = ['43', '45', '37'];
        if (validServiceTypes.includes(serviceType) && serviceItem && serviceItem.length === 4) {
          let unitScore = 0;
          for (let i = 7; i <= 15; i++) {
            if (row[i]) {
              const parsed = parseInt(row[i].trim(), 10);
              if (!isNaN(parsed) && parsed > 0 && parsed < 10000) {
                unitScore = parsed;
                break;
              }
            }
          }
          
          stmt.run(serviceType, serviceItem, serviceName, unitScore);
          importedCount++;
        }
      }
      return importedCount;
    });

    const count = insertMany(records);
    revalidatePath('/settings/master');
    return { success: true, count };
  } catch (error: any) {
    console.error("Failed to upload master CSV:", error);
    return { success: false, error: error.message || "CSVのインポートに失敗しました", count: 0 };
  }
}

export async function getMasterStats() {
  try {
    const role = await getRole();
    if (role !== 'admin') return null;
    
    const countRow = db.prepare(`SELECT COUNT(*) as count FROM service_code_master`).get() as { count: number };
    const latestRow = db.prepare(`SELECT created_at FROM service_code_master ORDER BY created_at DESC LIMIT 1`).get() as { created_at: string } | undefined;
    
    return {
      count: countRow.count,
      lastUpdated: latestRow?.created_at || null
    };
  } catch (error) {
    console.error("Failed to get master stats:", error);
    return null;
  }
}

export async function getOffsiteWorkplaces() {

    return db.prepare('SELECT * FROM offsite_workplaces WHERE deleted_at IS NULL ORDER BY created_at DESC').all() as any[];
  } catch (e) {
    console.error("Failed to get offsite workplaces:", e);
    return [];
  }

export async function addOffsiteWorkplace(formData: FormData) {
  try {
    const role = await getRole();
    if (role !== 'admin' && role !== 'service_manager') {
      return { success: false, error: '管理者またはサビ管権限が必要です' };
    }

    const name = formData.get('name') as string;
    const company_name = formData.get('company_name') as string;
    const address = formData.get('address') as string;
    const contact_person = formData.get('contact_person') as string;
    const contact_phone = formData.get('contact_phone') as string;

    if (!name) return { success: false, error: '就労先名が必要です' };

    db.prepare(`
      INSERT INTO offsite_workplaces (id, name, company_name, address, contact_person, contact_phone)
      VALUES (?, ?, ?, ?, ?, ?)
    `).run(randomUUID(), name, company_name, address, contact_person, contact_phone);

    try {
      revalidatePath('/settings/offsite');
    } catch (e) {}

    return { success: true };
  } catch (e: any) {
    console.error("Failed to add offsite workplace:", e);
    return { success: false, error: e.message || '施設外就労先の追加に失敗しました' };
  }
}

export async function deleteOffsiteWorkplace(id: string) {
  try {
    const role = await getRole();
    if (role !== 'admin' && role !== 'service_manager') {
      return { success: false, error: '管理者またはサビ管権限が必要です' };
    }

    await resolveServiceModeForTableRecord('offsite_workplaces', id);

    // Check if it's used
    const used = db.prepare('SELECT id FROM attendance_records WHERE offsite_workplace_id = ? AND deleted_at IS NULL LIMIT 1').get(id);
    if (used) {
      return { success: false, error: 'すでに実績記録（勤怠）で使用されているため削除できません。' };
    }

    db.prepare('UPDATE offsite_workplaces SET deleted_at = CURRENT_TIMESTAMP WHERE id = ?').run(id);

    try {
      revalidatePath('/settings/offsite');
    } catch (e) {}

    return { success: true };
  } catch (e: any) {
    console.error("Failed to delete offsite workplace:", e);
    return { success: false, error: e.message || '施設外就労先の削除に失敗しました' };
  }
}

              errorMsg={errorMsg}
              successMsg={successMsg}
            />
          )}

            />
          )}

          {serviceMode === 'houday' && (
            <SettingsHouday

            />
          )}

          {serviceMode === 'corporate' && (
            <SettingsCorporate

export async function getStaffList() {
  try {
    const role = await getRole();
    if (!role) return [];

    const staff = db.prepare(`
      SELECT s.*, d.name as department_name
      FROM staff s
      LEFT JOIN departments d ON s.department_id = d.id
      WHERE s.deleted_at IS NULL 
      ORDER BY 
        CASE WHEN s.hire_date IS NOT NULL AND s.hire_date != '' THEN 0 ELSE 1 END ASC,
        s.hire_date ASC,
        s.created_at ASC
    `).all() as any[];
    const ptoUsage = await getStaffPtoUsage();
    
    return staff.map(s => ({
      ...s,
      pto_used: ptoUsage[s.id] || 0
    }));
  } catch (error) {
    console.error("Failed to fetch staff:", error);
    return [];
  }
}

export async function getStaffPtoUsage() {
  try {

      WHERE is_holiday = 1 AND holiday_type = '有給休暇'
      GROUP BY staff_id
    `).all() as any[];
    
    const usageMap: Record<string, number> = {};
    for (const row of rows) {
      usageMap[row.staff_id] = row.used_days;
    }
    return usageMap;

    console.error("Failed to fetch PTO usage:", error);
    return {};
  }

export async function saveStaffShift(staffId: string, targetDate: string, data: { 
  scheduled_in: string | null; 
  scheduled_out: string | null; 
  is_holiday: boolean; 
  holiday_type?: string | null;
  is_other_facility?: boolean;
  other_facility_name?: string | null;
  is_offsite?: boolean | number;
  offsite_workplace_id?: string | null;
  offsite_workplace_name?: string | null;
  work_type?: string | null;
  offsite_start_time?: string | null;
  offsite_end_time?: string | null;
  in_facility_start_time?: string | null;
  in_facility_end_time?: string | null;
}) {
  try {
    const role = await getRole();
    if (!role) return { success: false, error: "認証されていません" };

    await resolveServiceModeForStaff(staffId);
    const safeStaffId = sanitizeForeignKeyId('staff', staffId);
    if (!safeStaffId) {
      return { success: false, error: "対象の職員が見つかりません" };
    }
    const safeWorkplaceId = sanitizeForeignKeyId('offsite_workplaces', data.offsite_workplace_id);

    let prevWasPto = false;
    let newIsPto = false;

    db.transaction(() => {
      const existing = db.prepare('SELECT id, is_holiday, holiday_type FROM staff_shifts WHERE staff_id = ? AND target_date = ?').get(safeStaffId, targetDate) as any;
      const isPtoType = (hType?: string | null) => Boolean(hType && ['有給休暇', '有給', '有休'].includes(hType));
      prevWasPto = Boolean(existing && existing.is_holiday === 1 && isPtoType(existing.holiday_type));
      newIsPto = Boolean(data.is_holiday && isPtoType(data.holiday_type));

      const isOffsite = data.is_offsite ? 1 : (data.work_type === 'out_facility' || data.work_type === 'both' ? 1 : 0);
      const workType = data.work_type || (isOffsite ? 'out_facility' : 'in_facility');
      
      if (existing) {
        db.prepare(`
          UPDATE staff_shifts 
          SET scheduled_in = ?, scheduled_out = ?, is_holiday = ?, holiday_type = ?, 
              is_other_facility = ?, other_facility_name = ?,
              is_offsite = ?, offsite_workplace_id = ?, offsite_workplace_name = ?,
              work_type = ?, offsite_start_time = ?, offsite_end_time = ?,
              in_facility_start_time = ?, in_facility_end_time = ?
          WHERE id = ?
        `).run(
          data.scheduled_in, 
          data.scheduled_out, 
          data.is_holiday ? 1 : 0, 
          data.holiday_type || null, 
          data.is_other_facility ? 1 : 0,
          data.other_facility_name || null,
          isOffsite,
          safeWorkplaceId,
          data.offsite_workplace_name || null,
          workType,
          data.offsite_start_time || null,
          data.offsite_end_time || null,
          data.in_facility_start_time || null,
          data.in_facility_end_time || null,
          existing.id
        );
      } else {
        db.prepare(`
          INSERT INTO staff_shifts (
            id, staff_id, target_date, scheduled_in, scheduled_out, is_holiday, holiday_type, 
            is_other_facility, other_facility_name, is_offsite, offsite_workplace_id, offsite_workplace_name,
            work_type, offsite_start_time, offsite_end_time, in_facility_start_time, in_facility_end_time
          )
          VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        `).run(
          randomUUID(), 
          safeStaffId, 
          targetDate, 
          data.scheduled_in, 
          data.scheduled_out, 
          data.is_holiday ? 1 : 0, 
          data.holiday_type || null, 
          data.is_other_facility ? 1 : 0,
          data.other_facility_name || null,
          isOffsite,
          safeWorkplaceId,
          data.offsite_workplace_name || null,
          workType,
          data.offsite_start_time || null,
          data.offsite_end_time || null,
          data.in_facility_start_time || null,
          data.in_facility_end_time || null
        );
      }
    })();

    // 職員有休の自動消費・返還連動
    try {
      if (!prevWasPto && newIsPto) {
        await consumeStaffPtoDays(safeStaffId, 1);
      } else if (prevWasPto && !newIsPto) {
        await restoreStaffPtoDays(safeStaffId, 1);
      }
    } catch (ptoErr) {
      console.error("職員有休自動消費/返還エラー:", ptoErr);
    }

    revalidatePath('/staff/shifts');
    revalidatePath('/staff');
    revalidatePath('/attendance');
    revalidatePath('/');
    return { success: true };
  } catch (error: any) {
    console.error("Failed to save shift:", error);
    return { success: false, error: error.message || "シフトの保存に失敗しました" };
  }
}

  const dailyStaffingAnalysis = useMemo(() => {

      inFacilityRequiredMinutes: number;
      inFacilityRequiredFte: number;
      inFacilityStaffMinutes: number;
      inFacilityStaffCount: number;
      isInFacilityCompliant: boolean;
      inFacilityShortageMinutes: number;

      totalRequiredMinutes: number;
      totalStaffMinutes: number;
      isOverallCompliant: boolean;
      clientCount: number;
      requiredMinutes: number;
      actualStaffMinutes: number;
      isCompliant: boolean;
      shortageMinutes: number;
    }> = {};

    daysArray.forEach(d => {
      // 1. 利用者数の集計（内勤 vs 施設外就労）

      let offsiteClientCount = 0;

            if (!cShift.is_holiday && (cShift.scheduled_in || cShift.work_type)) {
              isAttending = true;
              if (cShift.work_type === 'both') {
                offsiteClientCount++;
                inFacilityClientCount++;
              } else if (cShift.work_type === 'out_facility' || cShift.is_offsite === 1) {
                offsiteClientCount++;
              } else {
                inFacilityClientCount++;
              }
            }
          } else {
            try {
              const arr = JSON.parse(c.regular_holidays || '[]');
              const dObj = new Date(d.dateStr);
              if (!arr.includes(dObj.getDay().toString())) {
                isAttending = true;
                inFacilityClientCount++;
              }
            } catch(e) {}
          }
        });
      } else {
        const initialInfo = initialStaffingMap[d.dateStr];
        inFacilityClientCount = initialInfo?.inFacilityClientCount ?? initialInfo?.clientPresentCount ?? 0;
        offsiteClientCount = initialInfo?.offsiteClientCount ?? 0;
      }

      const totalClientCount = inFacilityClientCount + offsiteClientCount;

      // 2. 内勤の必要職員時間（施設外利用者は内勤必要人数の分母から除外）
      const inFacilityRequiredFte = inFacilityClientCount > 0 
        ? Math.round((inFacilityClientCount / ratioDenominator) * 100) / 100 
        : 0;
      const inFacilityRequiredMinutes = Math.round(inFacilityRequiredFte * dailyStandardMinutes);

      const totalRequiredFte = totalClientCount > 0
        ? Math.round((totalClientCount / ratioDenominator) * 100) / 100
        : 0;
      const totalRequiredMinutes = Math.round(totalRequiredFte * dailyStandardMinutes);

      // 3. 職員の勤務時間を集計（内勤直接処遇 vs 施設外引率職員）
      let inFacilityStaffMinutes = 0;
      let inFacilityStaffCount = 0;
      let offsiteStaffMinutes = 0;
      let offsiteStaffCount = 0;

      initialStaff.forEach(s => {
        if (!isDirectCareStaff(s)) return;

        let shift = shiftsData[s.id]?.[d.dateStr];
        if (!shift) {
          try {
            const arr = JSON.parse(s.regular_holidays || "[]");
            const dObj = new Date(d.dateStr);
            if (arr.includes(dObj.getDay().toString())) {
              shift = { is_holiday: 1, holiday_type: '公休' };
            } else {
              shift = {
                is_holiday: 0,
                scheduled_in: s.work_start_time || "09:00",
                scheduled_out: s.work_end_time || "18:00"
              };
            }
          } catch(e) {
            shift = {
              is_holiday: 0,
              scheduled_in: s.work_start_time || "09:00",
              scheduled_out: s.work_end_time || "18:00"
            };
          }

        const isOther = shift?.is_other_facility || ['GH', 'GH勤務', 'グループホーム', '他事業所', '他事業所勤務'].includes(shift?.holiday_type);

        // 施設外就労引率職員判定
        const isOffsiteStaff = shift?.is_offsite === 1 || ['施設外', '施設外引率', '引率'].includes(shift?.holiday_type);

        if (shift.scheduled_in && shift.scheduled_out) {
          const [sh, sm] = shift.scheduled_in.split(':').map(Number);
          const [eh, em] = shift.scheduled_out.split(':').map(Number);
          if (!isNaN(sh) && !isNaN(eh)) {
            let inM = sh * 60 + sm;
            let outM = eh * 60 + em;
            if (outM < inM) outM += 24 * 60;
            let dur = outM - inM;
            const breakM = shift.break_minutes ?? s.break_time_minutes ?? (dur >= 360 ? 60 : 0);
            dur -= breakM;
            if (s.is_concurrent && s.fte_ratio && s.fte_ratio < 1.0) {
              dur = Math.round(dur * s.fte_ratio);
            }
            if (dur > 0) {
              const hasSplitTimes = shift.work_type === 'both' || (shift.offsite_start_time && shift.offsite_end_time);

              if (hasSplitTimes && shift.offsite_start_time && shift.offsite_end_time) {
                // 施設外時間と内勤時間を正確に分割按分
                const [osh, osm] = shift.offsite_start_time.split(':').map(Number);
                const [oeh, oem] = shift.offsite_end_time.split(':').map(Number);
                let offsiteM = (oeh * 60 + oem) - (osh * 60 + osm);
                if (offsiteM < 0) offsiteM += 24 * 60;
                offsiteM = Math.min(dur, Math.max(0, offsiteM));
                const inFacilityM = Math.max(0, dur - offsiteM);

                if (offsiteM > 0) {
                  offsiteStaffMinutes += offsiteM;
                  offsiteStaffCount++;
                }
                if (inFacilityM > 0) {
                  inFacilityStaffMinutes += inFacilityM;
                  inFacilityStaffCount++;
                }
              } else if (isOffsiteStaff) {
                // 終日施設外就労引率職員（※内勤時間には加算しない）
                offsiteStaffMinutes += dur;
                offsiteStaffCount++;
              } else {
                // 終日内勤直接処遇職員
                inFacilityStaffMinutes += dur;
                inFacilityStaffCount++;
              }
            }
          }
        }
      });

      const totalStaffMinutes = inFacilityStaffMinutes + offsiteStaffMinutes;

      // 4. 判定ロジック
      // ① 内勤基準判定（時間充足 ＆ 開所中常時1名以上の直接処遇職員配置）
      const isInFacilityCompliant = inFacilityClientCount === 0 || 
        (inFacilityStaffCount >= 1 && inFacilityStaffMinutes >= inFacilityRequiredMinutes);
      const inFacilityShortageMinutes = Math.max(0, inFacilityRequiredMinutes - inFacilityStaffMinutes);

      // ② 施設外就労基準判定（施設外利用者がいる場合は現場専任引率職員1名以上必須）
      const isOffsiteCompliant = offsiteClientCount === 0 || offsiteStaffCount >= 1;

      // ③ 全体総合判定
      const isOverallCompliant = isInFacilityCompliant && isOffsiteCompliant && 
        (totalClientCount === 0 || totalStaffMinutes >= totalRequiredMinutes);

      result[d.dateStr] = {
        totalClientCount,
        inFacilityClientCount,
        offsiteClientCount,
        inFacilityRequiredMinutes,
        inFacilityRequiredFte,
        inFacilityStaffMinutes,
        inFacilityStaffCount,
        isInFacilityCompliant,
        inFacilityShortageMinutes,
        offsiteStaffCount,
        offsiteStaffMinutes,
        isOffsiteCompliant,
        totalRequiredMinutes,
        totalStaffMinutes,
        isOverallCompliant,
        // 互換用
        clientCount: totalClientCount,
        requiredMinutes: inFacilityRequiredMinutes,
        actualStaffMinutes: inFacilityStaffMinutes,
        isCompliant: isOverallCompliant,
        shortageMinutes: inFacilityShortageMinutes,
      };
    });

    return result;
  }, [daysArray, initialStaffingMap, shiftsData, initialStaff, ratioDenominator, dailyStandardMinutes, clients, clientShiftsMap]);

interface Props {
  currentDateStr: string;   // e.g. "2026-10-05"
  todayStr: string;         // e.g. "2026-10-05"
  basePath?: string;        // e.g. "/support-records/ledger"
}

export function SupportCalendarMonthPicker({ currentDateStr, todayStr, basePath = "/support-records/ledger" }: Props) {
  const router = useRouter();

  const targetDate = new Date(currentDateStr);
  const currentYear = targetDate.getFullYear();
  const currentMonth = targetDate.getMonth();
  const currentMonthStr = `${currentYear}-${String(currentMonth + 1).padStart(2, '0')}`;

  const prevMonthDate = new Date(currentYear, currentMonth - 1, 1);
  const nextMonthDate = new Date(currentYear, currentMonth + 1, 1);
  const prevMonthStr = `${prevMonthDate.getFullYear()}-${String(prevMonthDate.getMonth() + 1).padStart(2, '0')}-01`;
  const nextMonthStr = `${nextMonthDate.getFullYear()}-${String(nextMonthDate.getMonth() + 1).padStart(2, '0')}-01`;

  const getPastMonthStr = (monthsAgo: number) => {
    const d = new Date();
    d.setMonth(d.getMonth() - monthsAgo);
    return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-01`;
  };

  const handleMonthChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    if (e.target.value) {
      router.push(`${basePath}?date=${e.target.value}-01`);
    }
  };

  const handleJump = (dateStr: string) => {
    router.push(`${basePath}?date=${dateStr}`);
  };

  return (
    <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
      {/* 年月表示と前後ボタン */}
      <div className="flex items-center gap-2">
        <div className="flex items-center bg-white dark:bg-slate-900 border rounded-lg p-0.5 shadow-xs">
          <Button
            variant="ghost"
            size="sm"
            onClick={() => handleJump(prevMonthStr)}
            className="h-8 px-2 text-xs font-semibold text-slate-700 hover:text-slate-900"
            title="前月へ"
          >
            <ChevronLeft className="w-4 h-4 mr-0.5" />
            前月
          </Button>

          <Button
            variant={currentMonthStr === todayStr.substring(0, 7) ? "secondary" : "ghost"}
            size="sm"
            onClick={() => handleJump(todayStr)}
            className="h-8 px-2.5 text-xs font-bold"
            title="現在の月に戻る"
          >
            今月
          </Button>

          <Button
            variant="ghost"
            size="sm"
            onClick={() => handleJump(nextMonthStr)}
            className="h-8 px-2 text-xs font-semibold text-slate-700 hover:text-slate-900"
            title="次月へ"
          >
            次月
            <ChevronRight className="w-4 h-4 ml-0.5" />
          </Button>
        </div>

        {/* 年月ピッカー（直接年月を指定して過去へジャンプ） */}

export async function saveSupportRecord(formData: FormData): Promise<{ success: boolean; error?: string; warning?: string }> {
  try {
    const role = await getRole()
    if (!role) return { success: false, error: '認証されていません。ログインしてください。' }

    const parseResult = supportRecordSchema.safeParse({
      clientId: formData.get('clientId'),
      recordDate: formData.get('recordDate'),
      taskContent: formData.get('taskContent') as string || '',
      staffComment: formData.get('staffComment') as string || '',
      staffId: formData.get('staffId') as string || null,
    })

    if (!parseResult.success) {
      const msg = getFirstZodErrorMessage(parseResult.error);
      return { success: false, error: msg };
    }
    const parsedData = parseResult.data;
    await assertCanAccessClient(parsedData.clientId);

    const safeStaffId = sanitizeForeignKeyId('staff', parsedData.staffId);
    const temperature = formData.get('temperature') as string || '';
    const bloodPressure = formData.get('bloodPressure') as string || '';
    const sleep = formData.get('sleep') as string || '';
    const mood = formData.get('mood') as string || '';
    const serviceStartTime = formData.get('serviceStartTime') as string || null;
    const serviceEndTime = formData.get('serviceEndTime') as string || null;
    const serviceLocation = formData.get('serviceLocation') as string || null;

    // M-7: バイタルデータの範囲チェック
    if (temperature && temperature.trim() !== '') {
      const temp = parseFloat(temperature);
      if (isNaN(temp) || temp < 34.0 || temp > 42.0) {
        return { success: false, error: '体温は34.0～42.0℃の範囲で入力してください' };
      }
    }

    const vitalsJson = JSON.stringify({
      temperature,
      bloodPressure,
      sleep,
      mood
    });

    // 月次確定ロックチェック
    const yearMonth = parsedData.recordDate.substring(0, 7);
    if (await isMonthLocked(yearMonth)) {
      return { success: false, error: `【警告】${yearMonth}月は請求データが月次確定ロックされているため、支援記録を登録・更新できません。変更が必要な場合は管理者にロック解除を依頼してください。` };
    }

    // S-05: 個別支援計画の有効期間チェック（移行・テスト運用の利便性のため、ブロックせず警告通知にとどめて保存を許可）
    const activePlan = db.prepare(`
      SELECT id, plan_date, start_date, end_date, status
      FROM support_plans
      WHERE client_id = ? 
        AND status = 'active'
        AND start_date <= ?
        AND end_date >= ?
        AND deleted_at IS NULL
    `).get(parsedData.clientId, parsedData.recordDate, parsedData.recordDate) as any;

    let planWarning = '';
    if (!activePlan) {
      const client = db.prepare('SELECT full_name, service_start_date FROM clients WHERE id = ?').get(parsedData.clientId) as any;
      planWarning = `【注意・個別支援計画未作成】${client?.full_name || '対象者'}さんの記録日（${parsedData.recordDate}）に有効な個別支援計画が存在しません（未作成のまま本番請求すると「個別支援計画未作成減算（-30%）」の対象となります。後ほど計画を作成・有効化してください）。`;
    }

    // 実地指導・監査対応: 出勤実績との整合性チェック（移行・テスト運用の利便性のため警告通知にとどめて保存を許可）
    const attendance = db.prepare(`
      SELECT status, is_absence_addition 
      FROM attendance_records 
      WHERE client_id = ? AND record_date = ? AND deleted_at IS NULL
    `).get(parsedData.clientId, parsedData.recordDate) as any;

    let attendanceWarning = '';
    if (!attendance || (attendance.status !== 'present' && attendance.status !== 'out_facility' && attendance.is_absence_addition !== 1)) {
      const client = db.prepare('SELECT full_name FROM clients WHERE id = ?').get(parsedData.clientId) as any;
      attendanceWarning = `【注意・勤怠未登録】${client?.full_name || '対象者'}さんの記録日（${parsedData.recordDate}）の出勤実績が確認できません（後ほど出退勤打刻・利用実績を登録してください）。`;
    }

    // 法令専任チェック: サービス管理責任者、管理者、賃金向上達成指導員等の直接処遇外職種は直接支援記録の担当者として登録不可（警告通知）
    let staffWarning = '';
    if (safeStaffId) {
      const assignedStaff = db.prepare('SELECT name, job_category, role FROM staff WHERE id = ?').get(safeStaffId) as any;
      if (assignedStaff) {
        const nonDirectCategories = [
          'サービス管理責任者', '管理者', '管理者 兼 サービス管理責任者', 
          '賃金向上達成指導員', '視覚・聴覚言語障害者支援指導員', '重度者支援専門員', 
          '送迎ドライバー', '調理員', '事務員'
        ];
        const isDirectCare = (
          (assignedStaff.job_category === '職業指導員' || assignedStaff.job_category === '生活支援員') ||
          (assignedStaff.role === 'vocational_instructor' || assignedStaff.role === 'life_support_worker')
        ) && !nonDirectCategories.includes(assignedStaff.job_category);

        if (!isDirectCare) {
          staffWarning = `【注意・専任基準】担当の${assignedStaff.name}さん（${assignedStaff.job_category || assignedStaff.role}）は法令上の専任義務がある役職です（実地指導では職業指導員・生活支援員の記録が求められます）。`;
        }
      }
    }

    // 現場UX改善: 同日記録がある場合の追記対応
    const existing = db.prepare(`
      SELECT vitals, task_content, staff_comment 
      FROM support_records 
      WHERE client_id = ? AND record_date = ?
    `).get(parsedData.clientId, parsedData.recordDate) as any;

    let finalStaffComment = parsedData.staffComment;
    let finalTaskContent = parsedData.taskContent;
    const isAppend = formData.get('isAppend') === 'true';

    if (existing && isAppend) {
      if (existing.staff_comment && parsedData.staffComment && existing.staff_comment !== parsedData.staffComment) {
        finalStaffComment = `${existing.staff_comment}\n---\n【追記】${parsedData.staffComment}`;
      } else {
        finalStaffComment = parsedData.staffComment || existing.staff_comment || '';
      }
      if (!finalTaskContent && existing.task_content) {
        finalTaskContent = existing.task_content;
      }
    }

    const stmt = db.prepare(`
      INSERT INTO support_records (
        id, client_id, record_date, vitals, task_content, staff_comment, staff_id,
        service_start_time, service_end_time, service_location
      ) VALUES (
        ?, ?, ?, ?, ?, ?, ?, ?, ?, ?
      )
      ON CONFLICT(client_id, record_date) DO UPDATE SET
        vitals = excluded.vitals,
        task_content = excluded.task_content,
        staff_comment = excluded.staff_comment,
        staff_id = excluded.staff_id,
        service_start_time = excluded.service_start_time,
        service_end_time = excluded.service_end_time,
        service_location = excluded.service_location,
        deleted_at = NULL
    `);

    stmt.run(
      uuidv4(),
      parsedData.clientId,
      parsedData.recordDate,
      vitalsJson,
      finalTaskContent,
      finalStaffComment,
      safeStaffId,
      serviceStartTime,
      serviceEndTime,
      serviceLocation
    );

    // 監査ログに記録
    await logAudit({
      action: 'update',
      targetTable: 'support_records',
      targetId: `${parsedData.clientId}:${parsedData.recordDate}`,
      details: { clientId: parsedData.clientId, date: parsedData.recordDate }
    });

    try {
      revalidatePath('/support-records');
      revalidatePath('/support-records/ledger');
    } catch (e) {}
    const allWarnings = [planWarning, attendanceWarning, staffWarning].filter(Boolean).join(' / ');
    return {
      success: true,
      warning: allWarnings || undefined
    };
  } catch (error: any) {
    if (error && error.message && error.message.includes('NEXT_REDIRECT')) {
      throw error;
    }
    console.error('支援記録保存エラー:', error);
    return { success: false, error: error.message || '支援記録の保存に失敗しました' };
  }

    const defaultStartTime = client.work_start_time || '09:00';
    const defaultEndTime = client.work_end_time || '15:00';

    let recommendedStartTime = defaultStartTime;
    let recommendedEndTime = defaultEndTime;
    let recommendedLocation: '施設内' | '施設外' | '在宅' = '施設内';
    let recommendedSource: 'attendance' | 'shift' | 'default' = 'default';

    if (att && (att.clock_in || att.clock_out)) {
      recommendedStartTime = att.clock_in || shift?.scheduled_in || defaultStartTime;
      recommendedEndTime = att.clock_out || shift?.scheduled_out || defaultEndTime;
      recommendedSource = 'attendance';
      if (att.is_telework === 1) {
        recommendedLocation = '在宅';
      } else if (att.is_offsite === 1 || att.status === 'out_facility') {
        recommendedLocation = '施設外';
      } else {
        recommendedLocation = '施設内';
      }
    } else if (shift && (shift.scheduled_in || shift.scheduled_out)) {
      recommendedStartTime = shift.scheduled_in || defaultStartTime;
      recommendedEndTime = shift.scheduled_out || defaultEndTime;
      recommendedSource = 'shift';
      if (shift.work_type === 'out_facility' || shift.is_offsite === 1) {
        recommendedLocation = '施設外';
      } else {
        recommendedLocation = '施設内';
      }
    }

    return {
      hasAttendance: !!att,
      clockIn: att?.clock_in || null,
      clockOut: att?.clock_out || null,
      attendanceStatus: att?.status || null,
      isOffsite: att?.is_offsite === 1,
      isTelework: att?.is_telework === 1,
      isAbsenceAddition: att?.is_absence_addition === 1,
      hasShift: !!shift,
      scheduledIn: shift?.scheduled_in || null,
      scheduledOut: shift?.scheduled_out || null,
      shiftWorkType: shift?.work_type || null,
      defaultStartTime,
      defaultEndTime,
      recommendedStartTime,
      recommendedEndTime,
      recommendedLocation,
      recommendedSource,
      existingRecord: existing ? {
        id: existing.id,
        taskContent: existing.task_content || '',
        staffComment: existing.staff_comment || '',
        serviceStartTime: existing.service_start_time || null,
        serviceEndTime: existing.service_end_time || null,
        serviceLocation: existing.service_location || null,
        staffId: existing.staff_id || null,
        vitals: parsedVitals
      } : null
    };
  } catch (error) {
    console.error('スケジュール・打刻取得エラー:', error);
    return null;
  }
}

export async function updateSupportRecordInline(payload: {
  clientId: string;
  recordDate: string;
  serviceStartTime?: string | null;
  serviceEndTime?: string | null;
  serviceLocation?: string | null;
  temperature?: string;
  bloodPressure?: string;
  sleep?: string;
  mood?: string;
  taskContent: string;
  staffComment: string;
  staffId?: string | null;
}): Promise<{ success: boolean; error?: string; warning?: string }> {
  try {
    const role = await getRole();
    if (!role) return { success: false, error: 'ログインが必要です' };
    await assertCanAccessClient(payload.clientId);

    if (!payload.taskContent || payload.taskContent.trim() === '') {
      return { success: false, error: '作業内容を入力してください' };
    }

    // M-7: バイタルチェック
    if (payload.temperature && payload.temperature.trim() !== '') {
      const temp = parseFloat(payload.temperature);
      if (isNaN(temp) || temp < 34.0 || temp > 42.0) {
        return { success: false, error: '体温は34.0～42.0℃の範囲で入力してください' };
      }
    }

    // 月次確定ロック
    const yearMonth = payload.recordDate.substring(0, 7);
    if (await isMonthLocked(yearMonth)) {
      return { success: false, error: `【警告】${yearMonth}月は請求データが月次確定ロックされているため、支援記録を登録・更新できません。` };
    }

    // 個別支援計画チェック（移行・テスト運用の利便性のため、ブロックせず保存を許可）
    const activePlan = db.prepare(`
      SELECT id FROM support_plans
      WHERE client_id = ? AND status = 'active' AND start_date <= ? AND end_date >= ? AND deleted_at IS NULL
    `).get(payload.clientId, payload.recordDate, payload.recordDate) as any;

    // 勤怠チェック（移行・テスト運用の利便性のため、ブロックせず保存を許可）
    const attendance = db.prepare(`
      SELECT status, is_absence_addition FROM attendance_records
      WHERE client_id = ? AND record_date = ? AND deleted_at IS NULL
    `).get(payload.clientId, payload.recordDate) as any;

    let inlineWarning: string | undefined;
    if (!activePlan) {
      inlineWarning = '※有効な個別支援計画が未作成です（本番請求時は未作成減算対象）';
    } else if (!attendance || (attendance.status !== 'present' && attendance.status !== 'out_facility' && attendance.is_absence_addition !== 1)) {
      inlineWarning = '※出勤実績が未登録です';
    }

    const vitalsJson = JSON.stringify({
      temperature: payload.temperature || '',
      bloodPressure: payload.bloodPressure || '',
      sleep: payload.sleep || '',
      mood: payload.mood || ''
    });

    const safeStaffId = sanitizeForeignKeyId('staff', payload.staffId);

    const stmt = db.prepare(`
      INSERT INTO support_records (
        id, client_id, record_date, vitals, task_content, staff_comment, staff_id,
        service_start_time, service_end_time, service_location
      ) VALUES (
        ?, ?, ?, ?, ?, ?, ?, ?, ?, ?
      )
      ON CONFLICT(client_id, record_date) DO UPDATE SET
        vitals = excluded.vitals,
        task_content = excluded.task_content,
        staff_comment = excluded.staff_comment,
        staff_id = excluded.staff_id,
        service_start_time = excluded.service_start_time,
        service_end_time = excluded.service_end_time,
        service_location = excluded.service_location,
        deleted_at = NULL
    `);

    stmt.run(
      uuidv4(),
      payload.clientId,
      payload.recordDate,
      vitalsJson,
      payload.taskContent,
      payload.staffComment,
      safeStaffId,
      payload.serviceStartTime || null,
      payload.serviceEndTime || null,
      payload.serviceLocation || null
    );

    await logAudit({
      action: 'update',
      targetTable: 'support_records',
      targetId: `${payload.clientId}:${payload.recordDate}`,
      details: { clientId: payload.clientId, date: payload.recordDate }
    });

    try {
      revalidatePath('/support-records');
      revalidatePath('/support-records/ledger');
    } catch (e) {}
    return { success: true, warning: inlineWarning };
  } catch (err: any) {
    console.error('支援記録保存エラー:', err);
    return { success: false, error: err.message || '保存に失敗しました' };
  }
}
